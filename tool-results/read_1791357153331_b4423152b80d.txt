   458→   458→   *
   459→   459→   * AC-019: The contract includes objective, specification, constraints,
   460→   460→   * required output, acceptance criteria, and evidence.
   461→   461→   *
   462→   462→   * AC-021: Evidence is included so the receiver understands WHY decisions
   463→   463→   * were made, not just WHAT was decided.
   464→   464→   */
   465→   465→  generateHandoff(
   466→   466→    specification: SoftwareSpecification | ProductSpecification,
   467→   467→    targetAgent: "codex" | "antigravity" | "krea",
   468→   468→    constructor?: ConstructorSelection,
   469→   469→    architecture?: ProductArchitecture,
   470→   470→    evidence: ProductEvidence[] = []
   471→   471→  ): HandoffContract {

   473→   473→      ? specification.productName
   474→   474→      : specification.identity.name;
   475→   475→
   476→   476→    const isSoftware = "identity" in specification;
   477→   477→    const softwareSpec = isSoftware ? specification as SoftwareSpecification : undefined;
   478→   478→
   479→   479→    // Determine autonomy level based on target agent and evidence quality
   480→   480→    const autonomy = this.determineAutonomy(targetAgent, evidence);
   481→   481→
   482→   482→    // Determine approval requirements
   483→   483→    const approvals = this.determineApprovals(targetAgent, autonomy);
   484→   484→
   485→   485→    // Build constraints
   486→   486→    const outOfScope = architecture
   487→   487→      ? architecture.OUT.map((item) => item.description)
   488→   488→      : ["Features not in MUST list", "Multi-language support", "Mobile native apps"];

   491→   491→      handoff_id: randomUUID(),

   493→   493→      target_agent: targetAgent,

   498→   498→        format: architecture?.format || "unspecified",
   499→   499→      },
   500→   500→      input_specification: specification,
   501→   501→      constraints: {
   502→   502→        timebox: architecture?.mvp_definition.timebox || "4-6 weeks",

   504→   504→          "All acceptance criteria must pass",
   505→   505→          "No critical or high-severity bugs",
   506→   506→          "Code review completed",
   507→   507→        ],
   508→   508→        outOfScope,
   509→   509→      },
   510→   510→      required_output: {
   511→   511→        deliverables: this.inferDeliverables(targetAgent, architecture),
   512→   512→        format: this.inferOutputFormat(targetAgent),
   513→   513→        qualityStandards: [
   514→   514→          "All acceptance criteria met",
   515→   515→          "Test coverage >= 80%",

   517→   517→          "Documentation for all public APIs",

   519→   519→      },

   523→   523→        "User acceptance testing completed",
   524→   524→      ],

   527→   527→      risk: {
   528→   528→        level: this.assessHandoffRisk(targetAgent, evidence),
   529→   529→        factors: architecture?.risks || ["Unknown technical complexity"],
   530→   530→        mitigations: [
   531→   531→          "Phased delivery with milestones",
   532→   532→          "Regular progress check-ins",
   533→   533→          "Rollback plan for each phase",
   534→   534→        ],
   535→   535→      },
   536→   536→      autonomy,

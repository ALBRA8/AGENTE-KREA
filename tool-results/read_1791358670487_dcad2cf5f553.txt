     1→/**
     2→ * Feedback & Learning System for AGENTE-KREA
     3→ *
     4→ * Records feedback on agent executions and extracts learning.
     5→ * Distinguishes real success from partial success from failure.
     6→ * Never automatically converts any result into truth — all learning must be validated.
     7→ *
     8→ * Feedback types: SUCCESS, PARTIAL_SUCCESS, FAILURE, TECHNICAL_ERROR,
     9→ *   BAD_DECISION, BAD_HYPOTHESIS, BAD_TOOL, BAD_EXECUTION, UNKNOWN
    10→ */
    11→
    12→import { db } from "@/lib/db";
    13→import { MemoryManager, type MemoryType, type SourceType, type TruthLevel } from "@/lib/memory";
    14→import { SkillManager } from "@/lib/skills";
    15→
    16→// ─── Type Definitions ────────────────────────────────────────────────
    17→
    18→export type FeedbackType =
    19→  | "SUCCESS"
    20→  | "PARTIAL_SUCCESS"
    21→  | "FAILURE"
    22→  | "TECHNICAL_ERROR"
    23→  | "BAD_DECISION"
    24→  | "BAD_HYPOTHESIS"
    25→  | "BAD_TOOL"
    26→  | "BAD_EXECUTION"
    27→  | "UNKNOWN";
    28→
    29→export type ValidationStatus = "PENDING" | "VALIDATED" | "INVALIDATED";
    30→
    31→export interface RecordFeedbackOptions {
    32→  agentId?: string;
    33→  userId?: string;
    34→  input?: Record<string, unknown>;
    35→  result?: Record<string, unknown>;
    36→  feedbackText?: string;
    37→  confidence?: number;
    38→  evidence?: Record<string, unknown>;
    39→}
    40→
    41→export interface FeedbackAnalysis {
    42→  category: FeedbackType;
    43→  severity: "low" | "medium" | "high" | "critical";
    44→  rootCause?: string;
    45→  contributingFactors?: string[];
    46→  isRecoverable: boolean;
    47→  patterns?: string[];
    48→  recommendations?: string[];
    49→  estimatedConfidence: number;
    50→}
    51→
    52→export interface ExtractedLearning {
    53→  domain: string;
    54→  type: MemoryType;
    55→  content: Record<string, unknown>;
    56→  source: string;
    57→  sourceType: SourceType;
    58→  truthLevel: TruthLevel;
    59→  confidence: number;
    60→  needsValidation: boolean;
    61→  affectedSkills?: string[];
    62→  affectedMemories?: string[];
    63→}
    64→
    65→// ─── Feedback Category Severity ──────────────────────────────────────
    66→
    67→const FEEDBACK_SEVERITY: Record<FeedbackType, "low" | "medium" | "high" | "critical"> = {
    68→  SUCCESS: "low",
    69→  PARTIAL_SUCCESS: "medium",
    70→  FAILURE: "high",
    71→  TECHNICAL_ERROR: "high",
    72→  BAD_DECISION: "high",
    73→  BAD_HYPOTHESIS: "critical",
    74→  BAD_TOOL: "high",
    75→  BAD_EXECUTION: "critical",
    76→  UNKNOWN: "medium",
    77→};
    78→
    79→// ─── FeedbackManager ─────────────────────────────────────────────────
    80→
    81→export class FeedbackManager {
    82→  private defaultAgentId = "krea";
    83→  private memoryManager = new MemoryManager();
    84→  private skillManager = new SkillManager();
    85→
    86→  /**
    87→   * Record feedback on an execution.
    88→   * Creates a Feedback entry linked to the execution (if provided).
    89→   */
    90→  async record(
    91→    executionId: string | null,
    92→    feedbackType: FeedbackType,
    93→    feedbackText: string,
    94→    options: RecordFeedbackOptions = {}
    95→  ): Promise<{ id: string }> {
    96→    const agentId = options.agentId || this.defaultAgentId;
    97→
    98→    // Validate execution exists if provided
    99→    if (executionId) {
   100→      const execution = await db.execution.findUnique({ where: { id: executionId } });
   101→      if (!execution) throw new Error(`Execution ${executionId} not found`);
   102→    }
   103→
   104→    const feedback = await db.feedback.create({
   105→      data: {
   106→        agentId,
   107→        userId: options.userId || null,
   108→        executionId,
   109→        input: options.input ? JSON.stringify(options.input) : null,
   110→        result: options.result ? JSON.stringify(options.result) : null,
   111→        feedbackType,
   112→        feedbackText,
   113→        confidence: options.confidence ?? this.inferConfidence(feedbackType),
   114→        evidence: options.evidence ? JSON.stringify(options.evidence) : null,
   115→        validationStatus: "PENDING",
   116→      },
   117→    });
   118→
   119→    // Auto-analyze for non-success feedback
   120→    if (feedbackType !== "SUCCESS") {
   121→      await this.analyze(feedback.id);
   122→    }
   123→
   124→    return { id: feedback.id };
   125→  }
   126→
   127→  /**
   128→   * Automatically analyze feedback to determine root cause, severity, and patterns.
   129→   */
   130→  async analyze(feedbackId: string): Promise<FeedbackAnalysis> {
   131→    const feedback = await db.feedback.findUnique({ where: { id: feedbackId } });
   132→    if (!feedback) throw new Error(`Feedback ${feedbackId} not found`);
   133→
   134→    const analysis: FeedbackAnalysis = {
   135→      category: feedback.feedbackType as FeedbackType,
   136→      severity: FEEDBACK_SEVERITY[feedback.feedbackType as FeedbackType],
   137→      isRecoverable: true,
   138→      estimatedConfidence: feedback.confidence,
   139→    };
   140→
   141→    // Analyze based on feedback type
   142→    switch (feedback.feedbackType as FeedbackType) {
   143→      case "SUCCESS":
   144→        analysis.isRecoverable = true;
   145→        analysis.recommendations = ["Record successful pattern for reuse"];
   146→        break;
   147→
   148→      case "PARTIAL_SUCCESS":
   149→        analysis.isRecoverable = true;
   150→        analysis.rootCause = this.identifyPartialSuccessCause(feedback);
   151→        analysis.recommendations = [
   152→          "Review which steps succeeded and which failed",
   153→          "Consider adjusting parameters for the failing steps",
   154→        ];
   155→        break;
   156→
   157→      case "FAILURE":
   158→        analysis.isRecoverable = false;
   159→        analysis.rootCause = this.identifyFailureRootCause(feedback);
   160→        analysis.contributingFactors = this.identifyContributingFactors(feedback);
   161→        analysis.recommendations = [
   162→          "Review execution trace for error details",
   163→          "Consider alternative approaches",
   164→          "Check if prerequisites were met",
   165→        ];
   166→        break;
   167→
   168→      case "TECHNICAL_ERROR":
   169→        analysis.isRecoverable = true;
   170→        analysis.rootCause = "Technical/system error";
   171→        analysis.recommendations = [
   172→          "Retry with backoff",
   173→          "Check system health",
   174→          "Verify external service availability",
   175→        ];
   176→        break;
   177→
   178→      case "BAD_DECISION":
   179→        analysis.isRecoverable = false;
   180→        analysis.severity = "high";
   181→        analysis.rootCause = "Agent made an incorrect decision";
   182→        analysis.patterns = this.identifyDecisionPatterns(feedback);
   183→        analysis.recommendations = [
   184→          "Review decision criteria and thresholds",
   185→          "Update decision logic based on evidence",
   186→          "Add constraints to prevent similar decisions",
   187→        ];
   188→        break;
   189→
   190→      case "BAD_HYPOTHESIS":
   191→        analysis.isRecoverable = false;
   192→        analysis.severity = "critical";
   193→        analysis.rootCause = "Agent operated on an incorrect hypothesis";
   194→        analysis.recommendations = [
   195→          "Invalidate the hypothesis in memory",
   196→          "Record contradictory evidence",
   197→          "Adjust confidence for related hypotheses",
   198→        ];
   199→        break;
   200→
   201→      case "BAD_TOOL":
   202→        analysis.isRecoverable = true;
   203→        analysis.rootCause = "Wrong tool was selected or tool malfunctioned";
   204→        analysis.recommendations = [
   205→          "Review tool selection criteria",
   206→          "Consider alternative tools",
   207→          "Report tool issue",
   208→        ];
   209→        break;
   210→
   211→      case "BAD_EXECUTION":
   212→        analysis.isRecoverable = false;
   213→        analysis.severity = "critical";
   214→        analysis.rootCause = "Execution was flawed in structure or sequence";
   215→        analysis.recommendations = [
   216→          "Review execution procedure",
   217→          "Check step ordering and dependencies",
   218→          "Validate execution plan before running",
   219→        ];
   220→        break;
   221→
   222→      default:
   223→        analysis.recommendations = ["Investigate further", "Gather more evidence"];
   224→    }
   225→
   226→    // Store analysis in the feedback record
   227→    await db.feedback.update({
   228→      where: { id: feedbackId },
   229→      data: {
   230→        analysis: JSON.stringify(analysis),
   231→      },
   232→    });
   233→
   234→    return analysis;
   235→  }

   237→  /**
   238→   * Extract learning from feedback.
   239→   * CRITICAL: Never automatically convert any result into truth.
   240→   * All learning starts with PENDING validation and ESTIMATED/INFERRED truth level.
   241→   */
   242→  async learn(feedbackId: string): Promise<ExtractedLearning[]> {
   243→    const feedback = await db.feedback.findUnique({ where: { id: feedbackId } });
   244→    if (!feedback) throw new Error(`Feedback ${feedbackId} not found`);
   245→
   246→    // Ensure analysis exists
   247→    if (!feedback.analysis) {
   248→      await this.analyze(feedbackId);
   249→    }
   250→
   251→    const refreshedFeedback = await db.feedback.findUnique({ where: { id: feedbackId } });
   252→    if (!refreshedFeedback) throw new Error(`Feedback ${feedbackId} not found after analysis`);
   253→
   254→    const analysis: FeedbackAnalysis = refreshedFeedback.analysis
   255→      ? JSON.parse(refreshedFeedback.analysis)
   256→      : { category: refreshedFeedback.feedbackType as FeedbackType, severity: "medium", isRecoverable: true, estimatedConfidence: 0.5 };
   257→
   258→    const learnings: ExtractedLearning[] = [];
   259→
   260→    switch (refreshedFeedback.feedbackType as FeedbackType) {
   261→      case "SUCCESS": {
   262→        // Record successful pattern — but as ESTIMATED, not VERIFIED
   263→        learnings.push({
   264→          domain: this.inferDomain(refreshedFeedback),
   265→          type: "PROCEDURAL",
   266→          content: {
   267→            pattern: "successful_execution",
   268→            input: refreshedFeedback.input ? JSON.parse(refreshedFeedback.input) : null,
   269→            result: refreshedFeedback.result ? JSON.parse(refreshedFeedback.result) : null,
   270→            feedbackText: refreshedFeedback.feedbackText,
   271→          },
   272→          source: `feedback:${feedbackId}`,
   273→          sourceType: "FEEDBACK",
   274→          truthLevel: "ESTIMATED", // NOT verified — must be validated
   275→          confidence: 0.6, // Moderate confidence from single success
   276→          needsValidation: true,
   277→        });
   278→        break;
   279→      }
   280→
   281→      case "PARTIAL_SUCCESS": {
   282→        // Record what worked and what didn't
   283→        learnings.push({
   284→          domain: this.inferDomain(refreshedFeedback),
   285→          type: "EPISODIC",
   286→          content: {
   287→            pattern: "partial_success",
   288→            whatWorked: this.extractWhatWorked(refreshedFeedback),
   289→            whatFailed: this.extractWhatFailed(refreshedFeedback),
   290→            feedbackText: refreshedFeedback.feedbackText,
   291→          },
   292→          source: `feedback:${feedbackId}`,
   293→          sourceType: "FEEDBACK",
   294→          truthLevel: "ESTIMATED",
   295→          confidence: 0.4,
   296→          needsValidation: true,
   297→        });
   298→        break;
   299→      }
   300→
   301→      case "FAILURE":
   302→      case "BAD_EXECUTION": {
   303→        // Record failure as a negative pattern
   304→        learnings.push({
   305→          domain: this.inferDomain(refreshedFeedback),
   306→          type: "EPISODIC",
   307→          content: {
   308→            pattern: "failure",
   309→            rootCause: analysis.rootCause,
   310→            contributingFactors: analysis.contributingFactors,
   311→            feedbackText: refreshedFeedback.feedbackText,
   312→          },
   313→          source: `feedback:${feedbackId}`,
   314→          sourceType: "FEEDBACK",
   315→          truthLevel: "OBSERVED", // Failures are observable facts
   316→          confidence: 0.8, // High confidence in what DIDN'T work
   317→          needsValidation: false, // Failures don't need validation — they happened
   318→        });
   319→
   320→        // Also record a procedural learning about what to avoid
   321→        learnings.push({
   322→          domain: this.inferDomain(refreshedFeedback),
   323→          type: "PROCEDURAL",
   324→          content: {
   325→            pattern: "avoid",
   326→            whatToAvoid: refreshedFeedback.input ? JSON.parse(refreshedFeedback.input) : null,
   327→            reason: analysis.rootCause || refreshedFeedback.feedbackText,
   328→          },
   329→          source: `feedback:${feedbackId}`,
   330→          sourceType: "FEEDBACK",
   331→          truthLevel: "ESTIMATED",
   332→          confidence: 0.5,
   333→          needsValidation: true,
   334→        });
   335→        break;
   336→      }
   337→
   338→      case "BAD_DECISION": {
   339→        // Record the bad decision as a factual memory
   340→        learnings.push({
   341→          domain: "decision",
   342→          type: "FACTUAL",
   343→          content: {
   344→            pattern: "bad_decision",
   345→            decision: refreshedFeedback.input ? JSON.parse(refreshedFeedback.input) : null,
   346→            outcome: refreshedFeedback.result ? JSON.parse(refreshedFeedback.result) : null,
   347→            reason: refreshedFeedback.feedbackText,
   348→          },
   349→          source: `feedback:${feedbackId}`,
   350→          sourceType: "FEEDBACK",
   351→          truthLevel: "OBSERVED",
   352→          confidence: 0.7,
   353→          needsValidation: true, // Decision quality needs validation
   354→          affectedSkills: this.identifyAffectedSkills(refreshedFeedback),
   355→        });
   356→        break;
   357→      }
   358→
   359→      case "BAD_HYPOTHESIS": {
   360→        // CRITICAL: Invalidate the hypothesis in memory
   361→        learnings.push({
   362→          domain: this.inferDomain(refreshedFeedback),
   363→          type: "FACTUAL",
   364→          content: {
   365→            pattern: "invalidated_hypothesis",
   366→            hypothesis: refreshedFeedback.input ? JSON.parse(refreshedFeedback.input) : null,
   367→            contradictingEvidence: refreshedFeedback.result ? JSON.parse(refreshedFeedback.result) : null,
   368→            reason: refreshedFeedback.feedbackText,
   369→          },
   370→          source: `feedback:${feedbackId}`,
   371→          sourceType: "FEEDBACK",
   372→          truthLevel: "OBSERVED",
   373→          confidence: 0.9, // Very high confidence in contradiction
   374→          needsValidation: false, // Contradictory evidence is a fact
   375→        });
   376→        break;
   377→      }
   378→
   379→      case "BAD_TOOL": {
   380→        // Record tool failure
   381→        learnings.push({
   382→          domain: "tool",
   383→          type: "FACTUAL",
   384→          content: {
   385→            pattern: "bad_tool",
   386→            tool: refreshedFeedback.input ? JSON.parse(refreshedFeedback.input) : null,
   387→            issue: refreshedFeedback.feedbackText,
   388→            alternative: analysis.recommendations?.[0],
   389→          },
   390→          source: `feedback:${feedbackId}`,
   391→          sourceType: "FEEDBACK",
   392→          truthLevel: "ESTIMATED",
   393→          confidence: 0.5,
   394→          needsValidation: true,
   395→        });
   396→        break;
   397→      }
   398→
   399→      case "TECHNICAL_ERROR": {
   400→        // Record as episodic — technical errors are transient
   401→        learnings.push({
   402→          domain: "system",
   403→          type: "EPISODIC",
   404→          content: {
   405→            pattern: "technical_error",
   406→            error: refreshedFeedback.feedbackText,
   407→            recoverable: analysis.isRecoverable,
   408→          },
   409→          source: `feedback:${feedbackId}`,
   410→          sourceType: "FEEDBACK",
   411→          truthLevel: "OBSERVED",
   412→          confidence: 0.7,
   413→          needsValidation: false,
   414→        });
   415→        break;
   416→      }
   417→
   418→      default: {
   419→        learnings.push({
   420→          domain: this.inferDomain(refreshedFeedback),
   421→          type: "EPISODIC",
   422→          content: {
   423→            pattern: "unknown_feedback",
   424→            feedbackType: refreshedFeedback.feedbackType,
   425→            feedbackText: refreshedFeedback.feedbackText,
   426→          },
   427→          source: `feedback:${feedbackId}`,
   428→          sourceType: "FEEDBACK",
   429→          truthLevel: "UNKNOWN",
   430→          confidence: 0.3,
   431→          needsValidation: true,
   432→        });
   433→      }
   434→    }
   435→
   436→    // Store learning in the feedback record
   437→    await db.feedback.update({
   438→      where: { id: feedbackId },
   439→      data: {
   440→        learning: JSON.stringify(learnings),
   441→      },
   442→    });
   443→
   444→    return learnings;
   445→  }

   447→  /**
   448→   * Apply extracted learning to the system.
   449→   * Updates memory and/or skills based on the learning.
   450→   * CRITICAL: Only applies VALIDATED learning, or marks them as needing validation.
   451→   */
   452→  async applyLearning(learning: ExtractedLearning, agentId?: string): Promise<{
   453→    memoryId?: string;
   454→    skillsUpdated?: string[];
   455→    status: "applied" | "pending_validation";
   456→  }> {
   457→    const effectiveAgentId = agentId || this.defaultAgentId;
   458→    const skillsUpdated: string[] = [];
   459→
   460→    // Store in memory
   461→    const { id: memoryId } = await this.memoryManager.store(
   462→      effectiveAgentId,
   463→      learning.domain,
   464→      learning.type,
   465→      learning.content,
   466→      {
   467→        source: learning.source,
   468→        sourceType: learning.sourceType,
   469→        truthLevel: learning.truthLevel,
   470→        confidence: learning.confidence,
   471→      }
   472→    );
   473→
   474→    // If learning is about a bad hypothesis, invalidate related memories
   475→    if (learning.content.pattern === "invalidated_hypothesis" && learning.content.hypothesis) {
   476→      const relatedMemories = await this.memoryManager.recall({
   477→        domain: learning.domain,
   478→        searchText: JSON.stringify(learning.content.hypothesis).substring(0, 50),
   479→      });
   480→
   481→      for (const mem of relatedMemories) {
   482→        const m = mem as { id: string };
   483→        try {
   484→          await this.memoryManager.invalidate(m.id, "Invalidated by feedback: bad hypothesis");
   485→        } catch {
   486→          // Memory might already be invalidated
   487→        }
   488→      }
   489→    }
   490→
   491→    // If learning affects skills, update them
   492→    if (learning.affectedSkills && learning.affectedSkills.length > 0) {
   493→      for (const skillId of learning.affectedSkills) {
   494→        try {
   495→          await this.skillManager.improve(skillId, {
   496→            success: learning.content.pattern === "successful_execution",
   497→            partialSuccess: learning.content.pattern === "partial_success",
   498→            actualResult: learning.feedbackText,
   499→          });
   500→          skillsUpdated.push(skillId);
   501→        } catch {
   502→          // Skill might not exist or not be improvable
   503→        }
   504→      }
   505→    }
   506→
   507→    // Determine status
   508→    const status = learning.needsValidation ? "pending_validation" : "applied";
   509→
   510→    return { memoryId, skillsUpdated: skillsUpdated.length > 0 ? skillsUpdated : undefined, status };
   511→  }

   513→  /**
   514→   * Validate a feedback entry and its associated learning.
   515→   * Moves validation status from PENDING to VALIDATED or INVALIDATED.
   516→   */
   517→  async validate(feedbackId: string, isValid: boolean, reason?: string): Promise<void> {
   518→    const feedback = await db.feedback.findUnique({ where: { id: feedbackId } });
   519→    if (!feedback) throw new Error(`Feedback ${feedbackId} not found`);
   520→
   521→    const evidence = feedback.evidence ? JSON.parse(feedback.evidence) : {};
   522→    evidence.validationReason = reason;
   523→    evidence.validatedAt = new Date().toISOString();
   524→
   525→    await db.feedback.update({
   526→      where: { id: feedbackId },
   527→      data: {
   528→        validationStatus: isValid ? "VALIDATED" : "INVALIDATED",
   529→        evidence: JSON.stringify(evidence),
   530→      },
   531→    });
   532→
   533→    // If validated and has learning, apply it
   534→    if (isValid && feedback.learning) {
   535→      const learnings: ExtractedLearning[] = JSON.parse(feedback.learning);
   536→      for (const learning of learnings) {
   537→        await this.applyLearning(learning, feedback.agentId);
   538→      }
   539→    }
   540→  }

   559→  /**
   560→   * List feedback entries with optional filters.
   561→   */
   562→  async list(filters: {
   563→    agentId?: string;
   564→    userId?: string;
   565→    executionId?: string;
   566→    feedbackType?: FeedbackType;
   567→    validationStatus?: ValidationStatus;
   568→    limit?: number;
   569→  } = {}): Promise<unknown[]> {
   570→    const where: Record<string, unknown> = {};
   571→    if (filters.agentId) where.agentId = filters.agentId;
   572→    if (filters.userId) where.userId = filters.userId;
   573→    if (filters.executionId) where.executionId = filters.executionId;
   574→    if (filters.feedbackType) where.feedbackType = filters.feedbackType;
   575→    if (filters.validationStatus) where.validationStatus = filters.validationStatus;
   576→
   577→    const feedbacks = await db.feedback.findMany({
   578→      where,
   579→      orderBy: { createdAt: "desc" },
   580→      take: filters.limit ?? 50,
   581→    });
   582→
   583→    return feedbacks.map((f) => ({
   584→      ...f,
   585→      input: f.input ? JSON.parse(f.input) : null,
   586→      result: f.result ? JSON.parse(f.result) : null,
   587→      analysis: f.analysis ? JSON.parse(f.analysis) : null,
   588→      learning: f.learning ? JSON.parse(f.learning) : null,
   589→      evidence: f.evidence ? JSON.parse(f.evidence) : null,
   590→    }));
   591→  }

   593→  /**
   594→   * Get feedback statistics.
   595→   */
   596→  async getStats(agentId?: string): Promise<{
   597→    total: number;
   598→    byType: Record<string, number>;
   599→    byValidationStatus: Record<string, number>;
   600→    avgConfidence: number;
   601→    learningPending: number;
   602→    learningApplied: number;
   603→  }> {
   604→    const effectiveAgentId = agentId || this.defaultAgentId;
   605→
   606→    const feedbacks = await db.feedback.findMany({
   607→      where: { agentId: effectiveAgentId },
   608→    });
   609→
   610→    const byType: Record<string, number> = {};
   611→    const byValidationStatus: Record<string, number> = {};
   612→    let totalConfidence = 0;
   613→    let learningPending = 0;
   614→    let learningApplied = 0;
   615→
   616→    for (const f of feedbacks) {
   617→      byType[f.feedbackType] = (byType[f.feedbackType] || 0) + 1;
   618→      byValidationStatus[f.validationStatus] = (byValidationStatus[f.validationStatus] || 0) + 1;
   619→      totalConfidence += f.confidence;
   620→
   621→      if (f.validationStatus === "PENDING") learningPending++;
   622→      if (f.validationStatus === "VALIDATED") learningApplied++;
   623→    }
   624→
   625→    return {
   626→      total: feedbacks.length,
   627→      byType,
   628→      byValidationStatus,
   629→      avgConfidence: feedbacks.length > 0 ? totalConfidence / feedbacks.length : 0,
   630→      learningPending,
   631→      learningApplied,
   632→    };
   633→  }
   634→
   635→  // ─── Private Helpers ───────────────────────────────────────────────
   636→
   637→  /**
   638→   * Infer confidence from feedback type.
   639→   */
   640→  private inferConfidence(feedbackType: FeedbackType): number {
   641→    switch (feedbackType) {
   642→      case "SUCCESS": return 0.8;
   643→      case "PARTIAL_SUCCESS": return 0.5;
   644→      case "FAILURE": return 0.8; // We're confident it failed
   645→      case "TECHNICAL_ERROR": return 0.7;
   646→      case "BAD_DECISION": return 0.7;
   647→      case "BAD_HYPOTHESIS": return 0.9;
   648→      case "BAD_TOOL": return 0.6;
   649→      case "BAD_EXECUTION": return 0.8;
   650→      default: return 0.5;
   651→    }
   652→  }
   653→
   654→  /**
   655→   * Infer domain from feedback context.
   656→   */
   657→  private inferDomain(feedback: { executionId: string | null; feedbackType: string }): string {
   658→    if (feedback.executionId) return "execution";
   659→    switch (feedback.feedbackType) {
   660→      case "BAD_TOOL": return "tool";
   661→      case "BAD_DECISION": return "decision";
   662→      case "BAD_HYPOTHESIS": return "hypothesis";
   663→      case "TECHNICAL_ERROR": return "system";
   664→      default: return "general";
   665→    }
   666→  }
   667→
   668→  /**
   669→   * Identify the cause of a partial success.
   670→   */
   671→  private identifyPartialSuccessCause(feedback: { feedbackText: string; result: string | null }): string {
   672→    if (feedback.feedbackText) return feedback.feedbackText;
   673→    if (feedback.result) {
   674→      try {
   675→        const result = JSON.parse(feedback.result);
   676→        if (result.errors && Array.isArray(result.errors) && result.errors.length > 0) {
   677→          return `Partial success with errors: ${result.errors[0]}`;
   678→        }
   679→        if (result.stepsCompleted && result.stepsTotal && result.stepsCompleted < result.stepsTotal) {
   680→          return `Only ${result.stepsCompleted}/${result.stepsTotal} steps completed`;
   681→        }
   682→      } catch { /* ignore */ }
   683→    }
   684→    return "Unknown cause of partial success";
   685→  }
   686→
   687→  /**
   688→   * Identify root cause of a failure.
   689→   */
   690→  private identifyFailureRootCause(feedback: { feedbackText: string; result: string | null }): string {
   691→    if (feedback.feedbackText) return feedback.feedbackText;
   692→    if (feedback.result) {
   693→      try {
   694→        const result = JSON.parse(feedback.result);
   695→        if (result.error) return String(result.error);
   696→        if (result.errors && Array.isArray(result.errors) && result.errors.length > 0) {
   697→          return result.errors[0];
   698→        }
   699→      } catch { /* ignore */ }
   700→    }
   701→    return "Unknown root cause";
   702→  }
   703→
   704→  /**
   705→   * Identify contributing factors from feedback.
   706→   */
   707→  private identifyContributingFactors(feedback: { result: string | null; input: string | null }): string[] {
   708→    const factors: string[] = [];
   709→
   710→    if (feedback.input) {
   711→      try {
   712→        const input = JSON.parse(feedback.input);
   713→        if (input.tool) factors.push(`Tool: ${input.tool}`);
   714→        if (input.skill) factors.push(`Skill: ${input.skill}`);
   715→        if (input.params) factors.push("Custom parameters used");
   716→      } catch { /* ignore */ }
   717→    }
   718→
   719→    return factors;
   720→  }
   721→
   722→  /**
   723→   * Identify decision patterns from feedback.
   724→   */
   725→  private identifyDecisionPatterns(feedback: { feedbackText: string }): string[] {
   726→    const patterns: string[] = [];
   727→    const text = feedback.feedbackText.toLowerCase();
   728→
   729→    if (text.includes("timeout") || text.includes("too slow")) patterns.push("timeout_prone");
   730→    if (text.includes("wrong tool") || text.includes("incorrect tool")) patterns.push("tool_selection_error");
   731→    if (text.includes("insufficient") || text.includes("not enough")) patterns.push("resource_underestimation");
   732→    if (text.includes("oversized") || text.includes("too much")) patterns.push("resource_overestimation");
   733→    if (text.includes("wrong order") || text.includes("sequence")) patterns.push("ordering_error");
   734→
   735→    return patterns;
   736→  }
   737→
   738→  /**
   739→   * Extract what worked from partial success feedback.
   740→   */
   741→  private extractWhatWorked(feedback: { result: string | null }): string | null {
   742→    if (!feedback.result) return null;
   743→    try {
   744→      const result = JSON.parse(feedback.result);
   745→      if (result.stepsCompleted) return `${result.stepsCompleted} steps completed successfully`;
   746→      if (result.partialResult) return JSON.stringify(result.partialResult);
   747→    } catch { /* ignore */ }
   748→    return null;
   749→  }
   750→
   751→  /**
   752→   * Extract what failed from partial success feedback.
   753→   */
   754→  private extractWhatFailed(feedback: { result: string | null; feedbackText: string }): string | null {
   755→    if (feedback.feedbackText) return feedback.feedbackText;
   756→    if (!feedback.result) return null;
   757→    try {
   758→      const result = JSON.parse(feedback.result);
   759→      if (result.errors) return JSON.stringify(result.errors);
   760→      if (result.failedSteps) return `Failed at steps: ${result.failedSteps}`;
   761→    } catch { /* ignore */ }
   762→    return null;
   763→  }
   764→
   765→  /**
   766→   * Identify skills affected by this feedback.
   767→   */
   768→  private identifyAffectedSkills(feedback: { input: string | null; result: string | null }): string[] {
   769→    const skills: string[] = [];
   770→
   771→    if (feedback.input) {
   772→      try {
   773→        const input = JSON.parse(feedback.input);
   774→        if (input.skillId) skills.push(input.skillId);
   775→        if (input.skillName) skills.push(input.skillName);
   776→      } catch { /* ignore */ }
   777→    }
   778→
   779→    if (feedback.result) {
   780→      try {
   781→        const result = JSON.parse(feedback.result);
   782→        if (result.skillsUsed && Array.isArray(result.skillsUsed)) {
   783→          skills.push(...result.skillsUsed);
   784→        }
   785→      } catch { /* ignore */ }
   786→    }
   787→
   788→    return [...new Set(skills)];
   789→  }

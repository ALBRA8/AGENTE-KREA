   603→   603→   603→/* ════════════ Main App ════════════ */
   604→   604→   604→interface AppShellProps { onLogout: () => void; user?: User | null }
   605→   605→   605→
   606→   606→   606→export default function AppPage({ onLogout, user: initialUser }: AppShellProps) {
   607→   607→   607→  const [user, setUser] = useState<User | null>(() => {
   608→   608→   608→    if (initialUser) return initialUser;
   609→   609→   609→    try {
   610→   610→   610→      const saved = localStorage.getItem("p360_user");
   611→   611→   611→      return saved ? JSON.parse(saved) : null;
   612→   612→   612→    } catch { return null; }
   613→   613→   613→  });
   614→   614→   614→  const [page, setPage] = useState("dashboard");
   615→   615→   615→  const [sidebarOpen, setSidebarOpen] = useState(false);
   616→   616→   616→  const [loading, setLoading] = useState(false);
   617→   617→   617→  const [generations, setGenerations] = useState<Generation[]>([]);
   618→   618→   618→
   619→   619→   619→  const loadUser = useCallback(async () => {
   620→   620→   620→    const token = localStorage.getItem("p360_token");
   621→   621→   621→    if (!token) { if (!user) onLogout(); else setLoading(false); return; }
   622→   622→   622→    try {
   623→   623→   623→      const res = await fetch("/api/auth/me", { headers: { Authorization: `Bearer ${token}` } });
   624→   624→   624→      if (!res.ok) { if (!user) onLogout(); else setLoading(false); return; }
   625→   625→   625→      const data = await res.json();
   626→   626→   626→      setUser(data);
   627→   627→   627→      const gRes = await fetch("/api/generations?limit=10", { headers: { Authorization: `Bearer ${token}` } });
   628→   628→   628→      if (gRes.ok) setGenerations(await gRes.json());
   629→   629→   629→    } catch { if (!user) onLogout(); }
   630→   630→   630→    setLoading(false);
   631→   631→   631→  }, [user, onLogout]);
   632→   632→   632→
   633→   633→   633→  useEffect(() => {
   634→   634→   634→    const token = localStorage.getItem("p360_token");
   635→   635→   635→    if (token && user) {
   636→   636→   636→      fetch("/api/generations?limit=10", { headers: { Authorization: `Bearer ${token}` } })
   637→   637→   637→        .then(r => r.ok ? r.json() : [])
   638→   638→   638→        .then(setGenerations)
   639→   639→   639→        .catch(() => {});
   640→   640→   640→    }
   641→   641→   641→  }, []);
   642→   642→   642→
   643→   643→   643→  const logout = () => { localStorage.removeItem("p360_token"); localStorage.removeItem("p360_user"); onLogout(); };
   644→   644→   644→  const refreshUser = () => loadUser();
   645→   645→   645→  const onUpdateCredits = (c: number) => { if (user) setUser({ ...user, credits: c }); };
   646→   646→   646→
   647→   647→   647→  if (loading) return (
   648→   648→   648→    <div className="min-h-screen bg-[#080c16] flex items-center justify-center">
   649→   649→   649→      <Loader2 className="w-8 h-8 animate-spin text-[#3b82f6]" />
   650→   650→   650→    </div>
   651→   651→   651→  );
   652→   652→   652→
   653→   653→   653→  if (!user) return null;
   654→   654→   654→
   655→   655→   655→  const renderPage = () => {
   656→   656→   656→    switch (page) {
   657→   657→   657→      case "dashboard": return <Dashboard user={user} generations={generations} onNav={setPage} />;
   658→   658→   658→      case "prompts": return <PromptChat user={user} onUpdateCredits={onUpdateCredits} />;
   659→   659→   659→      case "images": return <ImageGenerator user={user} onUpdateCredits={onUpdateCredits} />;
   660→   660→   660→      case "copy": case "social": case "script": case "email": return <TextGenerator type={page} user={user} onUpdateCredits={onUpdateCredits} />;
   661→   661→   661→      case "voice": return <VoiceGenerator user={user} onUpdateCredits={onUpdateCredits} />;
   662→   662→   662→      case "ebook": return <EbookGenerator user={user} onUpdateCredits={onUpdateCredits} />;
   663→   663→   663→      case "subtitle": return <SubtitleGenerator user={user} onUpdateCredits={onUpdateCredits} />;
   664→   664→   664→      case "metrics": return <MetricsTracker userId={user.id} />;
   665→   665→   665→      case "library": return <Library user={user} />;
   666→   666→   666→      case "settings": return <SettingsPage user={user} />;
   667→   667→   667→      case "support": return <SupportPage />;
   668→   668→   668→      default: return <Dashboard user={user} generations={generations} onNav={setPage} />;
   669→   669→   669→    }
   670→   670→   670→  };
   671→   671→   671→
   672→   672→   672→  return (
   673→   673→   673→    <AuthCtx.Provider value={{ user, logout, refreshUser }}>
   674→   674→   674→      <div className="min-h-screen bg-[#080c16] flex">
   675→   675→   675→        <Sidebar active={page} onNav={setPage} open={sidebarOpen} onClose={() => setSidebarOpen(false)} user={user} />
   676→   676→   676→        <div className="flex-1 flex flex-col min-h-screen">
   677→   677→   677→          {/* Top bar */}
   678→   678→   678→          <header className="sticky top-0 z-30 flex items-center gap-4 px-4 sm:px-6 py-3 border-b border-white/[0.06] backdrop-blur-xl bg-[#080c16]/80">
   679→   679→   679→            <button onClick={() => setSidebarOpen(true)} className="lg:hidden text-white/50 hover:text-white">
   680→   680→   680→              <Menu className="w-5 h-5" />
   681→   681→   681→            </button>
   682→   682→   682→            <div className="flex-1" />
   683→   683→   683→            <div className="flex items-center gap-3">
   684→   684→   684→              <div className="flex items-center gap-1.5 px-3 py-1.5 rounded-lg bg-yellow-400/10 text-yellow-400 text-xs font-bold">
   685→   685→   685→                <Coins className="w-3.5 h-3.5" /> {user.credits}
   686→   686→   686→              </div>
   687→   687→   687→              <button onClick={logout} className="p-2 rounded-lg hover:bg-white/5 text-white/40 hover:text-white/70 transition-colors" title="Cerrar sesión">
   688→   688→   688→                <LogOut className="w-4 h-4" />
   689→   689→   689→              </button>
   690→   690→   690→            </div>
   691→   691→   691→          </header>
   692→   692→   692→          {/* Content */}
   693→   693→   693→          <main className="flex-1 p-4 sm:p-6 lg:p-8 overflow-y-auto">
   694→   694→   694→            {renderPage()}
   695→   695→   695→          </main>
   696→   696→   696→        </div>
   697→   697→   697→      </div>
   698→   698→   698→    </AuthCtx.Provider>
   699→   699→   699→  );

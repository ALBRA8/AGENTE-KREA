     1→/**
     2→ * Product Fit Engine — KREA V2 Product Brain
     3→ *
     4→ * Evaluates product opportunities across multiple dimensions to determine
     5→ * whether to BUILD, VALIDATE_FIRST, or DO_NOT_BUILD.
     6→ *
     7→ * Acceptance Criteria satisfied:
     8→ *   AC-010: No magic numbers — every score is explained with the variables evaluated
     9→ *   AC-007: Unknown evidence produces uncertain dimensions and VALIDATE_FIRST recommendation
    10→ *
    11→ * Scoring rules:
    12→ *   - High problem intensity + high demand + low competition → BUILD
    13→ *   - Good signals but uncertain evidence → VALIDATE_FIRST
    14→ *   - Low demand or high risk or poor evidence → DO_NOT_BUILD
    15→ *   - Always explain WHAT variables were evaluated and HOW
    16→ */
    17→

    19→import { TruthLevel, TRUTH_LEVEL_ORDER } from "@/contracts/evidence";

   119→// ─── Product Fit Engine ──────────────────────────────────────────────────────
   120→
   121→/**
   122→ * ProductFitEngine — Evaluates opportunities across 11 dimensions.
   123→ *
   124→ * Each dimension is scored 0-1 with explicit explanation of how
   125→ * the score was derived from evidence. No magic numbers (AC-010).
   126→ */
   127→export class ProductFitEngine {
   128→  /**
   129→   * evaluate — Produce a FitResult for an opportunity given evidence.
   130→   *
   131→   * Decision logic (AC-010 — every threshold explained):
   132→   *   - BUILD:       score >= 0.65 AND no critical uncertainties AND evidence_quality >= 0.5
   133→   *   - VALIDATE_FIRST: score >= 0.40 OR (score >= 0.30 AND has uncertain dimensions)
   134→   *   - DO_NOT_BUILD:  score < 0.40 AND no uncertain dimensions that could improve score
   135→   */
   136→  evaluate(opportunity: Opportunity, evidence: ProductEvidence[]): FitResult {
   137→    const evidenceByType = this.groupEvidenceByType(evidence);
   138→
   139→    // Evaluate each dimension
   140→    const dimensions: EvaluatedDimension[] = [
   141→      this.evaluateProblemIntensity(evidenceByType),
   142→      this.evaluateDemand(evidenceByType),
   143→      this.evaluatePaymentIntent(evidenceByType),
   144→      this.evaluateAudienceSize(evidenceByType),
   145→      this.evaluateCompetition(evidenceByType),
   146→      this.evaluateDifferentiation(evidenceByType),
   147→      this.evaluateFeasibility(evidenceByType, opportunity),
   148→      this.evaluateBuildTime(evidenceByType, opportunity),
   149→      this.evaluateEconomics(evidenceByType),
   150→      this.evaluateRisk(evidenceByType),
   151→      this.evaluateEvidenceQuality(evidence),
   152→    ];
   153→
   154→    // Calculate weighted overall score
   155→    const score = this.calculateWeightedScore(dimensions);
   156→
   157→    // Determine decision
   158→    const decision = this.determineDecision(score, dimensions);
   159→
   160→    // Collect uncertainties
   161→    const uncertainties = dimensions
   162→      .filter((d) => d.is_uncertain)
   163→      .map((d) => `${d.name}: ${d.explanation}`);
   164→
   165→    // Generate explanation
   166→    const explanation = this.generateExplanation(decision, score, dimensions, opportunity);
   167→
   168→    // Recommended actions
   169→    const recommended_actions = this.recommendActions(decision, dimensions, uncertainties);
   170→
   171→    return {
   172→      decision,
   173→      score: Math.round(score * 100) / 100,
   174→      dimensions,
   175→      explanation,
   176→      uncertainties,
   177→      recommended_actions,
   178→      evidence_used: evidence,
   179→      evaluatedAt: new Date().toISOString(),
   180→    };
   181→  }

   185→  /** Problem intensity: how painful is the problem? */
   186→  private evaluateProblemIntensity(
   187→    evidenceByType: Map<ProductEvidenceType, ProductEvidence[]>
   188→  ): EvaluatedDimension {
   189→    const problemEvidence = evidenceByType.get("PROBLEM") || [];
   190→    const needEvidence = evidenceByType.get("NEED") || [];
   191→    const allEvidence = [...problemEvidence, ...needEvidence];
   192→
   193→    if (allEvidence.length === 0) {
   194→      return {
   195→        name: "problem_intensity",
   196→        score: 0.3,
   197→        evidence: [],
   198→        confidence: 0,
   199→        is_uncertain: true,
   200→        explanation: "No PROBLEM or NEED evidence found. Default score 0.3 assigned due to absence of evidence (AC-007). REQUIERE_VALIDACION — research problem intensity before deciding.",
   201→      };
   202→    }
   203→
   204→    // Average confidence of problem evidence indicates intensity
   205→    const avgConfidence = allEvidence.reduce((s, e) => s + e.confidence, 0) / allEvidence.length;
   206→    const highConfidenceCount = allEvidence.filter((e) => e.confidence >= 0.7).length;
   207→    const ratio = highConfidenceCount / allEvidence.length;
   208→
   209→    // Score: weighted combination of average confidence and ratio of high-confidence evidence
   210→    const score = 0.6 * avgConfidence + 0.4 * ratio;
   211→    const isUncertain = avgConfidence < 0.5;
   212→
   213→    return {
   214→      name: "problem_intensity",
   215→      score: Math.round(score * 100) / 100,
   216→      evidence: allEvidence.map((e) => e.id),
   217→      confidence: avgConfidence,
   218→      is_uncertain: isUncertain,
   219→      explanation: isUncertain
   220→        ? `Problem intensity estimated at ${score.toFixed(2)} from ${allEvidence.length} evidence items (avg confidence: ${avgConfidence.toFixed(2)}, ${highConfidenceCount} high-confidence). Low confidence makes this uncertain — VALIDATE problem intensity with users.`
   221→        : `Problem intensity scored ${score.toFixed(2)}: ${allEvidence.length} evidence items, average confidence ${avgConfidence.toFixed(2)}, ${highConfidenceCount} items with confidence >= 0.7 (ratio: ${ratio.toFixed(2)}). Formula: 0.6 * avgConfidence + 0.4 * highRatio.`,
   222→    };
   223→  }
   224→
   225→  /** Demand: is there real demand? */
   226→  private evaluateDemand(
   227→    evidenceByType: Map<ProductEvidenceType, ProductEvidence[]>
   228→  ): EvaluatedDimension {
   229→    const demandEvidence = evidenceByType.get("DEMAND") || [];
   230→    const behaviorEvidence = evidenceByType.get("BEHAVIOR") || [];
   231→    const allEvidence = [...demandEvidence, ...behaviorEvidence];
   232→
   233→    if (allEvidence.length === 0) {
   234→      return {
   235→        name: "demand",
   236→        score: 0.2,
   237→        evidence: [],
   238→        confidence: 0,
   239→        is_uncertain: true,
   240→        explanation: "No DEMAND or BEHAVIOR evidence found. Default score 0.2 — cannot confirm demand without evidence (AC-007). REQUIERE_VALIDACION.",
   241→      };
   242→    }
   243→
   244→    const avgConfidence = allEvidence.reduce((s, e) => s + e.confidence, 0) / allEvidence.length;
   245→    const demandSpecific = demandEvidence.length / Math.max(allEvidence.length, 1);
   246→    const score = 0.5 * avgConfidence + 0.3 * demandSpecific + 0.2 * (allEvidence.length > 3 ? 1 : allEvidence.length / 3);
   247→
   248→    return {
   249→      name: "demand",
   250→      score: Math.round(score * 100) / 100,
   251→      evidence: allEvidence.map((e) => e.id),
   252→      confidence: avgConfidence,
   253→      is_uncertain: avgConfidence < 0.5 || demandEvidence.length === 0,
   254→      explanation: `Demand scored ${score.toFixed(2)}: ${demandEvidence.length} demand + ${behaviorEvidence.length} behavior evidence items. Formula: 0.5 * avgConfidence(${avgConfidence.toFixed(2)}) + 0.3 * demandRatio(${demandSpecific.toFixed(2)}) + 0.2 * volumeBonus.`,
   255→    };
   256→  }

   258→  /** Payment intent: will people pay? */
   259→  private evaluatePaymentIntent(
   260→    evidenceByType: Map<ProductEvidenceType, ProductEvidence[]>
   261→  ): EvaluatedDimension {
   262→    const demandEvidence = evidenceByType.get("DEMAND") || [];
   263→    const pricingEvidence = evidenceByType.get("PRICING") || [];
   264→    const allEvidence = [...demandEvidence, ...pricingEvidence];
   265→
   266→    if (pricingEvidence.length === 0 && demandEvidence.length === 0) {
   267→      return {
   268→        name: "payment_intent",
   269→        score: 0.3,
   270→        evidence: [],
   271→        confidence: 0,
   272→        is_uncertain: true,
   273→        explanation: "No PRICING or DEMAND evidence to assess payment intent. Default 0.3. REQUIERE_VALIDACION — validate willingness to pay.",
   274→      };
   275→    }
   276→
   277→    // Pricing evidence is the strongest signal for payment intent
   278→    const pricingSignal = pricingEvidence.length > 0
   279→      ? pricingEvidence.reduce((s, e) => s + e.confidence, 0) / pricingEvidence.length
   280→      : 0.3;
   281→    const demandSignal = demandEvidence.length > 0
   282→      ? demandEvidence.reduce((s, e) => s + e.confidence, 0) / demandEvidence.length
   283→      : 0.3;
   284→
   285→    const score = 0.6 * pricingSignal + 0.4 * demandSignal;
   286→
   287→    return {
   288→      name: "payment_intent",
   289→      score: Math.round(score * 100) / 100,
   290→      evidence: allEvidence.map((e) => e.id),
   291→      confidence: (pricingSignal + demandSignal) / 2,
   292→      is_uncertain: pricingEvidence.length === 0,
   293→      explanation: `Payment intent scored ${score.toFixed(2)}: pricing signal ${pricingSignal.toFixed(2)} from ${pricingEvidence.length} items, demand signal ${demandSignal.toFixed(2)} from ${demandEvidence.length} items. Formula: 0.6 * pricingSignal + 0.4 * demandSignal.${pricingEvidence.length === 0 ? " No pricing evidence — uncertain." : ""}`,
   294→    };
   295→  }

   326→    };
   327→  }

   329→  /** Competition: is the space crowded? (inverse — less competition = higher score) */
   330→  private evaluateCompetition(
   331→    evidenceByType: Map<ProductEvidenceType, ProductEvidence[]>
   332→  ): EvaluatedDimension {
   333→    const compEvidence = evidenceByType.get("COMPETITION") || [];
   334→    const productEvidence = evidenceByType.get("EXISTING_PRODUCT") || [];
   335→    const allEvidence = [...compEvidence, ...productEvidence];
   336→
   337→    if (allEvidence.length === 0) {
   338→      return {
   339→        name: "competition",
   340→        score: 0.5,
   341→        evidence: [],
   342→        confidence: 0,
   343→        is_uncertain: true,
   344→        explanation: "No COMPETITION or EXISTING_PRODUCT evidence. Cannot assess competitive landscape. Default 0.5 (neutral). REQUIERE_VALIDACION.",
   345→      };
   346→    }
   347→
   348→    // More competitors = lower score (inverse relationship)
   349→    const competitorCount = allEvidence.length;
   350→    const avgStrength = allEvidence.reduce((s, e) => s + e.confidence, 0) / allEvidence.length;
   351→
   352→    // Fewer competitors and weaker existing solutions → higher score
   353→    const crowdingFactor = Math.max(0, 1 - (competitorCount / 10));
   354→    const strengthFactor = Math.max(0, 1 - avgStrength);
   355→    const score = 0.5 * crowdingFactor + 0.5 * strengthFactor;
   356→
   357→    return {
   358→      name: "competition",
   359→      score: Math.round(score * 100) / 100,
   360→      evidence: allEvidence.map((e) => e.id),
   361→      confidence: 0.7,
   362→      is_uncertain: compEvidence.length === 0,
   363→      explanation: `Competition scored ${score.toFixed(2)} (inverse — higher = less competition): ${competitorCount} competitors/products found, avg strength ${avgStrength.toFixed(2)}. Crowding factor ${crowdingFactor.toFixed(2)}, strength factor ${strengthFactor.toFixed(2)}. Formula: 0.5 * crowding + 0.5 * strength.`,
   364→    };
   365→  }

   367→  /** Differentiation: can we stand out? */
   368→  private evaluateDifferentiation(
   369→    evidenceByType: Map<ProductEvidenceType, ProductEvidence[]>
   370→  ): EvaluatedDimension {
   371→    const diffEvidence = evidenceByType.get("DIFFERENTIATION") || [];
   372→    const compEvidence = evidenceByType.get("COMPETITION") || [];
   373→
   374→    if (diffEvidence.length === 0 && compEvidence.length === 0) {
   375→      return {
   376→        name: "differentiation",
   377→        score: 0.4,
   378→        evidence: [],
   379→        confidence: 0,
   380→        is_uncertain: true,
   381→        explanation: "No DIFFERENTIATION or COMPETITION evidence. Cannot assess differentiation potential. Default 0.4. REQUIERE_VALIDACION.",
   382→      };
   383→    }
   384→
   385→    const diffSignal = diffEvidence.length > 0
   386→      ? diffEvidence.reduce((s, e) => s + e.confidence, 0) / diffEvidence.length
   387→      : 0.3;
   388→    // Gaps in competition suggest differentiation opportunity
   389→    const gapSignal = compEvidence.length > 0
   390→      ? 0.5 // Competitors exist but gaps may too
   391→      : 0.7; // No competitors found — differentiation by default
   392→
   393→    const score = 0.6 * diffSignal + 0.4 * gapSignal;
   394→
   395→    return {
   396→      name: "differentiation",
   397→      score: Math.round(score * 100) / 100,
   398→      evidence: [...diffEvidence, ...compEvidence].map((e) => e.id),
   399→      confidence: diffEvidence.length > 0 ? diffSignal : 0.3,
   400→      is_uncertain: diffEvidence.length === 0,
   401→      explanation: `Differentiation scored ${score.toFixed(2)}: ${diffEvidence.length} differentiation evidence (signal: ${diffSignal.toFixed(2)}), gap signal ${gapSignal.toFixed(2)}. Formula: 0.6 * diffSignal + 0.4 * gapSignal.${diffEvidence.length === 0 ? " No explicit differentiation evidence — uncertain." : ""}`,
   402→    };
   403→  }

   405→  /** Feasibility: can we actually build it? */
   406→  private evaluateFeasibility(
   407→    evidenceByType: Map<ProductEvidenceType, ProductEvidence[]>,
   408→    opportunity: Opportunity
   409→  ): EvaluatedDimension {
   410→    // Assess feasibility based on product format
   411→    const format = (opportunity.format || "").toLowerCase();
   412→    let baseFeasibility = 0.5;
   413→
   414→    // Content products are highly feasible for KREA
   415→    if (["ebook", "guía", "guide", "workbook", "template", "checklist", "curso", "course"].some((f) => format.includes(f))) {
   416→      baseFeasibility = 0.85;
   417→    }
   418→    // Simple web apps are moderately feasible
   419→    else if (["app", "web app", "saas", "tool"].some((f) => format.includes(f))) {
   420→      baseFeasibility = 0.5;
   421→    }
   422→    // Complex infrastructure is less feasible
   423→    else if (["api", "platform", "infrastructure", "backend", "infra"].some((f) => format.includes(f))) {
   424→      baseFeasibility = 0.35;
   425→    }
   426→
   427→    const oppEvidence = evidenceByType.get("OPPORTUNITY") || [];
   428→    const evidenceBoost = oppEvidence.length > 0 ? 0.1 : 0;
   429→
   430→    const score = Math.min(1, baseFeasibility + evidenceBoost);
   431→
   432→    return {
   433→      name: "feasibility",
   434→      score: Math.round(score * 100) / 100,
   435→      evidence: oppEvidence.map((e) => e.id),
   436→      confidence: 0.7,
   437→      is_uncertain: format === "",
   438→      explanation: `Feasibility scored ${score.toFixed(2)}: base feasibility ${baseFeasibility.toFixed(2)} for format "${format || "unknown"}", evidence boost ${evidenceBoost}. Content products are more feasible for KREA; infrastructure requires delegation.`,
   439→    };
   440→  }

   478→    const demandEvidence = evidenceByType.get("DEMAND") || [];

   504→      confidence: (pricingSignal + demandSignal) / 2,
   505→      is_uncertain: pricingEvidence.length === 0,
   506→      explanation: `Economics scored ${score.toFixed(2)}: pricing signal ${pricingSignal.toFixed(2)}, demand signal ${demandSignal.toFixed(2)}. Formula: 0.5 * pricing + 0.5 * demand.${pricingEvidence.length === 0 ? " No pricing evidence — uncertain." : ""}`,
   507→    };
   508→  }

   510→  /** Risk: what could go wrong? (inverse — less risk = higher score) */
   511→  private evaluateRisk(
   512→    evidenceByType: Map<ProductEvidenceType, ProductEvidence[]>
   513→  ): EvaluatedDimension {
   514→    const compEvidence = evidenceByType.get("COMPETITION") || [];
   515→    const productEvidence = evidenceByType.get("EXISTING_PRODUCT") || [];
   516→
   517→    // Risk factors
   518→    const highCompetition = compEvidence.length >= 5;
   519→    const strongExisting = productEvidence.some((e) => e.confidence >= 0.8);
   520→    const lowEvidence = compEvidence.length + productEvidence.length < 2;
   521→
   522→    let riskScore = 0.6; // Base: moderate-low risk
   523→
   524→    if (highCompetition) riskScore -= 0.15;
   525→    if (strongExisting) riskScore -= 0.1;
   526→    if (lowEvidence) riskScore -= 0.2; // Unknown = risky
   527→
   528→    riskScore = Math.max(0, Math.min(1, riskScore));
   529→
   530→    return {
   531→      name: "risk",
   532→      score: Math.round(riskScore * 100) / 100,
   533→      evidence: [...compEvidence, ...productEvidence].map((e) => e.id),
   534→      confidence: 0.5,
   535→      is_uncertain: lowEvidence,
   536→      explanation: `Risk scored ${riskScore.toFixed(2)} (inverse — higher = less risk): base 0.6, highCompetition(${highCompetition}) -0.15, strongExisting(${strongExisting}) -0.1, lowEvidence(${lowEvidence}) -0.2. Less evidence = more risk.`,
   537→    };
   538→  }
   539→
   540→  /** Evidence quality: how good is our evidence overall? */
   541→  private evaluateEvidenceQuality(allEvidence: ProductEvidence[]): EvaluatedDimension {
   542→    if (allEvidence.length === 0) {
   543→      return {
   544→        name: "evidence_quality",
   545→        score: 0,
   546→        evidence: [],
   547→        confidence: 0,
   548→        is_uncertain: true,
   549→        explanation: "No evidence at all. Evidence quality is 0. REQUIERE_VALIDACION — research is required before any decision.",
   550→      };
   551→    }
   552→
   553→    const avgConfidence = allEvidence.reduce((s, e) => s + e.confidence, 0) / allEvidence.length;
   554→    const avgTruthLevel = allEvidence.reduce((s, e) => s + TRUTH_LEVEL_ORDER[e.truthLevel], 0) / allEvidence.length / 5;
   555→    const verifiedCount = allEvidence.filter((e) => e.verificationStatus === "VERIFIED_PASS").length;
   556→    const verifiedRatio = verifiedCount / allEvidence.length;
   557→
   558→    const score = 0.4 * avgConfidence + 0.3 * avgTruthLevel + 0.3 * (verifiedRatio > 0 ? verifiedRatio : avgConfidence * 0.5);
   559→
   560→    return {
   561→      name: "evidence_quality",
   562→      score: Math.round(score * 100) / 100,
   563→      evidence: allEvidence.map((e) => e.id),
   564→      confidence: avgConfidence,
   565→      is_uncertain: avgConfidence < 0.5,
   566→      explanation: `Evidence quality scored ${score.toFixed(2)}: ${allEvidence.length} items, avg confidence ${avgConfidence.toFixed(2)}, avg truth level ${avgTruthLevel.toFixed(2)}, ${verifiedCount} verified (${verifiedRatio.toFixed(2)}). Formula: 0.4 * confidence + 0.3 * truthLevel + 0.3 * verifiedRatio.`,
   567→    };
   568→  }

   572→  /** Calculate weighted overall score from dimensions */
   573→  private calculateWeightedScore(dimensions: EvaluatedDimension[]): number {
   574→    let totalWeight = 0;
   575→    let weightedSum = 0;
   576→
   577→    for (const dim of dimensions) {
   578→      const weight = DIMENSION_WEIGHTS[dim.name as DimensionName] || 0.05;
   579→      weightedSum += dim.score * weight;
   580→      totalWeight += weight;
   581→    }
   582→
   583→    return totalWeight > 0 ? weightedSum / totalWeight : 0;
   584→  }

   586→  /** Determine the fit decision based on score and dimensions */
   587→  private determineDecision(score: number, dimensions: EvaluatedDimension[]): FitDecision {
   588→    const evidenceQuality = dimensions.find((d) => d.name === "evidence_quality");
   589→    const uncertainCount = dimensions.filter((d) => d.is_uncertain).length;
   590→    const criticalUncertain = dimensions.filter(
   591→      (d) => d.is_uncertain && (d.name === "problem_intensity" || d.name === "demand" || d.name === "payment_intent")
   592→    ).length;
   593→
   594→    // DO_NOT_BUILD: very low score with no critical uncertainties that could change things
   595→    if (score < 0.30 && criticalUncertain === 0) {
   596→      return "DO_NOT_BUILD";
   597→    }
   598→
   599→    // VALIDATE_FIRST: good signals but uncertain evidence
   600→    if (score >= 0.30 && score < 0.65) {
   601→      return "VALIDATE_FIRST";
   602→    }
   603→
   604→    // VALIDATE_FIRST: decent score but critical uncertainties
   605→    if (score >= 0.65 && (criticalUncertain > 0 || (evidenceQuality && evidenceQuality.score < 0.5))) {
   606→      return "VALIDATE_FIRST";
   607→    }
   608→
   609→    // BUILD: strong score, good evidence, no critical uncertainties
   610→    if (score >= 0.65 && criticalUncertain === 0 && uncertainCount <= 3) {
   611→      return "BUILD";
   612→    }
   613→
   614→    // Default: validate if uncertain
   615→    if (uncertainCount > 3) {
   616→      return "VALIDATE_FIRST";
   617→    }
   618→
   619→    return "DO_NOT_BUILD";
   620→  }

   622→  /** Generate a human-readable explanation of the decision (AC-010) */
   623→  private generateExplanation(
   624→    decision: FitDecision,
   625→    score: number,
   626→    dimensions: EvaluatedDimension[],
   627→    opportunity: Opportunity
   628→  ): string {
   629→    const topDims = dimensions
   630→      .sort((a, b) => (DIMENSION_WEIGHTS[b.name as DimensionName] || 0) - (DIMENSION_WEIGHTS[a.name as DimensionName] || 0))
   631→      .slice(0, 5);
   632→
   633→    const uncertainDims = dimensions.filter((d) => d.is_uncertain);
   634→
   635→    const decisionExplanations: Record<FitDecision, string> = {
   636→      BUILD: `BUILD recommendation for "${opportunity.name}" (score: ${score.toFixed(2)}). Key factors: ${topDims.map((d) => `${d.name}=${d.score.toFixed(2)}`).join(", ")}. Evidence is sufficient and no critical uncertainties remain.`,
   637→      VALIDATE_FIRST: `VALIDATE_FIRST recommendation for "${opportunity.name}" (score: ${score.toFixed(2)}). Key factors: ${topDims.map((d) => `${d.name}=${d.score.toFixed(2)}`).join(", ")}. Uncertain dimensions: ${uncertainDims.map((d) => d.name).join(", ") || "none"}. Validation needed before committing to build.`,
   638→      DO_NOT_BUILD: `DO_NOT_BUILD recommendation for "${opportunity.name}" (score: ${score.toFixed(2)}). Key factors: ${topDims.map((d) => `${d.name}=${d.score.toFixed(2)}`).join(", ")}. The evidence does not support investment at this time.`,
   639→    };
   640→
   641→    return decisionExplanations[decision];
   642→  }

   644→  /** Recommend specific next actions based on the decision */
   645→  private recommendActions(
   646→    decision: FitDecision,
   647→    dimensions: EvaluatedDimension[],
   648→    uncertainties: string[]
   649→  ): string[] {
   650→    const actions: string[] = [];
   651→
   652→    if (decision === "BUILD") {
   653→      actions.push("Proceed to product architecture and specification");
   654→      actions.push("Define MVP scope (MUST items only)");
   655→      actions.push("Set up build tracking and success criteria");
   656→    } else if (decision === "VALIDATE_FIRST") {
   657→      // Recommend validation based on uncertain dimensions
   658→      const uncertain = dimensions.filter((d) => d.is_uncertain);
   659→      for (const dim of uncertain) {
   660→        switch (dim.name) {
   661→          case "problem_intensity":
   662→            actions.push("Validate problem intensity: interview 10+ potential users about their pain");
   663→            break;
   664→          case "demand":
   665→            actions.push("Validate demand: run a smoke test or landing page to measure interest");
   666→            break;
   667→          case "payment_intent":
   668→            actions.push("Validate payment intent: test pricing with pre-orders or deposit requests");
   669→            break;
   670→          case "audience_size":
   671→            actions.push("Validate audience size: research market size data and search volumes");
   672→            break;
   673→          case "competition":
   674→            actions.push("Validate competitive landscape: audit existing solutions and their gaps");
   675→            break;
   676→          case "differentiation":
   677→            actions.push("Validate differentiation: test unique value proposition with target audience");
   678→            break;
   679→          case "economics":
   680→            actions.push("Validate economics: estimate costs and revenue model with real data");
   681→            break;
   682→          default:
   683→            actions.push(`Validate ${dim.name}: gather more evidence to reduce uncertainty`);
   684→        }
   685→      }
   686→      if (actions.length === 0) {
   687→        actions.push("Gather more evidence for uncertain dimensions before deciding");
   688→      }
   689→    } else {
   690→      // DO_NOT_BUILD
   691→      actions.push("Document why this opportunity was rejected (evidence trail)");
   692→      actions.push("Set reminder to re-evaluate if market conditions change");
   693→      if (uncertainties.length > 0) {
   694→        actions.push("If new evidence becomes available, re-run fit evaluation");
   695→      }
   696→    }
   697→
   698→    return actions;
   699→  }

   703→  /** Group evidence items by their type */
   704→  private groupEvidenceByType(
   705→    evidence: ProductEvidence[]
   706→  ): Map<ProductEvidenceType, ProductEvidence[]> {
   707→    const map = new Map<ProductEvidenceType, ProductEvidence[]>();
   708→    for (const item of evidence) {

   710→      items.push(item);

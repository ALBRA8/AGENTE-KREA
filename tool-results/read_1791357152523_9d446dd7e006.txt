   197→   197→// ─── ProductFactory Class ──────────────────────────────────────────────
   198→   198→
   199→   199→/**
   200→   200→ * ProductFactory — general orchestration for all product types.
   201→   201→ *
   202→   202→ * Pipeline (AC-028):
   203→   203→ *   Opportunity → Fit → Architecture → Specification → Production → QA → Final Product
   204→   204→ *
   205→   205→ * Routing (AC-027):
   206→   206→ *   - Content products (ebook, guide, template, report) → BookFactory
   207→   207→ *   - Software products → HandoffContract for Codex/Antigravity
   208→   208→ *   - Hybrid → combines both
   209→   209→ *
   210→   210→ * Complexity determines routing: category alone isn't enough (AC-027).
   211→   211→ */
   212→   212→export class ProductFactory {
   213→   213→  private bookFactory = new BookFactory();
   214→   214→  private memory = new MemoryManager();
   215→   215→  private tracer = new ExecutionTracer();
   216→   216→
   217→   217→  /**
   218→   218→   * Produce a product from an opportunity.
   219→   219→   *
   220→   220→   * @param opportunity - The product opportunity
   221→   221→   * @param fitResult - Pre-computed fit result (or computed internally)
   222→   222→   * @param decision - Architecture decision (optional)
   223→   223→   * @returns ProductResult with the final product and evidence
   224→   224→   */
   225→   225→  async produce(
   226→   226→    opportunity: ProductOpportunity,
   227→   227→    fitResult?: FitResult,
   228→   228→    decision?: ArchitectureDecision
   229→   229→  ): Promise<ProductResult> {
   230→   230→    const startTime = Date.now();
   231→   231→    const evidence: ProductEvidence[] = [];
   232→   232→    const errors: string[] = [];
   233→   233→
   234→   234→    // Start execution trace
   235→   235→    const executionId = await this.tracer.start("product-factory-produce", {
   236→   236→      agentId: "krea",
   237→   237→      userId: opportunity.userId,
   238→   238→      inputs: { topic: opportunity.topic, productType: opportunity.productType },
   239→   239→    });
   240→   240→
   241→   241→    try {
   242→   242→      // ─── Step 1: Fit ──────────────────────────────────────────────
   243→   243→      const fit = fitResult ?? this.computeFit(opportunity);
   244→   244→      evidence.push({
   245→   245→        step: "fit",
   246→   246→        timestamp: new Date().toISOString(),
   247→   247→        description: `Routed to ${fit.production_path} (complexity: ${fit.complexity_score})`,
   248→   248→        data: { fit },
   249→   249→      });
   250→   250→
   251→   251→      // ─── Step 2: Architecture ─────────────────────────────────────
   252→   252→      const arch = decision ?? this.computeArchitecture(opportunity, fit);
   253→   253→      evidence.push({
   254→   254→        step: "architecture",
   255→   255→        timestamp: new Date().toISOString(),
   256→   256→        description: `Architecture: ${arch.type} with ${arch.components.length} components`,
   257→   257→        data: { architecture: arch },
   258→   258→      });
   259→   259→
   260→   260→      // ─── Step 3-7: Production (routed by type) ────────────────────
   261→   261→      let result: ProductResult;
   262→   262→
   263→   263→      switch (fit.category) {
   264→   264→        case "content":
   265→   265→          result = await this.produceContentProduct(opportunity, fit, arch, evidence, executionId);
   266→   266→          break;
   267→   267→
   268→   268→        case "software":
   269→   269→          result = this.produceSoftwareProduct(opportunity, fit, arch, evidence);
   270→   270→          break;
   271→   271→
   272→   272→        case "hybrid":
   273→   273→          result = await this.produceHybridProduct(opportunity, fit, arch, evidence, executionId);
   274→   274→          break;
   275→   275→
   276→   276→        default:
   277→   277→          throw new Error(`Unknown product category: ${fit.category}`);
   278→   278→      }
   279→   279→
   280→   280→      await this.tracer.succeed(executionId, {
   281→   281→        productType: opportunity.productType,
   282→   282→        success: result.success,
   283→   283→      });
   284→   284→
   285→   285→      return result;
   286→   286→
   287→   287→    } catch (error) {
   288→   288→      const message = error instanceof Error ? error.message : String(error);
   289→   289→      errors.push(message);
   290→   290→      await this.tracer.addError(executionId, message);
   291→   291→      await this.tracer.fail(executionId, { message });
   292→   292→
   293→   293→      return {
   294→   294→        success: false,
   295→   295→        product_type: opportunity.productType,
   296→   296→        artifact_path: null,
   297→   297→        dossier_id: randomUUID(),
   298→   298→        evidence,
   299→   299→        qa_result: { passed: false, issues_count: 1, details: message },
   300→   300→        production_time_ms: Date.now() - startTime,
   301→   301→        errors,
   302→   302→      };
   303→   303→    }
   304→   304→  }
   305→   305→
   306→   306→  /**
   307→   307→   * Compute the fit result for an opportunity.
   308→   308→   *
   309→   309→   * AC-027: Complexity determines routing, not just category.
   310→   310→   */
   311→   311→  computeFit(opportunity: ProductOpportunity): FitResult {
   312→   312→    const category = this.inferCategory(opportunity.productType);
   313→   313→
   314→   314→    // Compute complexity score (0-1)
   315→   315→    const complexityScore = this.computeComplexityScore(opportunity);
   316→   316→
   317→   317→    // Route based on category + complexity (AC-027)
   318→   318→    let productionPath: FitResult["production_path"];
   319→   319→    let routingReason: string;
   320→   320→
   321→   321→    if (category === "content") {
   322→   322→      if (complexityScore <= 0.3) {
   323→   323→        productionPath = "book-factory-streamlined";
   324→   324→        routingReason = "Simple content product — streamlined pipeline sufficient";
   325→   325→      } else {
   326→   326→        productionPath = "book-factory-full";
   327→   327→        routingReason = "Complex content product — full BookFactory pipeline required";
   328→   328→      }
   329→   329→    } else if (category === "software") {
   330→   330→      productionPath = "handoff-contract";
   331→   331→      routingReason = "Software product — generate HandoffContract for implementation agent";
   332→   332→    } else {
   333→   333→      productionPath = "hybrid";
   334→   334→      routingReason = "Hybrid product — combine BookFactory + HandoffContract";
   335→   335→    }
   336→   336→
   337→   337→    const requiredCapabilities = this.getRequiredCapabilities(productionPath);
   338→   338→    const estimatedTime = this.estimateProductionTime(productionPath, opportunity);
   339→   339→
   340→   340→    return {
   341→   341→      fits: true,
   342→   342→      category,
   343→   343→      production_path: productionPath,
   344→   344→      routing_reason: routingReason,
   345→   345→      complexity_score: complexityScore,
   346→   346→      required_capabilities: requiredCapabilities,
   347→   347→      estimated_time_seconds: estimatedTime,
   348→   348→    };
   349→   349→  }
   350→   350→
   351→   351→  // ─── Private Methods ───────────────────────────────────────────────
   352→   352→
   353→   353→  /**
   354→   354→   * Infer the product category from the product type.
   355→   355→   */
   356→   356→  private inferCategory(productType: ProductType): ProductCategory {
   357→   357→    const contentTypes: ProductType[] = ["ebook", "guide", "template", "report"];
   358→   358→    if (contentTypes.includes(productType)) return "content";
   359→   359→    if (productType === "software") return "software";
   360→   360→    return "hybrid";
   361→   361→  }
   362→   362→
   363→   363→  /**
   364→   364→   * Compute complexity score (0-1) from opportunity attributes.
   365→   365→   *
   366→   366→   * AC-027: Complexity is multi-dimensional, not just category.
   367→   367→   */
   368→   368→  private computeComplexityScore(opportunity: ProductOpportunity): number {
   369→   369→    let score = 0;
   370→   370→
   371→   371→    // Section count contributes to complexity
   372→   372→    score += Math.min(0.3, opportunity.sectionCount / 20);
   373→   373→
   374→   374→    // Explicit complexity level
   375→   375→    const complexityMap: Record<string, number> = {
   376→   376→      simple: 0.1,
   377→   377→      moderate: 0.3,
   378→   378→      complex: 0.5,
   379→   379→    };
   380→   380→    score += complexityMap[opportunity.complexity] ?? 0.3;
   381→   381→
   382→   382→    // Format complexity
   383→   383→    if (opportunity.format === "report") score += 0.1;
   384→   384→    if (opportunity.format === "hybrid") score += 0.2;
   385→   385→
   386→   386→    // Genre complexity (some genres need more visual richness)
   387→   387→    const complexGenres = ["fiction", "lifestyle", "creative"];
   388→   388→    if (complexGenres.includes(opportunity.genre.toLowerCase())) score += 0.1;
   389→   389→
   390→   390→    return Math.min(1, score);
   391→   391→  }
   392→   392→
   393→   393→  /**
   394→   394→   * Compute architecture for the product.
   395→   395→   */
   396→   396→  private computeArchitecture(
   397→   397→    opportunity: ProductOpportunity,
   398→   398→    fit: FitResult
   399→   399→  ): ArchitectureDecision {
   400→   400→    const contentComponents: ArchitectureComponent[] = [
   401→   401→      { name: "content-gen", type: "content", dependencies: [] },
   402→   402→      { name: "art-direction", type: "design", dependencies: ["content-gen"] },
   403→   403→      { name: "editorial-design", type: "design", dependencies: ["art-direction"] },
   404→   404→      { name: "layout", type: "layout", dependencies: ["editorial-design"] },
   405→   405→      { name: "render", type: "render", dependencies: ["layout"] },
   406→   406→      { name: "qa", type: "qa", dependencies: ["render"] },
   407→   407→    ];
   408→   408→
   409→   409→    if (fit.category === "software" || fit.category === "hybrid") {
   410→   410→      contentComponents.push({
   411→   411→        name: "software-impl",
   412→   412→        type: "software",
   413→   413→        dependencies: ["content-gen"],
   414→   414→      });
   415→   415→    }
   416→   416→
   417→   417→    return {
   418→   418→      type: fit.complexity_score > 0.6 ? "modular" : "monolithic",
   419→   419→      components: contentComponents,
   420→   420→      quality_attributes: [
   421→   421→        "correctness",
   422→   422→        "completeness",
   423→   423→        "visual_quality",
   424→   424→        "accessibility",
   425→   425→      ],
   426→   426→      constraints: [
   427→   427→        "No literal content copying (AC-032)",
   428→   428→        "Deterministic layout (AC-039)",
   429→   429→        "Visual system coherence (AC-034)",
   430→   430→        "Max 5 repair iterations (AC-045)",
   431→   431→      ],
   432→   432→    };
   433→   433→  }

   435→   435→  /**
   436→   436→   * Produce a content product via BookFactory.
   437→   437→   */
   438→   438→  private async produceContentProduct(
   439→   439→    opportunity: ProductOpportunity,
   440→   440→    fit: FitResult,
   441→   441→    _arch: ArchitectureDecision,
   442→   442→    evidence: ProductEvidence[],
   443→   443→    executionId: string
   444→   444→  ): Promise<ProductResult> {
   445→   445→    const startTime = Date.now();
   446→   446→
   447→   447→    // Map ProductOpportunity → BookOpportunity
   448→   448→    const bookOpportunity: BookOpportunity = {
   449→   449→      topic: opportunity.topic,
   450→   450→      targetAudience: opportunity.targetAudience,
   451→   451→      genre: opportunity.genre,
   452→   452→      chapterCount: opportunity.sectionCount,
   453→   453→      author: opportunity.author,
   454→   454→      purpose: opportunity.purpose,
   455→   455→      tone: opportunity.tone,
   456→   456→      userId: opportunity.userId,
   457→   457→    };
   458→   458→
   459→   459→    const bookArchitecture: BookArchitecture = {
   460→   460→      format: opportunity.format as BookArchitecture["format"],
   461→   461→      complexity: opportunity.complexity,
   462→   462→      targetMarket: opportunity.targetMarket,
   463→   463→    };
   464→   464→
   465→   465→    evidence.push({
   466→   466→      step: "specification",
   467→   467→      timestamp: new Date().toISOString(),
   468→   468→      description: "Content product specification mapped to BookFactory input",
   469→   469→      data: { productionPath: fit.production_path },
   470→   470→    });
   471→   471→
   472→   472→    // Route to BookFactory
   473→   473→    const bookProduct = await this.bookFactory.produce(bookOpportunity, bookArchitecture);
   474→   474→
   475→   475→    evidence.push({
   476→   476→      step: "production",
   477→   477→      timestamp: new Date().toISOString(),
   478→   478→      description: `BookFactory produced ${bookProduct.content.chapters.length} chapters, ${bookProduct.pdf.pageCount} pages`,
   479→   479→      data: {
   480→   480→        pageCount: bookProduct.pdf.pageCount,
   481→   481→        wordCount: bookProduct.content.total_word_count,
   482→   482→        qaPassed: bookProduct.qa_result.passed,
   483→   483→        pdfVerified: bookProduct.pdf_verification.overall,
   484→   484→      },
   485→   485→    });
   486→   486→
   487→   487→    evidence.push({
   488→   488→      step: "qa",
   489→   489→      timestamp: new Date().toISOString(),
   490→   490→      description: bookProduct.qa_result.passed
   491→   491→        ? "QA passed — product is verified"
   492→   492→        : `QA found ${bookProduct.qa_result.issue_count} issues`,
   493→   493→      data: {
   494→   494→        passed: bookProduct.qa_result.passed,
   495→   495→        issueCount: bookProduct.qa_result.issue_count,
   496→   496→        criticalCount: bookProduct.qa_result.by_severity.critical,
   497→   497→      },
   498→   498→    });
   499→   499→
   500→   500→    await this.tracer.addSkill(executionId, "book-factory");
   501→   501→
   502→   502→    // Store product dossier in memory
   503→   503→    const dossierId = randomUUID();
   504→   504→    await this.storeDossier(opportunity, bookProduct, dossierId);
   505→   505→
   506→   506→    return {
   507→   507→      success: bookProduct.success,
   508→   508→      product_type: opportunity.productType,
   509→   509→      artifact_path: bookProduct.pdf.filePath,
   510→   510→      dossier_id: dossierId,
   511→   511→      evidence,
   512→   512→      qa_result: {
   513→   513→        passed: bookProduct.qa_result.passed && bookProduct.pdf_verification.overall,
   514→   514→        issues_count: bookProduct.qa_result.issue_count,
   515→   515→        details: bookProduct.qa_result.passed

   526→   526→   * Produce a software product via HandoffContract.

   528→   528→  private produceSoftwareProduct(
   529→   529→    opportunity: ProductOpportunity,
   530→   530→    fit: FitResult,
   531→   531→    arch: ArchitectureDecision,
   532→   532→    evidence: ProductEvidence[]
   533→   533→  ): ProductResult {

   535→   535→      id: randomUUID(),

   547→   547→      constraints: arch.constraints,
   548→   548→      createdAt: new Date().toISOString(),
   549→   549→    };

   551→   551→    evidence.push({
   552→   552→      step: "specification",
   553→   553→      timestamp: new Date().toISOString(),
   554→   554→      description: `HandoffContract generated for ${contract.target_agent}`,
   555→   555→      data: { contractId: contract.id, targetAgent: contract.target_agent },

   557→   557→
   558→   558→    evidence.push({
   559→   559→      step: "production",
   560→   560→      timestamp: new Date().toISOString(),
   561→   561→      description: "Software specification ready for handoff — no rendering required",
   562→   562→      data: { features: contract.specification.features.length },
   563→   563→    });
   564→   564→
   565→   565→    evidence.push({
   566→   566→      step: "qa",
   567→   567→      timestamp: new Date().toISOString(),
   568→   568→      description: "Software products are QA'd by implementation agent",
   569→   569→      data: { passed: true },
   570→   570→    });
   571→   571→
   572→   572→    return {
   573→   573→      success: true,
   574→   574→      product_type: opportunity.productType,
   575→   575→      artifact_path: null,
   576→   576→      dossier_id: randomUUID(),
   577→   577→      evidence,
   578→   578→      qa_result: {
   579→   579→        passed: true,
   580→   580→        issues_count: 0,
   581→   581→        details: "HandoffContract ready — QA delegated to implementation agent",
   582→   582→      },
   583→   583→      handoff_contract: contract,
   584→   584→      production_time_ms: 0,
   585→   585→      errors: [],
   586→   586→    };
   587→   587→  }

   625→   625→      handoff_contract: softwareResult.handoff_contract,

   682→   682→    const baseTimes: Record<FitResult["production_path"], number> = {
   683→   683→      "book-factory-full": 60,
   684→   684→      "book-factory-streamlined": 30,
   685→   685→      "handoff-contract": 5,

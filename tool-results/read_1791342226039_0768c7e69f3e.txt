    40→    40→/* ─────────── Animated Counter ─────────── */
    41→    41→function AnimatedCounter({
    42→    42→  target,
    43→    43→  suffix = "",
    44→    44→  prefix = "",
    45→    45→}: {
    46→    46→  target: number;
    47→    47→  suffix?: string;
    48→    48→  prefix?: string;
    49→    49→}) {
    50→    50→  const [count, setCount] = useState(0);
    51→    51→  const ref = useRef<HTMLSpanElement>(null);
    52→    52→  const isInView = useInView(ref, { once: true });
    53→    53→
    54→    54→  useEffect(() => {
    55→    55→    if (!isInView) return;
    56→    56→    let start = 0;
    57→    57→    const duration = 2000;
    58→    58→    const step = target / (duration / 16);
    59→    59→    const timer = setInterval(() => {
    60→    60→      start += step;
    61→    61→      if (start >= target) {
    62→    62→        setCount(target);
    63→    63→        clearInterval(timer);
    64→    64→      } else {
    65→    65→        setCount(Math.floor(start));
    66→    66→      }
    67→    67→    }, 16);
    68→    68→    return () => clearInterval(timer);
    69→    69→  }, [isInView, target]);
    70→    70→
    71→    71→  return (
    72→    72→    <span ref={ref}>
    73→    73→      {prefix}
    74→    74→      {count.toLocaleString("es-ES")}
    75→    75→      {suffix}
    76→    76→    </span>
    77→    77→  );
    78→    78→}

    81→    81→function FadeInSection({

    87→    87→  className?: string;
    88→    88→  delay?: number;
    89→    89→}) {
    90→    90→  const ref = useRef<HTMLDivElement>(null);
    91→    91→  const isInView = useInView(ref, { once: true, margin: "-60px" });
    92→    92→
    93→    93→  return (
    94→    94→    <motion.div
    95→    95→      ref={ref}
    96→    96→      initial={{ opacity: 0, y: 40 }}
    97→    97→      animate={isInView ? { opacity: 1, y: 0 } : {}}
    98→    98→      transition={{ duration: 0.6, delay, ease: "easeOut" }}
    99→    99→      className={className}
   100→   100→    >
   101→   101→      {children}
   102→   102→    </motion.div>
   103→   103→  );

   997→   997→/* Fallback demo user saved to localStorage when auto-login API fails */
   998→   998→function saveFallbackUser() {
   999→   999→  const fallback = { id: "demo-fallback", name: "Usuario Demo", email: "demo@krea.ai", credits: 50, plan: "starter" };
  1000→  1000→  try { localStorage.setItem("p360_token", fallback.id); } catch {}
  1001→  1001→  try { localStorage.setItem("p360_user", JSON.stringify(fallback)); } catch {}
  1002→  1002→  return fallback;
  1003→  1003→}
  1004→  1004→
  1005→  1005→export default function Home() {
  1006→  1006→  const [view, setView] = useState<View>(() => {
  1007→  1007→    // Try to read from localStorage synchronously on init — go straight to app if token exists
  1008→  1008→    try {
  1009→  1009→      if (typeof window !== "undefined" && localStorage.getItem("p360_token")) {
  1010→  1010→        return "app";
  1011→  1011→      }
  1012→  1012→    } catch {}
  1013→  1013→    return "landing";
  1014→  1014→  });
  1015→  1015→  const [key, setKey] = useState(0);
  1016→  1016→
  1017→  1017→  async function onEnterApp() {
  1018→  1018→    try {
  1019→  1019→      const res = await fetch("/api/auth/auto-login", { method: "POST" });
  1020→  1020→      if (res.ok) {
  1021→  1021→        const data = await res.json();
  1022→  1022→        try { localStorage.setItem("p360_token", data.id); } catch {}
  1023→  1023→        try { localStorage.setItem("p360_user", JSON.stringify(data)); } catch {}
  1024→  1024→      } else {
  1025→  1025→        saveFallbackUser();
  1026→  1026→      }
  1027→  1027→    } catch {
  1028→  1028→      saveFallbackUser();
  1029→  1029→    }
  1030→  1030→    setView("app");
  1031→  1031→    setKey(k => k + 1);
  1032→  1032→  }
  1033→  1033→
  1034→  1034→  // Auto-login on first visit (no token in localStorage)
  1035→  1035→  useEffect(() => {
  1036→  1036→    if (view === "landing") {
  1037→  1037→      onEnterApp();
  1038→  1038→    }
  1039→  1039→  }, []); // eslint-disable-line react-hooks/exhaustive-deps
  1040→  1040→
  1041→  1041→  if (view === "app") {
  1042→  1042→    return (
  1043→  1043→      <ErrorBoundary>
  1044→  1044→        <AppShell key={key} onLogout={() => { setView("auth"); setKey(k => k + 1); }} />
  1045→  1045→      </ErrorBoundary>
  1046→  1046→    );
  1047→  1047→  }
  1048→  1048→
  1049→  1049→  if (view === "auth") {
  1050→  1050→    return (
  1051→  1051→      <ErrorBoundary>
  1052→  1052→        <AuthPage onLogin={() => { setView("app"); setKey(k => k + 1); }} />
  1053→  1053→      </ErrorBoundary>
  1054→  1054→    );
  1055→  1055→  }
  1056→  1056→
  1057→  1057→  // Landing view (briefly shown before auto-login completes)
  1058→  1058→  return (
  1059→  1059→    <ErrorBoundary>
  1060→  1060→      <LandingPage onEnterApp={onEnterApp} />
  1061→  1061→      <FloatingAppButton onClick={onEnterApp} />
  1062→  1062→    </ErrorBoundary>
  1063→  1063→  );
  1064→  1064→}

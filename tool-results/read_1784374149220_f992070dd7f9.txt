    11→// A project with a lot of accumulated images can take a while to fully
    12→// re-render after the picker dialog remounts post-upload — 45s wasn't

    16→const MEDIA_POLL_INTERVAL_MS = 2000;

    18→// final — it can still be mid-transcode/swap for a bit, and downloading
    19→// during that window produces a corrupt file. Give it a beat to settle.

    23→// Step-level markers only (current step, retry N/max, cooldown before the
    24→// next retry or scene) — not per-tick polling noise. Structured, not free
    25→// text: forwarded to the popup so it can render a live countdown for
    26→// `cooldownMs` and replace its whole status display on each new message
    27→// (so a fresh attempt visibly replaces the previous one instead of piling
    28→// up a scrollback).
    29→interface LogUpdate {
    30→  sceneNumber?: number;
    31→  step: string;
    32→  kind: LogKind;
    33→  attempt?: { current: number; max: number };
    34→  cooldownMs?: number;
    35→}

    42→function waitFor<T>(
    43→  check: () => T | null | undefined,
    44→  timeoutMs = WAIT_TIMEOUT_MS
    45→): Promise<T | null> {
    46→  const existing = check();
    47→  if (existing) return Promise.resolve(existing);
    48→
    49→  return new Promise((resolve) => {
    50→    const start = Date.now();
    51→    const interval = setInterval(() => {
    52→      const result = check();
    53→      if (result || aborted || Date.now() - start >= timeoutMs) {
    54→        clearInterval(interval);
    55→        resolve(result ?? null);
    56→      }
    57→    }, WAIT_INTERVAL_MS);
    58→  });
    59→}

    81→async function sleepAbortable(ms: number) {
    82→  const step = 500;
    83→  let waited = 0;
    84→  while (waited < ms && !aborted) {
    85→    await sleep(Math.min(step, ms - waited));
    86→    waited += step;
    87→  }
    88→}

    90→// A grid's own image count can still be climbing right after its dialog
    91→// mounts (React populating "This project" from scratch), which makes a
    92→// single count snapshot unreliable — it can look identical to "a fresh
    93→// upload just landed" when it's really just the initial render finishing.
    94→// Wait until the count hasn't changed for `quietMs` before trusting it.
    95→async function waitForStableCount(
    96→  getCount: () => number,
    97→  quietMs = 800,
    98→  timeoutMs = UPLOAD_WAIT_TIMEOUT_MS
    99→): Promise<number> {
   100→  const start = Date.now();
   101→  let lastCount = getCount();
   102→  let lastChangeAt = Date.now();
   103→  while (Date.now() - start < timeoutMs && !aborted) {
   104→    await sleep(WAIT_INTERVAL_MS);
   105→    const count = getCount();
   106→    if (count !== lastCount) {
   107→      lastCount = count;
   108→      lastChangeAt = Date.now();
   109→    } else if (Date.now() - lastChangeAt >= quietMs) {
   110→      return lastCount;
   111→    }
   112→  }
   113→  return lastCount;
   114→}

   116→// Radix triggers/menu items commonly open on `pointerdown`, not `click`.
   117→// `Element.click()` only fires a `click` event — it skips pointerdown/mouseup,
   118→// so it silently no-ops on those components. Dispatch the full sequence.
   119→// Some tap/drag libraries (grid tiles, Framer Motion-style press handlers)
   120→// validate the pointer coordinates against the element's own bounds instead
   121→// of trusting bubbled events blindly — a synthetic event with clientX/Y at
   122→// 0,0 can silently fail their hit-test. Center the coordinates on the real
   123→// element so those checks pass too.
   124→//
   125→// Every click is followed by a random 1-2s pause: firing this whole flow's
   126→// actions back-to-back (mode switch, upload, tile select, generate...) is
   127→// what was tripping vibes.ai's own rate limiting across scenes.
   128→async function simulateClick(el: HTMLElement) {
   129→  const rect = el.getBoundingClientRect();
   130→  const clientX = rect.left + rect.width / 2;
   131→  const clientY = rect.top + rect.height / 2;
   132→  const opts = { bubbles: true, cancelable: true, composed: true, clientX, clientY };
   133→  const pointerOpts = { ...opts, button: 0, pointerId: 1, isPrimary: true, pointerType: 'mouse' };
   134→  el.dispatchEvent(new PointerEvent('pointerdown', pointerOpts));
   135→  el.dispatchEvent(new MouseEvent('mousedown', opts));
   136→  el.dispatchEvent(new PointerEvent('pointerup', pointerOpts));
   137→  el.dispatchEvent(new MouseEvent('mouseup', opts));
   138→  el.dispatchEvent(new MouseEvent('click', { ...opts, detail: 1 }));
   139→  await sleep(1500 + Math.random() * 1500);
   140→}

   142→function dataURLtoFile(dataurl: string, filename: string): File {
   143→  const arr = dataurl.split(',');
   144→  const mimeMatch = arr[0].match(/:(.*?);/);
   145→  const mime = mimeMatch ? mimeMatch[1] : 'image/jpeg';
   146→  const bstr = atob(arr[1]);
   147→  let n = bstr.length;
   148→  const u8arr = new Uint8Array(n);
   149→  while (n--) u8arr[n] = bstr.charCodeAt(n);
   150→  return new File([u8arr], filename, { type: mime });
   151→}

   161→// vibes.ai's picker/upload dialogs aren't marked with role="dialog" in the
   162→// markup we inspected, so there's no reliable container selector. Instead we
   163→// locate the dialog's own heading text and climb ancestors until `isDone`
   164→// says we've reached a container that holds the whole dialog — that becomes
   165→// the scope for all further queries, so we don't accidentally hit the
   166→// blurred gallery sitting behind the modal.
   167→//
   168→// The default heuristic (>=2 img/button descendants) stops too early on
   169→// "Select start frame": that dialog has a header (Upload nav button + close
   170→// X — already 2 buttons) sitting as a SIBLING of the body that holds the
   171→// actual image grid, so the climb would lock onto the header alone and miss
   172→// the grid entirely. Callers that need the grid pass a stricter check.
   173→function resolveDialogByHeadingSync(
   174→  headingText: string,
   175→  isDone: (node: HTMLElement) => boolean
   176→): HTMLElement | null {
   177→  const heading = Array.from(document.querySelectorAll<HTMLElement>('div,span,h1,h2,h3')).find(
   178→    (el) => el.textContent?.trim() === headingText && el.children.length === 0
   179→  );
   180→  if (!heading) return null;
   181→  let node: HTMLElement | null = heading.parentElement;
   182→  for (let i = 0; i < 12 && node; i++) {
   183→    if (isDone(node)) return node;
   184→    node = node.parentElement;
   185→  }
   186→  return heading.parentElement;
   187→}

   203→// ── Mode switch (Video/Image/Lip sync toggle) ─────────────────────────────────
   204→
   205→// The composer's contenteditable swaps its aria-label between
   206→// "Describe an image..." and "Describe a video..." depending on which mode
   207→// (Video/Image/Lip sync) is currently active — this is the only reliable
   208→// signal for current mode, since the same label text ("Video"/"Image")
   209→// appears both on the mode-toggle trigger and inside its dropdown menu.
   210→function getCurrentMode(): 'image' | 'video' | null {
   211→  const composer = document.querySelector<HTMLElement>(ComposerSelectors.Input);
   212→  const label = composer?.getAttribute('aria-label') ?? composer?.getAttribute('title') ?? '';
   213→  if (label.includes('image')) return BatchModes.Image;
   214→  if (label.includes('video')) return BatchModes.Video;
   215→  return null;
   216→}

   218→// Trigger button for the mode dropdown looks identical (text-wise) to the
   219→// items inside its own menu ("Video" / "Image"). Distinguish it by: it's not
   220→// a menuitem, and it's wrapped by the radix dropdown trigger container
   221→// (aria-haspopup="menu"). The other such trigger on the toolbar is
   222→// "Ingredients", filtered out by the label check.
   223→function findModeTriggerButton(): HTMLButtonElement | null {
   224→  const buttons = Array.from(document.querySelectorAll<HTMLButtonElement>('button'));
   225→  return (
   226→    buttons.find((btn) => {
   227→      if (btn.getAttribute('role') === 'menuitem') return false;
   228→      const label = btn.querySelector('span')?.textContent?.trim();
   229→      if (label !== 'Video' && label !== 'Image') return false;
   230→      return !!btn.closest('[aria-haspopup="menu"]');
   231→    }) ?? null
   232→  );
   233→}

   243→  const wanted = target === BatchModes.Image ? 'Image' : 'Video';
   244→  const items = Array.from(menu.querySelectorAll<HTMLButtonElement>('button[role="menuitem"]'));

   254→async function ensureMode(target: 'image' | 'video'): Promise<boolean> {
   255→  if (getCurrentMode() === target) return true;
   256→
   257→  const menu = await openModeMenu();
   258→  if (!menu) return false;
   259→
   260→  if (!(await clickMenuItem(menu, target))) return false;
   261→
   262→  const result = await waitFor(() => (getCurrentMode() === target ? true : null));
   263→  return result === true;
   264→}

   266→// Right after the composer node appears (page just loaded, or mode switch
   267→// just swapped it), Lexical hasn't finished wiring its own event handlers
   268→// yet. execCommand('insertText') run in that window lands at the DOM level
   269→// but Lexical silently reverts it since the change didn't go through its own
   270→// input pipeline — so the Generate button never enables. Resolving against
   271→// the aria-label that matches the target mode (instead of "any composer")
   272→// guarantees we grab the post-switch node, and the retry loop below covers
   273→// the remaining "not wired up yet" race.
   274→async function getSettledComposer(
   275→  target: 'image' | 'video',
   276→  timeoutMs = 5000
   277→): Promise<HTMLElement | null> {
   278→  return waitFor(() => {
   279→    const composer = document.querySelector<HTMLElement>(ComposerSelectors.Input);
   280→    const label = composer?.getAttribute('aria-label') ?? composer?.getAttribute('title') ?? '';
   281→    const wanted = target === BatchModes.Image ? 'image' : 'video';
   282→    return composer && label.includes(wanted) ? composer : null;
   283→  }, timeoutMs);
   284→}

   286→async function fillComposer(composer: HTMLElement, prompt: string): Promise<boolean> {
   287→  const expected = prompt.trim();
   288→  for (let attempt = 0; attempt < 4; attempt++) {
   289→    composer.focus();
   290→    // Clear first: vibes.ai can leave stray text in the composer (e.g. the
   291→    // uploaded file's name, left over after attaching a start frame) —
   292→    // inserting on top of that without clearing would leave it mixed in
   293→    // with, or entirely masking, the real prompt.
   294→    document.execCommand('selectAll');
   295→    document.execCommand('insertText', false, prompt);
   296→    await sleep(200);
   297→    // Checking for "non-empty" isn't enough: if execCommand silently no-ops
   298→    // (the Lexical-not-wired-up-yet race this retry loop exists for) but the
   299→    // composer already had leftover text in it, a presence check would
   300→    // wrongly report success with the wrong text still sitting there. Only
   301→    // trust it once the composer's content actually matches what we tried
   302→    // to write.
   303→    if ((composer.textContent ?? '').trim() === expected) {
   304→      return true;
   305→    }
   306→  }
   307→  return false;
   308→}

   320→  if (findButtonByText(document, 'Add start frame')) return true;

   322→  const toggle = document.querySelector<HTMLButtonElement>(StartEndFrameSelectors.Toggle);
   323→  if (!toggle) return false;

   325→
   326→  const found = await waitFor(() => findButtonByText(document, 'Add start frame'));
   327→  return !!found;

   330→// Radix-style dialogs on this site close on Escape regardless of which one
   331→// is open, so this is a generic "give up cleanly" hatch — used whenever
   332→// attachStartFrame bails partway through, so a leftover open dialog doesn't
   333→// leak into the next scene's attempt (e.g. its stale tiles getting matched).
   334→function closeAnyOpenDialog() {
   335→  const opts = { bubbles: true, cancelable: true };
   336→  document.dispatchEvent(new KeyboardEvent('keydown', { key: 'Escape', code: 'Escape', ...opts }));
   337→  document.dispatchEvent(new KeyboardEvent('keyup', { key: 'Escape', code: 'Escape', ...opts }));
   338→}

   349→// The global "Upload failed" toast is a real server-side failure signal
   350→// (distinct from the always-instant-enabled local confirm button), and like
   351→// the generation-error toast, stays in the DOM after the fact — so this is
   352→// counted and compared against a per-attempt baseline rather than just
   353→// checked for existence, to avoid a stale toast from a previous attempt
   354→// falsely failing a later, actually-successful one.
   355→function countUploadFailedToasts(): number {
   356→  return Array.from(document.querySelectorAll<HTMLElement>('span')).filter(
   357→    (el) => el.textContent?.trim() === 'Upload failed'
   358→  ).length;

   361→const MAX_UPLOAD_ATTEMPTS = 5;
   362→const UPLOAD_RETRY_DELAY_MS = 12000;
   363→
   364→type UploadAttemptResult =
   365→  | { status: 'success'; pickerDialogAfter: HTMLElement; finalCount: number }
   366→  | { status: 'failed' }
   367→  | { status: 'aborted' };
   368→
   369→async function attemptUpload(
   370→  pickerDialog: HTMLElement,
   371→  imageBase64: string,
   372→  imageName: string,
   373→  baselineTotalImages: number
   374→): Promise<UploadAttemptResult> {
   375→  const uploadFailBaseline = countUploadFailedToasts();
   376→
   377→  const uploadNavBtn = findButtonByText(pickerDialog, 'Upload');
   378→  if (!uploadNavBtn) return { status: 'failed' };
   379→  await simulateClick(uploadNavBtn);
   380→
   381→  const uploadDialog = await findDialogByHeading('Upload images');
   382→  if (!uploadDialog) return { status: 'failed' };
   383→
   384→  const fileInput =
   385→    uploadDialog.querySelector<HTMLInputElement>('input[type="file"]') ??
   386→    document.querySelector<HTMLInputElement>('input[type="file"]');
   387→  if (!fileInput) return { status: 'failed' };
   388→
   389→  const file = dataURLtoFile(imageBase64, imageName);
   390→  const dt = new DataTransfer();
   391→  dt.items.add(file);
   392→  fileInput.files = dt.files;
   393→  fileInput.dispatchEvent(new Event('change', { bubbles: true }));
   394→
   395→  // The confirm button enables in the SAME tick as this dispatch — clicking
   396→  // it immediately means "attach file" and "confirm upload" fire back-to-back
   397→  // with zero gap, which is what was triggering vibes.ai's rate limiting.
   398→  // Force a real pause here regardless of how fast the button itself becomes
   399→  // clickable.
   400→  await sleep(2000 + Math.random() * 1500);
   401→
   402→  // Confirm as soon as it's clickable — this is the normal, expected click
   403→  // (it's what a real user would do too); it does not mean upload finished.
   404→  const confirmBtn = await waitFor(() => {
   405→    const btn = findButtonByText(uploadDialog, 'Upload');
   406→    return btn && !btn.disabled ? btn : null;
   407→  });
   408→  if (!confirmBtn) return { status: 'failed' };
   409→  await simulateClick(confirmBtn);
   410→
   411→  const pickerDialogAfter = await findDialogByHeading('Select start frame', hasAddToVideoButton);
   412→  if (!pickerDialogAfter) return { status: 'failed' };
   413→
   414→  // pickerDialogAfter can go stale mid-wait: vibes.ai can remount the
   415→  // "Select start frame" dialog again after this point (e.g. swapping the
   416→  // temp upload preview for the final CDN thumbnail), which detaches this
   417→  // reference from the live document — querying it further just replays
   418→  // whatever count it had at the moment of detachment, forever, instead of
   419→  // the real live state. Re-resolve the dialog fresh on every check instead
   420→  // of trusting one captured reference. A single stable reading isn't
   421→  // enough either: right after confirming, the grid can be legitimately
   422→  // stable AT the baseline for several seconds while the real upload is
   423→  // still in flight. Keep re-checking (each check itself waiting for its
   424→  // own settle window, capped short so we can also notice an "Upload
   425→  // failed" toast in between) until it stabilizes ABOVE baseline, a failure
   426→  // toast appears, or we run out of time.
   427→  const getLiveDialog = () => resolveDialogByHeadingSync('Select start frame', hasAddToVideoButton);
   428→  const getLiveCount = () => getLiveDialog()?.querySelectorAll('img[data-nimg="fill"]').length ?? 0;
   429→
   430→  const deadline = Date.now() + UPLOAD_WAIT_TIMEOUT_MS;
   431→  let finalCount = baselineTotalImages;
   432→  while (Date.now() < deadline) {
   433→    if (aborted) return { status: 'aborted' };
   434→    if (countUploadFailedToasts() > uploadFailBaseline) return { status: 'failed' };
   435→
   436→    finalCount = await waitForStableCount(getLiveCount, 800, Math.min(3000, deadline - Date.now()));
   437→    if (finalCount > baselineTotalImages) {
   438→      const liveDialog = getLiveDialog() ?? pickerDialogAfter;
   439→      return { status: 'success', pickerDialogAfter: liveDialog, finalCount };
   440→    }
   441→    if (countUploadFailedToasts() > uploadFailBaseline) return { status: 'failed' };
   442→    await sleep(500);
   443→  }
   444→  return { status: 'failed' };
   445→}

   447→async function attachStartFrame(
   448→  imageBase64: string,
   449→  imageName: string,
   450→  sceneNumber: number | undefined
   451→): Promise<boolean> {
   452→  log({ sceneNumber, step: 'Adjuntando start frame', kind: LogKinds.Info });
   453→  const result = await attachStartFrameAttempt(imageBase64, imageName, sceneNumber);
   454→  if (!result) {
   455→    log({ sceneNumber, step: 'No se pudo adjuntar el start frame', kind: LogKinds.Error });
   456→    closeAnyOpenDialog();
   457→    await sleep(300);
   458→  }
   459→  return result;
   460→}

   462→async function attachStartFrameAttempt(
   463→  imageBase64: string,
   464→  imageName: string,
   465→  sceneNumber: number | undefined
   466→): Promise<boolean> {
   467→  if (!(await ensureStartEndFramePanel())) return false;
   468→
   469→  const addStartBtn = findButtonByText(document, 'Add start frame');
   470→  if (!addStartBtn) return false;
   471→  await simulateClick(addStartBtn);
   472→
   473→  const pickerDialog = await findDialogByHeading('Select start frame', hasAddToVideoButton);
   474→  if (!pickerDialog) return false;
   475→
   476→  // The confirm "Upload" button inside the nested dialog enables the instant
   477→  // a local file is attached — it is NOT a signal that the real network
   478→  // upload finished (confirmed: it flips to disabled=false in the very same
   479→  // tick as the file input's change event, no network round-trip involved).
   480→  // The only trustworthy signal is the picker grid's own image count
   481→  // actually growing — but take that baseline only once the grid's own
   482→  // initial render has settled, or a still-populating "This project" tab
   483→  // looks identical to a fresh upload landing.
   484→  const baselineTotalImages = await waitForStableCount(
   485→    () => pickerDialog.querySelectorAll('img[data-nimg="fill"]').length
   486→  );
   487→
   488→  let uploadResult: UploadAttemptResult = { status: 'failed' };
   489→  for (let attempt = 1; attempt <= MAX_UPLOAD_ATTEMPTS; attempt++) {
   490→    if (aborted) {
   491→      uploadResult = { status: 'aborted' };
   492→      break;
   493→    }
   494→    log({
   495→      sceneNumber,
   496→      step: 'Subiendo start frame',
   497→      kind: LogKinds.Info,
   498→      attempt: { current: attempt, max: MAX_UPLOAD_ATTEMPTS },
   499→    });
   500→    uploadResult = await attemptUpload(pickerDialog, imageBase64, imageName, baselineTotalImages);
   501→    if (uploadResult.status !== 'failed') break;
   502→    if (attempt >= MAX_UPLOAD_ATTEMPTS) break;
   503→    log({
   504→      sceneNumber,
   505→      step: 'Subida falló, reintentando',
   506→      kind: LogKinds.Retry,
   507→      attempt: { current: attempt, max: MAX_UPLOAD_ATTEMPTS },
   508→      cooldownMs: UPLOAD_RETRY_DELAY_MS,
   509→    });
   510→    await sleepAbortable(UPLOAD_RETRY_DELAY_MS);
   511→
   512→    // A "failed" result can mean the grid-growth detection just missed a
   513→    // real, successful upload (e.g. its poll window ran out right as the
   514→    // file landed) rather than the upload itself actually failing —
   515→    // retrying blindly in that case re-uploads the same file and leaves a
   516→    // duplicate in the project. Re-check the live count before retrying: if
   517→    // it already grew past baseline, the previous attempt did land, so treat
   518→    // it as success instead of uploading again.
   519→    const liveDialog =
   520→      resolveDialogByHeadingSync('Select start frame', hasAddToVideoButton) ?? pickerDialog;
   521→    const liveCount = liveDialog.querySelectorAll('img[data-nimg="fill"]').length;
   522→    if (liveCount > baselineTotalImages) {
   523→      uploadResult = { status: 'success', pickerDialogAfter: liveDialog, finalCount: liveCount };
   524→      break;
   525→    }
   526→  }
   527→
   528→  if (uploadResult.status !== 'success') return false;
   529→  log({ sceneNumber, step: 'Start frame subido', kind: LogKinds.Success });
   530→
   531→  const { pickerDialogAfter } = uploadResult;
   532→
   533→  const selected = await selectFirstTile(pickerDialogAfter);
   534→  if (!selected) return false;
   535→
   536→  const addToVideoBtn = findButtonByText(pickerDialogAfter, 'Add to video');
   537→  if (!addToVideoBtn || addToVideoBtn.disabled) return false;
   538→  await simulateClick(addToVideoBtn);
   539→
   540→  return true;
   541→}
   542→
   543→// Selecting a tile just toggles a CSS class (blue border) picked up from
   544→// component state — there's no visible attribute we can poll for directly.
   545→// The "Add to video" button unlocking is the real, observable proof the
   546→// click landed. A fully synthetic pointer/mouse sequence (simulateClick) is
   547→// what this grid's tap handler actually needs — verified across every
   548→// observed run.
   549→async function selectFirstTile(pickerDialog: HTMLElement): Promise<boolean> {
   550→  const findFirstTile = (): HTMLElement | null => {
   551→    const img = pickerDialog.querySelector<HTMLImageElement>('img[data-nimg="fill"]');
   552→    if (!img) return null;
   553→    return img.closest<HTMLElement>('div') ?? img.parentElement;
   554→  };
   555→
   556→  const isAddToVideoEnabled = () => {
   557→    const btn = findButtonByText(pickerDialog, 'Add to video');
   558→    return !!btn && !btn.disabled;
   559→  };
   560→
   561→  const first = await waitFor(findFirstTile, UPLOAD_WAIT_TIMEOUT_MS);
   562→  if (!first) return false;
   563→
   564→  await simulateClick(first);
   565→  const result = await waitFor(() => (isAddToVideoEnabled() ? true : null), 2500);
   566→  return result === true;
   567→}

   569→// ── Gallery polling (shared by image and video generation) ───────────────────
   570→
   571→interface ReadyThumbnail {
   572→  mediaId: string;
   573→  batchId: string;
   574→  index: number;
   575→  url: string;
   576→}
   577→
   578→// data-analytics-media-id looks like "batch-<uuid>-content-<n>". The uuid
   579→// itself contains dashes, so split on the "-content-" separator instead of
   580→// trying to pattern-match the uuid shape. A card counts as "ready" once it
   581→// holds a loaded <img> (image mode) or a <video src> (video mode) instead of
   582→// the loading skeleton (a bare <canvas>).
   583→function getReadyThumbnails(): ReadyThumbnail[] {
   584→  const cards = Array.from(document.querySelectorAll<HTMLElement>(GallerySelectors.Thumbnail));
   585→  const result: ReadyThumbnail[] = [];
   586→  for (const card of cards) {
   587→    const mediaId = card.getAttribute('data-analytics-media-id');
   588→    if (!mediaId) continue;
   589→
   590→    const img = card.querySelector<HTMLImageElement>('img[data-nimg="fill"]');
   591→    const video = card.querySelector<HTMLVideoElement>('video[src]');
   592→
   593→    let url: string | null = null;
   594→    if (img && img.complete && img.naturalWidth > 0) url = img.src;
   595→    else if (video && video.src) url = video.src;
   596→    if (!url) continue;
   597→
   598→    const [batchId, indexPart] = mediaId.split('-content-');
   599→    if (!batchId || indexPart === undefined) continue;
   600→    result.push({ mediaId, batchId, index: Number(indexPart), url });
   601→  }
   602→  return result;
   603→}
   604→
   605→function findNewBatch(beforeIds: Set<string>): ReadyThumbnail[] | null {
   606→  const fresh = getReadyThumbnails().filter((t) => !beforeIds.has(t.mediaId));
   607→  const byBatch = new Map<string, ReadyThumbnail[]>();
   608→  for (const t of fresh) {
   609→    const arr = byBatch.get(t.batchId) ?? [];
   610→    arr.push(t);
   611→    byBatch.set(t.batchId, arr);
   612→  }
   613→  for (const arr of byBatch.values()) {
   614→    if (arr.length >= 4) return arr.sort((a, b) => a.index - b.index).slice(0, 4);
   615→  }
   616→  return null;
   617→}
   618→
   619→function getUrlByMediaId(mediaId: string): string | null {
   620→  const card = document.querySelector<HTMLElement>(
   621→    `[data-analytics-media-id="${CSS.escape(mediaId)}"]`
   622→  );
   623→  if (!card) return null;
   624→  const img = card.querySelector<HTMLImageElement>('img[data-nimg="fill"]');
   625→  const video = card.querySelector<HTMLVideoElement>('video[src]');
   626→  if (img && img.complete && img.naturalWidth > 0) return img.src;
   627→  if (video && video.src) return video.src;
   628→  return null;
   629→}
   630→
   631→// vibes.ai shows this inline card in place of a thumbnail when a generation
   632→// errors out (rate limit, content policy, etc.) — for either image or video.
   633→// It stays in the DOM even after a later retry succeeds, so a plain
   634→// "does one exist?" check would keep re-triggering on a stale card from an
   635→// earlier attempt. Counting them and comparing against a baseline taken
   636→// right before each attempt is what tells a NEW failure apart from an old one.
   637→function countGenerationErrors(): number {
   638→  return Array.from(document.querySelectorAll<HTMLElement>('span')).filter(
   639→    (el) => el.textContent?.trim() === "Couldn't generate"
   640→  ).length;
   641→}

   643→type MediaPollResult =
   644→  | { status: 'success'; urls: string[] }
   645→  | { status: 'error' }
   646→  | { status: 'timeout' }
   647→  | { status: 'aborted' };
   648→
   649→async function waitForMediaBatch(
   650→  beforeIds: Set<string>,
   651→  errorBaseline: number
   652→): Promise<MediaPollResult> {
   653→  for (let attempts = 0; attempts < MEDIA_POLL_MAX_ATTEMPTS; attempts++) {
   654→    if (aborted) return { status: 'aborted' };
   655→    if (countGenerationErrors() > errorBaseline) return { status: 'error' };
   656→
   657→    const batchMedia = findNewBatch(beforeIds);
   658→    if (batchMedia) {
   659→      await sleepAbortable(MEDIA_STABILIZE_MS);
   660→      if (aborted) return { status: 'aborted' };
   661→      // Re-read the src right before downloading — if the CDN swapped the
   662→      // file out from under the thumbnail during the settle window, this
   663→      // picks up the final one instead of the stale reference.
   664→      const urls = batchMedia.map((t) => getUrlByMediaId(t.mediaId) ?? t.url);
   665→      return { status: 'success', urls };
   666→    }
   667→
   668→    await sleep(MEDIA_POLL_INTERVAL_MS);
   669→  }
   670→  return { status: 'timeout' };
   671→}
   672→
   673→const MAX_GENERATION_ATTEMPTS = 5;
   674→const GENERATION_RETRY_DELAY_MS = 25000;
   675→
   676→// On "Couldn't generate", vibes.ai's own flakiness (not the prompt) is the
   677→// usual culprit — the same prompt often succeeds on a plain retry. Wait,
   678→// re-click Generate (reusing whatever's already in the composer), and only
   679→// give up after MAX_GENERATION_ATTEMPTS straight failures.
   680→async function generateWithRetries(
   681→  beforeIds: Set<string>,
   682→  sceneNumber: number,
   683→  clickGenerate: () => Promise<boolean>
   684→) {
   685→  let errorBaseline = countGenerationErrors();
   686→
   687→  for (let attempt = 1; attempt <= MAX_GENERATION_ATTEMPTS; attempt++) {
   688→    if (aborted) return;
   689→
   690→    log({
   691→      sceneNumber,
   692→      step: 'Esperando generación',
   693→      kind: LogKinds.Info,
   694→      attempt: { current: attempt, max: MAX_GENERATION_ATTEMPTS },
   695→      cooldownMs: MEDIA_POLL_MAX_ATTEMPTS * MEDIA_POLL_INTERVAL_MS,
   696→    });
   697→    const result = await waitForMediaBatch(beforeIds, errorBaseline);
   698→
   699→    if (result.status === 'aborted') return;
   700→
   701→    if (result.status === 'success') {
   702→      log({ sceneNumber, step: 'Media lista, descargando', kind: LogKinds.Success });
   703→      await browser.runtime.sendMessage({
   704→        action: Actions.DownloadMediaDirect,
   705→        urls: result.urls,
   706→        sceneNumber,
   707→      });
   708→      return;
   709→    }
   710→
   711→    if (result.status === 'timeout') {
   712→      const timeoutMinutes = (MEDIA_POLL_MAX_ATTEMPTS * MEDIA_POLL_INTERVAL_MS) / 60000;
   713→      log({
   714→        sceneNumber,
   715→        step: `Tiempo agotado esperando media (${timeoutMinutes} min)`,
   716→        kind: LogKinds.Error,
   717→      });
   718→      return;
   719→    }
   720→
   721→    // status === 'error'
   722→    if (attempt >= MAX_GENERATION_ATTEMPTS) {
   723→      log({
   724→        sceneNumber,
   725→        step: 'Generación falló, saltando escena',
   726→        kind: LogKinds.Error,
   727→        attempt: { current: attempt, max: MAX_GENERATION_ATTEMPTS },
   728→      });
   729→      await browser.runtime.sendMessage({ action: Actions.SceneFailed, sceneNumber });
   730→      return;
   731→    }
   732→
   733→    log({
   734→      sceneNumber,
   735→      step: 'Generación falló, reintentando',
   736→      kind: LogKinds.Retry,
   737→      attempt: { current: attempt, max: MAX_GENERATION_ATTEMPTS },
   738→      cooldownMs: GENERATION_RETRY_DELAY_MS,
   739→    });
   740→    await sleepAbortable(GENERATION_RETRY_DELAY_MS);
   741→    if (aborted) return;
   742→
   743→    // Reset the baseline to include the error(s) we just handled, so the
   744→    // NEXT attempt's check only fires on a genuinely new failure card
   745→    // instead of re-tripping on this one forever.
   746→    errorBaseline = countGenerationErrors();
   747→
   748→    const clicked = await clickGenerate();
   749→    if (!clicked) {
   750→      log({ sceneNumber, step: 'No se pudo reintentar, saltando escena', kind: LogKinds.Error });
   751→      await browser.runtime.sendMessage({ action: Actions.SceneFailed, sceneNumber });
   752→      return;
   753→    }
   754→  }
   755→}

   757→// ── Mode handlers ──────────────────────────────────────────────────────────────
   758→
   759→async function handleImageMode(
   760→  prompt: string,
   761→  sceneNumber: number | undefined,
   762→  sendResponse: (r: ContentResponse) => void
   763→) {
   764→  log({ sceneNumber, step: 'Generando imagen', kind: LogKinds.Info });
   765→  const switched = await ensureMode(BatchModes.Image);
   766→  if (!switched) {
   767→    sendResponse({ success: false, error: 'No se pudo activar el modo image.' });
   768→    return;
   769→  }
   770→
   771→  const composer = await getSettledComposer(BatchModes.Image);
   772→  if (!composer) {
   773→    sendResponse({ success: false, error: 'Composer no encontrado tras cambiar de modo.' });
   774→    return;
   775→  }
   776→
   777→  const filled = await fillComposer(composer, prompt);
   778→  if (!filled) {
   779→    sendResponse({ success: false, error: 'No se pudo escribir el prompt en el composer.' });
   780→    return;
   781→  }
   782→
   783→  const beforeIds = new Set(getReadyThumbnails().map((t) => t.mediaId));
   784→
   785→  const clickGenerate = async (): Promise<boolean> => {
   786→    // Defensive: a failed generation (or the composer picking up stray text
   787→    // from elsewhere on the page) can leave it empty or wrong ahead of a
   788→    // retry — re-fill unless it already holds exactly the prompt we want.
   789→    if ((composer.textContent ?? '').trim() !== prompt.trim()) {
   790→      if (!(await fillComposer(composer, prompt))) return false;
   791→    }
   792→    // A small buffer before hitting Generate — clicking the instant it
   793→    // enables, scene after scene, is what triggers vibes.ai's rate limiting.
   794→    await sleep(1300);
   795→    const btn = await waitForEnabledButton(GenerateButtonSelectors.Image);
   796→    if (!btn) return false;
   797→    await simulateClick(btn);
   798→    return true;
   799→  };
   800→
   801→  if (!(await clickGenerate())) {
   802→    sendResponse({ success: false, error: 'Botón Generate no disponible.' });
   803→    return;
   804→  }
   805→
   806→  sendResponse({ success: true, message: 'Enviado. Esperando imágenes...' });
   807→  if (sceneNumber !== undefined) generateWithRetries(beforeIds, sceneNumber, clickGenerate);
   808→}
   809→
   810→async function handleVideoMode(
   811→  prompt: string,
   812→  imageBase64: string | null,
   813→  imageName: string | null,
   814→  sceneNumber: number | undefined,
   815→  sendResponse: (r: ContentResponse) => void
   816→) {
   817→  log({ sceneNumber, step: 'Generando video', kind: LogKinds.Info });
   818→  const switched = await ensureMode(BatchModes.Video);
   819→  if (!switched) {
   820→    sendResponse({ success: false, error: 'No se pudo activar el modo video.' });
   821→    return;
   822→  }
   823→
   824→  if (imageBase64 && imageName) {
   825→    const attached = await attachStartFrame(imageBase64, imageName, sceneNumber);
   826→    if (!attached) {
   827→      sendResponse({ success: false, error: 'No se pudo adjuntar el start frame.' });
   828→      return;
   829→    }
   830→  }
   831→
   832→  const composer = await getSettledComposer(BatchModes.Video);
   833→  if (!composer) {
   834→    sendResponse({ success: false, error: 'Composer no encontrado tras cambiar de modo.' });
   835→    return;
   836→  }
   837→
   838→  const filled = await fillComposer(composer, prompt);
   839→  if (!filled) {
   840→    sendResponse({ success: false, error: 'No se pudo escribir el prompt en el composer.' });
   841→    return;
   842→  }
   843→
   844→  const beforeIds = new Set(getReadyThumbnails().map((t) => t.mediaId));
   845→
   846→  const clickGenerate = async (): Promise<boolean> => {
   847→    // Defensive: a failed generation can leave the composer empty. The
   848→    // start frame itself is untouched by a failed attempt, so no need to
   849→    // re-attach it here — only removed once this scene is fully done below.
   850→    if ((composer.textContent ?? '').trim().length === 0) {
   851→      if (!(await fillComposer(composer, prompt))) return false;
   852→    }
   853→    // A small buffer before hitting Generate — clicking the instant it
   854→    // enables, scene after scene, is what triggers vibes.ai's rate limiting.
   855→    await sleep(1300);
   856→    const btn = await waitForEnabledButton(GenerateButtonSelectors.Video);
   857→    if (!btn) return false;
   858→    await simulateClick(btn);
   859→    return true;
   860→  };
   861→
   862→  if (!(await clickGenerate())) {
   863→    sendResponse({ success: false, error: 'Botón Generate no disponible.' });
   864→    return;
   865→  }
   866→
   867→  sendResponse({ success: true, message: 'Enviado. Esperando videos...' });
   868→
   869→  if (sceneNumber !== undefined) {
   870→    await generateWithRetries(beforeIds, sceneNumber, clickGenerate);
   871→  }
   872→
   873→  // Only clear the start frame once we're fully done with this scene
   874→  // (success, timeout, or exhausted retries) — a retry needs it still
   875→  // attached to regenerate the same video. Left alone after that, the next
   876→  // scene would silently reuse THIS scene's reference image instead of its
   877→  // own.
   878→  await removeStartFrame();
   879→}

   881→export default defineContentScript({
   882→  matches: ['*://*.vibes.ai/*', '*://vibes.ai/*'],
   883→  main() {
   884→    browser.runtime.onMessage.addListener(
   885→      (message: ExtensionMessage, _sender, sendResponse: (r: ContentResponse) => void) => {
   886→        if (message.action === Actions.StopBatch) {
   887→          aborted = true;
   888→          return false;
   889→        }
   890→
   891→        if (message.action !== Actions.FillPrompt) return false;
   892→
   893→        // A fresh scene command means the batch is active again (e.g. a new
   894→        // batch started after a previous one was stopped) — clear any stale
   895→        // abort flag so this scene isn't dead on arrival.
   896→        aborted = false;
   897→
   898→        const { prompt, mediaType, imageBase64, imageName, sceneNumber } = message;
   899→
   900→        if (mediaType === BatchModes.Image) {
   901→          handleImageMode(prompt, sceneNumber, sendResponse);
   902→        } else {
   903→          handleVideoMode(prompt, imageBase64, imageName, sceneNumber, sendResponse);
   904→        }
   905→
   906→        return true;
   907→      }
   908→    );
   909→  },
   910→});

   356→// ─── EditorialDesign Class ─────────────────────────────────────────────
   357→
   358→/**
   359→ * EditorialDesign — produces editorial specifications and deterministic layouts.
   360→ *
   361→ * Two main operations:
   362→ * 1. design() — produces the EditorialSpecification (AC-038)
   363→ * 2. layout() — produces a deterministic LayoutResult (AC-039)
   364→ *
   365→ * Both operations derive from the same ArtDirectionSpec, ensuring
   366→ * visual system coherence throughout (AC-040).
   367→ */
   368→export class EditorialDesign {
   369→  /**
   370→   * Produce the editorial specification from art direction and book structure.
   371→   *
   372→   * @param artDirection - The visual system specification
   373→   * @param bookArchitecture - The book's editorial structure
   374→   * @returns Complete EditorialSpecification (AC-038)
   375→   */
   376→  design(
   377→    artDirection: ArtDirectionSpec,
   378→    bookArchitecture: BookArchitectureInput
   379→  ): EditorialSpecification {
   380→    const margins = this.computeMargins(artDirection);
   381→    const grid = this.buildGrid(artDirection);
   382→    const headingStyles = this.buildHeadingStyles(artDirection);
   383→    const paragraphStyle = this.buildParagraphStyle(artDirection);
   384→    const quoteStyle = this.buildQuoteStyle(artDirection);
   385→    const listStyle = this.buildListStyle(artDirection);
   386→    const tableStyle = this.buildTableStyle(artDirection);
   387→    const imageStyle = this.buildImageStyle(artDirection);
   388→    const captionStyle = this.buildCaptionStyle(artDirection);
   389→    const chapterStyle = this.buildChapterStyle(artDirection, headingStyles[0]);
   390→    const pageBreakRules = this.buildPageBreakRules();
   391→    const numbering = this.buildNumberingConfig();
   392→    const indexConfig = this.buildIndexConfig(bookArchitecture);
   393→    const hierarchyRules = this.buildHierarchyRules();
   394→
   395→    return {
   396→      id: randomUUID(),
   397→      createdAt: new Date().toISOString(),
   398→      visual_system_id: artDirection.visual_system.id,
   399→      margins,
   400→      grid,
   401→      hierarchy_rules: hierarchyRules,
   402→      heading_styles: headingStyles,
   403→      paragraph_style: paragraphStyle,
   404→      quote_style: quoteStyle,
   405→      list_style: listStyle,
   406→      table_style: tableStyle,
   407→      image_style: imageStyle,
   408→      caption_style: captionStyle,
   409→      chapter_style: chapterStyle,
   410→      page_break_rules: pageBreakRules,
   411→      numbering,
   412→      index_config: indexConfig,
   413→    };
   414→  }
   415→
   416→  /**
   417→   * Produce a deterministic layout from content and specifications.
   418→   *
   419→   * AC-039: Layout is deterministic — same inputs always produce same output.
   420→   * This is achieved by processing content sequentially with fixed rules,
   421→   * no randomization, and a deterministic page-breaking algorithm.
   422→   *
   423→   * @param content - The book content to lay out
   424→   * @param editorial - The editorial specification
   425→   * @param artDirection - The visual system (AC-040)
   426→   * @returns Deterministic LayoutResult (AC-039)
   427→   */
   428→  layout(
   429→    content: BookContent,
   430→    editorial: EditorialSpecification,
   431→    artDirection: ArtDirectionSpec
   432→  ): LayoutResult {
   433→    const inputHash = this.computeInputHash(content, editorial);
   434→    const { width } = artDirection.visual_system.page_dimensions;
   435→    const contentWidth = width - editorial.margins.inner - editorial.margins.outer;
   436→    const contentHeight = artDirection.visual_system.page_dimensions.height
   437→      - editorial.margins.top
   438→      - editorial.margins.bottom;
   439→
   440→    const pages: LayoutPage[] = [];
   441→    const toc: LayoutResult["table_of_contents"] = [];
   442→
   443→    let currentPage = 0;
   444→    let currentY = 0;
   445→    let currentChapter: number | null = null;
   446→    let currentPageElements: LayoutElement[] = [];
   447→    let remainingSpace = contentHeight;
   448→
   449→    // Process each chapter
   450→    for (const chapter of content.chapters) {
   451→      // Chapter opening — always starts new page
   452→      if (currentPage > 0 || currentPageElements.length > 0) {
   453→        // Finalize previous page
   454→        pages.push(this.finalizePage(
   455→          currentPage,
   456→          currentChapter,
   457→          currentPageElements,
   458→          remainingSpace,
   459→          editorial
   460→        ));
   461→        currentPage++;
   462→        currentPageElements = [];
   463→        currentY = 0;
   464→        remainingSpace = contentHeight;
   465→      }
   466→
   467→      currentChapter = chapter.chapter_number;
   468→
   469→      // Chapter heading
   470→      const chHeadingHeight = this.estimateElementHeight(
   471→        "heading",
   472→        chapter.title,
   473→        editorial.chapter_style.title_style,
   474→        contentWidth
   475→      );
   476→
   477→      toc.push({
   478→        title: chapter.title,
   479→        page: currentPage + 1, // 1-indexed for display
   480→        level: 1,
   481→      });
   482→
   483→      currentPageElements.push({
   484→        type: "heading",
   485→        page: currentPage,
   486→        y: currentY,
   487→        x: editorial.margins.inner,
   488→        width: contentWidth,
   489→        height: chHeadingHeight,
   490→        content_ref: `chapter-${chapter.chapter_number}-title`,
   491→        style: { ...editorial.chapter_style.title_style },
   492→      });
   493→
   494→      currentY += chHeadingHeight;
   495→      remainingSpace -= chHeadingHeight;
   496→
   497→      // Process sections
   498→      for (const section of chapter.sections) {
   499→        // Section heading
   500→        if (section.heading) {
   501→          const headingStyle = editorial.heading_styles.find(
   502→            (h) => h.level === section.level
   503→          ) ?? editorial.heading_styles[editorial.heading_styles.length - 1];
   504→
   505→          const headingHeight = this.estimateElementHeight(
   506→            "heading",
   507→            section.heading,
   508→            headingStyle,
   509→            contentWidth
   510→          );
   511→
   512→          if (headingHeight > remainingSpace) {
   513→            pages.push(this.finalizePage(
   514→              currentPage, currentChapter, currentPageElements, remainingSpace, editorial
   515→            ));
   516→            currentPage++;
   517→            currentPageElements = [];
   518→            currentY = 0;
   519→            remainingSpace = contentHeight;
   520→          }
   521→
   522→          toc.push({
   523→            title: section.heading,
   524→            page: currentPage + 1,
   525→            level: section.level,
   526→          });
   527→
   528→          currentPageElements.push({
   529→            type: "heading",
   530→            page: currentPage,
   531→            y: currentY,
   532→            x: editorial.margins.inner,
   533→            width: contentWidth,
   534→            height: headingHeight,
   535→            content_ref: `chapter-${chapter.chapter_number}-section-${section.level}`,
   536→            style: { ...headingStyle },
   537→          });
   538→
   539→          currentY += headingHeight;
   540→          remainingSpace -= headingHeight;
   541→        }
   542→
   543→        // Paragraphs
   544→        for (let pi = 0; pi < section.paragraphs.length; pi++) {
   545→          const para = section.paragraphs[pi];
   546→          const paraHeight = this.estimateElementHeight(
   547→            "paragraph",
   548→            para,
   549→            editorial.paragraph_style,
   550→            contentWidth
   551→          );
   552→
   553→          if (paraHeight > remainingSpace) {
   554→            pages.push(this.finalizePage(
   555→              currentPage, currentChapter, currentPageElements, remainingSpace, editorial
   556→            ));
   557→            currentPage++;
   558→            currentPageElements = [];
   559→            currentY = 0;
   560→            remainingSpace = contentHeight;
   561→          }
   562→
   563→          currentPageElements.push({
   564→            type: "paragraph",
   565→            page: currentPage,
   566→            y: currentY,
   567→            x: editorial.margins.inner,
   568→            width: contentWidth,
   569→            height: Math.min(paraHeight, remainingSpace),
   570→            content_ref: `chapter-${chapter.chapter_number}-para-${pi}`,
   571→            style: { ...editorial.paragraph_style },
   572→          });
   573→
   574→          currentY += paraHeight;
   575→          remainingSpace -= paraHeight;
   576→        }
   577→      }
   578→    }
   579→
   580→    // Finalize last page
   581→    if (currentPageElements.length > 0) {
   582→      pages.push(this.finalizePage(
   583→        currentPage, currentChapter, currentPageElements, remainingSpace, editorial
   584→      ));
   585→    }
   586→
   587→    return {
   588→      id: randomUUID(),
   589→      createdAt: new Date().toISOString(),
   590→      input_hash: inputHash,
   591→      page_count: pages.length,
   592→      pages,
   593→      table_of_contents: toc,
   594→      visual_system_id: artDirection.visual_system.id,
   595→    };
   596→  }
   597→
   598→  // ─── Private Methods ───────────────────────────────────────────────
   599→
   600→  /**
   601→   * Compute margins based on visual system and page size.
   602→   */
   603→  private computeMargins(artDirection: ArtDirectionSpec): Margins {
   604→    const base = artDirection.visual_system.base_unit;
   605→    // Traditional book margins: inner < outer (for gutter), top < bottom
   606→    return {
   607→      top: base * 7,    // 56pt
   608→      bottom: base * 9, // 72pt
   609→      inner: base * 6,  // 48pt (gutter)
   610→      outer: base * 8,  // 64pt
   611→    };
   612→  }
   613→
   614→  /**
   615→   * Build grid specification.
   616→   */
   617→  private buildGrid(artDirection: ArtDirectionSpec): GridSpec {
   618→    return {
   619→      columns: 1, // Single column for most books
   620→      column_gap: 0,
   621→      row_gap: artDirection.visual_system.base_unit,
   622→      baseline_grid: artDirection.typography.sizes.base * artDirection.typography.line_heights.normal,
   623→      asymmetric_margins: true,
   624→    };
   625→  }
   626→
   627→  /**
   628→   * Build heading styles from art direction hierarchy.
   629→   */
   630→  private buildHeadingStyles(artDirection: ArtDirectionSpec): HeadingStyleRule[] {
   631→    const h = artDirection.hierarchy;
   632→    const t = artDirection.typography;
   633→
   634→    return [
   635→      {
   636→        level: 1, font: h.h1.font, size: h.h1.size, weight: h.h1.weight,
   637→        color: h.h1.color, line_height: h.h1.line_height,
   638→        space_before: h.h1.margin_top, space_after: h.h1.margin_bottom,
   639→        keep_with_next: true, numbering_format: "chapter", numbering_prefix: "Chapter",
   640→      },

   856→  /**
   857→   * Estimate the height of a layout element in points.
   858→   * Used by the deterministic layout algorithm.
   859→   */
   860→  private estimateElementHeight(
   861→    type: "heading" | "paragraph",
   862→    text: string,
   863→    style: { size: number; line_height: number; space_before?: number; space_after?: number },
   864→    contentWidth: number
   865→  ): number {
   866→    // Estimate characters per line based on font size and width
   867→    const charsPerLine = Math.floor(contentWidth / (style.size * 0.5));
   868→    const lines = Math.max(1, Math.ceil(text.length / charsPerLine));
   869→    const lineHeight = style.size * style.line_height;
   870→    const height = lines * lineHeight;
   871→
   872→    const spaceBefore = ("space_before" in style && typeof style.space_before === "number")
   873→      ? style.space_before : 0;
   874→    const spaceAfter = ("space_after" in style && typeof style.space_after === "number")
   875→      ? style.space_after : 0;
   876→
   877→    return height + spaceBefore + spaceAfter;
   878→  }

   880→  /**
   881→   * Finalize a page with header/footer.
   882→   */
   883→  private finalizePage(
   884→    pageNum: number,
   885→    chapterNum: number | null,
   886→    elements: LayoutElement[],
   887→    remainingSpace: number,
   888→    editorial: EditorialSpecification
   889→  ): LayoutPage {
   890→    const headerText = chapterNum ? `Chapter ${chapterNum}` : null;
   891→    let footerText: string | null = null;
   892→
   893→    if (editorial.numbering.page_number_position.startsWith("footer")) {
   894→      footerText = String(pageNum + editorial.numbering.first_page_number);
   895→    }
   896→
   897→    return {
   898→      page_number: pageNum,
   899→      is_chapter_opening: elements.some((e) => e.type === "heading" && e.content_ref.includes("title")),
   900→      chapter_number: chapterNum,
   901→      elements,
   902→      remaining_space: remainingSpace,
   903→      header_text: headerText,
   904→      footer_text: footerText,
   905→    };
   906→  }

   908→  /**
   909→   * Compute a deterministic hash of layout inputs for reproducibility check.
   910→   * AC-039: Same inputs → same hash → same layout.
   911→   */
   912→  private computeInputHash(
   913→    content: BookContent,
   914→    editorial: EditorialSpecification
   915→  ): string {
   916→    // Simple deterministic hash — concatenate key inputs and hash
   917→    const data = JSON.stringify({
   918→      title: content.title,
   919→      chapterCount: content.chapters.length,
   920→      totalWords: content.total_word_count,
   921→      editorialId: editorial.id,
   922→      margins: editorial.margins,
   923→      grid: editorial.grid,
   924→    });
   925→
   926→    // Simple hash function (djb2)
   927→    let hash = 5381;
   928→    for (let i = 0; i < data.length; i++) {
   929→      hash = ((hash << 5) + hash + data.charCodeAt(i)) & 0xffffffff;
   930→    }
   931→    return hash.toString(16).padStart(8, "0");
   932→  }

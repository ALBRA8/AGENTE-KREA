     1→/**
     2→ * Tool Contract — ADN GENERAL DEL AGENTE V1.0
     3→ *
     4→ * Defines the tools the agent uses: their identity, purpose, schemas,
     5→ * permissions, risk, side effects, and execution tracking.
     6→ *
     7→ * Tools are the lowest-level primitives that capabilities compose.
     8→ * Each tool execution produces evidence for the provenance chain.

   228→// ─── Tool Registry ───────────────────────────────────────────────────────────
   229→
   230→/**
   231→ * ToolRegistry — Central registry for all tools available to the agent.
   232→ */
   233→export class ToolRegistry {
   234→  private tools: Map<string, ToolDefinition> = new Map();
   235→
   236→  /** Register a tool */
   237→  register(tool: ToolDefinition): void {
   238→    this.tools.set(tool.id, tool);
   239→  }
   240→
   241→  /** Register multiple tools */
   242→  registerAll(tools: ToolDefinition[]): void {
   243→    for (const tool of tools) {
   244→      this.register(tool);
   245→    }
   246→  }
   247→
   248→  /** Get a tool by ID */
   249→  get(id: string): ToolDefinition | undefined {
   250→    return this.tools.get(id);
   251→  }
   252→
   253→  /** Get all registered tools */
   254→  getAll(): ToolDefinition[] {
   255→    return Array.from(this.tools.values());
   256→  }
   257→
   258→  /** Check if a tool exists */
   259→  has(id: string): boolean {
   260→    return this.tools.has(id);
   261→  }
   262→
   263→  /** Get tools by category */
   264→  getByCategory(category: ToolCategory): ToolDefinition[] {
   265→    return this.getAll().filter((t) => t.category === category);
   266→  }
   267→
   268→  /** Get tools with a specific side effect */
   269→  getWithSideEffect(effect: SideEffect): ToolDefinition[] {
   270→    return this.getAll().filter((t) => t.sideEffects.includes(effect));
   271→  }
   272→
   273→  /** Get tools that require a specific permission */
   274→  getRequiringPermission(permission: string): ToolDefinition[] {
   275→    return this.getAll().filter((t) => t.permissions.includes(permission));
   276→  }
   277→
   278→  /** Get tools above a risk level */
   279→  getAboveRiskLevel(risk: ToolRiskLevel): ToolDefinition[] {
   280→    const order: Record<ToolRiskLevel, number> = { none: 0, low: 1, medium: 2, high: 3, critical: 4 };
   281→    return this.getAll().filter((t) => order[t.risk] > order[risk]);
   282→  }
   283→
   284→  /** Get all currently available tools */
   285→  getAvailable(): ToolDefinition[] {
   286→    return this.getAll().filter((t) => t.available);
   287→  }
   288→}

   290→// ─── Tool Executor ───────────────────────────────────────────────────────────
   291→
   292→/** Function type for tool execution */
   293→export type ToolExecutorFn = (inputs: Record<string, unknown>) => Promise<unknown>;
   294→
   295→/**
   296→ * ToolExecutor — Executes tools with full evidence tracking and auditing.
   297→ * Wraps each execution with timing, evidence capture, and audit logging.
   298→ */
   299→export class ToolExecutor {
   300→  private registry: ToolRegistry;
   301→  private executors: Map<string, ToolExecutorFn> = new Map();
   302→  private auditLog: ToolAuditRecord[] = [];
   303→
   304→  constructor(registry: ToolRegistry) {
   305→    this.registry = registry;
   306→  }
   307→
   308→  /** Register an executor function for a tool */
   309→  registerExecutor(toolId: string, executor: ToolExecutorFn): void {
   310→    this.executors.set(toolId, executor);
   311→  }
   312→
   313→  /**
   314→   * Execute a tool with full tracking.
   315→   * Produces evidence and audit records for every execution.
   316→   */
   317→  async execute(
   318→    toolId: string,
   319→    inputs: Record<string, unknown>,
   320→    initiatedBy: string,
   321→    executionId?: string,
   322→  ): Promise<ToolExecutionResult> {
   323→    const execId = executionId || randomUUID();
   324→    const tool = this.registry.get(toolId);
   325→
   326→    if (!tool) {
   327→      return this.createFailedResult(execId, toolId, 0, 0, `Tool not found: ${toolId}`, initiatedBy);
   328→    }
   329→
   330→    if (!tool.available) {
   331→      return this.createFailedResult(execId, toolId, 0, 0, `Tool not available: ${toolId}`, initiatedBy);
   332→    }
   333→
   334→    const executor = this.executors.get(toolId);
   335→    if (!executor) {
   336→      return this.createFailedResult(execId, toolId, 0, 0, `No executor registered for tool: ${toolId}`, initiatedBy);
   337→    }
   338→
   339→    const startedAt = new Date().toISOString();
   340→    const startTime = Date.now();
   341→    const evidence: ToolEvidence[] = [];
   342→
   343→    try {
   344→      // Execute with timeout
   345→      const output = await this.executeWithTimeout(executor, inputs, tool.timeoutMs);
   346→      const durationMs = Date.now() - startTime;
   347→      const endedAt = new Date().toISOString();
   348→
   349→      // Create evidence
   350→      evidence.push({
   351→        evidenceId: randomUUID(),
   352→        toolId,
   353→        executionId: execId,
   354→        timestamp: endedAt,
   355→        observation: `Tool ${toolId} executed successfully`,
   356→        confidence: 1.0,
   357→        truthLevel: "OBSERVED",
   358→        source: `tool:${toolId}`,
   359→      });
   360→
   361→      // Create audit record
   362→      const audit: ToolAuditRecord = {
   363→        auditId: randomUUID(),
   364→        toolId,
   365→        executionId: execId,
   366→        initiatedBy,
   367→        startedAt,
   368→        endedAt,
   369→        durationMs,
   370→        success: true,
   371→        sideEffectsObserved: tool.sideEffects,
   372→        evidenceIds: evidence.map((e) => e.evidenceId),
   373→        creditsConsumed: tool.sideEffects.includes("consumes_credits") ? 1 : 0,
   374→        inputsRedacted: tool.handlesSensitiveData,
   375→        outputsRedacted: false,
   376→      };
   377→
   378→      this.auditLog.push(audit);
   379→
   380→      return {
   381→        executionId: execId,
   382→        toolId,
   383→        success: true,
   384→        output,
   385→        evidence,
   386→        audit,
   387→        durationMs,
   388→        creditsConsumed: audit.creditsConsumed,
   389→      };
   390→    } catch (error) {
   391→      const durationMs = Date.now() - startTime;
   392→      const endedAt = new Date().toISOString();
   393→      const errorMessage = error instanceof Error ? error.message : String(error);
   394→
   395→      // Create failed evidence
   396→      evidence.push({
   397→        evidenceId: randomUUID(),
   398→        toolId,
   399→        executionId: execId,
   400→        timestamp: endedAt,
   401→        observation: `Tool ${toolId} execution failed: ${errorMessage}`,
   402→        confidence: 0,
   403→        truthLevel: "OBSERVED",
   404→        source: `tool:${toolId}`,
   405→      });
   406→
   407→      // Create audit record for failure
   408→      const audit: ToolAuditRecord = {
   409→        auditId: randomUUID(),
   410→        toolId,
   411→        executionId: execId,
   412→        initiatedBy,
   413→        startedAt,
   414→        endedAt,
   415→        durationMs,
   416→        success: false,
   417→        error: errorMessage,
   418→        sideEffectsObserved: [],
   419→        evidenceIds: evidence.map((e) => e.evidenceId),
   420→        creditsConsumed: 0,
   421→        inputsRedacted: tool.handlesSensitiveData,
   422→        outputsRedacted: true,
   423→      };
   424→
   425→      this.auditLog.push(audit);
   426→
   427→      return {
   428→        executionId: execId,
   429→        toolId,
   430→        success: false,
   431→        output: null,
   432→        error: errorMessage,
   433→        evidence,
   434→        audit,
   435→        durationMs,
   436→        creditsConsumed: 0,
   437→      };
   438→    }
   439→  }
   440→
   441→  /** Get the audit log */
   442→  getAuditLog(): ToolAuditRecord[] {
   443→    return [...this.auditLog];
   444→  }
   445→
   446→  /** Get audit records for a specific execution */
   447→  getAuditByExecution(executionId: string): ToolAuditRecord[] {
   448→    return this.auditLog.filter((a) => a.executionId === executionId);
   449→  }
   450→
   451→  /** Clear old audit records (keep last N) */
   452→  pruneAuditLog(keepLast: number): void {
   453→    if (this.auditLog.length > keepLast) {
   454→      this.auditLog = this.auditLog.slice(-keepLast);
   455→    }
   456→  }
   457→
   458→  // ── Private helpers ───────────────────────────────────────────────────
   459→
   460→  private async executeWithTimeout(
   461→    executor: ToolExecutorFn,
   462→    inputs: Record<string, unknown>,
   463→    timeoutMs: number,
   464→  ): Promise<unknown> {
   465→    return Promise.race([
   466→      executor(inputs),
   467→      new Promise<never>((_, reject) =>
   468→        setTimeout(() => reject(new Error(`Tool execution timed out after ${timeoutMs}ms`)), timeoutMs)
   469→      ),
   470→    ]);
   471→  }
   472→
   473→  private createFailedResult(
   474→    executionId: string,
   475→    toolId: string,
   476→    durationMs: number,
   477→    creditsConsumed: number,
   478→    error: string,
   479→    initiatedBy: string,
   480→  ): ToolExecutionResult {
   481→    const now = new Date().toISOString();
   482→    const audit: ToolAuditRecord = {
   483→      auditId: randomUUID(),
   484→      toolId,
   485→      executionId,
   486→      initiatedBy,
   487→      startedAt: now,
   488→      endedAt: now,
   489→      durationMs,
   490→      success: false,
   491→      error,
   492→      sideEffectsObserved: [],
   493→      evidenceIds: [],
   494→      creditsConsumed,
   495→      inputsRedacted: false,
   496→      outputsRedacted: true,
   497→    };
   498→
   499→    this.auditLog.push(audit);
   500→
   501→    return {
   502→      executionId,
   503→      toolId,
   504→      success: false,
   505→      output: null,
   506→      error,
   507→      evidence: [],
   508→      audit,
   509→      durationMs,
   510→      creditsConsumed,
   511→    };
   512→  }
   513→}

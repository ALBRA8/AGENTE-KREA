   206→// ─── VisualQA Class ────────────────────────────────────────────────────
   207→
   208→/**
   209→ * VisualQA — inspects PDF/layout results for visual quality issues.
   210→ *
   211→ * Three main operations:
   212→ * 1. inspect() — detect all visual quality issues (AC-043)
   213→ * 2. inspectPDF() — inspect an actual PDF file for rendering-level issues (FASE 10)
   214→ * 3. diagnose() — determine root cause for each issue
   215→ * 4. repair() — attempt to fix an issue
   216→ *
   217→ * AC-043: Never trust the renderer output blindly.
   218→ * AC-044: Every issue has severity, location, description, suggested_fix.
   219→ * AC-045: Repair loop with max 5 iterations.
   220→ */
   221→export class VisualQA {
   222→  /**
   223→   * Inspect a PDF/layout result for visual quality issues.
   224→   *
   225→   * AC-043: This inspection is MANDATORY. The renderer finishing does NOT
   226→   * mean the PDF is done — we must actively verify visual quality.
   227→   *
   228→   * FASE 10: This is a LAYOUT-LEVEL inspection only. It checks the layout
   229→   * JSON, not the actual rendered PDF. For PDF-level QA, use inspectPDF().
   230→   *
   231→   * @param pdfSpec - The PDF specification and layout to inspect
   232→   * @param content - The source content that was rendered
   233→   * @param editorial - The editorial specification used
   234→   * @returns QAResult with all detected issues
   235→   */
   236→  inspect(
   237→    pdfSpec: PDFSpecForQA,
   238→    content: BookContent,
   239→    editorial: EditorialSpecification
   240→  ): QAResult {
   241→    const startTime = Date.now();
   242→    const issues: QAIssue[] = [];
   243→
   244→    // Run all inspection checks
   245→    issues.push(...this.checkCutText(pdfSpec, content, editorial));
   246→    issues.push(...this.checkOverflow(pdfSpec, editorial));
   247→    issues.push(...this.checkOffPage(pdfSpec));
   248→    issues.push(...this.checkEmptyPages(pdfSpec));
   249→    issues.push(...this.checkBadHierarchy(pdfSpec, content));
   250→    issues.push(...this.checkContrastProblems(pdfSpec, editorial));
   251→    issues.push(...this.checkTypographicInconsistency(pdfSpec, editorial));
   252→    issues.push(...this.checkNumberingErrors(pdfSpec, content));
   253→    issues.push(...this.checkLegibilityProblems(pdfSpec, editorial));
   254→
   255→    // Compute summaries
   256→    const byCategory: Record<QAIssueCategory, number> = {
   257→      cut_text: 0, overflow: 0, off_page: 0, distorted_images: 0,
   258→      empty_pages: 0, bad_hierarchy: 0, contrast_problems: 0,
   259→      typographic_inconsistency: 0, cover_errors: 0, numbering_errors: 0,
   260→      index_errors: 0, legibility_problems: 0,
   261→    };
   262→
   263→    const bySeverity: Record<QAIssueSeverity, number> = {
   264→      critical: 0, warning: 0, info: 0,
   265→    };
   266→
   267→    for (const issue of issues) {
   268→      byCategory[issue.category]++;
   269→      bySeverity[issue.severity]++;
   270→    }
   271→
   272→    const duration = Date.now() - startTime;
   273→
   274→    return {
   275→      passed: bySeverity.critical === 0,
   276→      issue_count: issues.length,
   277→      issues,
   278→      by_category: byCategory,
   279→      by_severity: bySeverity,
   280→      inspectedAt: new Date().toISOString(),
   281→      inspection_duration_ms: duration,
   282→      layout_level_only: true,
   283→      qa_scope_note: "Layout-level QA only. PDF-level QA requires renderToPDF + inspectPDF.",
   284→    };
   285→  }
   286→
   287→  /**
   288→   * Inspect an actual PDF file for rendering-level issues.
   289→   *
   290→   * FASE 10: This method checks the real PDF file on disk, not just the
   291→   * layout JSON. It can detect rendering defects that layout-level checks
   292→   * cannot.
   293→   *
   294→   * @param filePath - Absolute path to the PDF file
   295→   * @param expectedPageCount - Expected number of pages
   296→   * @returns PDFInspectionResult with check results
   297→   */
   298→  inspectPDF(
   299→    filePath: string,
   300→    expectedPageCount?: number
   301→  ): PDFInspectionResult {
   302→    const result: PDFInspectionResult = {
   303→      inspected: false,
   304→      issues: [],
   305→      checks: {
   306→        file_exists: "NOT_VERIFIED",
   307→        valid_pdf: "NOT_VERIFIED",
   308→        page_count_matches: "NOT_VERIFIED",
   309→        text_extractable: "NOT_VERIFIED",
   310→        no_rendering_defects: "NOT_VERIFIED",
   311→      },
   312→    };
   313→
   314→    // ── Check: File exists ──────────────────────────────────────────
   315→    if (!existsSync(filePath)) {
   316→      result.not_inspected_reason = `No PDF file to inspect: ${filePath} does not exist`;
   317→      result.checks.file_exists = "FAIL";
   318→      result.checks.valid_pdf = "NOT_VERIFIED";
   319→      result.checks.page_count_matches = "NOT_VERIFIED";
   320→      result.checks.text_extractable = "NOT_VERIFIED";
   321→      result.checks.no_rendering_defects = "NOT_VERIFIED";
   322→
   323→      // Add an issue for the missing file
   324→      result.issues.push({
   325→        id: "qa-pdf-missing-file",
   326→        category: "empty_pages",
   327→        severity: "critical",
   328→        location: { page: 0 },
   329→        description: `PDF file does not exist at expected path: ${filePath}`,
   330→        suggested_fix: "Run renderToPDF() before inspectPDF()",
   331→        auto_repairable: false,
   332→        status: "FAIL",
   333→      });
   334→      return result;
   335→    }
   336→
   337→    result.checks.file_exists = "PASS";
   338→    result.inspected = true;
   339→
   340→    // ── Check: Valid PDF (read header) ──────────────────────────────
   341→    let fileContent: Buffer;
   342→    try {
   343→      const stat = statSync(filePath);
   344→      result.sizeBytes = stat.size;
   345→
   346→      if (stat.size === 0) {
   347→        result.checks.valid_pdf = "FAIL";
   348→        result.issues.push({
   349→          id: "qa-pdf-empty-file",
   350→          category: "empty_pages",
   351→          severity: "critical",
   352→          location: { page: 0 },
   353→          description: `PDF file is empty (0 bytes): ${filePath}`,
   354→          suggested_fix: "Re-render the PDF — the renderer produced an empty file",
   355→          auto_repairable: false,
   356→          status: "FAIL",
   357→        });
   358→        return result;
   359→      }
   360→
   361→      fileContent = readFileSync(filePath);
   362→
   363→      // Check %PDF header
   364→      const header = fileContent.subarray(0, 5).toString("utf-8");
   365→      if (header.startsWith("%PDF")) {
   366→        result.checks.valid_pdf = "PASS";
   367→        result.validHeader = true;
   368→      } else {
   369→        result.checks.valid_pdf = "FAIL";
   370→        result.validHeader = false;
   371→        result.issues.push({
   372→          id: "qa-pdf-invalid-header",
   373→          category: "empty_pages",
   374→          severity: "critical",
   375→          location: { page: 0 },
   376→          description: `File does not have a valid PDF header: expected %PDF, got "${header}"`,
   377→          suggested_fix: "Re-render the PDF — the output may be corrupted or not a PDF",
   378→          auto_repairable: false,
   379→          status: "FAIL",
   380→        });
   381→        return result;
   382→      }
   383→    } catch (err) {
   384→      result.checks.valid_pdf = "NOT_VERIFIED";
   385→      result.not_inspected_reason = `Cannot read PDF file: ${err instanceof Error ? err.message : String(err)}`;
   386→      return result;
   387→    }
   388→
   389→    // ── Check: Page count ───────────────────────────────────────────
   390→    // Quick page count by scanning for /Type /Page (not /Pages) objects
   391→    const contentStr = fileContent.toString("binary");
   392→    const pageTypeCount = contentStr.split("/Type /Page").length - 1;
   393→    const pagesTreeCount = contentStr.split("/Type /Pages").length - 1;
   394→    const pageCount = Math.max(1, pageTypeCount - pagesTreeCount);
   395→    result.pageCount = pageCount;
   396→
   397→    if (expectedPageCount !== undefined) {
   398→      if (pageCount === expectedPageCount) {
   399→        result.checks.page_count_matches = "PASS";
   400→      } else {
   401→        result.checks.page_count_matches = "FAIL";
   402→        result.issues.push({
   403→          id: "qa-pdf-page-count-mismatch",
   404→          category: "numbering_errors",
   405→          severity: "warning",
   406→          location: { page: 0 },
   407→          description: `PDF has ${pageCount} pages, expected ${expectedPageCount}`,
   408→          suggested_fix: "Check page break algorithm or re-render with correct spec",
   409→          auto_repairable: false,
   410→          status: "FAIL",
   411→        });
   412→      }
   413→    } else {
   414→      result.checks.page_count_matches = "NOT_VERIFIED";
   415→      result.issues.push({
   416→        id: "qa-pdf-page-count-unknown",
   417→        category: "numbering_errors",
   418→        severity: "info",
   419→        location: { page: 0 },
   420→        description: `PDF has ${pageCount} pages but no expected count was provided for comparison`,
   421→        suggested_fix: "Provide expectedPageCount to inspectPDF() for verification",
   422→        auto_repairable: false,
   423→        status: "NOT_VERIFIED",
   424→        not_verified_reason: "No expectedPageCount provided",
   425→      });
   426→    }
   427→
   428→    // ── Check: Text extractability ──────────────────────────────────
   429→    // Quick check: look for text stream objects in the PDF
   430→    const textStreamCount = contentStr.split("/Subtype /Form").length - 1;
   431→    const hasTextStreams = contentStr.includes("BT\n") || contentStr.includes("BT ");
   432→    if (hasTextStreams || textStreamCount > 0) {
   433→      result.checks.text_extractable = "PASS";
   434→    } else {
   435→      // Could be a scanned/image-only PDF, or we just can't detect text
   436→      result.checks.text_extractable = "NOT_VERIFIED";
   437→      result.issues.push({
   438→        id: "qa-pdf-text-not-detected",
   439→        category: "legibility_problems",
   440→        severity: "warning",
   441→        location: { page: 0 },
   442→        description: "Could not detect text streams in PDF — file may be image-only or use non-standard text encoding",
   443→        suggested_fix: "Use a PDF text extraction tool (pdftotext) to verify text content",
   444→        auto_repairable: false,
   445→        status: "NOT_VERIFIED",
   446→        not_verified_reason: "No PDF text extraction library available",
   447→      });
   448→    }
   449→
   450→    // ── Check: No rendering defects ─────────────────────────────────
   451→    // Without a visual renderer, we can only check for structural issues
   452→    // Check for common PDF error markers
   453→    const hasXref = contentStr.includes("xref") || contentStr.includes("/XRef");
   454→    const hasTrailer = contentStr.includes("trailer") || contentStr.includes("/Root");
   455→    if (hasXref && hasTrailer) {
   456→      result.checks.no_rendering_defects = "NOT_VERIFIED";
   457→      result.issues.push({
   458→        id: "qa-pdf-render-not-verified",
   459→        category: "distorted_images",
   460→        severity: "info",
   461→        location: { page: 0 },
   462→        description: "PDF structure appears valid but rendering defects cannot be detected without visual inspection",
   463→        suggested_fix: "Open the PDF in a viewer and visually verify rendering quality",
   464→        auto_repairable: false,
   465→        status: "NOT_VERIFIED",
   466→        not_verified_reason: "No visual PDF renderer available for defect detection",
   467→      });
   468→    } else {
   469→      result.checks.no_rendering_defects = "FAIL";
   470→      result.issues.push({
   471→        id: "qa-pdf-corrupt-structure",
   472→        category: "empty_pages",
   473→        severity: "critical",
   474→        location: { page: 0 },
   475→        description: `PDF is missing critical structure: ${!hasXref ? "no xref table" : ""} ${!hasTrailer ? "no trailer/root" : ""}`,
   476→        suggested_fix: "Re-render the PDF — the file may be truncated or corrupted",
   477→        auto_repairable: false,
   478→        status: "FAIL",
   479→      });
   480→    }
   481→
   482→    return result;
   483→  }
   484→
   485→  /**
   486→   * Diagnose the root cause of QA issues.
   487→   *
   488→   * @param issues - The issues to diagnose
   489→   * @returns Array of diagnoses with repair strategies
   490→   */
   491→  diagnose(issues: QAIssue[]): Diagnosis[] {
   492→    return issues.map((issue) => this.diagnoseSingle(issue));
   493→  }
   494→
   495→  /**
   496→   * Attempt to repair a single QA issue.
   497→   *
   498→   * @param issue - The issue to repair
   499→   * @param content - Current book content
   500→   * @param editorial - Current editorial specification
   501→   * @returns RepairedContent with the repair result
   502→   */
   503→  repair(
   504→    issue: QAIssue,
   505→    content: BookContent,
   506→    editorial: EditorialSpecification
   507→  ): RepairedContent {
   508→    if (!issue.auto_repairable) {
   509→      return {
   510→        repaired: false,
   511→        issue_id: issue.id,
   512→        repair_description: "Issue is not auto-repairable — requires manual intervention",
   513→        remaining_issues: [issue.id],
   514→      };
   515→    }
   516→
   517→    switch (issue.category) {
   518→      case "cut_text":
   519→        return this.repairCutText(issue, content, editorial);
   520→      case "overflow":
   521→        return this.repairOverflow(issue, content, editorial);
   522→      case "empty_pages":
   523→        return this.repairEmptyPages(issue, content, editorial);
   524→      case "bad_hierarchy":
   525→        return this.repairBadHierarchy(issue, content, editorial);
   526→      case "numbering_errors":
   527→        return this.repairNumberingErrors(issue, content, editorial);
   528→      default:
   529→        return {
   530→          repaired: false,
   531→          issue_id: issue.id,
   532→          repair_description: `Auto-repair not implemented for category: ${issue.category}`,
   533→          remaining_issues: [issue.id],
   534→        };
   535→    }
   536→  }

   538→  // ─── Inspection Checks ─────────────────────────────────────────────
   539→
   540→  /**
   541→   * Check for cut/truncated text.
   542→   */
   543→  private checkCutText(
   544→    pdfSpec: PDFSpecForQA,
   545→    content: BookContent,
   546→    editorial: EditorialSpecification
   547→  ): QAIssue[] {
   548→    const issues: QAIssue[] = [];
   549→
   550→    for (const page of pdfSpec.layout.pages) {
   551→      for (const element of page.elements) {
   552→        if (element.type === "paragraph" || element.type === "heading") {
   553→          // Check if element height was constrained by remaining space
   554→          const contentRef = element.content_ref;
   555→          const sourceContent = this.findSourceContent(content, contentRef);
   556→
   557→          if (sourceContent && element.height < this.estimateMinHeight(sourceContent, editorial)) {
   558→            issues.push({
   559→              id: `qa-cut-${page.page_number}-${element.content_ref}`,
   560→              category: "cut_text",
   561→              severity: "critical",
   562→              location: {
   563→                page: page.page_number,
   564→                element_ref: element.content_ref,

  1003→  /**
  1004→   * Repair bad heading hierarchy.
  1005→   *
  1006→   * FASE 10 fix: This now applies a REAL repair — it inserts intermediate
  1007→   * heading levels to fill gaps in the hierarchy.
  1008→   *
  1009→   * Example: If hierarchy goes H1 → H3, we insert an H2 section with
  1010→   * a generated heading to bridge the gap.
  1011→   */
  1012→  private repairBadHierarchy(
  1013→    issue: QAIssue,
  1014→    content: BookContent,
  1015→    editorial: EditorialSpecification
  1016→  ): RepairedContent {
  1017→    // Parse the chapter and level from the issue ID
  1018→    // Format: qa-hierarchy-{chapterNumber}-{level}
  1019→    const match = issue.id.match(/qa-hierarchy-(\d+)-(\d+)/);
  1020→    if (!match) {
  1021→      return {
  1022→        repaired: false,
  1023→        issue_id: issue.id,
  1024→        repair_description: "Cannot parse hierarchy issue — unable to determine which chapter and level to repair",
  1025→        remaining_issues: [issue.id],
  1026→      };
  1027→    }
  1028→
  1029→    const chapterNum = parseInt(match[1], 10);
  1030→    const skippedLevel = parseInt(match[2], 10);
  1031→
  1032→    // Find the chapter with the hierarchy issue
  1033→    const updatedContent: BookContent = {
  1034→      ...content,
  1035→      chapters: content.chapters.map((ch) => {
  1036→        if (ch.chapter_number !== chapterNum) return ch;
  1037→
  1038→        // Insert intermediate heading levels to fill the gap
  1039→        // We walk through sections and when we find a skip, we insert
  1040→        // a bridging section at the missing level
  1041→        const repairedSections: typeof ch.sections = [];
  1042→        let lastLevel = 0;
  1043→
  1044→        for (const section of ch.sections) {
  1045→          if (section.heading && section.level > lastLevel + 1 && lastLevel > 0) {
  1046→            // Insert intermediate sections for each skipped level
  1047→            for (let level = lastLevel + 1; level < section.level; level++) {
  1048→              repairedSections.push({
  1049→                heading: `Section ${level}`,
  1050→                level,
  1051→                paragraphs: [],
  1052→              });
  1053→            }
  1054→          }
  1055→          repairedSections.push(section);
  1056→          if (section.heading) {
  1057→            lastLevel = section.level;
  1058→          }
  1059→        }
  1060→
  1061→        return {
  1062→          ...ch,
  1063→          sections: repairedSections,
  1064→        };
  1065→      }),
  1066→    };
  1067→
  1068→    return {
  1069→      repaired: true,
  1070→      issue_id: issue.id,
  1071→      repair_description: `Inserted intermediate heading levels to bridge hierarchy gap in chapter ${chapterNum} (up to level ${skippedLevel})`,
  1072→      updated_content: updatedContent,
  1073→      remaining_issues: [],
  1074→    };
  1075→  }

  1114→      if (chapter) {

  1130→  private estimateMinHeight(text: string, editorial: EditorialSpecification): number {
  1131→    const charsPerLine = 60; // conservative estimate

  1136→  /**
  1137→   * Estimate WCAG contrast ratio (simplified).
  1138→   */
  1139→  private estimateContrast(foreground: string, background: string): number {
  1140→    const fgLum = this.relativeLuminance(foreground);
  1141→    const bgLum = this.relativeLuminance(background);
  1142→    const lighter = Math.max(fgLum, bgLum);
  1143→    const darker = Math.min(fgLum, bgLum);
  1144→    return (lighter + 0.05) / (darker + 0.05);
  1145→  }
  1146→
  1147→  private relativeLuminance(hex: string): number {
  1148→    const r = this.sRGBtoLinear(parseInt(hex.slice(1, 3), 16) / 255);
  1149→    const g = this.sRGBtoLinear(parseInt(hex.slice(3, 5), 16) / 255);

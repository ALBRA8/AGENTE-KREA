     1→/**
     2→ * Product Factory — KREA Book Factory Layer
     3→ *
     4→ * General orchestration layer that routes product production based on type
     5→ * and complexity. Content products (eBook, guide, template) are routed to
     6→ * BookFactory; software products generate HandoffContracts for Codex/
     7→ * Antigravity; hybrid products combine both.
     8→ *
     9→ * Satisfies:
    10→ *   AC-027 — Complexity determines routing: category alone isn't enough
    11→ *   AC-028 — Pipeline: Opportunity → Fit → Architecture → Specification →
    12→ *            Production → QA → Final Product
    13→ *
    14→ * The Product Factory is the top-level entry point for all product creation.
    15→ */
    16→

   197→// ─── ProductFactory Class ──────────────────────────────────────────────
   198→
   199→/**
   200→ * ProductFactory — general orchestration for all product types.
   201→ *
   202→ * Pipeline (AC-028):
   203→ *   Opportunity → Fit → Architecture → Specification → Production → QA → Final Product
   204→ *
   205→ * Routing (AC-027):
   206→ *   - Content products (ebook, guide, template, report) → BookFactory
   207→ *   - Software products → HandoffContract for Codex/Antigravity
   208→ *   - Hybrid → combines both
   209→ *
   210→ * Complexity determines routing: category alone isn't enough (AC-027).
   211→ */
   212→export class ProductFactory {
   213→  private bookFactory = new BookFactory();
   214→  private memory = new MemoryManager();
   215→  private tracer = new ExecutionTracer();
   216→
   217→  /**
   218→   * Produce a product from an opportunity.
   219→   *
   220→   * @param opportunity - The product opportunity
   221→   * @param fitResult - Pre-computed fit result (or computed internally)
   222→   * @param decision - Architecture decision (optional)
   223→   * @returns ProductResult with the final product and evidence
   224→   */
   225→  async produce(
   226→    opportunity: ProductOpportunity,
   227→    fitResult?: FitResult,
   228→    decision?: ArchitectureDecision
   229→  ): Promise<ProductResult> {
   230→    const startTime = Date.now();
   231→    const evidence: ProductEvidence[] = [];
   232→    const errors: string[] = [];
   233→
   234→    // Start execution trace
   235→    const executionId = await this.tracer.start("product-factory-produce", {
   236→      agentId: "krea",
   237→      userId: opportunity.userId,
   238→      inputs: { topic: opportunity.topic, productType: opportunity.productType },
   239→    });
   240→
   241→    try {
   242→      // ─── Step 1: Fit ──────────────────────────────────────────────
   243→      const fit = fitResult ?? this.computeFit(opportunity);
   244→      evidence.push({
   245→        step: "fit",
   246→        timestamp: new Date().toISOString(),
   247→        description: `Routed to ${fit.production_path} (complexity: ${fit.complexity_score})`,
   248→        data: { fit },
   249→      });
   250→
   251→      // ─── Step 2: Architecture ─────────────────────────────────────
   252→      const arch = decision ?? this.computeArchitecture(opportunity, fit);
   253→      evidence.push({
   254→        step: "architecture",
   255→        timestamp: new Date().toISOString(),
   256→        description: `Architecture: ${arch.type} with ${arch.components.length} components`,
   257→        data: { architecture: arch },
   258→      });
   259→
   260→      // ─── Step 3-7: Production (routed by type) ────────────────────
   261→      let result: ProductResult;
   262→
   263→      switch (fit.category) {
   264→        case "content":
   265→          result = await this.produceContentProduct(opportunity, fit, arch, evidence, executionId);
   266→          break;
   267→
   268→        case "software":
   269→          result = this.produceSoftwareProduct(opportunity, fit, arch, evidence);
   270→          break;
   271→
   272→        case "hybrid":
   273→          result = await this.produceHybridProduct(opportunity, fit, arch, evidence, executionId);
   274→          break;
   275→
   276→        default:
   277→          throw new Error(`Unknown product category: ${fit.category}`);
   278→      }
   279→
   280→      await this.tracer.succeed(executionId, {
   281→        productType: opportunity.productType,
   282→        success: result.success,
   283→      });
   284→
   285→      return result;
   286→
   287→    } catch (error) {
   288→      const message = error instanceof Error ? error.message : String(error);
   289→      errors.push(message);
   290→      await this.tracer.addError(executionId, message);
   291→      await this.tracer.fail(executionId, { message });
   292→
   293→      return {
   294→        success: false,
   295→        product_type: opportunity.productType,
   296→        artifact_path: null,
   297→        dossier_id: randomUUID(),
   298→        evidence,
   299→        qa_result: { passed: false, issues_count: 1, details: message },
   300→        production_time_ms: Date.now() - startTime,
   301→        errors,
   302→      };
   303→    }
   304→  }
   305→
   306→  /**
   307→   * Compute the fit result for an opportunity.
   308→   *
   309→   * AC-027: Complexity determines routing, not just category.
   310→   */
   311→  computeFit(opportunity: ProductOpportunity): FitResult {
   312→    const category = this.inferCategory(opportunity.productType);
   313→
   314→    // Compute complexity score (0-1)
   315→    const complexityScore = this.computeComplexityScore(opportunity);
   316→
   317→    // Route based on category + complexity (AC-027)
   318→    let productionPath: FitResult["production_path"];
   319→    let routingReason: string;
   320→
   321→    if (category === "content") {
   322→      if (complexityScore <= 0.3) {
   323→        productionPath = "book-factory-streamlined";
   324→        routingReason = "Simple content product — streamlined pipeline sufficient";
   325→      } else {
   326→        productionPath = "book-factory-full";
   327→        routingReason = "Complex content product — full BookFactory pipeline required";
   328→      }
   329→    } else if (category === "software") {
   330→      productionPath = "handoff-contract";
   331→      routingReason = "Software product — generate HandoffContract for implementation agent";
   332→    } else {
   333→      productionPath = "hybrid";
   334→      routingReason = "Hybrid product — combine BookFactory + HandoffContract";
   335→    }
   336→
   337→    const requiredCapabilities = this.getRequiredCapabilities(productionPath);
   338→    const estimatedTime = this.estimateProductionTime(productionPath, opportunity);
   339→
   340→    return {
   341→      fits: true,
   342→      category,
   343→      production_path: productionPath,
   344→      routing_reason: routingReason,
   345→      complexity_score: complexityScore,
   346→      required_capabilities: requiredCapabilities,
   347→      estimated_time_seconds: estimatedTime,
   348→    };
   349→  }
   350→
   351→  // ─── Private Methods ───────────────────────────────────────────────
   352→
   353→  /**
   354→   * Infer the product category from the product type.
   355→   */
   356→  private inferCategory(productType: ProductType): ProductCategory {
   357→    const contentTypes: ProductType[] = ["ebook", "guide", "template", "report"];
   358→    if (contentTypes.includes(productType)) return "content";
   359→    if (productType === "software") return "software";
   360→    return "hybrid";
   361→  }
   362→
   363→  /**
   364→   * Compute complexity score (0-1) from opportunity attributes.
   365→   *
   366→   * AC-027: Complexity is multi-dimensional, not just category.
   367→   */
   368→  private computeComplexityScore(opportunity: ProductOpportunity): number {
   369→    let score = 0;
   370→
   371→    // Section count contributes to complexity
   372→    score += Math.min(0.3, opportunity.sectionCount / 20);
   373→
   374→    // Explicit complexity level
   375→    const complexityMap: Record<string, number> = {
   376→      simple: 0.1,
   377→      moderate: 0.3,
   378→      complex: 0.5,
   379→    };
   380→    score += complexityMap[opportunity.complexity] ?? 0.3;
   381→
   382→    // Format complexity
   383→    if (opportunity.format === "report") score += 0.1;
   384→    if (opportunity.format === "hybrid") score += 0.2;
   385→
   386→    // Genre complexity (some genres need more visual richness)
   387→    const complexGenres = ["fiction", "lifestyle", "creative"];
   388→    if (complexGenres.includes(opportunity.genre.toLowerCase())) score += 0.1;
   389→
   390→    return Math.min(1, score);
   391→  }

   393→  /**
   394→   * Compute architecture for the product.
   395→   */
   396→  private computeArchitecture(
   397→    opportunity: ProductOpportunity,
   398→    fit: FitResult
   399→  ): ArchitectureDecision {
   400→    const contentComponents: ArchitectureComponent[] = [
   401→      { name: "content-gen", type: "content", dependencies: [] },
   402→      { name: "art-direction", type: "design", dependencies: ["content-gen"] },
   403→      { name: "editorial-design", type: "design", dependencies: ["art-direction"] },
   404→      { name: "layout", type: "layout", dependencies: ["editorial-design"] },
   405→      { name: "render", type: "render", dependencies: ["layout"] },
   406→      { name: "qa", type: "qa", dependencies: ["render"] },
   407→    ];
   408→
   409→    if (fit.category === "software" || fit.category === "hybrid") {
   410→      contentComponents.push({
   411→        name: "software-impl",
   412→        type: "software",
   413→        dependencies: ["content-gen"],
   414→      });
   415→    }
   416→
   417→    return {
   418→      type: fit.complexity_score > 0.6 ? "modular" : "monolithic",
   419→      components: contentComponents,
   420→      quality_attributes: [
   421→        "correctness",
   422→        "completeness",
   423→        "visual_quality",
   424→        "accessibility",
   425→      ],
   426→      constraints: [
   427→        "No literal content copying (AC-032)",
   428→        "Deterministic layout (AC-039)",
   429→        "Visual system coherence (AC-034)",
   430→        "Max 5 repair iterations (AC-045)",
   431→      ],
   432→    };
   433→  }

   435→  /**
   436→   * Produce a content product via BookFactory.
   437→   */
   438→  private async produceContentProduct(
   439→    opportunity: ProductOpportunity,
   440→    fit: FitResult,
   441→    _arch: ArchitectureDecision,
   442→    evidence: ProductEvidence[],
   443→    executionId: string
   444→  ): Promise<ProductResult> {
   445→    const startTime = Date.now();
   446→
   447→    // Map ProductOpportunity → BookOpportunity
   448→    const bookOpportunity: BookOpportunity = {
   449→      topic: opportunity.topic,
   450→      targetAudience: opportunity.targetAudience,
   451→      genre: opportunity.genre,
   452→      chapterCount: opportunity.sectionCount,
   453→      author: opportunity.author,
   454→      purpose: opportunity.purpose,
   455→      tone: opportunity.tone,
   456→      userId: opportunity.userId,
   457→    };
   458→
   459→    const bookArchitecture: BookArchitecture = {
   460→      format: opportunity.format as BookArchitecture["format"],
   461→      complexity: opportunity.complexity,
   462→      targetMarket: opportunity.targetMarket,
   463→    };
   464→
   465→    evidence.push({
   466→      step: "specification",
   467→      timestamp: new Date().toISOString(),
   468→      description: "Content product specification mapped to BookFactory input",
   469→      data: { productionPath: fit.production_path },
   470→    });
   471→
   472→    // Route to BookFactory
   473→    const bookProduct = await this.bookFactory.produce(bookOpportunity, bookArchitecture);
   474→
   475→    evidence.push({
   476→      step: "production",
   477→      timestamp: new Date().toISOString(),
   478→      description: `BookFactory produced ${bookProduct.content.chapters.length} chapters, ${bookProduct.pdf.pageCount} pages`,
   479→      data: {
   480→        pageCount: bookProduct.pdf.pageCount,
   481→        wordCount: bookProduct.content.total_word_count,
   482→        qaPassed: bookProduct.qa_result.passed,
   483→        pdfVerified: bookProduct.pdf_verification.overall,
   484→      },
   485→    });
   486→
   487→    evidence.push({
   488→      step: "qa",
   489→      timestamp: new Date().toISOString(),
   490→      description: bookProduct.qa_result.passed
   491→        ? "QA passed — product is verified"
   492→        : `QA found ${bookProduct.qa_result.issue_count} issues`,
   493→      data: {
   494→        passed: bookProduct.qa_result.passed,
   495→        issueCount: bookProduct.qa_result.issue_count,
   496→        criticalCount: bookProduct.qa_result.by_severity.critical,
   497→      },
   498→    });
   499→
   500→    await this.tracer.addSkill(executionId, "book-factory");
   501→
   502→    // Store product dossier in memory
   503→    const dossierId = randomUUID();
   504→    await this.storeDossier(opportunity, bookProduct, dossierId);
   505→
   506→    return {
   507→      success: bookProduct.success,
   508→      product_type: opportunity.productType,
   509→      artifact_path: bookProduct.pdf.filePath,
   510→      dossier_id: dossierId,
   511→      evidence,
   512→      qa_result: {
   513→        passed: bookProduct.qa_result.passed && bookProduct.pdf_verification.overall,
   514→        issues_count: bookProduct.qa_result.issue_count,
   515→        details: bookProduct.qa_result.passed
   516→          ? "All QA checks passed"
   517→          : `${bookProduct.qa_result.by_severity.critical} critical, ${bookProduct.qa_result.by_severity.warning} warnings`,
   518→      },
   519→      book_product: bookProduct,
   520→      production_time_ms: Date.now() - startTime,
   521→      errors: bookProduct.errors,
   522→    };
   523→  }

   525→  /**
   526→   * Produce a software product via HandoffContract.
   527→   */
   528→  private produceSoftwareProduct(
   529→    opportunity: ProductOpportunity,
   530→    fit: FitResult,
   531→    arch: ArchitectureDecision,
   532→    evidence: ProductEvidence[]
   533→  ): ProductResult {
   534→    const contract: HandoffContract = {
   535→      id: randomUUID(),
   536→      opportunity_id: opportunity.id,
   537→      target_agent: opportunity.complexity === "complex" ? "codex" : "antigravity",
   538→      specification: {
   539→        name: opportunity.topic,
   540→        description: `${opportunity.purpose} — ${opportunity.targetAudience}`,
   541→        features: this.inferFeatures(opportunity),
   542→        tech_stack: this.inferTechStack(opportunity),
   543→        architecture: arch,
   544→        complexity: opportunity.complexity,
   545→      },
   546→      acceptance_criteria: this.inferAcceptanceCriteria(opportunity),
   547→      constraints: arch.constraints,
   548→      createdAt: new Date().toISOString(),
   549→    };
   550→
   551→    evidence.push({
   552→      step: "specification",
   553→      timestamp: new Date().toISOString(),
   554→      description: `HandoffContract generated for ${contract.target_agent}`,
   555→      data: { contractId: contract.id, targetAgent: contract.target_agent },
   556→    });
   557→
   558→    evidence.push({
   559→      step: "production",
   560→      timestamp: new Date().toISOString(),
   561→      description: "Software specification ready for handoff — no rendering required",
   562→      data: { features: contract.specification.features.length },
   563→    });
   564→
   565→    evidence.push({
   566→      step: "qa",
   567→      timestamp: new Date().toISOString(),
   568→      description: "Software products are QA'd by implementation agent",
   569→      data: { passed: true },
   570→    });
   571→
   572→    return {
   573→      success: true,
   574→      product_type: opportunity.productType,
   575→      artifact_path: null,
   576→      dossier_id: randomUUID(),
   577→      evidence,
   578→      qa_result: {
   579→        passed: true,
   580→        issues_count: 0,
   581→        details: "HandoffContract ready — QA delegated to implementation agent",
   582→      },
   583→      handoff_contract: contract,
   584→      production_time_ms: 0,
   585→      errors: [],
   586→    };
   587→  }

   589→  /**
   590→   * Produce a hybrid product (content + software).
   591→   */
   592→  private async produceHybridProduct(
   593→    opportunity: ProductOpportunity,
   594→    fit: FitResult,
   595→    arch: ArchitectureDecision,
   596→    evidence: ProductEvidence[],
   597→    executionId: string
   598→  ): Promise<ProductResult> {
   599→    const startTime = Date.now();
   600→
   601→    // Produce the content component
   602→    const contentResult = await this.produceContentProduct(
   603→      opportunity, fit, arch, evidence, executionId
   604→    );
   605→
   606→    // Produce the software component
   607→    const softwareResult = this.produceSoftwareProduct(
   608→      opportunity, fit, arch, evidence
   609→    );
   610→
   611→    const combinedSuccess = contentResult.success && softwareResult.success;
   612→
   613→    return {
   614→      success: combinedSuccess,
   615→      product_type: opportunity.productType,
   616→      artifact_path: contentResult.artifact_path,
   617→      dossier_id: randomUUID(),
   618→      evidence,
   619→      qa_result: {
   620→        passed: combinedSuccess,
   621→        issues_count: contentResult.qa_result.issues_count + softwareResult.qa_result.issues_count,
   622→        details: `Content: ${contentResult.qa_result.details} | Software: ${softwareResult.qa_result.details}`,
   623→      },
   624→      book_product: contentResult.book_product,
   625→      handoff_contract: softwareResult.handoff_contract,
   626→      production_time_ms: Date.now() - startTime,
   627→      errors: [...contentResult.errors, ...softwareResult.errors],
   628→    };
   629→  }

   631→  /**
   632→   * Get required capabilities for a production path.
   633→   */

   675→  /**
   676→   * Estimate production time in seconds.
   677→   */
   678→  private estimateProductionTime(
   679→    path: FitResult["production_path"],
   680→    opportunity: ProductOpportunity
   681→  ): number {
   682→    const baseTimes: Record<FitResult["production_path"], number> = {
   683→      "book-factory-full": 60,
   684→      "book-factory-streamlined": 30,
   685→      "handoff-contract": 5,
   686→      "hybrid": 65,
   687→    };
   688→
   689→    const base = baseTimes[path] ?? 60;
   690→    // Scale by section count (more sections = more time)
   691→    const scale = 1 + (opportunity.sectionCount / 10) * 0.5;
   692→    return Math.round(base * scale);
   693→  }

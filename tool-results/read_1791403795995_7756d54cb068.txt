    85→    85→function statusColor(status: string): string {
    86→    86→  const s = status.toUpperCase();
    87→    87→  if (["COMPLETED", "GO", "PUBLISHED", "ACKNOWLEDGED"].includes(s)) return "text-emerald-400";
    88→    88→  if (["FAILED", "NO_GO", "ARCHIVED"].includes(s)) return "text-red-400";
    89→    89→  if (["RUNNING", "PRODUCING", "SENT", "CONDITIONAL_GO"].includes(s)) return "text-amber-400";
    90→    90→  if (["PENDING", "CREATED", "IDEA", "READY"].includes(s)) return "text-blue-400";
    91→    91→  return "text-white/50";
    92→    92→}
    93→    93→
    94→    94→function statusBg(status: string): string {
    95→    95→  const s = status.toUpperCase();
    96→    96→  if (["COMPLETED", "GO", "PUBLISHED", "ACKNOWLEDGED"].includes(s)) return "bg-emerald-400/10 text-emerald-400 border-emerald-400/20";
    97→    97→  if (["FAILED", "NO_GO", "ARCHIVED"].includes(s)) return "bg-red-400/10 text-red-400 border-red-400/20";
    98→    98→  if (["RUNNING", "PRODUCING", "SENT", "CONDITIONAL_GO"].includes(s)) return "bg-amber-400/10 text-amber-400 border-amber-400/20";
    99→    99→  if (["PENDING", "CREATED", "IDEA", "READY"].includes(s)) return "bg-blue-400/10 text-blue-400 border-blue-400/20";
   100→   100→  return "bg-white/5 text-white/50 border-white/10";
   101→   101→}

   865→   865→function HandoffSection() {
   866→   866→  const [handoffs, setHandoffs] = useState<Handoff[]>([]);
   867→   867→  const [loading, setLoading] = useState(true);
   868→   868→  const [error, setError] = useState("");
   869→   869→  const [acting, setActing] = useState<string | null>(null);
   870→   870→
   871→   871→  const load = useCallback(async () => {
   872→   872→    setLoading(true);
   873→   873→    setError("");
   874→   874→    try {
   875→   875→      const token = getToken();
   876→   876→      const res = await fetch("/api/product/handoff", {
   877→   877→        headers: { Authorization: `Bearer ${token}` },
   878→   878→      });
   879→   879→      if (!res.ok) throw new Error("Error al cargar handoffs");
   880→   880→      const data = await res.json();
   881→   881→      setHandoffs(Array.isArray(data) ? data : data.handoffs || []);
   882→   882→    } catch (e: any) {
   883→   883→      setError(e.message || "Error de conexión");
   884→   884→    } finally {
   885→   885→      setLoading(false);
   886→   886→    }
   887→   887→  }, []);
   888→   888→
   889→   889→  // eslint-disable-next-line react-hooks/set-state-in-effect
   890→   890→  useEffect(() => { load(); }, [load]);
   891→   891→
   892→   892→  const performAction = async (handoffId: string, action: string) => {
   893→   893→    setActing(handoffId);
   894→   894→    try {
   895→   895→      const token = getToken();
   896→   896→      const res = await fetch(`/api/product/handoff/${handoffId}`, {
   897→   897→        method: "POST",
   898→   898→        headers: { "Content-Type": "application/json", Authorization: `Bearer ${token}` },
   899→   899→        body: JSON.stringify({ action }),
   900→   900→      });
   901→   901→      if (!res.ok) {
   902→   902→        const data = await res.json();
   903→   903→        throw new Error(data.error || `Error al ejecutar ${action}`);
   904→   904→      }
   905→   905→      load();
   906→   906→    } catch (e: any) {
   907→   907→      setError(e.message);
   908→   908→    } finally {
   909→   909→      setActing(null);
   910→   910→    }
   911→   911→  };
   912→   912→
   913→   913→  const getActions = (status: string): { action: string; label: string; icon: React.ComponentType<{ className?: string }> }[] => {
   914→   914→    const s = status.toUpperCase();
   915→   915→    if (s === "CREATED") return [{ action: "ready", label: "Marcar Listo", icon: CheckCircle2 }];
   916→   916→    if (s === "READY") return [{ action: "send", label: "Enviar", icon: Send }];

   918→   918→    if (s === "ACKNOWLEDGED") return [{ action: "complete", label: "Completar", icon: CheckCircle2 }];
   919→   919→    return [];
   920→   920→  };
   921→   921→
   922→   922→  return (
   923→   923→    <div>
   924→   924→      <SectionHeader
   925→   925→        icon={Send}
   926→   926→        title="Handoff"
   927→   927→        subtitle="Contratos de handoff para implementación externa de productos software"
   928→   928→        actions={
   929→   929→          <button
   930→   930→            onClick={load}
   931→   931→            className="p-2 rounded-lg bg-white/5 hover:bg-white/10 text-white/50 hover:text-white transition-colors"
   932→   932→          >
   933→   933→            <RefreshCw className="w-4 h-4" />
   934→   934→          </button>
   935→   935→        }
   936→   936→      />
   937→   937→
   938→   938→      {/* Status summary */}
   939→   939→      {!loading && !error && handoffs.length > 0 && (
   940→   940→        <div className="grid grid-cols-2 sm:grid-cols-4 gap-3 mb-6">
   941→   941→          {[
   942→   942→            { label: "Creados", count: handoffs.filter(h => h.status === "CREATED").length, icon: Circle, color: "text-blue-400" },
   943→   943→            { label: "Listos", count: handoffs.filter(h => h.status === "READY").length, icon: CheckCircle2, color: "text-amber-400" },
   944→   944→            { label: "Enviados", count: handoffs.filter(h => ["SENT", "ACKNOWLEDGED"].includes(h.status)).length, icon: Send, color: "text-purple-400" },
   945→   945→            { label: "Completados", count: handoffs.filter(h => h.status === "COMPLETED").length, icon: CheckCircle2, color: "text-emerald-400" },
   946→   946→          ].map((s) => (
   947→   947→            <div key={s.label} className="p-3 rounded-xl bg-[#0f1629] border border-white/[0.06] text-center">
   948→   948→              <s.icon className={`w-5 h-5 mx-auto mb-1 ${s.color}`} />
   949→   949→              <p className="text-xl font-extrabold">{s.count}</p>
   950→   950→              <p className="text-[10px] text-white/40">{s.label}</p>
   951→   951→            </div>
   952→   952→          ))}
   953→   953→        </div>
   954→   954→      )}
   955→   955→
   956→   956→      {error ? (

   962→   962→      ) : handoffs.length === 0 ? (
   963→   963→        <EmptyState
   964→   964→          icon={Send}
   965→   965→          title="Sin handoffs"
   966→   966→          subtitle="Los handoffs se crean para productos software que requieren implementación externa"
   967→   967→        />
   968→   968→      ) : (
   969→   969→        <div className="space-y-3">
   970→   970→          {handoffs.map((h, i) => {
   971→   971→            const actions = getActions(h.status);
   972→   972→            return (
   973→   973→              <motion.div
   974→   974→                key={h.id}
   975→   975→                initial={{ opacity: 0, y: 20 }}
   976→   976→                animate={{ opacity: 1, y: 0 }}
   977→   977→                transition={{ delay: i * 0.05 }}
   978→   978→                className="p-4 rounded-2xl bg-[#0f1629] border border-white/[0.06] hover:border-white/10 transition-all"
   979→   979→              >
   980→   980→                <div className="flex items-center justify-between mb-3">

   982→   982→                    <div className={`w-10 h-10 rounded-xl flex items-center justify-center ${
   983→   983→                      h.status === "COMPLETED" ? "bg-emerald-400/10" :
   984→   984→                      h.status === "FAILED" ? "bg-red-400/10" :
   985→   985→                      h.status === "SENT" || h.status === "ACKNOWLEDGED" ? "bg-purple-400/10" :
   986→   986→                      h.status === "READY" ? "bg-amber-400/10" :
   987→   987→                      "bg-blue-400/10"
   988→   988→                    }`}>
   989→   989→                      <Send className={`w-5 h-5 ${
   990→   990→                        h.status === "COMPLETED" ? "text-emerald-400" :
   991→   991→                        h.status === "FAILED" ? "text-red-400" :
   992→   992→                        h.status === "SENT" || h.status === "ACKNOWLEDGED" ? "text-purple-400" :
   993→   993→                        h.status === "READY" ? "text-amber-400" :

   995→   995→                      }`} />
   996→   996→                    </div>

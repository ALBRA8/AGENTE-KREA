   604→/* ════════════ Main App ════════════ */
   605→interface AppShellProps { onLogout: () => void; user?: User | null }
   606→
   607→const FALLBACK_USER: User = { id: "demo-fallback", name: "Usuario Demo", email: "demo@krea.ai", credits: 50, plan: "starter" };
   608→
   609→export default function AppPage({ onLogout, user: initialUser }: AppShellProps) {
   610→  const [user, setUser] = useState<User>(() => {
   611→    if (initialUser) return initialUser;
   612→    try {
   613→      const saved = localStorage.getItem("p360_user");
   614→      if (saved) {
   615→        const parsed = JSON.parse(saved);
   616→        if (parsed && parsed.id) return parsed;
   617→      }
   618→    } catch {}
   619→    // Always ensure a user exists — save fallback to localStorage
   620→    localStorage.setItem("p360_token", FALLBACK_USER.id);
   621→    localStorage.setItem("p360_user", JSON.stringify(FALLBACK_USER));
   622→    return FALLBACK_USER;
   623→  });
   624→  const [page, setPage] = useState("dashboard");
   625→  const [sidebarOpen, setSidebarOpen] = useState(false);
   626→  const [generations, setGenerations] = useState<Generation[]>([]);
   627→
   628→  useEffect(() => {
   629→    const token = localStorage.getItem("p360_token");
   630→    if (token) {
   631→      fetch("/api/generations?limit=10", { headers: { Authorization: `Bearer ${token}` } })
   632→        .then(r => r.ok ? r.json() : [])
   633→        .then(setGenerations)
   634→        .catch(() => {});
   635→    }
   636→  }, []);
   637→
   638→  const logout = () => { localStorage.removeItem("p360_token"); localStorage.removeItem("p360_user"); onLogout(); };
   639→  const refreshUser = async () => {
   640→    const token = localStorage.getItem("p360_token");
   641→    if (!token) return;
   642→    try {
   643→      const res = await fetch("/api/auth/me", { headers: { Authorization: `Bearer ${token}` } });
   644→      if (res.ok) setUser(await res.json());
   645→    } catch {}
   646→  };
   647→  const onUpdateCredits = (c: number) => { setUser(prev => ({ ...prev, credits: c })); };
   648→
   649→  const renderPage = () => {
   650→    switch (page) {
   651→      case "dashboard": return <Dashboard user={user} generations={generations} onNav={setPage} />;
   652→      case "prompts": return <PromptChat user={user} onUpdateCredits={onUpdateCredits} />;
   653→      case "images": return <ImageGenerator user={user} onUpdateCredits={onUpdateCredits} />;
   654→      case "copy": case "social": case "script": case "email": return <TextGenerator type={page} user={user} onUpdateCredits={onUpdateCredits} />;
   655→      case "voice": return <VoiceGenerator user={user} onUpdateCredits={onUpdateCredits} />;
   656→      case "ebook": return <EbookGenerator user={user} onUpdateCredits={onUpdateCredits} />;
   657→      case "subtitle": return <SubtitleGenerator user={user} onUpdateCredits={onUpdateCredits} />;
   658→      case "metrics": return <MetricsTracker userId={user.id} />;
   659→      case "library": return <Library user={user} />;
   660→      case "settings": return <SettingsPage user={user} />;
   661→      case "support": return <SupportPage />;
   662→      case "adn": return <AdnDashboard user={user} />;
   663→      default: return <Dashboard user={user} generations={generations} onNav={setPage} />;
   664→    }
   665→  };
   666→
   667→  return (
   668→    <AuthCtx.Provider value={{ user, logout, refreshUser }}>
   669→      <div className="min-h-screen bg-[#080c16] flex">
   670→        <Sidebar active={page} onNav={setPage} open={sidebarOpen} onClose={() => setSidebarOpen(false)} user={user} />
   671→        <div className="flex-1 flex flex-col min-h-screen">
   672→          {/* Top bar */}
   673→          <header className="sticky top-0 z-30 flex items-center gap-4 px-4 sm:px-6 py-3 border-b border-white/[0.06] backdrop-blur-xl bg-[#080c16]/80">
   674→            <button onClick={() => setSidebarOpen(true)} className="lg:hidden text-white/50 hover:text-white">
   675→              <Menu className="w-5 h-5" />
   676→            </button>
   677→            <div className="flex-1" />
   678→            <div className="flex items-center gap-3">
   679→              <div className="flex items-center gap-1.5 px-3 py-1.5 rounded-lg bg-yellow-400/10 text-yellow-400 text-xs font-bold">
   680→                <Coins className="w-3.5 h-3.5" /> {user.credits}
   681→              </div>
   682→              <button onClick={logout} className="p-2 rounded-lg hover:bg-white/5 text-white/40 hover:text-white/70 transition-colors" title="Cerrar sesión">
   683→                <LogOut className="w-4 h-4" />
   684→              </button>
   685→            </div>
   686→          </header>
   687→          {/* Content */}
   688→          <main className="flex-1 p-4 sm:p-6 lg:p-8 overflow-y-auto">
   689→            {renderPage()}
   690→          </main>
   691→        </div>
   692→      </div>
   693→    </AuthCtx.Provider>

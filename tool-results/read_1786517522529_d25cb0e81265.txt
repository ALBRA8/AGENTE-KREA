   472→            </div>

   474→        )}
   475→      </div>
   476→    </div>

   480→/* ════════════ Subtitle Generator ════════════ */
   481→function SubtitleGenerator({ user, onUpdateCredits }: { user: User; onUpdateCredits: (c: number) => void }) {
   482→  const [url, setUrl] = useState("");
   483→  const [loading, setLoading] = useState(false);
   484→  const [result, setResult] = useState("");
   485→  const [error, setError] = useState("");
   486→
   487→  const generate = async () => {
   488→    setLoading(true); setError(""); setResult("");
   489→    try {
   490→      const res = await fetch("/api/generate/subtitle", {
   491→        method: "POST", headers: { "Content-Type": "application/json", Authorization: `Bearer ${user.id}` },
   492→        body: JSON.stringify({ videoUrl: url || undefined }),
   493→      });
   494→      const data = await res.json();
   495→      if (!res.ok) { setError(data.error); return; }
   496→      setResult(data.srt); onUpdateCredits(data.credits);
   497→    } catch { setError("Error de conexión"); }
   498→    finally { setLoading(false); }
   499→  };
   500→
   501→  return (
   502→    <div>
   503→      <div className="mb-6">
   504→        <h1 className="text-2xl font-extrabold mb-1">Generador de <span className="gradient-text">Subtítulos</span></h1>
   505→        <p className="text-sm text-white/50">Crea subtítulos SRT profesionales. Costo: <span className="text-yellow-400 font-semibold">2 créditos</span></p>
   506→      </div>
   507→      <div className="max-w-2xl space-y-4">
   508→        <div>
   509→          <label className="text-xs text-white/50 mb-1.5 block">URL del video (opcional — si lo dejas vacío se genera un template)</label>
   510→          <input type="url" value={url} onChange={(e) => setUrl(e.target.value)} className="w-full px-4 py-3 rounded-xl bg-white/5 border border-white/10 text-white text-sm placeholder:text-white/30 focus:outline-none focus:border-[#3b82f6]" placeholder="https://youtube.com/watch?v=..." />
   511→        </div>
   512→        <button onClick={generate} disabled={loading} className="w-full py-3.5 rounded-xl bg-gradient-to-b from-[#3b82f6] to-[#1e40af] text-white font-bold text-sm shadow-[0_10px_24px_rgba(30,64,175,0.45)] hover:brightness-110 active:scale-[0.98] transition-all disabled:opacity-50 flex items-center justify-center gap-2">
   513→          {loading ? <><Loader2 className="w-4 h-4 animate-spin" /> Generando...</> : <><Subtitles className="w-4 h-4" /> Generar subtítulos (-2 créditos)</>}
   514→        </button>
   515→        {error && <p className="text-red-400 text-xs">{error}</p>}
   516→        {result && (
   517→          <div className="rounded-2xl border border-white/[0.06] bg-[#0f1629]">
   518→            <div className="flex items-center justify-between p-3 border-b border-white/[0.06]">
   519→              <span className="text-xs text-white/40">Subtítulos SRT</span>
   520→              <button onClick={() => navigator.clipboard.writeText(result)} className="text-xs text-[#3b82f6] flex items-center gap-1 hover:underline"><Copy className="w-3 h-3" /> Copiar SRT</button>
   521→            </div>
   522→            <pre className="p-4 text-xs text-white/70 whitespace-pre-wrap max-h-[400px] overflow-y-auto">{result}</pre>
   523→          </div>
   524→        )}
   525→      </div>
   526→    </div>
   527→  );
   528→}

   530→/* ════════════ Library ════════════ */
   531→function Library({ user }: { user: User }) {
   532→  const [gens, setGens] = useState<Generation[]>([]);
   533→  const [filter, setFilter] = useState("all");
   534→  const [loading, setLoading] = useState(true);
   535→
   536→  const load = async () => {
   537→    setLoading(true);
   538→    try {
   539→      const params = filter === "all" ? "" : `?type=${filter}`;
   540→      const res = await fetch(`/api/generations${params}`, { headers: { Authorization: `Bearer ${user.id}` } });
   541→      if (res.ok) setGens(await res.json());
   542→    } catch { /* ignore */ }
   543→    setLoading(false);
   544→  };
   545→
   546→  useEffect(() => { load(); }, [filter]);
   547→
   548→  const typeFilters = [
   549→    { id: "all", label: "Todos" },
   550→    { id: "image", label: "Imágenes" },
   551→    { id: "text", label: "Textos" },
   552→    { id: "voice", label: "Voces" },
   553→    { id: "ebook", label: "eBooks" },
   554→    { id: "subtitle", label: "Subtítulos" },
   555→  ];
   556→
   557→  return (
   558→    <div>
   559→      <div className="mb-6">
   560→        <h1 className="text-2xl font-extrabold mb-1">Mi <span className="gradient-text">Biblioteca</span></h1>
   561→        <p className="text-sm text-white/50">Historial de todas tus creaciones</p>
   562→      </div>
   563→      <div className="flex gap-2 mb-6 overflow-x-auto pb-2">
   564→        {typeFilters.map((f) => (
   565→          <button key={f.id} onClick={() => setFilter(f.id)} className={`px-4 py-2 rounded-lg text-xs font-medium whitespace-nowrap transition-all ${filter === f.id ? "bg-[#3b82f6] text-white" : "bg-white/5 text-white/50 hover:bg-white/10"}`}>{f.label}</button>
   566→        ))}
   567→      </div>
   568→      {loading ? (
   569→        <div className="flex items-center justify-center py-20 text-white/30"><Loader2 className="w-6 h-6 animate-spin mr-2" /> Cargando...</div>
   570→      ) : gens.length === 0 ? (
   571→        <div className="text-center py-20 text-white/20"><LibraryIcon className="w-12 h-12 mx-auto mb-3 opacity-50" /><p className="text-sm">No hay creaciones todavía</p></div>
   572→      ) : (
   573→        <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-4">
   574→          {gens.map((g) => (
   575→            <div key={g.id} className="rounded-2xl border border-white/[0.06] bg-[#0f1629] overflow-hidden">
   576→              {g.type === "image" && g.result.startsWith("/") ? (
   577→                <img src={g.result} alt="" className="w-full aspect-square object-cover" />
   578→              ) : g.type === "voice" && g.result.startsWith("/") ? (
   579→                <div className="p-4 bg-white/[0.02]"><audio controls src={g.result} className="w-full" /></div>
   580→              ) : (
   581→                <div className="p-4 max-h-40 overflow-y-auto"><pre className="text-xs text-white/50 whitespace-pre-wrap">{g.result.substring(0, 500)}</pre></div>
   582→              )}
   583→              <div className="p-3 border-t border-white/[0.06]">
   584→                <p className="text-sm font-medium truncate">{g.title}</p>
   585→                <div className="flex items-center justify-between mt-1">
   586→                  <p className="text-[10px] text-white/30 flex items-center gap-1"><Clock className="w-3 h-3" /> {new Date(g.createdAt).toLocaleDateString("es-ES")}</p>
   587→                  <span className="text-[10px] text-white/30">-{g.credits} <Coins className="inline w-3 h-3" /></span>
   588→                </div>
   589→              </div>
   590→            </div>
   591→          ))}
   592→        </div>
   593→      )}
   594→    </div>
   595→  );
   596→}

   598→/* ════════════ Main App ════════════ */
   599→interface AppShellProps { onLogout: () => void; user?: User | null }
   600→
   601→export default function AppPage({ onLogout, user: initialUser }: AppShellProps) {
   602→  const [user, setUser] = useState<User | null>(() => {
   603→    if (initialUser) return initialUser;
   604→    try {
   605→      const saved = localStorage.getItem("p360_user");
   606→      return saved ? JSON.parse(saved) : null;
   607→    } catch { return null; }
   608→  });
   609→  const [page, setPage] = useState("dashboard");
   610→  const [sidebarOpen, setSidebarOpen] = useState(false);
   611→  const [loading, setLoading] = useState(false);
   612→  const [generations, setGenerations] = useState<Generation[]>([]);
   613→
   614→  const loadUser = useCallback(async () => {
   615→    const token = localStorage.getItem("p360_token");
   616→    if (!token) { if (!user) onLogout(); else setLoading(false); return; }
   617→    try {
   618→      const res = await fetch("/api/auth/me", { headers: { Authorization: `Bearer ${token}` } });
   619→      if (!res.ok) { if (!user) onLogout(); else setLoading(false); return; }
   620→      const data = await res.json();
   621→      setUser(data);
   622→      const gRes = await fetch("/api/generations?limit=10", { headers: { Authorization: `Bearer ${token}` } });
   623→      if (gRes.ok) setGenerations(await gRes.json());
   624→    } catch { if (!user) onLogout(); }
   625→    setLoading(false);
   626→  }, [user, onLogout]);
   627→
   628→  useEffect(() => {
   629→    const token = localStorage.getItem("p360_token");
   630→    if (token && user) {
   631→      fetch("/api/generations?limit=10", { headers: { Authorization: `Bearer ${token}` } })
   632→        .then(r => r.ok ? r.json() : [])
   633→        .then(setGenerations)
   634→        .catch(() => {});
   635→    }
   636→  }, []);
   637→
   638→  const logout = () => { localStorage.removeItem("p360_token"); localStorage.removeItem("p360_user"); onLogout(); };
   639→  const refreshUser = () => loadUser();
   640→  const onUpdateCredits = (c: number) => { if (user) setUser({ ...user, credits: c }); };
   641→
   642→  if (loading) return (
   643→    <div className="min-h-screen bg-[#080c16] flex items-center justify-center">
   644→      <Loader2 className="w-8 h-8 animate-spin text-[#3b82f6]" />
   645→    </div>
   646→  );
   647→
   648→  if (!user) return null;
   649→
   650→  const renderPage = () => {
   651→    switch (page) {
   652→      case "dashboard": return <Dashboard user={user} generations={generations} onNav={setPage} />;
   653→      case "prompts": return <PromptChat user={user} onUpdateCredits={onUpdateCredits} />;
   654→      case "images": return <ImageGenerator user={user} onUpdateCredits={onUpdateCredits} />;
   655→      case "copy": case "social": case "script": case "email": return <TextGenerator type={page} user={user} onUpdateCredits={onUpdateCredits} />;
   656→      case "voice": return <VoiceGenerator user={user} onUpdateCredits={onUpdateCredits} />;
   657→      case "ebook": return <EbookGenerator user={user} onUpdateCredits={onUpdateCredits} />;
   658→      case "subtitle": return <SubtitleGenerator user={user} onUpdateCredits={onUpdateCredits} />;
   659→      case "library": return <Library user={user} />;
   660→      case "settings": return <SettingsPage user={user} />;
   661→      case "support": return <SupportPage />;
   662→      default: return <Dashboard user={user} generations={generations} onNav={setPage} />;
   663→    }
   664→  };
   665→
   666→  return (
   667→    <AuthCtx.Provider value={{ user, logout, refreshUser }}>
   668→      <div className="min-h-screen bg-[#080c16] flex">
   669→        <Sidebar active={page} onNav={setPage} open={sidebarOpen} onClose={() => setSidebarOpen(false)} user={user} />
   670→        <div className="flex-1 flex flex-col min-h-screen">
   671→          {/* Top bar */}
   672→          <header className="sticky top-0 z-30 flex items-center gap-4 px-4 sm:px-6 py-3 border-b border-white/[0.06] backdrop-blur-xl bg-[#080c16]/80">
   673→            <button onClick={() => setSidebarOpen(true)} className="lg:hidden text-white/50 hover:text-white">
   674→              <Menu className="w-5 h-5" />
   675→            </button>
   676→            <div className="flex-1" />
   677→            <div className="flex items-center gap-3">
   678→              <div className="flex items-center gap-1.5 px-3 py-1.5 rounded-lg bg-yellow-400/10 text-yellow-400 text-xs font-bold">
   679→                <Coins className="w-3.5 h-3.5" /> {user.credits}
   680→              </div>
   681→              <button onClick={logout} className="p-2 rounded-lg hover:bg-white/5 text-white/40 hover:text-white/70 transition-colors" title="Cerrar sesión">
   682→                <LogOut className="w-4 h-4" />
   683→              </button>
   684→            </div>
   685→          </header>
   686→          {/* Content */}
   687→          <main className="flex-1 p-4 sm:p-6 lg:p-8 overflow-y-auto">
   688→            {renderPage()}
   689→          </main>
   690→        </div>
   691→      </div>
   692→    </AuthCtx.Provider>
   693→  );
   694→}

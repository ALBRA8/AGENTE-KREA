   546→      if (res.ok) setGens(await res.json());
   547→    } catch { /* ignore */ }
   548→    setLoading(false);
   549→  };
   550→
   551→  useEffect(() => { load(); }, [filter]);
   552→
   553→  const typeFilters = [
   554→    { id: "all", label: "Todos" },
   555→    { id: "image", label: "Imágenes" },
   556→    { id: "text", label: "Textos" },
   557→    { id: "voice", label: "Voces" },
   558→    { id: "ebook", label: "eBooks" },
   559→    { id: "subtitle", label: "Subtítulos" },
   560→  ];
   561→
   562→  return (
   563→    <div>
   564→      <div className="mb-6">
   565→        <h1 className="text-2xl font-extrabold mb-1">Mi <span className="gradient-text">Biblioteca</span></h1>
   566→        <p className="text-sm text-white/50">Historial de todas tus creaciones</p>
   567→      </div>
   568→      <div className="flex gap-2 mb-6 overflow-x-auto pb-2">
   569→        {typeFilters.map((f) => (
   570→          <button key={f.id} onClick={() => setFilter(f.id)} className={`px-4 py-2 rounded-lg text-xs font-medium whitespace-nowrap transition-all ${filter === f.id ? "bg-[#3b82f6] text-white" : "bg-white/5 text-white/50 hover:bg-white/10"}`}>{f.label}</button>
   571→        ))}
   572→      </div>
   573→      {loading ? (
   574→        <div className="flex items-center justify-center py-20 text-white/30"><Loader2 className="w-6 h-6 animate-spin mr-2" /> Cargando...</div>
   575→      ) : gens.length === 0 ? (
   576→        <div className="text-center py-20 text-white/20"><LibraryIcon className="w-12 h-12 mx-auto mb-3 opacity-50" /><p className="text-sm">No hay creaciones todavía</p></div>
   577→      ) : (
   578→        <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-4">
   579→          {gens.map((g) => (
   580→            <div key={g.id} className="rounded-2xl border border-white/[0.06] bg-[#0f1629] overflow-hidden">
   581→              {g.type === "image" && g.result.startsWith("/") ? (
   582→                <img src={g.result} alt="" className="w-full aspect-square object-cover" />
   583→              ) : g.type === "voice" && g.result.startsWith("/") ? (
   584→                <div className="p-4 bg-white/[0.02]"><audio controls src={g.result} className="w-full" /></div>
   585→              ) : (
   586→                <div className="p-4 max-h-40 overflow-y-auto"><pre className="text-xs text-white/50 whitespace-pre-wrap">{g.result.substring(0, 500)}</pre></div>
   587→              )}
   588→              <div className="p-3 border-t border-white/[0.06]">
   589→                <p className="text-sm font-medium truncate">{g.title}</p>
   590→                <div className="flex items-center justify-between mt-1">
   591→                  <p className="text-[10px] text-white/30 flex items-center gap-1"><Clock className="w-3 h-3" /> {new Date(g.createdAt).toLocaleDateString("es-ES")}</p>
   592→                  <span className="text-[10px] text-white/30">-{g.credits} <Coins className="inline w-3 h-3" /></span>
   593→                </div>
   594→              </div>
   595→            </div>
   596→          ))}
   597→        </div>
   598→      )}
   599→    </div>
   600→  );
   601→}

   603→/* ════════════ Main App ════════════ */
   604→interface AppShellProps { onLogout: () => void; user?: User | null }
   605→
   606→export default function AppPage({ onLogout, user: initialUser }: AppShellProps) {
   607→  const [user, setUser] = useState<User | null>(() => {
   608→    if (initialUser) return initialUser;
   609→    try {
   610→      const saved = localStorage.getItem("p360_user");
   611→      return saved ? JSON.parse(saved) : null;
   612→    } catch { return null; }
   613→  });
   614→  const [page, setPage] = useState("dashboard");
   615→  const [sidebarOpen, setSidebarOpen] = useState(false);
   616→  const [loading, setLoading] = useState(false);
   617→  const [generations, setGenerations] = useState<Generation[]>([]);
   618→
   619→  const loadUser = useCallback(async () => {
   620→    const token = localStorage.getItem("p360_token");
   621→    if (!token) { if (!user) onLogout(); else setLoading(false); return; }
   622→    try {
   623→      const res = await fetch("/api/auth/me", { headers: { Authorization: `Bearer ${token}` } });
   624→      if (!res.ok) { if (!user) onLogout(); else setLoading(false); return; }
   625→      const data = await res.json();
   626→      setUser(data);
   627→      const gRes = await fetch("/api/generations?limit=10", { headers: { Authorization: `Bearer ${token}` } });
   628→      if (gRes.ok) setGenerations(await gRes.json());
   629→    } catch { if (!user) onLogout(); }
   630→    setLoading(false);
   631→  }, [user, onLogout]);
   632→
   633→  useEffect(() => {
   634→    const token = localStorage.getItem("p360_token");
   635→    if (token && user) {
   636→      fetch("/api/generations?limit=10", { headers: { Authorization: `Bearer ${token}` } })
   637→        .then(r => r.ok ? r.json() : [])
   638→        .then(setGenerations)
   639→        .catch(() => {});
   640→    }
   641→  }, []);
   642→
   643→  const logout = () => { localStorage.removeItem("p360_token"); localStorage.removeItem("p360_user"); onLogout(); };
   644→  const refreshUser = () => loadUser();
   645→  const onUpdateCredits = (c: number) => { if (user) setUser({ ...user, credits: c }); };
   646→
   647→  if (loading) return (
   648→    <div className="min-h-screen bg-[#080c16] flex items-center justify-center">
   649→      <Loader2 className="w-8 h-8 animate-spin text-[#3b82f6]" />
   650→    </div>
   651→  );
   652→
   653→  if (!user) return null;
   654→
   655→  const renderPage = () => {
   656→    switch (page) {
   657→      case "dashboard": return <Dashboard user={user} generations={generations} onNav={setPage} />;
   658→      case "prompts": return <PromptChat user={user} onUpdateCredits={onUpdateCredits} />;
   659→      case "images": return <ImageGenerator user={user} onUpdateCredits={onUpdateCredits} />;
   660→      case "copy": case "social": case "script": case "email": return <TextGenerator type={page} user={user} onUpdateCredits={onUpdateCredits} />;
   661→      case "voice": return <VoiceGenerator user={user} onUpdateCredits={onUpdateCredits} />;
   662→      case "ebook": return <EbookGenerator user={user} onUpdateCredits={onUpdateCredits} />;
   663→      case "subtitle": return <SubtitleGenerator user={user} onUpdateCredits={onUpdateCredits} />;
   664→      case "metrics": return <MetricsTracker userId={user.id} />;
   665→      case "library": return <Library user={user} />;
   666→      case "settings": return <SettingsPage user={user} />;
   667→      case "support": return <SupportPage />;
   668→      default: return <Dashboard user={user} generations={generations} onNav={setPage} />;
   669→    }
   670→  };
   671→
   672→  return (
   673→    <AuthCtx.Provider value={{ user, logout, refreshUser }}>
   674→      <div className="min-h-screen bg-[#080c16] flex">
   675→        <Sidebar active={page} onNav={setPage} open={sidebarOpen} onClose={() => setSidebarOpen(false)} user={user} />
   676→        <div className="flex-1 flex flex-col min-h-screen">
   677→          {/* Top bar */}
   678→          <header className="sticky top-0 z-30 flex items-center gap-4 px-4 sm:px-6 py-3 border-b border-white/[0.06] backdrop-blur-xl bg-[#080c16]/80">
   679→            <button onClick={() => setSidebarOpen(true)} className="lg:hidden text-white/50 hover:text-white">
   680→              <Menu className="w-5 h-5" />
   681→            </button>
   682→            <div className="flex-1" />
   683→            <div className="flex items-center gap-3">
   684→              <div className="flex items-center gap-1.5 px-3 py-1.5 rounded-lg bg-yellow-400/10 text-yellow-400 text-xs font-bold">
   685→                <Coins className="w-3.5 h-3.5" /> {user.credits}
   686→              </div>
   687→              <button onClick={logout} className="p-2 rounded-lg hover:bg-white/5 text-white/40 hover:text-white/70 transition-colors" title="Cerrar sesión">
   688→                <LogOut className="w-4 h-4" />
   689→              </button>
   690→            </div>
   691→          </header>
   692→          {/* Content */}
   693→          <main className="flex-1 p-4 sm:p-6 lg:p-8 overflow-y-auto">
   694→            {renderPage()}
   695→          </main>
   696→        </div>
   697→      </div>
   698→    </AuthCtx.Provider>
   699→  );
   700→}

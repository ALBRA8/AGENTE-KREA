     1→/**
     2→ * Visual QA — KREA Book Factory Layer
     3→ *
     4→ * Inspects PDF/layout results for visual quality issues.
     5→ * KREA CANNOT declare a PDF done just because the renderer finished (AC-043).
     6→ * Every artifact must be inspected, issues diagnosed, and repairs applied
     7→ * before the product is considered final.
     8→ *
     9→ * Satisfies:
    10→ *   AC-043 — KREA cannot declare PDF done just because renderer finished
    11→ *   AC-044 — QAIssue categories with severity, location, description, suggested_fix
    12→ *   AC-045 — Repair loop: DETECT → DIAGNOSE → REPAIR → RENDER → VERIFY (max 5 iterations)
    13→ */
    14→

    19→// ─── QA Issue Categories ───────────────────────────────────────────────
    20→
    21→/**
    22→ * All categories of visual issues that VisualQA checks for.
    23→ * Each category maps to a specific visual quality concern.
    24→ */
    25→export type QAIssueCategory =
    26→  | "cut_text"              // Text is cut off / truncated
    27→  | "overflow"              // Content overflows its container
    28→  | "off_page"              // Content placed outside page boundaries
    29→  | "distorted_images"      // Images with wrong aspect ratio or distortion
    30→  | "empty_pages"           // Pages with no meaningful content
    31→  | "bad_hierarchy"         // Heading hierarchy is broken or inconsistent
    32→  | "contrast_problems"     // Text/background contrast below WCAG AA
    33→  | "typographic_inconsistency" // Inconsistent fonts, sizes, or weights
    34→  | "cover_errors"          // Cover doesn't match interior or has issues
    35→  | "numbering_errors"      // Page, chapter, or figure numbering is wrong
    36→  | "index_errors"          // Index is missing, incomplete, or incorrect
    37→  | "legibility_problems";  // General legibility issues (too small, etc.)
    38→

   152→// ─── VisualQA Class ────────────────────────────────────────────────────
   153→
   154→/**
   155→ * VisualQA — inspects PDF/layout results for visual quality issues.
   156→ *
   157→ * Three main operations:
   158→ * 1. inspect() — detect all visual quality issues (AC-043)
   159→ * 2. diagnose() — determine root cause for each issue
   160→ * 3. repair() — attempt to fix an issue
   161→ *
   162→ * AC-043: Never trust the renderer output blindly.
   163→ * AC-044: Every issue has severity, location, description, suggested_fix.
   164→ * AC-045: Repair loop with max 5 iterations.
   165→ */
   166→export class VisualQA {
   167→  /**
   168→   * Inspect a PDF/layout result for visual quality issues.
   169→   *
   170→   * AC-043: This inspection is MANDATORY. The renderer finishing does NOT
   171→   * mean the PDF is done — we must actively verify visual quality.
   172→   *
   173→   * @param pdfSpec - The PDF specification and layout to inspect
   174→   * @param content - The source content that was rendered
   175→   * @param editorial - The editorial specification used
   176→   * @returns QAResult with all detected issues
   177→   */
   178→  inspect(
   179→    pdfSpec: PDFSpecForQA,
   180→    content: BookContent,
   181→    editorial: EditorialSpecification
   182→  ): QAResult {
   183→    const startTime = Date.now();
   184→    const issues: QAIssue[] = [];
   185→
   186→    // Run all inspection checks
   187→    issues.push(...this.checkCutText(pdfSpec, content, editorial));
   188→    issues.push(...this.checkOverflow(pdfSpec, editorial));
   189→    issues.push(...this.checkOffPage(pdfSpec));
   190→    issues.push(...this.checkEmptyPages(pdfSpec));
   191→    issues.push(...this.checkBadHierarchy(pdfSpec, content));
   192→    issues.push(...this.checkContrastProblems(pdfSpec, editorial));
   193→    issues.push(...this.checkTypographicInconsistency(pdfSpec, editorial));
   194→    issues.push(...this.checkNumberingErrors(pdfSpec, content));
   195→    issues.push(...this.checkLegibilityProblems(pdfSpec, editorial));
   196→
   197→    // Compute summaries
   198→    const byCategory: Record<QAIssueCategory, number> = {
   199→      cut_text: 0, overflow: 0, off_page: 0, distorted_images: 0,
   200→      empty_pages: 0, bad_hierarchy: 0, contrast_problems: 0,
   201→      typographic_inconsistency: 0, cover_errors: 0, numbering_errors: 0,
   202→      index_errors: 0, legibility_problems: 0,
   203→    };
   204→
   205→    const bySeverity: Record<QAIssueSeverity, number> = {
   206→      critical: 0, warning: 0, info: 0,
   207→    };
   208→
   209→    for (const issue of issues) {
   210→      byCategory[issue.category]++;
   211→      bySeverity[issue.severity]++;
   212→    }
   213→
   214→    const duration = Date.now() - startTime;
   215→
   216→    return {
   217→      passed: bySeverity.critical === 0,
   218→      issue_count: issues.length,
   219→      issues,
   220→      by_category: byCategory,
   221→      by_severity: bySeverity,
   222→      inspectedAt: new Date().toISOString(),
   223→      inspection_duration_ms: duration,
   224→    };
   225→  }
   226→
   227→  /**
   228→   * Diagnose the root cause of QA issues.
   229→   *
   230→   * @param issues - The issues to diagnose
   231→   * @returns Array of diagnoses with repair strategies
   232→   */
   233→  diagnose(issues: QAIssue[]): Diagnosis[] {
   234→    return issues.map((issue) => this.diagnoseSingle(issue));
   235→  }
   236→
   237→  /**
   238→   * Attempt to repair a single QA issue.
   239→   *
   240→   * @param issue - The issue to repair
   241→   * @param content - Current book content
   242→   * @param editorial - Current editorial specification
   243→   * @returns RepairedContent with the repair result
   244→   */
   245→  repair(
   246→    issue: QAIssue,
   247→    content: BookContent,
   248→    editorial: EditorialSpecification
   249→  ): RepairedContent {
   250→    if (!issue.auto_repairable) {
   251→      return {
   252→        repaired: false,
   253→        issue_id: issue.id,
   254→        repair_description: "Issue is not auto-repairable — requires manual intervention",
   255→        remaining_issues: [issue.id],
   256→      };
   257→    }
   258→
   259→    switch (issue.category) {
   260→      case "cut_text":
   261→        return this.repairCutText(issue, content, editorial);
   262→      case "overflow":
   263→        return this.repairOverflow(issue, content, editorial);
   264→      case "empty_pages":
   265→        return this.repairEmptyPages(issue, content, editorial);
   266→      case "bad_hierarchy":
   267→        return this.repairBadHierarchy(issue, content, editorial);
   268→      case "numbering_errors":
   269→        return this.repairNumberingErrors(issue, content, editorial);
   270→      default:
   271→        return {
   272→          repaired: false,
   273→          issue_id: issue.id,
   274→          repair_description: `Auto-repair not implemented for category: ${issue.category}`,
   275→          remaining_issues: [issue.id],
   276→        };
   277→    }
   278→  }
   279→
   280→  // ─── Inspection Checks ─────────────────────────────────────────────
   281→
   282→  /**
   283→   * Check for cut/truncated text.
   284→   */
   285→  private checkCutText(
   286→    pdfSpec: PDFSpecForQA,
   287→    content: BookContent,
   288→    editorial: EditorialSpecification
   289→  ): QAIssue[] {
   290→    const issues: QAIssue[] = [];
   291→
   292→    for (const page of pdfSpec.layout.pages) {
   293→      for (const element of page.elements) {
   294→        if (element.type === "paragraph" || element.type === "heading") {
   295→          // Check if element height was constrained by remaining space
   296→          const contentRef = element.content_ref;
   297→          const sourceContent = this.findSourceContent(content, contentRef);
   298→
   299→          if (sourceContent && element.height < this.estimateMinHeight(sourceContent, editorial)) {
   300→            issues.push({
   301→              id: `qa-cut-${page.page_number}-${element.content_ref}`,
   302→              category: "cut_text",
   303→              severity: "critical",
   304→              location: {
   305→                page: page.page_number,
   306→                element_ref: element.content_ref,
   307→                bounds: { x: element.x, y: element.y, width: element.width, height: element.height },
   308→              },
   309→              description: `Text appears truncated on page ${page.page_number + 1}: element height (${element.height.toFixed(1)}pt) is less than estimated minimum`,
   310→              suggested_fix: "Reduce font size, increase line height, or split content across pages",
   311→              auto_repairable: true,
   312→            });
   313→          }
   314→        }
   315→      }
   316→    }
   317→
   318→    return issues;
   319→  }
   320→
   321→  /**
   322→   * Check for content overflow.
   323→   */
   324→  private checkOverflow(
   325→    pdfSpec: PDFSpecForQA,
   326→    editorial: EditorialSpecification
   327→  ): QAIssue[] {
   328→    const issues: QAIssue[] = [];
   329→    const pageHeight = pdfSpec.layout.pages[0]
   330→      ? editorial.margins.top + editorial.margins.bottom +
   331→        pdfSpec.layout.pages[0].remaining_space +
   332→        pdfSpec.layout.pages[0].elements.reduce((sum, e) => sum + e.height, 0)
   333→      : 0;
   334→
   335→    for (const page of pdfSpec.layout.pages) {
   336→      for (const element of page.elements) {
   337→        const elementBottom = element.y + element.height;
   338→        const contentBottom = pageHeight - editorial.margins.bottom;
   339→
   340→        if (elementBottom > contentBottom) {
   341→          issues.push({
   342→            id: `qa-overflow-${page.page_number}-${element.content_ref}`,
   343→            category: "overflow",
   344→            severity: "critical",
   345→            location: {
   346→              page: page.page_number,
   347→              element_ref: element.content_ref,
   348→            },
   349→            description: `Content overflows page bottom by ${(elementBottom - contentBottom).toFixed(1)}pt on page ${page.page_number + 1}`,
   350→            suggested_fix: "Move overflowing content to next page or reduce element size",
   351→            auto_repairable: true,
   352→          });
   353→        }
   354→      }
   355→    }
   356→
   357→    return issues;
   358→  }
   359→
   360→  /**
   361→   * Check for content placed outside page boundaries.
   362→   */
   363→  private checkOffPage(pdfSpec: PDFSpecForQA): QAIssue[] {
   364→    const issues: QAIssue[] = [];
   365→
   366→    for (const page of pdfSpec.layout.pages) {
   367→      for (const element of page.elements) {
   368→        if (element.x < 0 || element.y < 0) {
   369→          issues.push({
   370→            id: `qa-offpage-${page.page_number}-${element.content_ref}`,
   371→            category: "off_page",
   372→            severity: "critical",
   373→            location: {
   374→              page: page.page_number,
   375→              element_ref: element.content_ref,
   376→              bounds: { x: element.x, y: element.y, width: element.width, height: element.height },
   377→            },
   378→            description: `Element placed off-page at (${element.x.toFixed(1)}, ${element.y.toFixed(1)}) on page ${page.page_number + 1}`,
   379→            suggested_fix: "Adjust margins or reposition element within page bounds",
   380→            auto_repairable: true,
   381→          });
   382→        }
   383→      }
   384→    }
   385→
   386→    return issues;
   387→  }

   389→  /**
   390→   * Check for empty pages (pages with no meaningful content).
   391→   */
   392→  private checkEmptyPages(pdfSpec: PDFSpecForQA): QAIssue[] {
   393→    const issues: QAIssue[] = [];
   394→
   395→    for (const page of pdfSpec.layout.pages) {
   396→      const contentElements = page.elements.filter(
   397→        (e) => e.type !== "page_break" && e.type !== "decoration"
   398→      );
   399→
   400→      if (contentElements.length === 0 && pdfSpec.layout.pages.length > 1) {
   401→        issues.push({
   402→          id: `qa-empty-${page.page_number}`,
   403→          category: "empty_pages",
   404→          severity: "warning",
   405→          location: { page: page.page_number },
   406→          description: `Page ${page.page_number + 1} has no meaningful content`,
   407→          suggested_fix: "Remove empty page or redistribute content from adjacent pages",
   408→          auto_repairable: true,
   409→        });
   410→      }
   411→    }
   412→
   413→    return issues;
   414→  }

   416→  /**
   417→   * Check for broken heading hierarchy.
   418→   */
   419→  private checkBadHierarchy(
   420→    pdfSpec: PDFSpecForQA,
   421→    content: BookContent
   422→  ): QAIssue[] {
   423→    const issues: QAIssue[] = [];
   424→    let lastHeadingLevel = 0;
   425→
   426→    for (const chapter of content.chapters) {
   427→      for (const section of chapter.sections) {
   428→        if (section.heading) {
   429→          // Check for skipped levels (e.g., h1 → h3)
   430→          if (lastHeadingLevel > 0 && section.level > lastHeadingLevel + 1) {
   431→            issues.push({
   432→              id: `qa-hierarchy-${chapter.chapter_number}-${section.level}`,
   433→              category: "bad_hierarchy",
   434→              severity: "warning",
   435→              location: {
   436→                page: 0,
   437→                chapter: chapter.chapter_number,
   438→              },
   439→              description: `Heading hierarchy skip: level ${lastHeadingLevel} to ${section.level} in chapter ${chapter.chapter_number}`,
   440→              suggested_fix: "Insert intermediate heading level or adjust hierarchy",
   441→              auto_repairable: true,
   442→            });
   443→          }
   444→          lastHeadingLevel = section.level;
   445→        }
   446→      }
   447→    }
   448→
   449→    return issues;
   450→  }

   452→  /**
   453→   * Check for contrast problems.
   454→   */
   455→  private checkContrastProblems(
   456→    pdfSpec: PDFSpecForQA,
   457→    editorial: EditorialSpecification
   458→  ): QAIssue[] {
   459→    const issues: QAIssue[] = [];
   460→
   461→    // Check if body text contrast meets minimum
   462→    const bodyContrast = this.estimateContrast(
   463→      editorial.paragraph_style.color,
   464→      "#ffffff" // Assume white background
   465→    );
   466→
   467→    if (bodyContrast < 4.5) {
   468→      issues.push({
   469→        id: "qa-contrast-body",
   470→        category: "contrast_problems",
   471→        severity: "critical",
   472→        location: { page: 0 },
   473→        description: `Body text contrast ratio (${bodyContrast.toFixed(2)}:1) below WCAG AA minimum (4.5:1)`,
   474→        suggested_fix: "Increase text color darkness or adjust background color",
   475→        auto_repairable: true,
   476→      });
   477→    }
   478→
   479→    return issues;
   480→  }

   482→  /**
   483→   * Check for typographic inconsistency.
   484→   */
   485→  private checkTypographicInconsistency(
   486→    pdfSpec: PDFSpecForQA,
   487→    editorial: EditorialSpecification
   488→  ): QAIssue[] {
   489→    const issues: QAIssue[] = [];
   490→    const usedFonts = new Set<string>();
   491→    const usedSizes = new Set<number>();
   492→
   493→    // Collect all fonts and sizes used
   494→    for (const heading of editorial.heading_styles) {
   495→      usedFonts.add(heading.font);
   496→      usedSizes.add(heading.size);
   497→    }
   498→    usedFonts.add(editorial.paragraph_style.font);
   499→    usedSizes.add(editorial.paragraph_style.size);
   500→
   501→    // Flag if too many different fonts (more than 3 is usually a problem)
   502→    if (usedFonts.size > 3) {
   503→      issues.push({
   504→        id: "qa-type-too-many-fonts",
   505→        category: "typographic_inconsistency",
   506→        severity: "warning",
   507→        location: { page: 0 },
   508→        description: `${usedFonts.size} different fonts used — recommend maximum 3 for coherence`,
   509→        suggested_fix: "Reduce font variety to heading + body + mono maximum",
   510→        auto_repairable: false,
   511→      });
   512→    }
   513→
   514→    return issues;
   515→  }

   517→  /**
   518→   * Check for numbering errors.
   519→   */
   520→  private checkNumberingErrors(
   521→    pdfSpec: PDFSpecForQA,
   522→    content: BookContent
   523→  ): QAIssue[] {
   524→    const issues: QAIssue[] = [];
   525→
   526→    // Verify chapter numbers are sequential
   527→    for (let i = 1; i < content.chapters.length; i++) {
   528→      const prev = content.chapters[i - 1].chapter_number;
   529→      const curr = content.chapters[i].chapter_number;
   530→
   531→      if (curr !== prev + 1) {
   532→        issues.push({
   533→          id: `qa-numbering-ch-${curr}`,
   534→          category: "numbering_errors",
   535→          severity: "critical",
   536→          location: { page: 0, chapter: curr },
   537→          description: `Chapter numbering gap: expected ${prev + 1}, found ${curr}`,
   538→          suggested_fix: "Renumber chapters sequentially",
   539→          auto_repairable: true,
   540→        });
   541→      }
   542→    }
   543→
   544→    return issues;
   545→  }

   547→  /**
   548→   * Check for legibility problems.
   549→   */
   550→  private checkLegibilityProblems(
   551→    pdfSpec: PDFSpecForQA,
   552→    editorial: EditorialSpecification
   553→  ): QAIssue[] {
   554→    const issues: QAIssue[] = [];
   555→
   556→    // Check body text size (minimum 9pt for print, 12pt for ebook)
   557→    if (editorial.paragraph_style.size < 9) {
   558→      issues.push({
   559→        id: "qa-legibility-body-size",
   560→        category: "legibility_problems",
   561→        severity: "critical",
   562→        location: { page: 0 },
   563→        description: `Body text size (${editorial.paragraph_style.size}pt) is below minimum readable size`,
   564→        suggested_fix: "Increase body text size to at least 9pt (print) or 12pt (ebook)",
   565→        auto_repairable: true,
   566→      });
   567→    }
   568→
   569→    // Check line height (minimum 1.2 for readability)
   570→    if (editorial.paragraph_style.line_height < 1.2) {
   571→      issues.push({
   572→        id: "qa-legibility-line-height",
   573→        category: "legibility_problems",
   574→        severity: "warning",
   575→        location: { page: 0 },
   576→        description: `Line height (${editorial.paragraph_style.line_height}) is below minimum readable value (1.2)`,
   577→        suggested_fix: "Increase line height to at least 1.2 for comfortable reading",
   578→        auto_repairable: true,
   579→      });
   580→    }
   581→
   582→    return issues;
   583→  }

   585→  // ─── Diagnosis ─────────────────────────────────────────────────────
   586→
   587→  /**
   588→   * Diagnose a single issue.
   589→   */
   590→  private diagnoseSingle(issue: QAIssue): Diagnosis {
   591→    const diagnoses: Partial<Record<QAIssueCategory, Diagnosis>> = {
   592→      cut_text: {
   593→        issue_id: issue.id,
   594→        root_cause: "insufficient_space",
   595→        explanation: "Text was cut because the available space on the page was insufficient to contain it",
   596→        repair_strategy: "reflow",
   597→        repair_params: { action: "split_or_shrink" },
   598→        confidence: 0.85,
   599→      },

   773→  /**
   774→   * Find source content by reference key.
   775→   */
   776→  private findSourceContent(
   777→    content: BookContent,
   778→    contentRef: string
   779→  ): string | null {
   780→    const match = contentRef.match(/chapter-(\d+)-para-(\d+)/);
   781→    if (match) {
   782→      const chapterNum = parseInt(match[1], 10);
   783→      const paraIdx = parseInt(match[2], 10);
   784→      const chapter = content.chapters.find((c) => c.chapter_number === chapterNum);
   785→      if (chapter) {
   786→        let paraCount = 0;
   787→        for (const section of chapter.sections) {
   788→          for (let i = 0; i < section.paragraphs.length; i++) {
   789→            if (paraCount === paraIdx) return section.paragraphs[i];
   790→            paraCount++;
   791→          }
   792→        }
   793→      }
   794→    }
   795→    return null;
   796→  }

   798→  /**
   799→   * Estimate minimum height for a text element.
   800→   */
   801→  private estimateMinHeight(text: string, editorial: EditorialSpecification): number {
   802→    const charsPerLine = 60; // conservative estimate
   803→    const lines = Math.max(1, Math.ceil(text.length / charsPerLine));
   804→    return lines * editorial.paragraph_style.size * editorial.paragraph_style.line_height;
   805→  }

   807→  /**
   808→   * Estimate WCAG contrast ratio (simplified).
   809→   */
   810→  private estimateContrast(foreground: string, background: string): number {
   811→    const fgLum = this.relativeLuminance(foreground);
   812→    const bgLum = this.relativeLuminance(background);
   813→    const lighter = Math.max(fgLum, bgLum);
   814→    const darker = Math.min(fgLum, bgLum);
   815→    return (lighter + 0.05) / (darker + 0.05);
   816→  }
   817→
   818→  private relativeLuminance(hex: string): number {
   819→    const r = this.sRGBtoLinear(parseInt(hex.slice(1, 3), 16) / 255);
   820→    const g = this.sRGBtoLinear(parseInt(hex.slice(3, 5), 16) / 255);
   821→    const b = this.sRGBtoLinear(parseInt(hex.slice(5, 7), 16) / 255);
   822→    return 0.2126 * r + 0.7152 * g + 0.0722 * b;
   823→  }
   824→
   825→  private sRGBtoLinear(c: number): number {
   826→    return c <= 0.04045 ? c / 12.92 : Math.pow((c + 0.055) / 1.055, 2.4);
   827→  }

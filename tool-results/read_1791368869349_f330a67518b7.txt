   266→   266→
   267→   267→export default function AdnDashboard({ user }: { user: User }) {
   268→   268→  const [activeTab, setActiveTab] = useState<TabId>("identity");
   269→   269→  const [loading, setLoading] = useState(true);
   270→   270→  const [doctorRunning, setDoctorRunning] = useState(false);
   271→   271→
   272→   272→  /* Data state */
   273→   273→  const [identity, setIdentity] = useState<Identity | null>(null);
   274→   274→  const [mission, setMission] = useState<Mission | null>(null);
   275→   275→  const [capabilities, setCapabilities] = useState<Capability[]>([]);
   276→   276→  const [doctor, setDoctor] = useState<DoctorResult | null>(null);
   277→   277→  const [memory, setMemory] = useState<MemoryStats[]>([]);
   278→   278→  const [skills, setSkills] = useState<Skill[]>([]);
   279→   279→  const [observability, setObservability] = useState<Observability | null>(null);
   280→   280→  const [mcp, setMcp] = useState<McpData | null>(null);
   281→   281→  const [security, setSecurity] = useState<SecurityData | null>(null);
   282→   282→  const [agentComm, setAgentComm] = useState<AgentComm | null>(null);
   283→   283→
   284→   284→  /* ── Fetch helpers ── */
   285→   285→  const fetchJson = useCallback(
   286→   286→    async (path: string): Promise<unknown> => {
   287→   287→      try {
   288→   288→        const res = await fetch(path);
   289→   289→        if (!res.ok) return null;
   290→   290→        return await res.json();
   291→   291→      } catch {
   292→   292→        return null;
   293→   293→      }
   294→   294→    },
   295→   295→    []
   296→   296→  );
   297→   297→
   298→   298→  /* ── Load all data ── */
   299→   299→  const loadAll = useCallback(async () => {
   300→   300→    setLoading(true);
   301→   301→    const [
   302→   302→      identRes,
   303→   303→      missionRes,
   304→   304→      capRes,
   305→   305→      docRes,
   306→   306→      memRes,
   307→   307→      skillsRes,
   308→   308→      obsRes,
   309→   309→      mcpRes,
   310→   310→      secRes,
   311→   311→      commRes,
   312→   312→    ] = await Promise.all([
   313→   313→      fetchJson("/api/adn/identity"),
   314→   314→      fetchJson("/api/adn/mission"),
   315→   315→      fetchJson("/api/adn/capabilities"),
   316→   316→      fetchJson("/api/adn/doctor"),
   317→   317→      fetchJson("/api/adn/memory?limit=50"),
   318→   318→      fetchJson("/api/adn/skills"),
   319→   319→      fetchJson("/api/adn/observability?range=24h"),
   320→   320→      fetchJson("/api/adn/mcp"),
   321→   321→      fetchJson("/api/adn/security"),
   322→   322→      fetchJson("/api/adn/agent-comm?limit=10"),
   323→   323→    ]);
   324→   324→
   325→   325→    setIdentity((identRes as Identity) ?? null);
   326→   326→    setMission((missionRes as Mission) ?? null);
   327→   327→    setCapabilities(
   328→   328→      Array.isArray(capRes) ? (capRes as Capability[]) : (capRes as { capabilities?: Capability[] })?.capabilities ?? []
   329→   329→    );
   330→   330→    setDoctor((docRes as DoctorResult) ?? null);
   331→   331→    setMemory(
   332→   332→      Array.isArray(memRes) ? (memRes as MemoryStats[]) : (memRes as { types?: MemoryStats[] })?.types ?? []
   333→   333→    );
   334→   334→    setSkills(
   335→   335→      Array.isArray(skillsRes) ? (skillsRes as Skill[]) : (skillsRes as { skills?: Skill[] })?.skills ?? []
   336→   336→    );
   337→   337→    setObservability((obsRes as Observability) ?? null);
   338→   338→    setMcp((mcpRes as McpData) ?? null);
   339→   339→    setSecurity((secRes as SecurityData) ?? null);
   340→   340→    setAgentComm((commRes as AgentComm) ?? null);
   341→   341→    setLoading(false);
   342→   342→  }, [fetchJson]);

   346→   346→  }, [loadAll]);

   438→   438→          )}
   439→   439→        </SectionCard>
   440→   440→      </div>

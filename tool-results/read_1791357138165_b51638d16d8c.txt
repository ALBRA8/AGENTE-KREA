   453→  /**
   454→   * generateHandoff — Create a handoff contract for agent delegation.
   455→   *
   456→   * AC-018: Handoff is reproducible — the receiver can understand the full

   458→   *
   459→   * AC-019: The contract includes objective, specification, constraints,
   460→   * required output, acceptance criteria, and evidence.
   461→   *
   462→   * AC-021: Evidence is included so the receiver understands WHY decisions
   463→   * were made, not just WHAT was decided.
   464→   */
   465→  generateHandoff(
   466→    specification: SoftwareSpecification | ProductSpecification,
   467→    targetAgent: "codex" | "antigravity" | "krea",
   468→    constructor?: ConstructorSelection,
   469→    architecture?: ProductArchitecture,
   470→    evidence: ProductEvidence[] = []
   471→  ): HandoffContract {
   472→    const productName = "productName" in specification
   473→      ? specification.productName
   474→      : specification.identity.name;
   475→
   476→    const isSoftware = "identity" in specification;
   477→    const softwareSpec = isSoftware ? specification as SoftwareSpecification : undefined;
   478→
   479→    // Determine autonomy level based on target agent and evidence quality
   480→    const autonomy = this.determineAutonomy(targetAgent, evidence);
   481→
   482→    // Determine approval requirements
   483→    const approvals = this.determineApprovals(targetAgent, autonomy);
   484→
   485→    // Build constraints
   486→    const outOfScope = architecture
   487→      ? architecture.OUT.map((item) => item.description)
   488→      : ["Features not in MUST list", "Multi-language support", "Mobile native apps"];

   490→    return {
   491→      handoff_id: randomUUID(),
   492→      source_agent: "krea",
   493→      target_agent: targetAgent,

   496→        name: productName,

   507→        ],
   508→        outOfScope,
   509→      },
   510→      required_output: {
   511→        deliverables: this.inferDeliverables(targetAgent, architecture),
   512→        format: this.inferOutputFormat(targetAgent),
   513→        qualityStandards: [
   514→          "All acceptance criteria met",
   515→          "Test coverage >= 80%",
   516→          "No regressions in existing functionality",
   517→          "Documentation for all public APIs",
   518→        ],
   519→      },
   520→      acceptance_criteria: architecture?.success_criteria || [
   521→        "Product meets all MUST requirements",
   522→        "All validation tests pass",
   523→        "User acceptance testing completed",
   524→      ],
   525→      evidence,
   526→      dependencies: architecture?.dependencies || [],
   527→      risk: {
   528→        level: this.assessHandoffRisk(targetAgent, evidence),
   529→        factors: architecture?.risks || ["Unknown technical complexity"],

   531→          "Phased delivery with milestones",

   533→          "Rollback plan for each phase",
   534→        ],
   535→      },
   536→      autonomy,
   537→      approvals,
   538→      status: "draft",
   539→      handedOffAt: new Date().toISOString(),
   540→    };
   541→  }

   545→  /** Build constructor-independent features from architecture (AC-016) */
   546→  private buildNeutralFeatures(architecture: ProductArchitecture): FeatureSpec[] {
   547→    const features: FeatureSpec[] = [];
   548→
   549→    for (const item of architecture.MUST) {
   550→      features.push(this.moscowToFeature(item, "MUST"));
   551→    }
   552→    for (const item of architecture.SHOULD) {
   553→      features.push(this.moscowToFeature(item, "SHOULD"));
   554→    }
   555→    for (const item of architecture.COULD) {
   556→      features.push(this.moscowToFeature(item, "COULD"));
   557→    }
   558→
   559→    return features;
   560→  }
   561→
   562→  /** Convert MoSCoW item to feature specification */
   563→  private moscowToFeature(item: MoscowItem, priority: "MUST" | "SHOULD" | "COULD"): FeatureSpec {
   564→    return {
   565→      name: item.description.substring(0, 80),
   566→      description: item.description,
   567→      userStories: [`As a user, I want ${item.description.toLowerCase()} so that ${item.rationale.toLowerCase()}`],
   568→      acceptanceCriteria: [
   569→        `Feature "${item.description}" is fully functional`,
   570→        `Feature passes all validation tests`,
   571→      ],
   572→      priority,
   573→      dependencies: [],
   574→      evidenceIds: item.evidenceIds,
   575→    };
   576→  }

   578→  /** Build software-specific features (AC-017) */
   579→  private buildSoftwareFeatures(architecture: ProductArchitecture): SoftwareFeature[] {
   580→    const features: SoftwareFeature[] = [];
   581→
   582→    const allItems = [
   583→      ...architecture.MUST.map((i) => ({ ...i, priority: "MUST" as const })),
   584→      ...architecture.SHOULD.map((i) => ({ ...i, priority: "SHOULD" as const })),
   585→      ...architecture.COULD.map((i) => ({ ...i, priority: "COULD" as const })),
   586→    ];
   587→
   588→    for (const item of allItems) {
   589→      features.push({
   590→        name: item.description.substring(0, 80),
   591→        description: item.description,
   592→        userStories: [`Given a user in the target audience, When they need ${item.description.toLowerCase()}, Then the product provides ${item.rationale.toLowerCase()}`],
   593→        acceptanceCriteria: [
   594→          `Feature works end-to-end`,
   595→          `Error handling is graceful`,
   596→          `Feature is accessible`,
   597→        ],
   598→        priority: item.priority,
   599→        technicalNotes: `Implementation approach determined by constructor. Effort: ${item.effort || "TBD"}.`,
   600→        dependencies: [],
   601→      });
   602→    }
   603→
   604→    return features;
   605→  }

   607→  /** Infer product type from format */
   608→  private inferProductType(format: string): string {
   609→    const lower = format.toLowerCase();
   610→    if (["app", "saas", "web app", "platform"].some((f) => lower.includes(f))) return "web_application";
   611→    if (["api", "backend", "service"].some((f) => lower.includes(f))) return "api_service";
   612→    if (["ebook", "guide", "template"].some((f) => lower.includes(f))) return "digital_product";
   613→    if (["course", "curso"].some((f) => lower.includes(f))) return "course";
   614→    return "product";
   615→  }

   621→      `Evaluator — comparing solutions before adopting`,
   622→    ];
   623→  }

   625→  /** Infer authentication requirement */
   626→  private inferAuthRequirement(format: string): string {

   628→    if (["ebook", "guide", "template", "course"].some((f) => lower.includes(f))) {
   629→      return "Email-based or social login for purchase access";

   643→  /** Build functional requirements */
   644→  private buildFunctionalRequirements(architecture: ProductArchitecture): string[] {
   645→    const requirements: string[] = [];
   646→
   647→    for (const item of architecture.MUST) {

    41→/* ─────────── Animated Counter ─────────── */
    42→function AnimatedCounter({
    43→  target,
    44→  suffix = "",
    45→  prefix = "",
    46→}: {
    47→  target: number;
    48→  suffix?: string;
    49→  prefix?: string;
    50→}) {
    51→  const [count, setCount] = useState(0);
    52→  const ref = useRef<HTMLSpanElement>(null);
    53→  const isInView = useInView(ref, { once: true });
    54→
    55→  useEffect(() => {
    56→    if (!isInView) return;
    57→    let start = 0;
    58→    const duration = 2000;
    59→    const step = target / (duration / 16);
    60→    const timer = setInterval(() => {
    61→      start += step;
    62→      if (start >= target) {
    63→        setCount(target);
    64→        clearInterval(timer);
    65→      } else {
    66→        setCount(Math.floor(start));
    67→      }
    68→    }, 16);
    69→    return () => clearInterval(timer);
    70→  }, [isInView, target]);
    71→
    72→  return (
    73→    <span ref={ref}>
    74→      {prefix}
    75→      {count.toLocaleString("es-ES")}
    76→      {suffix}
    77→    </span>
    78→  );
    79→}

    81→/* ─────────── Fade In Section ─────────── */
    82→function FadeInSection({
    83→  children,
    84→  className = "",
    85→  delay = 0,
    86→}: {
    87→  children: React.ReactNode;
    88→  className?: string;
    89→  delay?: number;
    90→}) {
    91→  const ref = useRef<HTMLDivElement>(null);
    92→  const isInView = useInView(ref, { once: true, margin: "-60px" });
    93→
    94→  return (
    95→    <motion.div
    96→      ref={ref}
    97→      initial={{ opacity: 0, y: 40 }}
    98→      animate={isInView ? { opacity: 1, y: 0 } : {}}
    99→      transition={{ duration: 0.6, delay, ease: "easeOut" }}
   100→      className={className}
   101→    >
   102→      {children}
   103→    </motion.div>
   104→  );
   105→}

   313→   313→  /** Build the complete ProductArchitecture from parsed parts */
   314→   314→  private buildArchitecture(
   315→   315→    opportunity: Opportunity,
   316→   316→    decision: ProductDecision,
   317→   317→    evidence: ProductEvidence[],
   318→   318→    parsed: Partial<ProductArchitecture>
   319→   319→  ): ProductArchitecture {
   320→   320→    const evidenceIds = evidence.map((e) => e.id);
   321→   321→
   322→   322→    // JTBD
   323→   323→    const jtbd: JobToBeDone[] = (parsed.jtbd || []).map((j: Record<string, unknown>) => ({
   324→   324→      statement: `When ${j.situation || "context"}, I want to ${j.motivation || "action"}, so I can ${j.outcome || "result"}`,
   325→   325→      situation: String(j.situation || ""),
   326→   326→      motivation: String(j.motivation || ""),
   327→   327→      outcome: String(j.outcome || ""),
   328→   328→      priority: (["primary", "secondary", "tertiary"].includes(String(j.priority)) ? j.priority : "primary") as JobToBeDone["priority"],
   329→   329→    }));
   330→   330→
   331→   331→    if (jtbd.length === 0) {
   332→   332→      jtbd.push({
   333→   333→        statement: `When facing ${opportunity.problem}, I want to ${opportunity.solution}, so I can solve my problem effectively`,
   334→   334→        situation: opportunity.problem,
   335→   335→        motivation: opportunity.solution,
   336→   336→        outcome: "Effective problem resolution",
   337→   337→        priority: "primary",
   338→   338→      });
   339→   339→    }
   340→   340→
   341→   341→    // MoSCoW items (AC-015)
   342→   342→    const MUST: MoscowItem[] = this.buildMoscowItems(parsed.MUST, [
   343→   343→      {
   344→   344→        description: `Core problem solution: ${opportunity.solution}`,
   345→   345→        rationale: "Solves the primary problem identified in the opportunity",
   346→   346→        evidenceIds,
   347→   347→        effort: "weeks",
   348→   348→        inMvp: true,
   349→   349→      },
   350→   350→    ], true, evidenceIds);
   351→   351→
   352→   352→    const SHOULD: MoscowItem[] = this.buildMoscowItems(parsed.SHOULD, [], false, evidenceIds);
   353→   353→
   354→   354→    const COULD: MoscowItem[] = this.buildMoscowItems(parsed.COULD, [], false, evidenceIds);
   355→   355→
   356→   356→    // OUT items — AC-015: explicitly excluded to prevent scope creep
   357→   357→    const OUT: MoscowItem[] = this.buildMoscowItems(parsed.OUT, [
   358→   358→      {
   359→   359→        description: "Features not related to the core problem statement",
   360→   360→        rationale: "Scope boundary — anything not directly addressing the core problem is OUT to prevent scope creep (AC-015)",
   361→   361→        evidenceIds: [],
   362→   362→        effort: "weeks",
   363→   363→        inMvp: false,
   364→   364→      },
   365→   365→      {
   366→   366→        description: "Multi-language support (unless core problem is language-specific)",
   367→   367→        rationale: "Internationalization is a post-MVP concern. Including it now violates scope control (AC-015)",
   368→   368→        evidenceIds: [],
   369→   369→        effort: "months",
   370→   370→        inMvp: false,
   371→   371→      },
   372→   372→    ], false, evidenceIds);
   373→   373→
   374→   374→    // MVP definition — subset of MUST items only (AC-015)
   375→   375→    const mvpItems = MUST.filter((item) => item.inMvp);
   376→   376→    const mvp_definition = {
   377→   377→      scope: `MVP: Core problem solution for ${opportunity.audience}`,
   378→   378→      items: mvpItems.length > 0 ? mvpItems : MUST.slice(0, 1),
   379→   379→      timebox: this.estimateMvpTimebox(opportunity, decision),
   380→   380→      successMetrics: (parsed.success_criteria as string[]) || [
   381→   381→        "Users can solve the core problem end-to-end",
   382→   382→        "Positive feedback from target audience",
   383→   383→        "Engagement metrics meet minimum threshold",
   384→   384→      ],
   385→   385→    };
   386→   386→
   387→   387→    return {
   388→   388→      client: opportunity.audience,
   389→   389→      problem: opportunity.problem,
   390→   390→      jtbd,
   391→   391→      desired_outcome: parsed.desired_outcome || `Effectively solve ${opportunity.problem} for ${opportunity.audience}`,
   392→   392→      value_proposition: parsed.value_proposition || `${opportunity.solution} — the most effective way for ${opportunity.audience} to solve ${opportunity.problem}`,
   393→   393→      format: opportunity.format || "unspecified",
   394→   394→      MUST,
   395→   395→      SHOULD,
   396→   396→      COULD,
   397→   397→      OUT,
   398→   398→      ux_description: parsed.ux_description || `Clean, focused interface that guides ${opportunity.audience} through solving ${opportunity.problem} with minimal friction`,
   399→   399→      differentiation: parsed.differentiation || this.inferDifferentiation(evidence),
   400→   400→      monetization: parsed.monetization || this.inferMonetization(opportunity, evidence),
   401→   401→      mvp_definition,
   402→   402→      dependencies: this.inferDependencies(opportunity, decision),
   403→   403→      risks: (parsed.risks as string[]) || this.inferRisks(evidence),
   404→   404→      assumptions: (parsed.assumptions as string[]) || [
   405→   405→        `The problem (${opportunity.problem}) is real and painful for ${opportunity.audience}`,
   406→   406→        `The proposed solution (${opportunity.solution}) effectively addresses the problem`,
   407→   407→        `${opportunity.audience} is willing to adopt a new solution`,
   408→   408→      ],
   409→   409→      unknowns: (parsed.unknowns as string[]) || decision.uncertainties,
   410→   410→      validation_requirements: this.buildValidationRequirements(decision, evidence),
   411→   411→      build_requirements: this.buildBuildRequirements(opportunity, decision, MUST),
   412→   412→      success_criteria: (parsed.success_criteria as string[]) || [
   413→   413→        "Core problem is solved for target audience",
   414→   414→        "User satisfaction score >= 7/10",
   415→   415→        "Retention rate >= 40% after 30 days",
   416→   416→        "Time-to-value < 5 minutes for new users",
   417→   417→      ],
   418→   418→      architectedAt: new Date().toISOString(),
   419→   419→    };
   420→   420→  }

   422→   422→  /** Build MoSCoW items from parsed data with defaults */
   423→   423→  private buildMoscowItems(
   424→   424→    parsedItems: unknown,
   425→   425→    defaults: MoscowItem[],
   426→   426→    isMust: boolean,
   427→   427→    evidenceIds: string[]
   428→   428→  ): MoscowItem[] {
   429→   429→    if (Array.isArray(parsedItems) && parsedItems.length > 0) {
   430→   430→      return parsedItems.map((item: Record<string, unknown>) => ({
   431→   431→        description: String(item.description || ""),
   432→   432→        rationale: String(item.rationale || "Derived from product analysis"),
   433→   433→        evidenceIds: (item.evidenceIds as string[]) || evidenceIds.slice(0, 3),
   434→   434→        effort: (["hours", "days", "weeks", "months"].includes(String(item.effort)) ? item.effort : "days") as MoscowItem["effort"],
   435→   435→        inMvp: isMust ? (item.inMvp !== false) : false,
   436→   436→      }));
   437→   437→    }
   438→   438→    return defaults;
   439→   439→  }

   441→   441→  /** Estimate MVP timebox based on product format */
   442→   442→  private estimateMvpTimebox(opportunity: Opportunity, decision: ProductDecision): string {
   443→   443→    const format = (opportunity.format || "").toLowerCase();
   444→   444→
   445→   445→    if (["ebook", "guía", "guide", "template"].some((f) => format.includes(f))) {
   446→   446→      return "1-2 weeks";
   447→   447→    }
   448→   448→    if (["workbook", "checklist", "course"].some((f) => format.includes(f))) {
   449→   449→      return "2-3 weeks";
   450→   450→    }
   451→   451→    if (["app", "web app", "tool", "landing page"].some((f) => format.includes(f))) {
   452→   452→      return "4-6 weeks";
   453→   453→    }
   454→   454→    if (["saas", "platform", "api"].some((f) => format.includes(f))) {
   455→   455→      return "8-12 weeks";
   456→   456→    }
   457→   457→    return "4-6 weeks";
   458→   458→  }

   460→   460→  /** Infer differentiation from evidence */
   461→   461→  private inferDifferentiation(evidence: ProductEvidence[]): string {
   462→   462→    const diffEvidence = evidence.filter((e) => e.type === "DIFFERENTIATION");
   463→   463→    if (diffEvidence.length > 0) {
   464→   464→      return diffEvidence.map((e) => e.content).join("; ");
   465→   465→    }
   466→   466→    return "Unique approach to solving the identified problem — differentiation to be refined based on competitive analysis";
   467→   467→  }

   469→   469→  /** Infer monetization model from opportunity and evidence */

   473→   473→      return `Based on market pricing evidence: ${pricingEvidence[0].content}`;
   474→   474→    }
   475→   475→

     1→/**

     3→ * ADN GENERAL DEL AGENTE V1.0
     4→ *
     5→ * Pipeline: AUDIT → DIAGNOSE → VERIFY → REPORT → SAFE_FIX (only when explicitly authorized)
     6→ *
     7→ * 15 real checks:
     8→ *   1. runtime        — process memory, uptime, event loop lag
     9→ *   2. database       — connection, integrity, user count, migration status
    10→ *   3. apis           — all /api routes responding
    11→ *   4. providers      — ZAI SDK available, adapters functional
    12→ *   5. mcp            — MCP server/client status
    13→ *   6. memory         — memory system functional, count by type
    14→ *   7. skills         — skill registry status, active skills count
    15→ *   8. permissions    — policy engine loaded, rules count
    16→ *   9. authentication — auth flow working, no plaintext passwords
    17→ *  10. integrations   — agent-to-agent comm status
    18→ *  11. configuration  — env vars, required config present
    19→ *  12. dependencies   — critical packages installed
    20→ *  13. health         — overall health score (computed)
    21→ *  14. persistence    — DB writable, file storage writable
    22→ *  15. critical_workflows — test a simple generation flow
    23→ */

    31→
    32→export type CheckStatus = "HEALTHY" | "DEGRADED" | "UNHEALTHY" | "UNKNOWN";
    33→export type TruthLevel = "VERIFIED" | "OBSERVED" | "ESTIMATED" | "UNKNOWN";
    34→
    35→export interface HealthCheckResultV2 {
    36→  name: string;
    37→  category: string;
    38→  status: CheckStatus;
    39→  latencyMs?: number;
    40→  details?: string;
    41→  error?: string;
    42→  evidence: {
    43→    checkedAt: string;
    44→    truthLevel: TruthLevel;
    45→    source: string;
    46→  };
    47→  metrics?: Record<string, number | string>;
    48→}
    49→
    50→export interface SafeFix {
    51→  diagnosis: string;
    52→  action: string;
    53→  before: string;
    54→  after: string;
    55→  timestamp: string;
    56→  executionId: string;
    57→  result: "SUCCESS" | "FAILED" | "SKIPPED";
    58→}
    59→
    60→export interface DoctorReportV2 {
    61→  agentId: string;
    62→  version: string;
    63→  timestamp: string;
    64→  overallStatus: CheckStatus;
    65→  checks: HealthCheckResultV2[];
    66→  healthScore: number; // 0-100
    67→  summary: string;
    68→  recommendations: string[];
    69→  safeFixes: SafeFix[];
    70→  pipeline: string;
    71→}
    72→
    73→export interface DoctorV2Options {
    74→  authorizeFixes?: boolean;
    75→  skipChecks?: string[];
    76→  timeout?: number;
    77→}

    87→
    88→  private options: DoctorV2Options;

    90→  constructor(options: DoctorV2Options = {}) {
    91→    this.options = {

    94→      timeout: 10000,

    99→  /**
   100→   * Run the full diagnostic pipeline.
   101→   */
   102→  async runFullCheck(): Promise<DoctorReportV2> {
   103→    const allChecks = [
   104→      this.checkRuntime(),
   105→      this.checkDatabase(),
   106→      this.checkAPIs(),
   107→      this.checkProviders(),
   108→      this.checkMCP(),
   109→      this.checkMemory(),
   110→      this.checkSkills(),
   111→      this.checkPermissions(),
   112→      this.checkAuthentication(),
   113→      this.checkIntegrations(),
   114→      this.checkConfiguration(),
   115→      this.checkDependencies(),
   116→      this.checkPersistence(),
   117→      this.checkCriticalWorkflows(),
   118→    ];
   119→
   120→    // Filter skipped checks
   121→    const filteredChecks = allChecks.filter((check) => {
   122→      // We can't know the name before execution, so we run all
   123→      // and filter after if needed
   124→      return true;
   125→    });
   126→
   127→    const results = await Promise.allSettled(filteredChecks);
   128→    const checks: HealthCheckResultV2[] = results.map((r, i) => {
   129→      if (r.status === "fulfilled") return r.value;
   130→      return {
   131→        name: `check_${i}`,
   132→        category: "unknown",
   133→        status: "UNKNOWN" as CheckStatus,
   134→        error: r.reason?.message || "Unknown error",
   135→        evidence: {
   136→          checkedAt: new Date().toISOString(),
   137→          truthLevel: "UNKNOWN" as TruthLevel,
   138→          source: "doctor-v2",
   139→        },
   140→      };
   141→    });
   142→
   143→    // Filter out skipped checks
   144→    const activeChecks = checks.filter(
   145→      (c) => !this.options.skipChecks?.includes(c.name)
   146→    );
   147→
   148→    // Compute overall status
   149→    const unhealthy = activeChecks.filter((c) => c.status === "UNHEALTHY").length;
   150→    const degraded = activeChecks.filter((c) => c.status === "DEGRADED").length;
   151→    const healthy = activeChecks.filter((c) => c.status === "HEALTHY").length;
   152→
   153→    const overallStatus: CheckStatus =
   154→      unhealthy > 0 ? "UNHEALTHY" : degraded > 0 ? "DEGRADED" : "HEALTHY";
   155→
   156→    // Compute health score (weighted)
   157→    const healthScore = Math.round(
   158→      (healthy * 100 + degraded * 50 + unhealthy * 0) / activeChecks.length
   159→    );
   160→
   161→    // Generate recommendations
   162→    const recommendations = this.generateRecommendations(activeChecks);
   163→
   164→    // Run safe fixes only if authorized
   165→    const safeFixes: SafeFix[] = this.options.authorizeFixes
   166→      ? await this.runSafeFixes(activeChecks)
   167→      : [];
   168→
   169→    // Add the computed health check
   170→    const healthCheck: HealthCheckResultV2 = {
   171→      name: "health",
   172→      category: "system",
   173→      status: overallStatus,
   174→      details: `Score: ${healthScore}/100 — ${healthy} healthy, ${degraded} degraded, ${unhealthy} unhealthy`,
   175→      metrics: {
   176→        score: healthScore,
   177→        healthy,
   178→        degraded,
   179→        unhealthy,
   180→        total: activeChecks.length,
   181→      },
   182→      evidence: {
   183→        checkedAt: new Date().toISOString(),
   184→        truthLevel: "VERIFIED",
   185→        source: "doctor-v2",
   186→      },
   187→    };
   188→
   189→    return {
   190→      agentId: this.agentId,
   191→      version: this.version,
   192→      timestamp: new Date().toISOString(),
   193→      overallStatus,
   194→      checks: [...activeChecks, healthCheck],
   195→      healthScore,
   196→      summary: `${activeChecks.length + 1} checks: ${healthy} healthy, ${degraded} degraded, ${unhealthy} unhealthy (score: ${healthScore}/100)`,
   197→      recommendations,
   198→      safeFixes,
   199→      pipeline: this.pipeline,
   200→    };
   201→  }

   203→  // ──────────────────────────────────────────────
   204→  // CHECK 1: Runtime
   205→  // ──────────────────────────────────────────────
   206→  async checkRuntime(): Promise<HealthCheckResultV2> {
   207→    const start = Date.now();
   208→    try {
   209→      const mem = process.memoryUsage();
   210→      const uptime = process.uptime();
   211→      const rssMB = Math.round(mem.rss / 1024 / 1024);
   212→      const heapUsedMB = Math.round(mem.heapUsed / 1024 / 1024);
   213→      const heapTotalMB = Math.round(mem.heapTotal / 1024 / 1024);
   214→      const heapRatio = mem.heapUsed / mem.heapTotal;
   215→
   216→      // Event loop lag estimation
   217→      const loopStart = Date.now();
   218→      await new Promise((resolve) => setImmediate(resolve));
   219→      const loopLag = Date.now() - loopStart;
   220→
   221→      const isMemoryHigh = heapRatio > 0.9;
   222→      const isLoopSlow = loopLag > 50;
   223→
   224→      return {
   225→        name: "runtime",
   226→        category: "system",
   227→        status: isMemoryHigh ? "UNHEALTHY" : isLoopSlow ? "DEGRADED" : "HEALTHY",
   228→        latencyMs: Date.now() - start,
   229→        details: `RSS: ${rssMB}MB, Heap: ${heapUsedMB}/${heapTotalMB}MB (${Math.round(heapRatio * 100)}%), Uptime: ${Math.round(uptime)}s, EventLoop lag: ${loopLag}ms`,
   230→        metrics: {
   231→          rssMB,
   232→          heapUsedMB,
   233→          heapTotalMB,
   234→          heapRatio: Math.round(heapRatio * 100),
   235→          uptimeSec: Math.round(uptime),
   236→          eventLoopLagMs: loopLag,
   237→        },
   238→        evidence: {
   239→          checkedAt: new Date().toISOString(),
   240→          truthLevel: "VERIFIED",
   241→          source: "process.memoryUsage",
   242→        },
   243→      };
   244→    } catch (e: any) {
   245→      return {
   246→        name: "runtime",
   247→        category: "system",
   248→        status: "UNHEALTHY",
   249→        latencyMs: Date.now() - start,
   250→        error: e.message,
   251→        evidence: {
   252→          checkedAt: new Date().toISOString(),
   253→          truthLevel: "VERIFIED",
   254→          source: "doctor-v2",
   255→        },
   256→      };
   257→    }
   258→  }

   263→  async checkDatabase(): Promise<HealthCheckResultV2> {
   264→    const start = Date.now();
   265→    try {
   266→      // Basic connectivity
   267→      await db.$queryRaw`SELECT 1`;
   268→
   269→      // Integrity check (SQLite specific)
   270→      const integrityResult = await db.$queryRawUnsafe<
   271→        { integrity_check: string }[]
   272→      >("PRAGMA integrity_check");
   273→      const integrityOk = integrityResult[0]?.integrity_check === "ok";
   274→
   275→      // Counts
   276→      const userCount = await db.user.count();
   277→      const generationCount = await db.generation.count();
   278→      const memoryCount = await db.memory.count();
   279→
   280→      // Migration status — check if all models are accessible
   281→      let migrationOk = true;
   282→      try {
   283→        await db.auditEvent.count();
   284→        await db.agentInteraction.count();
   285→        await db.skill.count();
   286→        await db.execution.count();
   287→        await db.feedback.count();
   288→        await db.session.count();
   289→      } catch {
   290→        migrationOk = false;
   291→      }
   292→
   293→      const status: CheckStatus =
   294→        !integrityOk ? "UNHEALTHY" : !migrationOk ? "DEGRADED" : "HEALTHY";
   295→
   296→      return {
   297→        name: "database",
   298→        category: "persistence",
   299→        status,
   300→        latencyMs: Date.now() - start,
   301→        details: `SQLite ${integrityOk ? "integrity OK" : "INTEGRITY FAIL"}, ${userCount} users, ${generationCount} generations, ${memoryCount} memories, migrations ${migrationOk ? "OK" : "INCOMPLETE"}`,
   302→        metrics: {

   304→          generationCount,

   306→          integrityOk: integrityOk ? 1 : 0,
   307→          migrationsOk: migrationOk ? 1 : 0,

   309→        evidence: {
   310→          checkedAt: new Date().toISOString(),
   311→          truthLevel: "VERIFIED",
   312→          source: "prisma",
   313→        },
   314→      };
   315→    } catch (e: any) {

   317→        name: "database",

   344→
   345→    let healthy = 0;
   346→    let degraded = 0;
   347→    let unhealthy = 0;
   348→    const details: string[] = [];
   349→
   350→    for (const route of routes) {
   351→      try {
   352→        const res = await fetch(`http://localhost:3000${route.path}`, {
   353→          method: route.method,
   354→          signal: AbortSignal.timeout(5000),
   355→        });
   356→        if (res.ok) {
   357→          healthy++;
   358→          details.push(`${route.path}: OK`);
   359→        } else {
   360→          degraded++;
   361→          details.push(`${route.path}: HTTP ${res.status}`);
   362→        }
   363→      } catch {
   364→        unhealthy++;
   365→        details.push(`${route.path}: UNREACHABLE`);
   366→      }
   367→    }
   368→
   369→    return {
   370→      name: "apis",
   371→      category: "network",
   372→      status: unhealthy > 0 ? "UNHEALTHY" : degraded > 0 ? "DEGRADED" : "HEALTHY",
   373→      latencyMs: Date.now() - start,
   374→      details: `${routes.length} routes: ${healthy} ok, ${degraded} degraded, ${unhealthy} unreachable`,
   375→      metrics: { healthy, degraded, unhealthy, total: routes.length },
   376→      evidence: {
   377→        checkedAt: new Date().toISOString(),
   378→        truthLevel: "OBSERVED",
   379→        source: "fetch",
   380→      },

   981→  // ──────────────────────────────────────────────
   982→  // Recommendations Engine
   983→  // ──────────────────────────────────────────────
   984→  private generateRecommendations(checks: HealthCheckResultV2[]): string[] {
   985→    const recs: string[] = [];
   986→
   987→    for (const check of checks) {
   988→      if (check.status === "HEALTHY") continue;
   989→
   990→      switch (check.name) {
   991→        case "runtime":
   992→          recs.push("Consider restarting the process — high memory usage detected");
   993→          break;
   994→        case "database":
   995→          if (check.error) recs.push("Database unreachable — check DATABASE_URL and file permissions");
   996→          recs.push("Run PRAGMA integrity_check on SQLite to verify database integrity");
   997→          break;
   998→        case "apis":
   999→          recs.push("One or more API routes are not responding — check Next.js dev server logs");
  1000→          break;
  1001→        case "providers":
  1002→          recs.push("Z-AI SDK initialization failed — verify API keys and network access");
  1003→          break;
  1004→        case "mcp":
  1005→          recs.push("MCP modules not loaded — ensure mcp-server.ts and mcp-client.ts are available");
  1006→          break;
  1007→        case "memory":
  1008→          recs.push("No memories stored yet — this is normal for a fresh install");
  1009→          break;
  1010→        case "skills":
  1011→          recs.push("No skills registered — skill registry is empty");
  1012→          break;
  1013→        case "permissions":
  1014→          recs.push("Security module not loaded — permission checks will use defaults");
  1015→          break;
  1016→        case "authentication":
  1017→          if (check.metrics?.weakPasswordCount && Number(check.metrics.weakPasswordCount) > 0) {
  1018→            recs.push("⚠️ CRITICAL: Users with unhashed passwords detected — hash all passwords immediately");
  1019→          }
  1020→          break;
  1021→        case "integrations":
  1022→          recs.push("Agent communication module not available — inter-agent delegation disabled");
  1023→          break;
  1024→        case "configuration":
  1025→          recs.push("Missing environment variables — check .env file");
  1026→          break;
  1027→        case "dependencies":
  1028→          recs.push("Missing critical dependencies — run bun install");
  1029→          break;
  1030→        case "persistence":
  1031→          recs.push("Storage not writable — check file system permissions");
  1032→          break;
  1033→        case "critical_workflows":
  1034→          recs.push("Critical workflows failing — review individual check results for details");
  1035→          break;
  1036→      }
  1037→    }
  1038→
  1039→    if (recs.length === 0 && checks.some((c) => c.status === "DEGRADED")) {
  1040→      recs.push("System is degraded but functional — monitor for further degradation");
  1041→    }
  1042→
  1043→    return recs;
  1044→  }

  1047→  // Safe Fixes (only when authorized)
  1048→  // ──────────────────────────────────────────────
  1049→  private async runSafeFixes(checks: HealthCheckResultV2[]): Promise<SafeFix[]> {
  1050→    const fixes: SafeFix[] = [];
  1051→    const executionId = crypto.randomUUID();
  1052→
  1053→    for (const check of checks) {
  1054→      if (check.status === "HEALTHY") continue;
  1055→
  1056→      switch (check.name) {
  1057→        case "persistence": {
  1058→          // Try to create missing directories
  1059→          try {
  1060→            const fs = await import("fs/promises");
  1061→            const path = await import("path");
  1062→            const genDir = path.join(process.cwd(), "public", "generated");
  1063→            const before = "Directory may not exist";
  1064→            await fs.mkdir(genDir, { recursive: true });
  1065→            fixes.push({
  1066→              diagnosis: "public/generated/ directory may not exist",
  1067→              action: "Created public/generated/ directory recursively",
  1068→              before,
  1069→              after: "Directory exists and is writable",
  1070→              timestamp: new Date().toISOString(),
  1071→              executionId,
  1072→              result: "SUCCESS",
  1073→            });
  1074→          } catch (e: any) {
  1075→            fixes.push({
  1076→              diagnosis: "Cannot create public/generated/ directory",
  1077→              action: "Attempted mkdir recursive",
  1078→              before: "Directory missing",
  1079→              after: "Still missing",
  1080→              timestamp: new Date().toISOString(),
  1081→              executionId,
  1082→              result: "FAILED",
  1083→            });
  1084→          }
  1085→          break;
  1086→        }
  1087→
  1088→        case "configuration": {
  1089→          // Set DATABASE_URL if missing
  1090→          if (!process.env.DATABASE_URL) {
  1091→            const before = "DATABASE_URL not set";
  1092→            process.env.DATABASE_URL = `file:${process.cwd()}/db/custom.db`;
  1093→            fixes.push({
  1094→              diagnosis: "DATABASE_URL environment variable not set",
  1095→              action: "Set DATABASE_URL to default SQLite path",
  1096→              before,
  1097→              after: process.env.DATABASE_URL,
  1098→              timestamp: new Date().toISOString(),
  1099→              executionId,
  1100→              result: "SUCCESS",
  1101→            });
  1102→          }
  1103→          break;
  1104→        }
  1105→
  1106→        // For other checks, we only diagnose — no auto-fix
  1107→        default: {
  1108→          fixes.push({
  1109→            diagnosis: check.error || check.details || `${check.name} is ${check.status}`,
  1110→            action: "No auto-fix available — manual intervention required",
  1111→            before: check.status,
  1112→            after: check.status,
  1113→            timestamp: new Date().toISOString(),
  1114→            executionId,
  1115→            result: "SKIPPED",
  1116→          });
  1117→          break;
  1118→        }
  1119→      }
  1120→    }
  1121→
  1122→    return fixes;
  1123→  }

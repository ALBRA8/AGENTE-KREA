    71→function formatDate(d: string): string {
    72→  try {
    73→    return new Date(d).toLocaleDateString("es-ES", {
    74→      day: "2-digit",
    75→      month: "short",
    76→      year: "numeric",
    77→      hour: "2-digit",
    78→      minute: "2-digit",
    79→    });
    80→  } catch {
    81→    return d;
    82→  }

    85→function statusColor(status: string): string {
    86→  const s = status.toUpperCase();
    87→  if (["COMPLETED", "GO", "PUBLISHED", "ACKNOWLEDGED"].includes(s)) return "text-emerald-400";
    88→  if (["FAILED", "NO_GO", "ARCHIVED"].includes(s)) return "text-red-400";
    89→  if (["RUNNING", "PRODUCING", "SENT", "CONDITIONAL_GO"].includes(s)) return "text-amber-400";
    90→  if (["PENDING", "CREATED", "IDEA", "READY"].includes(s)) return "text-blue-400";
    91→  return "text-white/50";
    92→}
    93→
    94→function statusBg(status: string): string {
    95→  const s = status.toUpperCase();
    96→  if (["COMPLETED", "GO", "PUBLISHED", "ACKNOWLEDGED"].includes(s)) return "bg-emerald-400/10 text-emerald-400 border-emerald-400/20";
    97→  if (["FAILED", "NO_GO", "ARCHIVED"].includes(s)) return "bg-red-400/10 text-red-400 border-red-400/20";
    98→  if (["RUNNING", "PRODUCING", "SENT", "CONDITIONAL_GO"].includes(s)) return "bg-amber-400/10 text-amber-400 border-amber-400/20";
    99→  if (["PENDING", "CREATED", "IDEA", "READY"].includes(s)) return "bg-blue-400/10 text-blue-400 border-blue-400/20";
   100→  return "bg-white/5 text-white/50 border-white/10";
   101→}
   102→
   103→function fitScoreColor(score: number): string {
   104→  if (score >= 0.7) return "text-emerald-400";
   105→  if (score >= 0.4) return "text-amber-400";
   106→  return "text-red-400";
   107→}

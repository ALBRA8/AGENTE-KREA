    40→/* ─────────── Animated Counter ─────────── */
    41→function AnimatedCounter({
    42→  target,
    43→  suffix = "",
    44→  prefix = "",
    45→}: {
    46→  target: number;
    47→  suffix?: string;
    48→  prefix?: string;
    49→}) {
    50→  const [count, setCount] = useState(0);
    51→  const ref = useRef<HTMLSpanElement>(null);
    52→  const isInView = useInView(ref, { once: true });
    53→
    54→  useEffect(() => {
    55→    if (!isInView) return;
    56→    let start = 0;
    57→    const duration = 2000;
    58→    const step = target / (duration / 16);
    59→    const timer = setInterval(() => {
    60→      start += step;
    61→      if (start >= target) {
    62→        setCount(target);
    63→        clearInterval(timer);
    64→      } else {
    65→        setCount(Math.floor(start));
    66→      }
    67→    }, 16);
    68→    return () => clearInterval(timer);
    69→  }, [isInView, target]);
    70→
    71→  return (
    72→    <span ref={ref}>
    73→      {prefix}
    74→      {count.toLocaleString("es-ES")}
    75→      {suffix}
    76→    </span>
    77→  );
    78→}

    80→/* ─────────── Fade In Section ─────────── */
    81→function FadeInSection({
    82→  children,
    83→  className = "",
    84→  delay = 0,
    85→}: {
    86→  children: React.ReactNode;
    87→  className?: string;
    88→  delay?: number;
    89→}) {
    90→  const ref = useRef<HTMLDivElement>(null);
    91→  const isInView = useInView(ref, { once: true, margin: "-60px" });
    92→
    93→  return (
    94→    <motion.div
    95→      ref={ref}
    96→      initial={{ opacity: 0, y: 40 }}
    97→      animate={isInView ? { opacity: 1, y: 0 } : {}}
    98→      transition={{ duration: 0.6, delay, ease: "easeOut" }}
    99→      className={className}
   100→    >
   101→      {children}
   102→    </motion.div>
   103→  );
   104→}

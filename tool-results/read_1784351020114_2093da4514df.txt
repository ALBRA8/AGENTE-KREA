    37→/* ─────────── Animated Counter ─────────── */
    38→function AnimatedCounter({
    39→  target,
    40→  suffix = "",
    41→  prefix = "",
    42→}: {
    43→  target: number;
    44→  suffix?: string;
    45→  prefix?: string;
    46→}) {
    47→  const [count, setCount] = useState(0);
    48→  const ref = useRef<HTMLSpanElement>(null);
    49→  const isInView = useInView(ref, { once: true });
    50→
    51→  useEffect(() => {
    52→    if (!isInView) return;
    53→    let start = 0;
    54→    const duration = 2000;
    55→    const step = target / (duration / 16);
    56→    const timer = setInterval(() => {
    57→      start += step;
    58→      if (start >= target) {
    59→        setCount(target);
    60→        clearInterval(timer);
    61→      } else {
    62→        setCount(Math.floor(start));
    63→      }
    64→    }, 16);
    65→    return () => clearInterval(timer);
    66→  }, [isInView, target]);
    67→
    68→  return (
    69→    <span ref={ref}>
    70→      {prefix}
    71→      {count.toLocaleString("es-ES")}
    72→      {suffix}
    73→    </span>
    74→  );
    75→}

    77→/* ─────────── Fade In Section ─────────── */
    78→function FadeInSection({
    79→  children,
    80→  className = "",
    81→  delay = 0,
    82→}: {
    83→  children: React.ReactNode;
    84→  className?: string;
    85→  delay?: number;
    86→}) {
    87→  const ref = useRef<HTMLDivElement>(null);
    88→  const isInView = useInView(ref, { once: true, margin: "-60px" });
    89→
    90→  return (
    91→    <motion.div
    92→      ref={ref}
    93→      initial={{ opacity: 0, y: 40 }}
    94→      animate={isInView ? { opacity: 1, y: 0 } : {}}
    95→      transition={{ duration: 0.6, delay, ease: "easeOut" }}
    96→      className={className}
    97→    >
    98→      {children}
    99→    </motion.div>
   100→  );
   101→}

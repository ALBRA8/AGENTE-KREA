     1→     1→/**
     2→     2→ * Feedback & Learning System for AGENTE-KREA
     3→     3→ *
     4→     4→ * Records feedback on agent executions and extracts learning.
     5→     5→ * Distinguishes real success from partial success from failure.
     6→     6→ * Never automatically converts any result into truth — all learning must be validated.
     7→     7→ *
     8→     8→ * Feedback types: SUCCESS, PARTIAL_SUCCESS, FAILURE, TECHNICAL_ERROR,
     9→     9→ *   BAD_DECISION, BAD_HYPOTHESIS, BAD_TOOL, BAD_EXECUTION, UNKNOWN
    10→    10→ */
    11→    11→
    12→    12→import { db } from "@/lib/db";
    13→    13→import { MemoryManager, type MemoryType, type SourceType, type TruthLevel } from "@/lib/memory";
    14→    14→import { SkillManager } from "@/lib/skills";
    15→    15→
    16→    16→// ─── Type Definitions ────────────────────────────────────────────────
    17→    17→
    18→    18→export type FeedbackType =
    19→    19→  | "SUCCESS"
    20→    20→  | "PARTIAL_SUCCESS"
    21→    21→  | "FAILURE"
    22→    22→  | "TECHNICAL_ERROR"
    23→    23→  | "BAD_DECISION"
    24→    24→  | "BAD_HYPOTHESIS"
    25→    25→  | "BAD_TOOL"
    26→    26→  | "BAD_EXECUTION"
    27→    27→  | "UNKNOWN";
    28→    28→
    29→    29→export type ValidationStatus = "PENDING" | "VALIDATED" | "INVALIDATED";
    30→    30→
    31→    31→export interface RecordFeedbackOptions {
    32→    32→  agentId?: string;
    33→    33→  userId?: string;
    34→    34→  input?: Record<string, unknown>;
    35→    35→  result?: Record<string, unknown>;
    36→    36→  feedbackText?: string;
    37→    37→  confidence?: number;
    38→    38→  evidence?: Record<string, unknown>;
    39→    39→}
    40→    40→
    41→    41→export interface FeedbackAnalysis {
    42→    42→  category: FeedbackType;
    43→    43→  severity: "low" | "medium" | "high" | "critical";
    44→    44→  rootCause?: string;
    45→    45→  contributingFactors?: string[];
    46→    46→  isRecoverable: boolean;
    47→    47→  patterns?: string[];
    48→    48→  recommendations?: string[];
    49→    49→  estimatedConfidence: number;
    50→    50→}
    51→    51→
    52→    52→export interface ExtractedLearning {
    53→    53→  domain: string;
    54→    54→  type: MemoryType;
    55→    55→  content: Record<string, unknown>;
    56→    56→  source: string;
    57→    57→  sourceType: SourceType;
    58→    58→  truthLevel: TruthLevel;
    59→    59→  confidence: number;
    60→    60→  needsValidation: boolean;
    61→    61→  affectedSkills?: string[];
    62→    62→  affectedMemories?: string[];
    63→    63→}
    64→    64→
    65→    65→// ─── Feedback Category Severity ──────────────────────────────────────
    66→    66→
    67→    67→const FEEDBACK_SEVERITY: Record<FeedbackType, "low" | "medium" | "high" | "critical"> = {
    68→    68→  SUCCESS: "low",
    69→    69→  PARTIAL_SUCCESS: "medium",
    70→    70→  FAILURE: "high",
    71→    71→  TECHNICAL_ERROR: "high",
    72→    72→  BAD_DECISION: "high",
    73→    73→  BAD_HYPOTHESIS: "critical",
    74→    74→  BAD_TOOL: "high",
    75→    75→  BAD_EXECUTION: "critical",
    76→    76→  UNKNOWN: "medium",
    77→    77→};
    78→    78→
    79→    79→// ─── FeedbackManager ─────────────────────────────────────────────────
    80→    80→
    81→    81→export class FeedbackManager {
    82→    82→  private defaultAgentId = "krea";
    83→    83→  private memoryManager = new MemoryManager();
    84→    84→  private skillManager = new SkillManager();
    85→    85→
    86→    86→  /**
    87→    87→   * Record feedback on an execution.
    88→    88→   * Creates a Feedback entry linked to the execution (if provided).
    89→    89→   */
    90→    90→  async record(
    91→    91→    executionId: string | null,
    92→    92→    feedbackType: FeedbackType,
    93→    93→    feedbackText: string,
    94→    94→    options: RecordFeedbackOptions = {}
    95→    95→  ): Promise<{ id: string }> {
    96→    96→    const agentId = options.agentId || this.defaultAgentId;
    97→    97→
    98→    98→    // Validate execution exists if provided
    99→    99→    if (executionId) {
   100→   100→      const execution = await db.execution.findUnique({ where: { id: executionId } });
   101→   101→      if (!execution) throw new Error(`Execution ${executionId} not found`);
   102→   102→    }
   103→   103→
   104→   104→    const feedback = await db.feedback.create({
   105→   105→      data: {
   106→   106→        agentId,
   107→   107→        userId: options.userId || null,
   108→   108→        executionId,
   109→   109→        input: options.input ? JSON.stringify(options.input) : null,
   110→   110→        result: options.result ? JSON.stringify(options.result) : null,
   111→   111→        feedbackType,
   112→   112→        feedbackText,
   113→   113→        confidence: options.confidence ?? this.inferConfidence(feedbackType),
   114→   114→        evidence: options.evidence ? JSON.stringify(options.evidence) : null,
   115→   115→        validationStatus: "PENDING",
   116→   116→      },
   117→   117→    });
   118→   118→
   119→   119→    // Auto-analyze for non-success feedback
   120→   120→    if (feedbackType !== "SUCCESS") {
   121→   121→      await this.analyze(feedback.id);
   122→   122→    }
   123→   123→
   124→   124→    return { id: feedback.id };
   125→   125→  }
   126→   126→
   127→   127→  /**
   128→   128→   * Automatically analyze feedback to determine root cause, severity, and patterns.
   129→   129→   */
   130→   130→  async analyze(feedbackId: string): Promise<FeedbackAnalysis> {
   131→   131→    const feedback = await db.feedback.findUnique({ where: { id: feedbackId } });
   132→   132→    if (!feedback) throw new Error(`Feedback ${feedbackId} not found`);
   133→   133→
   134→   134→    const analysis: FeedbackAnalysis = {
   135→   135→      category: feedback.feedbackType as FeedbackType,
   136→   136→      severity: FEEDBACK_SEVERITY[feedback.feedbackType as FeedbackType],
   137→   137→      isRecoverable: true,
   138→   138→      estimatedConfidence: feedback.confidence,
   139→   139→    };
   140→   140→
   141→   141→    // Analyze based on feedback type
   142→   142→    switch (feedback.feedbackType as FeedbackType) {
   143→   143→      case "SUCCESS":
   144→   144→        analysis.isRecoverable = true;
   145→   145→        analysis.recommendations = ["Record successful pattern for reuse"];
   146→   146→        break;
   147→   147→
   148→   148→      case "PARTIAL_SUCCESS":
   149→   149→        analysis.isRecoverable = true;
   150→   150→        analysis.rootCause = this.identifyPartialSuccessCause(feedback);
   151→   151→        analysis.recommendations = [
   152→   152→          "Review which steps succeeded and which failed",
   153→   153→          "Consider adjusting parameters for the failing steps",
   154→   154→        ];
   155→   155→        break;
   156→   156→
   157→   157→      case "FAILURE":
   158→   158→        analysis.isRecoverable = false;
   159→   159→        analysis.rootCause = this.identifyFailureRootCause(feedback);
   160→   160→        analysis.contributingFactors = this.identifyContributingFactors(feedback);
   161→   161→        analysis.recommendations = [
   162→   162→          "Review execution trace for error details",
   163→   163→          "Consider alternative approaches",
   164→   164→          "Check if prerequisites were met",
   165→   165→        ];
   166→   166→        break;
   167→   167→
   168→   168→      case "TECHNICAL_ERROR":
   169→   169→        analysis.isRecoverable = true;
   170→   170→        analysis.rootCause = "Technical/system error";
   171→   171→        analysis.recommendations = [
   172→   172→          "Retry with backoff",
   173→   173→          "Check system health",
   174→   174→          "Verify external service availability",
   175→   175→        ];
   176→   176→        break;
   177→   177→
   178→   178→      case "BAD_DECISION":
   179→   179→        analysis.isRecoverable = false;
   180→   180→        analysis.severity = "high";
   181→   181→        analysis.rootCause = "Agent made an incorrect decision";
   182→   182→        analysis.patterns = this.identifyDecisionPatterns(feedback);
   183→   183→        analysis.recommendations = [
   184→   184→          "Review decision criteria and thresholds",
   185→   185→          "Update decision logic based on evidence",
   186→   186→          "Add constraints to prevent similar decisions",
   187→   187→        ];
   188→   188→        break;
   189→   189→
   190→   190→      case "BAD_HYPOTHESIS":
   191→   191→        analysis.isRecoverable = false;
   192→   192→        analysis.severity = "critical";
   193→   193→        analysis.rootCause = "Agent operated on an incorrect hypothesis";
   194→   194→        analysis.recommendations = [
   195→   195→          "Invalidate the hypothesis in memory",
   196→   196→          "Record contradictory evidence",
   197→   197→          "Adjust confidence for related hypotheses",
   198→   198→        ];
   199→   199→        break;
   200→   200→
   201→   201→      case "BAD_TOOL":
   202→   202→        analysis.isRecoverable = true;
   203→   203→        analysis.rootCause = "Wrong tool was selected or tool malfunctioned";
   204→   204→        analysis.recommendations = [
   205→   205→          "Review tool selection criteria",
   206→   206→          "Consider alternative tools",
   207→   207→          "Report tool issue",
   208→   208→        ];
   209→   209→        break;
   210→   210→
   211→   211→      case "BAD_EXECUTION":
   212→   212→        analysis.isRecoverable = false;
   213→   213→        analysis.severity = "critical";
   214→   214→        analysis.rootCause = "Execution was flawed in structure or sequence";
   215→   215→        analysis.recommendations = [
   216→   216→          "Review execution procedure",
   217→   217→          "Check step ordering and dependencies",
   218→   218→          "Validate execution plan before running",
   219→   219→        ];
   220→   220→        break;
   221→   221→
   222→   222→      default:
   223→   223→        analysis.recommendations = ["Investigate further", "Gather more evidence"];
   224→   224→    }
   225→   225→
   226→   226→    // Store analysis in the feedback record
   227→   227→    await db.feedback.update({
   228→   228→      where: { id: feedbackId },
   229→   229→      data: {
   230→   230→        analysis: JSON.stringify(analysis),
   231→   231→      },
   232→   232→    });
   233→   233→
   234→   234→    return analysis;
   235→   235→  }
   236→   236→
   237→   237→  /**
   238→   238→   * Extract learning from feedback.
   239→   239→   * CRITICAL: Never automatically convert any result into truth.
   240→   240→   * All learning starts with PENDING validation and ESTIMATED/INFERRED truth level.
   241→   241→   */
   242→   242→  async learn(feedbackId: string): Promise<ExtractedLearning[]> {
   243→   243→    const feedback = await db.feedback.findUnique({ where: { id: feedbackId } });
   244→   244→    if (!feedback) throw new Error(`Feedback ${feedbackId} not found`);
   245→   245→
   246→   246→    // Ensure analysis exists
   247→   247→    if (!feedback.analysis) {
   248→   248→      await this.analyze(feedbackId);
   249→   249→    }
   250→   250→
   251→   251→    const refreshedFeedback = await db.feedback.findUnique({ where: { id: feedbackId } });
   252→   252→    if (!refreshedFeedback) throw new Error(`Feedback ${feedbackId} not found after analysis`);
   253→   253→
   254→   254→    const analysis: FeedbackAnalysis = refreshedFeedback.analysis
   255→   255→      ? JSON.parse(refreshedFeedback.analysis)
   256→   256→      : { category: refreshedFeedback.feedbackType as FeedbackType, severity: "medium", isRecoverable: true, estimatedConfidence: 0.5 };
   257→   257→
   258→   258→    const learnings: ExtractedLearning[] = [];
   259→   259→
   260→   260→    switch (refreshedFeedback.feedbackType as FeedbackType) {
   261→   261→      case "SUCCESS": {
   262→   262→        // Record successful pattern — but as ESTIMATED, not VERIFIED
   263→   263→        learnings.push({
   264→   264→          domain: this.inferDomain(refreshedFeedback),
   265→   265→          type: "PROCEDURAL",
   266→   266→          content: {
   267→   267→            pattern: "successful_execution",
   268→   268→            input: refreshedFeedback.input ? JSON.parse(refreshedFeedback.input) : null,
   269→   269→            result: refreshedFeedback.result ? JSON.parse(refreshedFeedback.result) : null,
   270→   270→            feedbackText: refreshedFeedback.feedbackText,
   271→   271→          },
   272→   272→          source: `feedback:${feedbackId}`,
   273→   273→          sourceType: "FEEDBACK",
   274→   274→          truthLevel: "ESTIMATED", // NOT verified — must be validated
   275→   275→          confidence: 0.6, // Moderate confidence from single success
   276→   276→          needsValidation: true,
   277→   277→        });
   278→   278→        break;
   279→   279→      }
   280→   280→
   281→   281→      case "PARTIAL_SUCCESS": {
   282→   282→        // Record what worked and what didn't
   283→   283→        learnings.push({
   284→   284→          domain: this.inferDomain(refreshedFeedback),
   285→   285→          type: "EPISODIC",
   286→   286→          content: {
   287→   287→            pattern: "partial_success",
   288→   288→            whatWorked: this.extractWhatWorked(refreshedFeedback),
   289→   289→            whatFailed: this.extractWhatFailed(refreshedFeedback),
   290→   290→            feedbackText: refreshedFeedback.feedbackText,
   291→   291→          },
   292→   292→          source: `feedback:${feedbackId}`,
   293→   293→          sourceType: "FEEDBACK",
   294→   294→          truthLevel: "ESTIMATED",
   295→   295→          confidence: 0.4,
   296→   296→          needsValidation: true,
   297→   297→        });
   298→   298→        break;
   299→   299→      }
   300→   300→
   301→   301→      case "FAILURE":
   302→   302→      case "BAD_EXECUTION": {
   303→   303→        // Record failure as a negative pattern
   304→   304→        learnings.push({
   305→   305→          domain: this.inferDomain(refreshedFeedback),
   306→   306→          type: "EPISODIC",
   307→   307→          content: {
   308→   308→            pattern: "failure",
   309→   309→            rootCause: analysis.rootCause,
   310→   310→            contributingFactors: analysis.contributingFactors,
   311→   311→            feedbackText: refreshedFeedback.feedbackText,
   312→   312→          },
   313→   313→          source: `feedback:${feedbackId}`,
   314→   314→          sourceType: "FEEDBACK",
   315→   315→          truthLevel: "OBSERVED", // Failures are observable facts
   316→   316→          confidence: 0.8, // High confidence in what DIDN'T work
   317→   317→          needsValidation: false, // Failures don't need validation — they happened
   318→   318→        });
   319→   319→
   320→   320→        // Also record a procedural learning about what to avoid
   321→   321→        learnings.push({
   322→   322→          domain: this.inferDomain(refreshedFeedback),
   323→   323→          type: "PROCEDURAL",
   324→   324→          content: {
   325→   325→            pattern: "avoid",
   326→   326→            whatToAvoid: refreshedFeedback.input ? JSON.parse(refreshedFeedback.input) : null,
   327→   327→            reason: analysis.rootCause || refreshedFeedback.feedbackText,
   328→   328→          },
   329→   329→          source: `feedback:${feedbackId}`,
   330→   330→          sourceType: "FEEDBACK",
   331→   331→          truthLevel: "ESTIMATED",
   332→   332→          confidence: 0.5,
   333→   333→          needsValidation: true,
   334→   334→        });
   335→   335→        break;
   336→   336→      }
   337→   337→
   338→   338→      case "BAD_DECISION": {
   339→   339→        // Record the bad decision as a factual memory
   340→   340→        learnings.push({
   341→   341→          domain: "decision",
   342→   342→          type: "FACTUAL",
   343→   343→          content: {
   344→   344→            pattern: "bad_decision",
   345→   345→            decision: refreshedFeedback.input ? JSON.parse(refreshedFeedback.input) : null,
   346→   346→            outcome: refreshedFeedback.result ? JSON.parse(refreshedFeedback.result) : null,
   347→   347→            reason: refreshedFeedback.feedbackText,
   348→   348→          },
   349→   349→          source: `feedback:${feedbackId}`,
   350→   350→          sourceType: "FEEDBACK",
   351→   351→          truthLevel: "OBSERVED",
   352→   352→          confidence: 0.7,
   353→   353→          needsValidation: true, // Decision quality needs validation
   354→   354→          affectedSkills: this.identifyAffectedSkills(refreshedFeedback),
   355→   355→        });
   356→   356→        break;
   357→   357→      }
   358→   358→
   359→   359→      case "BAD_HYPOTHESIS": {
   360→   360→        // CRITICAL: Invalidate the hypothesis in memory
   361→   361→        learnings.push({
   362→   362→          domain: this.inferDomain(refreshedFeedback),
   363→   363→          type: "FACTUAL",
   364→   364→          content: {
   365→   365→            pattern: "invalidated_hypothesis",
   366→   366→            hypothesis: refreshedFeedback.input ? JSON.parse(refreshedFeedback.input) : null,
   367→   367→            contradictingEvidence: refreshedFeedback.result ? JSON.parse(refreshedFeedback.result) : null,
   368→   368→            reason: refreshedFeedback.feedbackText,
   369→   369→          },
   370→   370→          source: `feedback:${feedbackId}`,
   371→   371→          sourceType: "FEEDBACK",
   372→   372→          truthLevel: "OBSERVED",
   373→   373→          confidence: 0.9, // Very high confidence in contradiction
   374→   374→          needsValidation: false, // Contradictory evidence is a fact
   375→   375→        });
   376→   376→        break;
   377→   377→      }
   378→   378→
   379→   379→      case "BAD_TOOL": {
   380→   380→        // Record tool failure
   381→   381→        learnings.push({
   382→   382→          domain: "tool",
   383→   383→          type: "FACTUAL",
   384→   384→          content: {
   385→   385→            pattern: "bad_tool",
   386→   386→            tool: refreshedFeedback.input ? JSON.parse(refreshedFeedback.input) : null,
   387→   387→            issue: refreshedFeedback.feedbackText,
   388→   388→            alternative: analysis.recommendations?.[0],
   389→   389→          },
   390→   390→          source: `feedback:${feedbackId}`,
   391→   391→          sourceType: "FEEDBACK",
   392→   392→          truthLevel: "ESTIMATED",
   393→   393→          confidence: 0.5,
   394→   394→          needsValidation: true,
   395→   395→        });
   396→   396→        break;
   397→   397→      }
   398→   398→
   399→   399→      case "TECHNICAL_ERROR": {
   400→   400→        // Record as episodic — technical errors are transient
   401→   401→        learnings.push({
   402→   402→          domain: "system",
   403→   403→          type: "EPISODIC",
   404→   404→          content: {
   405→   405→            pattern: "technical_error",
   406→   406→            error: refreshedFeedback.feedbackText,
   407→   407→            recoverable: analysis.isRecoverable,
   408→   408→          },
   409→   409→          source: `feedback:${feedbackId}`,
   410→   410→          sourceType: "FEEDBACK",
   411→   411→          truthLevel: "OBSERVED",
   412→   412→          confidence: 0.7,
   413→   413→          needsValidation: false,
   414→   414→        });
   415→   415→        break;
   416→   416→      }
   417→   417→
   418→   418→      default: {
   419→   419→        learnings.push({
   420→   420→          domain: this.inferDomain(refreshedFeedback),
   421→   421→          type: "EPISODIC",
   422→   422→          content: {
   423→   423→            pattern: "unknown_feedback",
   424→   424→            feedbackType: refreshedFeedback.feedbackType,
   425→   425→            feedbackText: refreshedFeedback.feedbackText,
   426→   426→          },
   427→   427→          source: `feedback:${feedbackId}`,
   428→   428→          sourceType: "FEEDBACK",
   429→   429→          truthLevel: "UNKNOWN",
   430→   430→          confidence: 0.3,
   431→   431→          needsValidation: true,
   432→   432→        });
   433→   433→      }
   434→   434→    }
   435→   435→
   436→   436→    // Store learning in the feedback record
   437→   437→    await db.feedback.update({
   438→   438→      where: { id: feedbackId },
   439→   439→      data: {
   440→   440→        learning: JSON.stringify(learnings),
   441→   441→      },
   442→   442→    });
   443→   443→
   444→   444→    return learnings;
   445→   445→  }
   446→   446→
   447→   447→  /**
   448→   448→   * Apply extracted learning to the system.
   449→   449→   * Updates memory and/or skills based on the learning.
   450→   450→   * CRITICAL: Only applies VALIDATED learning, or marks them as needing validation.
   451→   451→   */
   452→   452→  async applyLearning(learning: ExtractedLearning, agentId?: string): Promise<{
   453→   453→    memoryId?: string;
   454→   454→    skillsUpdated?: string[];
   455→   455→    status: "applied" | "pending_validation";
   456→   456→  }> {
   457→   457→    const effectiveAgentId = agentId || this.defaultAgentId;
   458→   458→    const skillsUpdated: string[] = [];
   459→   459→
   460→   460→    // Store in memory
   461→   461→    const { id: memoryId } = await this.memoryManager.store(
   462→   462→      effectiveAgentId,
   463→   463→      learning.domain,
   464→   464→      learning.type,
   465→   465→      learning.content,
   466→   466→      {
   467→   467→        source: learning.source,
   468→   468→        sourceType: learning.sourceType,
   469→   469→        truthLevel: learning.truthLevel,
   470→   470→        confidence: learning.confidence,
   471→   471→      }
   472→   472→    );
   473→   473→
   474→   474→    // If learning is about a bad hypothesis, invalidate related memories
   475→   475→    if (learning.content.pattern === "invalidated_hypothesis" && learning.content.hypothesis) {
   476→   476→      const relatedMemories = await this.memoryManager.recall({
   477→   477→        domain: learning.domain,
   478→   478→        searchText: JSON.stringify(learning.content.hypothesis).substring(0, 50),
   479→   479→      });
   480→   480→
   481→   481→      for (const mem of relatedMemories) {
   482→   482→        const m = mem as { id: string };
   483→   483→        try {
   484→   484→          await this.memoryManager.invalidate(m.id, "Invalidated by feedback: bad hypothesis");
   485→   485→        } catch {
   486→   486→          // Memory might already be invalidated
   487→   487→        }
   488→   488→      }
   489→   489→    }
   490→   490→
   491→   491→    // If learning affects skills, update them
   492→   492→    if (learning.affectedSkills && learning.affectedSkills.length > 0) {
   493→   493→      for (const skillId of learning.affectedSkills) {
   494→   494→        try {
   495→   495→          await this.skillManager.improve(skillId, {
   496→   496→            success: learning.content.pattern === "successful_execution",
   497→   497→            partialSuccess: learning.content.pattern === "partial_success",
   498→   498→            actualResult: learning.feedbackText,
   499→   499→          });
   500→   500→          skillsUpdated.push(skillId);
   501→   501→        } catch {
   502→   502→          // Skill might not exist or not be improvable
   503→   503→        }
   504→   504→      }
   505→   505→    }
   506→   506→
   507→   507→    // Determine status
   508→   508→    const status = learning.needsValidation ? "pending_validation" : "applied";
   509→   509→
   510→   510→    return { memoryId, skillsUpdated: skillsUpdated.length > 0 ? skillsUpdated : undefined, status };
   511→   511→  }

   513→   513→  /**
   514→   514→   * Validate a feedback entry and its associated learning.
   515→   515→   * Moves validation status from PENDING to VALIDATED or INVALIDATED.
   516→   516→   */
   517→   517→  async validate(feedbackId: string, isValid: boolean, reason?: string): Promise<void> {
   518→   518→    const feedback = await db.feedback.findUnique({ where: { id: feedbackId } });
   519→   519→    if (!feedback) throw new Error(`Feedback ${feedbackId} not found`);
   520→   520→
   521→   521→    const evidence = feedback.evidence ? JSON.parse(feedback.evidence) : {};
   522→   522→    evidence.validationReason = reason;
   523→   523→    evidence.validatedAt = new Date().toISOString();
   524→   524→
   525→   525→    await db.feedback.update({
   526→   526→      where: { id: feedbackId },
   527→   527→      data: {
   528→   528→        validationStatus: isValid ? "VALIDATED" : "INVALIDATED",
   529→   529→        evidence: JSON.stringify(evidence),
   530→   530→      },
   531→   531→    });
   532→   532→
   533→   533→    // If validated and has learning, apply it
   534→   534→    if (isValid && feedback.learning) {
   535→   535→      const learnings: ExtractedLearning[] = JSON.parse(feedback.learning);
   536→   536→      for (const learning of learnings) {
   537→   537→        await this.applyLearning(learning, feedback.agentId);
   538→   538→      }
   539→   539→    }
   540→   540→  }

   596→   596→  async getStats(agentId?: string): Promise<{
   597→   597→    total: number;
   598→   598→    byType: Record<string, number>;
   599→   599→    byValidationStatus: Record<string, number>;
   600→   600→    avgConfidence: number;
   601→   601→    learningPending: number;
   602→   602→    learningApplied: number;
   603→   603→  }> {
   604→   604→    const effectiveAgentId = agentId || this.defaultAgentId;
   605→   605→
   606→   606→    const feedbacks = await db.feedback.findMany({
   607→   607→      where: { agentId: effectiveAgentId },
   608→   608→    });
   609→   609→
   610→   610→    const byType: Record<string, number> = {};
   611→   611→    const byValidationStatus: Record<string, number> = {};
   612→   612→    let totalConfidence = 0;
   613→   613→    let learningPending = 0;
   614→   614→    let learningApplied = 0;
   615→   615→
   616→   616→    for (const f of feedbacks) {
   617→   617→      byType[f.feedbackType] = (byType[f.feedbackType] || 0) + 1;
   618→   618→      byValidationStatus[f.validationStatus] = (byValidationStatus[f.validationStatus] || 0) + 1;
   619→   619→      totalConfidence += f.confidence;
   620→   620→
   621→   621→      if (f.validationStatus === "PENDING") learningPending++;
   622→   622→      if (f.validationStatus === "VALIDATED") learningApplied++;
   623→   623→    }
   624→   624→
   625→   625→    return {
   626→   626→      total: feedbacks.length,
   627→   627→      byType,
   628→   628→      byValidationStatus,
   629→   629→      avgConfidence: feedbacks.length > 0 ? totalConfidence / feedbacks.length : 0,
   630→   630→      learningPending,
   631→   631→      learningApplied,
   632→   632→    };
   633→   633→  }

   640→   640→  private inferConfidence(feedbackType: FeedbackType): number {
   641→   641→    switch (feedbackType) {
   642→   642→      case "SUCCESS": return 0.8;
   643→   643→      case "PARTIAL_SUCCESS": return 0.5;
   644→   644→      case "FAILURE": return 0.8; // We're confident it failed
   645→   645→      case "TECHNICAL_ERROR": return 0.7;
   646→   646→      case "BAD_DECISION": return 0.7;
   647→   647→      case "BAD_HYPOTHESIS": return 0.9;
   648→   648→      case "BAD_TOOL": return 0.6;

   651→   651→    }

   668→   668→  /**
   669→   669→   * Identify the cause of a partial success.
   670→   670→   */
   671→   671→  private identifyPartialSuccessCause(feedback: { feedbackText: string; result: string | null }): string {
   672→   672→    if (feedback.feedbackText) return feedback.feedbackText;
   673→   673→    if (feedback.result) {
   674→   674→      try {
   675→   675→        const result = JSON.parse(feedback.result);
   676→   676→        if (result.errors && Array.isArray(result.errors) && result.errors.length > 0) {
   677→   677→          return `Partial success with errors: ${result.errors[0]}`;
   678→   678→        }
   679→   679→        if (result.stepsCompleted && result.stepsTotal && result.stepsCompleted < result.stepsTotal) {
   680→   680→          return `Only ${result.stepsCompleted}/${result.stepsTotal} steps completed`;
   681→   681→        }
   682→   682→      } catch { /* ignore */ }
   683→   683→    }
   684→   684→    return "Unknown cause of partial success";
   685→   685→  }

   689→   689→   */
   690→   690→  private identifyFailureRootCause(feedback: { feedbackText: string; result: string | null }): string {
   691→   691→    if (feedback.feedbackText) return feedback.feedbackText;
   692→   692→    if (feedback.result) {
   693→   693→      try {
   694→   694→        const result = JSON.parse(feedback.result);

   696→   696→        if (result.errors && Array.isArray(result.errors) && result.errors.length > 0) {
   697→   697→          return result.errors[0];
   698→   698→        }
   699→   699→      } catch { /* ignore */ }
   700→   700→    }
   701→   701→    return "Unknown root cause";
   702→   702→  }

   725→   725→  private identifyDecisionPatterns(feedback: { feedbackText: string }): string[] {
   726→   726→    const patterns: string[] = [];
   727→   727→    const text = feedback.feedbackText.toLowerCase();
   728→   728→

   731→   731→    if (text.includes("insufficient") || text.includes("not enough")) patterns.push("resource_underestimation");
   732→   732→    if (text.includes("oversized") || text.includes("too much")) patterns.push("resource_overestimation");
   733→   733→    if (text.includes("wrong order") || text.includes("sequence")) patterns.push("ordering_error");
   734→   734→
   735→   735→    return patterns;

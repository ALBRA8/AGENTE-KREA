   629→   629→        .then(r => r.ok ? r.json() : [])
   630→   630→        .then(setGenerations)
   631→   631→        .catch(() => {});
   632→   632→    }
   633→   633→  }, []);
   634→   634→
   635→   635→  const logout = () => { localStorage.removeItem("p360_token"); localStorage.removeItem("p360_user"); onLogout(); };
   636→   636→  const refreshUser = async () => {
   637→   637→    const token = localStorage.getItem("p360_token");
   638→   638→    if (!token) return;
   639→   639→    try {
   640→   640→      const res = await fetch("/api/auth/me", { headers: { Authorization: `Bearer ${token}` } });
   641→   641→      if (res.ok) setUser(await res.json());
   642→   642→    } catch {}
   643→   643→  };
   644→   644→  const onUpdateCredits = (c: number) => { setUser(prev => ({ ...prev, credits: c })); };
   645→   645→
   646→   646→  const renderPage = () => {
   647→   647→    switch (page) {
   648→   648→      case "dashboard": return <Dashboard user={user} generations={generations} onNav={setPage} />;
   649→   649→      case "prompts": return <PromptChat user={user} onUpdateCredits={onUpdateCredits} />;
   650→   650→      case "images": return <ImageGenerator user={user} onUpdateCredits={onUpdateCredits} />;
   651→   651→      case "copy": case "social": case "script": case "email": return <TextGenerator type={page} user={user} onUpdateCredits={onUpdateCredits} />;
   652→   652→      case "voice": return <VoiceGenerator user={user} onUpdateCredits={onUpdateCredits} />;
   653→   653→      case "ebook": return <EbookGenerator user={user} onUpdateCredits={onUpdateCredits} />;
   654→   654→      case "subtitle": return <SubtitleGenerator user={user} onUpdateCredits={onUpdateCredits} />;
   655→   655→      case "metrics": return <MetricsTracker userId={user.id} />;
   656→   656→      case "library": return <Library user={user} />;
   657→   657→      case "settings": return <SettingsPage user={user} />;
   658→   658→      case "support": return <SupportPage />;
   659→   659→      default: return <Dashboard user={user} generations={generations} onNav={setPage} />;
   660→   660→    }
   661→   661→  };
   662→   662→
   663→   663→  return (
   664→   664→    <AuthCtx.Provider value={{ user, logout, refreshUser }}>
   665→   665→      <div className="min-h-screen bg-[#080c16] flex">
   666→   666→        <Sidebar active={page} onNav={setPage} open={sidebarOpen} onClose={() => setSidebarOpen(false)} user={user} />
   667→   667→        <div className="flex-1 flex flex-col min-h-screen">
   668→   668→          {/* Top bar */}
   669→   669→          <header className="sticky top-0 z-30 flex items-center gap-4 px-4 sm:px-6 py-3 border-b border-white/[0.06] backdrop-blur-xl bg-[#080c16]/80">
   670→   670→            <button onClick={() => setSidebarOpen(true)} className="lg:hidden text-white/50 hover:text-white">
   671→   671→              <Menu className="w-5 h-5" />
   672→   672→            </button>
   673→   673→            <div className="flex-1" />
   674→   674→            <div className="flex items-center gap-3">
   675→   675→              <div className="flex items-center gap-1.5 px-3 py-1.5 rounded-lg bg-yellow-400/10 text-yellow-400 text-xs font-bold">
   676→   676→                <Coins className="w-3.5 h-3.5" /> {user.credits}
   677→   677→              </div>
   678→   678→              <button onClick={logout} className="p-2 rounded-lg hover:bg-white/5 text-white/40 hover:text-white/70 transition-colors" title="Cerrar sesión">
   679→   679→                <LogOut className="w-4 h-4" />
   680→   680→              </button>
   681→   681→            </div>
   682→   682→          </header>
   683→   683→          {/* Content */}
   684→   684→          <main className="flex-1 p-4 sm:p-6 lg:p-8 overflow-y-auto">
   685→   685→            {renderPage()}
   686→   686→          </main>
   687→   687→        </div>
   688→   688→      </div>
   689→   689→    </AuthCtx.Provider>
   690→   690→  );
   691→   691→}

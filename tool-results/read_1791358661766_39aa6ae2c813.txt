    16→ *   IDEA → RESEARCHING → VALIDATING → APPROVED → ARCHITECTING → SPECIFIED →
    17→ *   BUILDING → QA → READY → LAUNCHED → ITERATING → PAUSED → KILLED → ARCHIVED

   344→// ─── JSON Field Helpers ───────────────────────────────────────────────────────
   345→
   346→/** Safely parse a JSON field from DB, returning fallback on failure */
   347→function parseJsonField<T>(raw: string | null | undefined, fallback: T): T {
   348→  if (!raw) return fallback;
   349→  try {
   350→    return JSON.parse(raw) as T;
   351→  } catch {
   352→    return fallback;
   353→  }
   354→}
   355→
   356→/** Safely stringify a value for DB storage */
   357→function stringifyField(value: unknown): string | null {
   358→  if (value === null || value === undefined) return null;
   359→  return JSON.stringify(value);
   360→}

   364→/**
   365→ * Convert a Prisma ProductDossier row to the application-level ProductDossier object.
   366→ * All JSON fields are parsed from their string representations.
   367→ */
   368→function rowToDossier(row: {
   369→  id: string;
   370→  title: string;
   371→  description: string;
   372→  domain: string | null;
   373→  status: string;
   374→  userId: string | null;
   375→  opportunity: string | null;
   376→  evidence: string | null;
   377→  fitResult: string | null;
   378→  decision: string | null;
   379→  architecture: string | null;
   380→  specification: string | null;
   381→  handoffs: string | null;
   382→  productionData: string | null;
   383→  qaResults: string | null;
   384→  feedback: string | null;
   385→  learnings: string | null;
   386→  economics: string | null;
   387→  version: number;
   388→  changes: string | null;
   389→  format: string | null;
   390→  priority: string;
   391→  createdAt: Date;
   392→  updatedAt: Date;
   393→}): ProductDossier {
   394→  return {
   395→    id: row.id,
   396→    title: row.title,
   397→    description: row.description,
   398→    domain: row.domain ?? "",
   399→    format: row.format ?? undefined,
   400→    status: row.status as DossierStatus,
   401→    userId: row.userId ?? undefined,
   402→    opportunity: parseJsonField<ProductOpportunity>(row.opportunity, {
   403→      name: row.title,
   404→      domain: row.domain ?? "",
   405→      problem: "",
   406→      solution: "",
   407→      audience: "",
   408→    }),
   409→    evidence: parseJsonField<ProductEvidence[]>(row.evidence, []),
   410→    fitResult: parseJsonField<FitResult | null>(row.fitResult, null),
   411→    decision: parseJsonField<ProductDecision | null>(row.decision, null),
   412→    architecture: parseJsonField<ProductArchitecture | null>(row.architecture, null),
   413→    specification: parseJsonField<ProductSpecification | null>(row.specification, null),
   414→    softwareSpecification: null, // stored within specification JSON or separately
   415→    handoffs: parseJsonField<HandoffContract[]>(row.handoffs, []),
   416→    productionStatus: parseJsonField<string>(row.productionData, "not_started"),
   417→    qaResults: parseJsonField<QAResult[]>(row.qaResults, []),
   418→    feedback: parseJsonField<ProductFeedback[]>(row.feedback, []),
   419→    learnings: parseJsonField<ProductLearning[]>(row.learnings, []),
   420→    economics: parseJsonField<EconomicsEstimate | null>(row.economics, null),
   421→    version: row.version,
   422→    changes: parseJsonField<DossierChange[]>(row.changes, []),
   423→    priority: row.priority as "LOW" | "MEDIUM" | "HIGH" | "CRITICAL",
   424→    createdAt: row.createdAt.toISOString(),
   425→    updatedAt: row.updatedAt.toISOString(),
   426→  };
   427→}

   429→/**
   430→ * Build the Prisma update payload from application-level updates.
   431→ * Only non-undefined fields are included in the update.
   432→ */
   433→function buildUpdatePayload(
   434→  updates: DossierUpdates,
   435→  changes: DossierChange[],
   436→  version: number
   437→): Record<string, unknown> {
   438→  const payload: Record<string, unknown> = {
   439→    version: version + 1,
   440→    changes: stringifyField(changes),
   441→  };
   442→
   443→  if (updates.title !== undefined) payload.title = updates.title;
   444→  if (updates.description !== undefined) payload.description = updates.description;
   445→  if (updates.domain !== undefined) payload.domain = updates.domain;
   446→  if (updates.format !== undefined) payload.format = updates.format;
   447→  if (updates.opportunity !== undefined) payload.opportunity = stringifyField(updates.opportunity);
   448→  if (updates.fitResult !== undefined) payload.fitResult = stringifyField(updates.fitResult);
   449→  if (updates.decision !== undefined) payload.decision = stringifyField(updates.decision);
   450→  if (updates.architecture !== undefined) payload.architecture = stringifyField(updates.architecture);
   451→  if (updates.specification !== undefined) payload.specification = stringifyField(updates.specification);
   452→  if (updates.softwareSpecification !== undefined) payload.specification = stringifyField(updates.softwareSpecification);
   453→  if (updates.economics !== undefined) payload.economics = stringifyField(updates.economics);
   454→  if (updates.productionStatus !== undefined) payload.productionData = stringifyField(updates.productionStatus);
   455→  if (updates.priority !== undefined) payload.priority = updates.priority;
   456→
   457→  return payload;
   458→}

   460→// ─── Product Dossier Manager ──────────────────────────────────────────────────
   461→
   462→/**
   463→ * ProductDossierManager — Master product lifecycle manager (DB-backed).
   464→ *
   465→ * AC-023: Lifecycle transitions are validated — only valid state changes allowed.
   466→ * AC-053: Each dossier's memory domain is `product:{id}:dossier` for isolation.
   467→ *
   468→ * All operations read/write directly to the Prisma ProductDossier model.
   469→ * No in-memory state — fully durable across process restarts.
   470→ */
   471→export class ProductDossierManager {
   472→  /**
   473→   * create — Start a new product dossier.
   474→   *
   475→   * Creates a dossier in IDEA status with the given opportunity.
   476→   * Persists directly to DB. Returns the created dossier.
   477→   */
   478→  async create(
   479→    title: string,
   480→    description: string,
   481→    domain: string,
   482→    format?: string,
   483→    userId?: string
   484→  ): Promise<ProductDossier> {
   485→    const now = new Date();
   486→
   487→    const opportunity: ProductOpportunity = {
   488→      name: title,
   489→      domain,
   490→      problem: description,
   491→      solution: "",
   492→      audience: "",
   493→      format,
   494→    };
   495→
   496→    const row = await db.productDossier.create({
   497→      data: {
   498→        title,
   499→        description,
   500→        domain,
   501→        format: format ?? null,
   502→        status: "IDEA",
   503→        userId: userId ?? null,
   504→        opportunity: stringifyField(opportunity),
   505→        evidence: stringifyField([]),
   506→        fitResult: null,
   507→        decision: null,
   508→        architecture: null,
   509→        specification: null,
   510→        handoffs: stringifyField([]),
   511→        productionData: stringifyField("not_started"),
   512→        qaResults: stringifyField([]),
   513→        feedback: stringifyField([]),
   514→        learnings: stringifyField([]),
   515→        economics: null,
   516→        version: 1,
   517→        changes: stringifyField([]),
   518→        priority: "MEDIUM",
   519→        createdAt: now,
   520→        updatedAt: now,
   521→      },
   522→    });
   523→
   524→    return rowToDossier(row);
   525→  }
   526→
   527→  /**
   528→   * get — Retrieve a complete dossier from DB.
   529→   *
   530→   * Reads directly from the database. No in-memory cache.
   531→   */
   532→  async get(id: string): Promise<ProductDossier | null> {
   533→    const row = await db.productDossier.findUnique({ where: { id } });
   534→    if (!row) return null;
   535→    return rowToDossier(row);
   536→  }
   537→
   538→  /**
   539→   * update — Update dossier fields.
   540→   *
   541→   * Reads from DB, records each changed field in the version history,
   542→   * then writes the updated dossier back to DB.
   543→   */
   544→  async update(
   545→    id: string,
   546→    updates: DossierUpdates,
   547→    changedBy: string = "krea",
   548→    reason: string = "Update"
   549→  ): Promise<ProductDossier | null> {
   550→    const row = await db.productDossier.findUnique({ where: { id } });
   551→    if (!row) return null;
   552→
   553→    const dossier = rowToDossier(row);
   554→    const now = new Date().toISOString();
   555→
   556→    // Build change records for each updated field
   557→    const newChanges: DossierChange[] = [];
   558→    for (const [field, newValue] of Object.entries(updates)) {
   559→      if (newValue === undefined) continue;
   560→      const oldValue = (dossier as Record<string, unknown>)[field];
   561→      newChanges.push({
   562→        id: randomUUID(),
   563→        field,
   564→        oldValue,
   565→        newValue,
   566→        changedBy,
   567→        timestamp: now,
   568→        reason,
   569→      });
   570→    }
   571→
   572→    // Merge with existing changes
   573→    const allChanges = [...dossier.changes, ...newChanges];
   574→
   575→    // Build and execute DB update
   576→    const payload = buildUpdatePayload(updates, allChanges, dossier.version);
   577→    const updatedRow = await db.productDossier.update({
   578→      where: { id },
   579→      data: payload,
   580→    });
   581→
   582→    return rowToDossier(updatedRow);
   583→  }
   584→
   585→  /**
   586→   * transition — Transition dossier to a new lifecycle status.
   587→   *
   588→   * AC-023: Only valid transitions are allowed. Invalid transitions
   589→   * throw an error with the list of valid next states.
   590→   */
   591→  async transition(
   592→    id: string,
   593→    newStatus: DossierStatus,
   594→    changedBy: string = "krea",
   595→    reason: string = ""
   596→  ): Promise<ProductDossier | null> {
   597→    const row = await db.productDossier.findUnique({ where: { id } });
   598→    if (!row) return null;
   599→
   600→    const currentStatus = row.status as DossierStatus;
   601→
   602→    // Validate transition (AC-023)
   603→    if (!VALID_TRANSITIONS[currentStatus].includes(newStatus)) {
   604→      throw new Error(
   605→        `Invalid transition: ${currentStatus} → ${newStatus}. ` +
   606→        `Valid transitions from ${currentStatus}: ${VALID_TRANSITIONS[currentStatus].join(", ")}`
   607→      );
   608→    }
   609→
   610→    const now = new Date().toISOString();
   611→    const existingChanges = parseJsonField<DossierChange[]>(row.changes, []);
   612→
   613→    existingChanges.push({
   614→      id: randomUUID(),
   615→      field: "status",
   616→      oldValue: currentStatus,
   617→      newValue: newStatus,
   618→      changedBy,
   619→      timestamp: now,
   620→      reason: reason || `Transition from ${currentStatus} to ${newStatus}`,
   621→    });
   622→
   623→    const updatedRow = await db.productDossier.update({
   624→      where: { id },
   625→      data: {
   626→        status: newStatus,
   627→        version: row.version + 1,
   628→        changes: stringifyField(existingChanges),
   629→      },
   630→    });
   631→
   632→    return rowToDossier(updatedRow);
   633→  }
   634→
   635→  /**
   636→   * addEvidence — Append evidence to the dossier.
   637→   *
   638→   * Reads current evidence array from DB, appends new items, writes back.
   639→   */
   640→  async addEvidence(
   641→    id: string,
   642→    evidence: ProductEvidence[]
   643→  ): Promise<ProductDossier | null> {
   644→    const row = await db.productDossier.findUnique({ where: { id } });
   645→    if (!row) return null;
   646→
   647→    const existing = parseJsonField<ProductEvidence[]>(row.evidence, []);
   648→    existing.push(...evidence);
   649→
   650→    const existingChanges = parseJsonField<DossierChange[]>(row.changes, []);
   651→    existingChanges.push({
   652→      id: randomUUID(),
   653→      field: "evidence",
   654→      oldValue: existing.length - evidence.length,
   655→      newValue: existing.length,
   656→      changedBy: "krea",

   663→      data: {
   664→        evidence: stringifyField(existing),
   665→        version: row.version + 1,
   666→        changes: stringifyField(existingChanges),
   667→      },
   668→    });

   768→  /**
   769→   * list — List dossiers with optional filters.
   770→   *
   771→   * Queries DB with Prisma where clauses. Supports status, domain,
   772→   * format, userId, name search, date ranges, and pagination.
   773→   */
   774→  async list(filters: DossierFilters = {}): Promise<ProductDossier[]> {
   775→    const where: Record<string, unknown> = {};
   776→
   777→    if (filters.status) where.status = filters.status;
   778→    if (filters.domain) where.domain = filters.domain;
   779→    if (filters.format) where.format = filters.format;
   780→    if (filters.userId) where.userId = filters.userId;
   781→    if (filters.nameContains) {
   782→      where.title = { contains: filters.nameContains, mode: "insensitive" };
   783→    }
   784→    if (filters.createdAfter || filters.createdBefore) {
   785→      const createdAt: Record<string, Date> = {};
   786→      if (filters.createdAfter) createdAt.gte = new Date(filters.createdAfter);
   787→      if (filters.createdBefore) createdAt.lte = new Date(filters.createdBefore);
   788→      where.createdAt = createdAt;
   789→    }
   790→
   791→    const offset = filters.offset || 0;
   792→    const limit = filters.limit || 50;
   793→
   794→    const rows = await db.productDossier.findMany({
   795→      where,
   796→      orderBy: { updatedAt: "desc" },
   797→      skip: offset,
   798→      take: limit,
   799→    });
   800→
   801→    return rows.map(rowToDossier);
   802→  }

   818→  /**
   819→   * Get dossier statistics across all dossiers in DB.
   820→   */
   821→  async getStats(): Promise<{
   822→    total: number;
   823→    byStatus: Record<DossierStatus, number>;
   824→    byDomain: Record<string, number>;
   825→  }> {
   826→    const all = await db.productDossier.findMany({
   827→      select: { status: true, domain: true },
   828→    });
   829→
   830→    const byStatus: Record<DossierStatus, number> = {
   831→      IDEA: 0, RESEARCHING: 0, VALIDATING: 0, APPROVED: 0,
   832→      ARCHITECTING: 0, SPECIFIED: 0, BUILDING: 0, QA: 0,
   833→      READY: 0, LAUNCHED: 0, ITERATING: 0, PAUSED: 0,
   834→      KILLED: 0, ARCHIVED: 0,
   835→    };
   836→    const byDomain: Record<string, number> = {};
   837→
   838→    for (const d of all) {
   839→      const status = d.status as DossierStatus;
   840→      if (status in byStatus) byStatus[status]++;
   841→      const domain = d.domain ?? "unknown";
   842→      byDomain[domain] = (byDomain[domain] || 0) + 1;
   843→    }
   844→
   845→    return { total: all.length, byStatus, byDomain };
   846→  }

   849→   * Delete a dossier from DB.
   850→   *
   851→   * Only allowed if the dossier is in KILLED or ARCHIVED status.
   852→   */
   853→  async delete(id: string): Promise<boolean> {
   854→    const row = await db.productDossier.findUnique({
   855→      where: { id },
   856→      select: { status: true },
   857→    });
   858→    if (!row) return false;
   859→
   860→    const status = row.status as DossierStatus;
   861→    if (status !== "KILLED" && status !== "ARCHIVED") {
   862→      throw new Error(
   863→        `Cannot delete dossier in ${status} status. ` +
   864→        `Must be KILLED or ARCHIVED.`
   865→      );
   866→    }
   867→
   868→    await db.productDossier.delete({ where: { id } });
   869→    return true;
   870→  }

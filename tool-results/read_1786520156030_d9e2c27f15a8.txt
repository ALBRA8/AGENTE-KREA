   405→   604→interface AppShellProps { onLogout: () => void; user?: User | null }
   406→   605→
   407→   606→export default function AppPage({ onLogout, user: initialUser }: AppShellProps) {
   408→   607→  const [user, setUser] = useState<User | null>(() => {
   409→   608→    if (initialUser) return initialUser;
   410→   609→    try {
   411→   610→      const saved = localStorage.getItem("p360_user");
   412→   611→      return saved ? JSON.parse(saved) : null;
   413→   612→    } catch { return null; }
   414→   613→  });
   415→   614→  const [page, setPage] = useState("dashboard");
   416→   615→  const [sidebarOpen, setSidebarOpen] = useState(false);
   417→   616→  const [loading, setLoading] = useState(false);
   418→   617→  const [generations, setGenerations] = useState<Generation[]>([]);
   419→   618→
   420→   619→  const loadUser = useCallback(async () => {
   421→   620→    const token = localStorage.getItem("p360_token");
   422→   621→    if (!token) { if (!user) onLogout(); else setLoading(false); return; }
   423→   622→    try {
   424→   623→      const res = await fetch("/api/auth/me", { headers: { Authorization: `Bearer ${token}` } });
   425→   624→      if (!res.ok) { if (!user) onLogout(); else setLoading(false); return; }
   426→   625→      const data = await res.json();
   427→   626→      setUser(data);
   428→   627→      const gRes = await fetch("/api/generations?limit=10", { headers: { Authorization: `Bearer ${token}` } });
   429→   628→      if (gRes.ok) setGenerations(await gRes.json());
   430→   629→    } catch { if (!user) onLogout(); }
   431→   630→    setLoading(false);
   432→   631→  }, [user, onLogout]);
   433→   632→
   434→   633→  useEffect(() => {
   435→   634→    const token = localStorage.getItem("p360_token");
   436→   635→    if (token && user) {
   437→   636→      fetch("/api/generations?limit=10", { headers: { Authorization: `Bearer ${token}` } })
   438→   637→        .then(r => r.ok ? r.json() : [])
   439→   638→        .then(setGenerations)
   440→   639→        .catch(() => {});
   441→   640→    }
   442→   641→  }, []);
   443→   642→
   444→   643→  const logout = () => { localStorage.removeItem("p360_token"); localStorage.removeItem("p360_user"); onLogout(); };
   445→   644→  const refreshUser = () => loadUser();
   446→   645→  const onUpdateCredits = (c: number) => { if (user) setUser({ ...user, credits: c }); };
   447→   646→
   448→   647→  if (loading) return (
   449→   648→    <div className="min-h-screen bg-[#080c16] flex items-center justify-center">
   450→   649→      <Loader2 className="w-8 h-8 animate-spin text-[#3b82f6]" />
   451→   650→    </div>
   452→   651→  );
   453→   652→
   454→   653→  if (!user) return null;
   455→   654→
   456→   655→  const renderPage = () => {
   457→   656→    switch (page) {
   458→   657→      case "dashboard": return <Dashboard user={user} generations={generations} onNav={setPage} />;
   459→   658→      case "prompts": return <PromptChat user={user} onUpdateCredits={onUpdateCredits} />;
   460→   659→      case "images": return <ImageGenerator user={user} onUpdateCredits={onUpdateCredits} />;
   461→   660→      case "copy": case "social": case "script": case "email": return <TextGenerator type={page} user={user} onUpdateCredits={onUpdateCredits} />;
   462→   661→      case "voice": return <VoiceGenerator user={user} onUpdateCredits={onUpdateCredits} />;
   463→   662→      case "ebook": return <EbookGenerator user={user} onUpdateCredits={onUpdateCredits} />;
   464→   663→      case "subtitle": return <SubtitleGenerator user={user} onUpdateCredits={onUpdateCredits} />;
   465→   664→      case "metrics": return <MetricsTracker userId={user.id} />;
   466→   665→      case "library": return <Library user={user} />;
   467→   666→      case "settings": return <SettingsPage user={user} />;
   468→   667→      case "support": return <SupportPage />;
   469→   668→      default: return <Dashboard user={user} generations={generations} onNav={setPage} />;
   470→   669→    }
   471→   670→  };
   472→   671→
   473→   672→  return (
   474→   673→    <AuthCtx.Provider value={{ user, logout, refreshUser }}>
   475→   674→      <div className="min-h-screen bg-[#080c16] flex">
   476→   675→        <Sidebar active={page} onNav={setPage} open={sidebarOpen} onClose={() => setSidebarOpen(false)} user={user} />
   477→   676→        <div className="flex-1 flex flex-col min-h-screen">
   478→   677→          {/* Top bar */}
   479→   678→          <header className="sticky top-0 z-30 flex items-center gap-4 px-4 sm:px-6 py-3 border-b border-white/[0.06] backdrop-blur-xl bg-[#080c16]/80">
   480→   679→            <button onClick={() => setSidebarOpen(true)} className="lg:hidden text-white/50 hover:text-white">
   481→   680→              <Menu className="w-5 h-5" />
   482→   681→            </button>
   483→   682→            <div className="flex-1" />
   484→   683→            <div className="flex items-center gap-3">
   485→   684→              <div className="flex items-center gap-1.5 px-3 py-1.5 rounded-lg bg-yellow-400/10 text-yellow-400 text-xs font-bold">
   486→   685→                <Coins className="w-3.5 h-3.5" /> {user.credits}
   487→   686→              </div>
   488→   687→              <button onClick={logout} className="p-2 rounded-lg hover:bg-white/5 text-white/40 hover:text-white/70 transition-colors" title="Cerrar sesión">
   489→   688→                <LogOut className="w-4 h-4" />
   490→   689→              </button>
   491→   690→            </div>
   492→   691→          </header>
   493→   692→          {/* Content */}
   494→   693→          <main className="flex-1 p-4 sm:p-6 lg:p-8 overflow-y-auto">
   495→   694→            {renderPage()}
   496→   695→          </main>
   497→   696→        </div>
   498→   697→      </div>
   499→   698→    </AuthCtx.Provider>
   500→   699→  );

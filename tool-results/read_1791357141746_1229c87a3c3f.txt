   154→// ─── BookFactory Class ─────────────────────────────────────────────────
   155→
   156→/**
   157→ * BookFactory — orchestrates the full eBook production pipeline.
   158→ *
   159→ * Pipeline (AC-029):
   160→ *   1. bookArchitecture — define editorial structure
   161→ *   2. content — generate chapter content with ZAI
   162→ *   3. artDirection — define visual system
   163→ *   4. coverDesign — specify cover
   164→ *   5. editorialDesign — margins, grid, typography, hierarchy
   165→ *   6. layout — produce deterministic layout
   166→ *   7. pdf — generate PDF artifact
   167→ *   8. visualQA — inspect result
   168→ *   9. repair — fix issues if found (AC-045: max 5 iterations)
   169→ *   10. finalProduct — verified artifact
   170→ *
   171→ * Each step produces evidence and is traceable via ExecutionTracer.
   172→ */
   173→export class BookFactory {
   174→  private artDirection = new ArtDirection();
   175→  private coverDesign = new CoverDesign();
   176→  private editorialDesign = new EditorialDesign();
   177→  private pdfFactory = new PDFFactory();
   178→  private visualQA = new VisualQA();
   179→  private memory = new MemoryManager();
   180→  private tracer = new ExecutionTracer();
   181→
   182→  /**
   183→   * Produce a complete book product from an opportunity and architecture.
   184→   *
   185→   * @param opportunity - The book opportunity/topic
   186→   * @param architecture - The product architecture
   187→   * @returns BookProduct — the verified book artifact
   188→   */
   189→  async produce(
   190→    opportunity: Opportunity,
   191→    architecture: Architecture
   192→  ): Promise<BookProduct> {
   193→    const startTime = Date.now();
   194→    const errors: string[] = [];
   195→    const pipelineSteps: PipelineStepEvidence[] = [];
   196→
   197→    const pipelineState: PipelineState = {
   198→      current_step: "init",
   199→      steps: pipelineSteps,
   200→      in_repair: false,
   201→      repair_iteration: 0,
   202→      max_repair_iterations: 5,
   203→    };
   204→
   205→    // Start execution trace
   206→    const executionId = await this.tracer.start("book-factory-produce", {
   207→      agentId: "krea",
   208→      userId: opportunity.userId,
   209→      inputs: { topic: opportunity.topic, genre: opportunity.genre },
   210→      skillsUsed: ["book-factory"],
   211→    });
   212→
   213→    // Mutable state for repair loop
   214→    let bookArchitecture: BookArchitectureInput | null = null;
   215→    let content: BookContent | null = null;
   216→    let artDir: ArtDirectionSpec | null = null;
   217→    let cover: CoverSpecification | null = null;
   218→    let editorial: EditorialSpecification | null = null;
   219→    let layout: LayoutResult | null = null;
   220→    let pdfResult: PDFResult | null = null;
   221→    let qaResult: QAResult | null = null;
   222→    let pdfVerification: PDFVerification | null = null;
   223→
   224→    try {
   225→      // ─── Step 1: Book Architecture ────────────────────────────────
   226→      pipelineState.current_step = "bookArchitecture";
   227→      bookArchitecture = await this.runStep(
   228→        "bookArchitecture", 1, pipelineSteps, executionId,
   229→        () => this.defineBookArchitecture(opportunity)
   230→      );
   231→
   232→      // ─── Step 2: Content Generation ───────────────────────────────
   233→      pipelineState.current_step = "content";
   234→      content = await this.runStep(
   235→        "content", 2, pipelineSteps, executionId,
   236→        () => this.generateContent(opportunity, bookArchitecture!, executionId)
   237→      );
   238→
   239→      // ─── Step 3: Art Direction ────────────────────────────────────
   240→      pipelineState.current_step = "artDirection";
   241→      const archInput: ArchitectureInput = {
   242→        productType: "book",
   243→        format: architecture.format,
   244→        complexity: architecture.complexity,
   245→        targetMarket: architecture.targetMarket,
   246→      };
   247→      artDir = await this.runStep(
   248→        "artDirection", 3, pipelineSteps, executionId,
   249→        () => Promise.resolve(this.artDirection.define(archInput, bookArchitecture!))
   250→      );
   251→
   252→      // ─── Step 4: Cover Design ─────────────────────────────────────
   253→      pipelineState.current_step = "coverDesign";
   254→      cover = await this.runStep(
   255→        "coverDesign", 4, pipelineSteps, executionId,
   256→        () => Promise.resolve(this.coverDesign.design(artDir!, bookArchitecture!))
   257→      );
   258→
   259→      // ─── Step 5: Editorial Design ────────────────────────────────
   260→      pipelineState.current_step = "editorialDesign";
   261→      editorial = await this.runStep(
   262→        "editorialDesign", 5, pipelineSteps, executionId,
   263→        () => Promise.resolve(this.editorialDesign.design(artDir!, bookArchitecture!))
   264→      );
   265→
   266→      // ─── Step 6: Layout ──────────────────────────────────────────
   267→      pipelineState.current_step = "layout";
   268→      layout = await this.runStep(
   269→        "layout", 6, pipelineSteps, executionId,
   270→        () => Promise.resolve(this.editorialDesign.layout(content!, editorial!, artDir!))
   271→      );
   272→
   273→      // ─── Step 7: PDF Generation ──────────────────────────────────
   274→      pipelineState.current_step = "pdf";
   275→      pdfResult = await this.runStep(
   276→        "pdf", 7, pipelineSteps, executionId,
   277→        () => Promise.resolve(this.pdfFactory.generatePDF(
   278→          layout!, content!, cover!, editorial!, artDir!
   279→        ))
   280→      );
   281→
   282→      // ─── Step 8: Visual QA ──────────────────────────────────────
   283→      pipelineState.current_step = "visualQA";
   284→      qaResult = await this.runStep(
   285→        "visualQA", 8, pipelineSteps, executionId,
   286→        () => Promise.resolve(this.visualQA.inspect(
   287→          { page_count: pdfResult!.pageCount, layout: layout!, content: content! },
   288→          content!,
   289→          editorial!
   290→        ))
   291→      );
   292→
   293→      // ─── Step 9: Repair Loop (AC-045) ──────────────────────────
   294→      if (!qaResult.passed) {
   295→        pipelineState.in_repair = true;
   296→        const repairResult = await this.executeRepairLoop(
   297→          qaResult, content, editorial, artDir, layout, cover,
   298→          pipelineState, pipelineSteps, executionId
   299→        );
   300→        content = repairResult.content;
   301→        editorial = repairResult.editorial;
   302→        layout = repairResult.layout;
   303→        cover = repairResult.cover;
   304→        qaResult = repairResult.qaResult;
   305→
   306→        // Re-generate PDF after repair
   307→        pdfResult = this.pdfFactory.generatePDF(
   308→          layout, content, cover, editorial, artDir
   309→        );
   310→      }
   311→
   312→      // ─── Step 10: Final Verification ───────────────────────────
   313→      pipelineState.current_step = "finalProduct";
   314→      pdfVerification = this.pdfFactory.verify(
   315→        pdfResult,
   316→        layout.page_count,
   317→        qaResult
   318→      );
   319→
   320→      // Record step
   321→      pipelineSteps.push({
   322→        step: "finalProduct",
   323→        step_number: 10,
   324→        started_at: new Date().toISOString(),
   325→        completed_at: new Date().toISOString(),
   326→        duration_ms: 0,
   327→        success: pdfVerification.overall,
   328→        output_id: randomUUID(),
   329→      });
   330→
   331→      // Store in memory for future reference
   332→      await this.storeProductMemory(opportunity, content, executionId);
   333→
   334→      // Mark execution as succeeded
   335→      await this.tracer.succeed(executionId, {
   336→        pageCount: layout.page_count,
   337→        qaPassed: qaResult.passed,
   338→        pdfVerified: pdfVerification.overall,
   339→      });
   340→
   341→    } catch (error) {
   342→      const message = error instanceof Error ? error.message : String(error);
   343→      errors.push(message);
   344→      await this.tracer.addError(executionId, message);
   345→      await this.tracer.fail(executionId, { message });
   346→    }
   347→
   348→    return {
   349→      success: pdfVerification?.overall ?? false,
   350→      title: bookArchitecture?.title ?? opportunity.topic,
   351→      subtitle: bookArchitecture?.subtitle ?? "",
   352→      author: opportunity.author,
   353→      content: content ?? { title: "", subtitle: "", author: "", chapters: [], total_word_count: 0 },
   354→      art_direction: artDir ?? {} as ArtDirectionSpec,
   355→      cover: cover ?? {} as CoverSpecification,
   356→      editorial: editorial ?? {} as EditorialSpecification,
   357→      layout: layout ?? { id: "", createdAt: "", input_hash: "", page_count: 0, pages: [], table_of_contents: [], visual_system_id: "" },
   358→      pdf: pdfResult ?? { success: false, filePath: null, fileName: "", pageCount: 0, sizeBytes: 0, checksum: "", generatedAt: "", spec: {} as PDFResult["spec"], errors: [] },
   359→      pdf_verification: pdfVerification ?? {} as PDFVerification,
   360→      qa_result: qaResult ?? { passed: false, issue_count: 0, issues: [], by_category: {} as QAResult["by_category"], by_severity: {} as QAResult["by_severity"], inspectedAt: "", inspection_duration_ms: 0 },
   361→      pipeline: pipelineState,
   362→      production_time_ms: Date.now() - startTime,
   363→      repair_iterations_used: pipelineState.repair_iteration,
   364→      errors,
   365→    };
   366→  }
   367→
   368→  // ─── Pipeline Steps ───────────────────────────────────────────────
   369→
   370→  /**
   371→   * Step 1: Define the book's editorial architecture.
   372→   * Derives structure from the opportunity (topic, audience, genre, etc.)
   373→   */
   374→  private async defineBookArchitecture(
   375→    opportunity: Opportunity
   376→  ): Promise<BookArchitectureInput> {
   377→    // Generate chapter titles using ZAI
   378→    const chapterTitles = await this.generateChapterTitles(opportunity);
   379→
   380→    return {
   381→      title: opportunity.topic,
   382→      subtitle: opportunity.subtitle ?? `A comprehensive guide for ${opportunity.targetAudience}`,
   383→      author: opportunity.author,
   384→      genre: opportunity.genre,
   385→      audience: opportunity.targetAudience,
   386→      chapterCount: opportunity.chapterCount,
   387→      chapterTitles,
   388→      tone: opportunity.tone,
   389→      purpose: opportunity.purpose,
   390→    };
   391→  }
   392→
   393→  /**
   394→   * Step 2: Generate chapter content using ZAI SDK.
   395→   *
   396→   * AC-032: Model principles and structure only — no literal copying.
   397→   * The ZAI generates original content based on the topic and structure.
   398→   */
   399→  private async generateContent(
   400→    opportunity: Opportunity,
   401→    bookArchitecture: BookArchitectureInput,
   402→    executionId: string
   403→  ): Promise<BookContent> {
   404→    const zai = await ZAI.create();
   405→    const chapters: ChapterContent[] = [];
   406→
   407→    for (let i = 0; i < bookArchitecture.chapterCount; i++) {
   408→      const chapterTitle = bookArchitecture.chapterTitles[i] ?? `Chapter ${i + 1}`;
   409→
   410→      const prompt = `Write a comprehensive, original chapter titled "${chapterTitle}" for a book about "${opportunity.topic}".
   411→
   412→Target audience: ${opportunity.targetAudience}
   413→Genre: ${opportunity.genre}
   414→Tone: ${opportunity.tone}
   415→Purpose: ${opportunity.purpose}
   416→
   417→Requirements:
   418→- Write ORIGINAL content that models principles and structure (AC-032)
   419→- Include 3-5 major sections with subheadings
   420→- Each section should have 2-3 substantial paragraphs
   421→- Include practical insights the reader can apply
   422→- Use clear, engaging language appropriate for the audience
   423→- Do NOT copy or closely paraphrase existing sources
   424→
   425→Format the response as JSON with this structure:
   426→{
   427→  "sections": [
   428→    {
   429→      "heading": "Section Title",
   430→      "level": 2,
   431→      "paragraphs": ["paragraph 1 text...", "paragraph 2 text..."]
   432→    }
   433→  ]
   434→}`;
   435→
   436→      try {
   437→        const response = await zai.chat.completions.create({
   438→          messages: [
   439→            {
   440→              role: "system",
   441→              content: `You are a professional author writing original book content. You model principles and structure without copying literal content from references. You write in ${opportunity.tone} tone for ${opportunity.targetAudience}. Always respond with valid JSON.`,
   442→            },
   443→            { role: "user", content: prompt },
   444→          ],
   445→        });
   446→
   447→        const rawContent = response.choices[0]?.message?.content ?? "{}";
   448→
   449→        // Parse the generated content
   450→        let sections: SectionContent[];
   451→        try {
   452→          // Extract JSON from the response (may be wrapped in markdown code blocks)
   453→          const jsonMatch = rawContent.match(/\{[\s\S]*\}/);
   454→          const parsed = JSON.parse(jsonMatch?.[0] ?? "{}");
   455→          sections = Array.isArray(parsed.sections)
   456→            ? parsed.sections.map((s: Record<string, unknown>) => ({
   457→                heading: String(s.heading ?? ""),
   458→                level: Number(s.level ?? 2),
   459→                paragraphs: Array.isArray(s.paragraphs)
   460→                  ? s.paragraphs.map((p: unknown) => String(p))
   461→                  : [],
   462→              }))
   463→            : [];
   464→        } catch {
   465→          // Fallback: treat entire response as a single section
   466→          sections = [{
   467→            heading: chapterTitle,
   468→            level: 2,
   469→            paragraphs: [rawContent],
   470→          }];
   471→        }
   472→
   473→        const wordCount = sections.reduce(
   474→          (sum, s) => sum + s.paragraphs.reduce((ps, p) => ps + p.split(/\s+/).length, 0),
   475→          0
   476→        );
   477→
   478→        chapters.push({
   479→          chapter_number: i + 1,
   480→          title: chapterTitle,
   481→          sections,
   482→          word_count: wordCount,
   483→        });
   484→
   485→        await this.tracer.addTool(executionId, "zai-chat-completions");
   486→      } catch (error) {
   487→        const msg = error instanceof Error ? error.message : String(error);
   488→        await this.tracer.addError(executionId, `Chapter ${i + 1} generation failed: ${msg}`);
   489→
   490→        // Add placeholder chapter on failure
   491→        chapters.push({
   492→          chapter_number: i + 1,
   493→          title: chapterTitle,
   494→          sections: [{
   495→            heading: chapterTitle,
   496→            level: 2,
   497→            paragraphs: [`Content for ${chapterTitle} will be regenerated.`],
   498→          }],
   499→          word_count: 10,
   500→        });
   501→      }
   502→    }
   503→
   504→    const totalWordCount = chapters.reduce((sum, ch) => sum + ch.word_count, 0);
   505→
   506→    return {
   507→      title: bookArchitecture.title,
   508→      subtitle: bookArchitecture.subtitle,
   509→      author: bookArchitecture.author,
   510→      chapters,
   511→      total_word_count: totalWordCount,
   512→    };
   513→  }

   515→  /**
   516→   * Generate chapter titles using ZAI.
   517→   */
   518→  private async generateChapterTitles(
   519→    opportunity: Opportunity
   520→  ): Promise<string[]> {
   521→    try {
   522→      const zai = await ZAI.create();
   523→
   524→      const response = await zai.chat.completions.create({
   525→        messages: [
   526→          {
   527→            role: "system",
   528→            content: "You are a book architect. Generate compelling, original chapter titles. Respond with a JSON array of strings only.",
   529→          },
   530→          {
   531→            role: "user",
   532→            content: `Generate ${opportunity.chapterCount} chapter titles for a book about "${opportunity.topic}" for ${opportunity.targetAudience}. The tone should be ${opportunity.tone}. Each title should be concise (2-6 words) and descriptive. Return as JSON array: ["Title 1", "Title 2", ...]`,
   533→          },
   534→        ],
   535→      });
   536→
   537→      const raw = response.choices[0]?.message?.content ?? "[]";
   538→      const jsonMatch = raw.match(/\[[\s\S]*\]/);
   539→      const parsed = JSON.parse(jsonMatch?.[0] ?? "[]");
   540→
   541→      if (Array.isArray(parsed) && parsed.length >= opportunity.chapterCount) {
   542→        return parsed.slice(0, opportunity.chapterCount).map(String);
   543→      }
   544→
   545→      // Fallback if ZAI returned wrong number of titles
   546→      return this.generateFallbackTitles(opportunity);
   547→    } catch {
   548→      return this.generateFallbackTitles(opportunity);
   549→    }
   550→  }
   551→
   552→  /**
   553→   * Fallback chapter titles when ZAI is unavailable.
   554→   */
   555→  private generateFallbackTitles(opportunity: Opportunity): string[] {
   556→    const titles: string[] = [];
   557→    for (let i = 1; i <= opportunity.chapterCount; i++) {
   558→      titles.push(`Chapter ${i}: ${opportunity.topic} — Part ${i}`);
   559→    }
   560→    return titles;
   561→  }
   562→
   563→  // ─── Repair Loop (AC-045) ─────────────────────────────────────────
   564→
   565→  /**
   566→   * Execute the repair loop: DETECT → DIAGNOSE → REPAIR → RENDER → VERIFY
   567→   *
   568→   * AC-045: Maximum 5 iterations. Each iteration:
   569→   *   1. DETECT — issues already detected in qaResult
   570→   *   2. DIAGNOSE — determine root causes
   571→   *   3. REPAIR — apply fixes
   572→   *   4. RENDER — re-layout and re-render
   573→   *   5. VERIFY — re-inspect
   574→   */
   575→  private async executeRepairLoop(
   576→    qaResult: QAResult,
   577→    content: BookContent,
   578→    editorial: EditorialSpecification,
   579→    artDir: ArtDirectionSpec,
   580→    layout: LayoutResult,
   581→    cover: CoverSpecification,
   582→    pipelineState: PipelineState,
   583→    pipelineSteps: PipelineStepEvidence[],
   584→    executionId: string
   585→  ): Promise<{
   586→    content: BookContent;
   587→    editorial: EditorialSpecification;
   588→    layout: LayoutResult;
   589→    cover: CoverSpecification;
   590→    qaResult: QAResult;
   591→  }> {
   592→    let currentContent = content;
   593→    let currentEditorial = editorial;
   594→    let currentLayout = layout;
   595→    let currentCover = cover;
   596→    let currentQA = qaResult;
   597→
   598→    // Only repair critical and auto-repairable issues
   599→    const repairableIssues = currentQA.issues.filter(
   600→      (i) => i.severity === "critical" && i.auto_repairable
   601→    );
   602→
   603→    if (repairableIssues.length === 0) {
   604→      // No auto-repairable issues — skip repair
   605→      return {
   606→        content: currentContent,
   607→        editorial: currentEditorial,
   608→        layout: currentLayout,
   609→        cover: currentCover,
   610→        qaResult: currentQA,
   611→      };
   612→    }
   613→
   614→    for (let iteration = 1; iteration <= pipelineState.max_repair_iterations; iteration++) {
   615→      pipelineState.repair_iteration = iteration;
   616→      pipelineState.current_step = `repair-${iteration}`;
   617→
   618→      const iterStart = Date.now();
   619→
   620→      // ── DETECT (already done) ──────────────────────────────────────
   621→      const criticalIssues = currentQA.issues.filter(
   622→        (i) => i.severity === "critical" && i.auto_repairable
   623→      );
   624→
   625→      if (criticalIssues.length === 0) {
   626→        // All critical issues resolved
   627→        break;
   628→      }
   629→
   630→      // ── DIAGNOSE ───────────────────────────────────────────────────
   631→      const diagnoses = this.visualQA.diagnose(criticalIssues);
   632→
   633→      // ── REPAIR ─────────────────────────────────────────────────────
   634→      for (const diagnosis of diagnoses) {
   635→        const issue = criticalIssues.find((i) => i.id === diagnosis.issue_id);
   636→        if (!issue) continue;
   637→
   638→        const repairResult: RepairedContent = this.visualQA.repair(
   639→          issue, currentContent, currentEditorial
   640→        );
   641→
   642→        if (repairResult.repaired) {
   643→          if (repairResult.updated_content) {
   644→            currentContent = repairResult.updated_content;
   645→          }
   646→          if (repairResult.updated_editorial) {
   647→            currentEditorial = repairResult.updated_editorial;
   648→          }
   649→        }
   650→
   651→        await this.tracer.addEvidence(executionId, {
   652→          repair_iteration: iteration,
   653→          issue_id: issue.id,
   654→          repair_strategy: diagnosis.repair_strategy,
   655→          repaired: repairResult.repaired,
   656→        });
   657→      }
   658→
   659→      // ── RENDER ─────────────────────────────────────────────────────
   660→      currentLayout = this.editorialDesign.layout(
   661→        currentContent, currentEditorial, artDir
   662→      );
   663→
   664→      // ── VERIFY ─────────────────────────────────────────────────────
   665→      currentQA = this.visualQA.inspect(
   666→        { page_count: currentLayout.page_count, layout: currentLayout, content: currentContent },
   667→        currentContent,
   668→        currentEditorial
   669→      );
   670→
   671→      // Record step evidence
   672→      pipelineSteps.push({
   673→        step: `repair-${iteration}`,
   674→        step_number: 9,
   675→        started_at: new Date(iterStart).toISOString(),
   676→        completed_at: new Date().toISOString(),
   677→        duration_ms: Date.now() - iterStart,
   678→        success: currentQA.passed,
   679→        output_id: randomUUID(),
   680→      });
   681→
   682→      // If QA now passes, exit repair loop
   683→      if (currentQA.passed) {
   684→        break;
   685→      }
   686→    }
   687→
   688→    return {
   689→      content: currentContent,
   690→      editorial: currentEditorial,
   691→      layout: currentLayout,
   692→      cover: currentCover,
   693→      qaResult: currentQA,
   694→    };
   695→  }

   699→  /**
   700→   * Run a single pipeline step with timing and error handling.
   701→   */
   702→  private async runStep<T>(
   703→    stepName: string,
   704→    stepNumber: number,
   705→    pipelineSteps: PipelineStepEvidence[],
   706→    executionId: string,
   707→    fn: () => Promise<T>
   708→  ): Promise<T> {
   709→    const start = Date.now();
   710→    try {
   711→      const result = await fn();
   712→      pipelineSteps.push({
   713→        step: stepName,
   714→        step_number: stepNumber,
   715→        started_at: new Date(start).toISOString(),
   716→        completed_at: new Date().toISOString(),
   717→        duration_ms: Date.now() - start,
   718→        success: true,
   719→        output_id: randomUUID(),
   720→      });
   721→      await this.tracer.addSkill(executionId, `book-factory-${stepName}`);
   722→      return result;
   723→    } catch (error) {
   724→      const message = error instanceof Error ? error.message : String(error);
   725→      pipelineSteps.push({
   726→        step: stepName,
   727→        step_number: stepNumber,
   728→        started_at: new Date(start).toISOString(),
   729→        completed_at: new Date().toISOString(),
   730→        duration_ms: Date.now() - start,
   731→        success: false,
   732→        output_id: randomUUID(),
   733→        error: message,
   734→      });
   735→      await this.tracer.addError(executionId, `${stepName}: ${message}`);
   736→      throw error;
   737→    }
   738→  }

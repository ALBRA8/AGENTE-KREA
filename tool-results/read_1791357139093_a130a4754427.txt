   146→/**

   149→ * AC-015: The MUST/SHOULD/COULD/OUT boundary is explicit and enforced.

   153→type ArchZAIInstance = Awaited<ReturnType<typeof ZAI.create>>;

   155→export class ProductArchitectureEngine {
   156→  private memory: MemoryManager;
   157→  private tracer: ExecutionTracer;
   158→  private zai: ArchZAIInstance | null = null;

   160→  constructor(memory?: MemoryManager, tracer?: ExecutionTracer) {

   165→  private async getZAI(): Promise<ArchZAIInstance> {
   166→    if (!this.zai) {
   167→      this.zai = await ZAI.create();

   172→  /**
   173→   * architect — Design the product architecture.
   174→   *
   175→   * Produces a complete ProductArchitecture with:
   176→   *   - JTBD framework
   177→   *   - Value proposition
   178→   *   - MoSCoW scope (AC-015)
   179→   *   - MVP definition
   180→   *   - Validation and build requirements
   181→   *   - Success criteria
   182→   */
   183→  async architect(
   184→    opportunity: Opportunity,
   185→    decision: ProductDecision,
   186→    evidence: ProductEvidence[]
   187→  ): Promise<ProductArchitecture> {
   188→    const executionId = await this.tracer.start("product_architecture:architect", {
   189→      inputs: { opportunity: opportunity.name, decision: decision.decision },
   190→    });
   191→
   192→    try {
   193→      const zai = await this.getZAI();
   194→
   195→      // AI-powered architecture design
   196→      const response = await zai.chat.completions.create({
   197→        messages: [
   198→          {
   199→            role: "system",
   200→            content: `You are a product architect. Design a complete product architecture for the given opportunity.
   201→
   202→Your response MUST be valid JSON with this structure:
   203→{
   204→  "jtbd": [{ "situation": "...", "motivation": "...", "outcome": "...", "priority": "primary|secondary|tertiary" }],
   205→  "value_proposition": "...",
   206→  "desired_outcome": "...",
   207→  "ux_description": "...",
   208→  "differentiation": "...",
   209→  "monetization": "...",
   210→  "MUST": [{ "description": "...", "rationale": "...", "effort": "hours|days|weeks", "inMvp": true }],
   211→  "SHOULD": [{ "description": "...", "rationale": "...", "effort": "hours|days|weeks", "inMvp": false }],
   212→  "COULD": [{ "description": "...", "rationale": "...", "effort": "hours|days|weeks", "inMvp": false }],
   213→  "OUT": [{ "description": "...", "rationale": "...", "effort": "hours|days|weeks", "inMvp": false }],
   214→  "risks": ["..."],
   215→  "assumptions": ["..."],
   216→  "unknowns": ["..."],
   217→  "success_criteria": ["..."]
   218→}
   219→
   220→CRITICAL RULES (AC-015 - Scope Control):
   221→1. MUST items are NON-NEGOTIABLE for launch — without them, the product doesn't solve the core problem
   222→2. SHOULD items are important but the product launches without them
   223→3. COULD items are nice-to-have — only if time permits
   224→4. OUT items are EXPLICITLY EXCLUDED — they prevent scope creep
   225→5. Every OUT item needs a rationale explaining WHY it's excluded
   226→6. MVP MUST be a subset of MUST items only
   227→7. Include at least 2 OUT items to demonstrate scope boundary awareness`,
   228→          },
   229→          {
   230→            role: "user",
   231→            content: `Opportunity: ${opportunity.name}
   232→Domain: ${opportunity.domain}
   233→Problem: ${opportunity.problem}
   234→Solution: ${opportunity.solution}
   235→Audience: ${opportunity.audience}
   236→Format: ${opportunity.format || "unspecified"}
   237→Decision: ${decision.decision}
   238→
   239→Evidence summary:
   240→${evidence.slice(0, 10).map((e) => `- [${e.type}] ${e.content.substring(0, 100)}`).join("\n")}
   241→
   242→Design the product architecture with explicit scope boundaries.`,
   243→          },
   244→        ],
   245→      });
   246→
   247→      const archText = response.choices[0]?.message?.content || "";
   248→      const parsed = this.parseArchitectureResponse(archText);
   249→
   250→      // Build the complete architecture
   251→      const architecture = this.buildArchitecture(
   252→        opportunity,
   253→        decision,
   254→        evidence,
   255→        parsed
   256→      );
   257→
   258→      // Persist to memory
   259→      await this.memory.store(
   260→        "krea",
   261→        `product:${opportunity.domain}:architecture`,
   262→        "SEMANTIC",
   263→        {
   264→          opportunity: opportunity.name,
   265→          decision: decision.decision,
   266→          mustCount: architecture.MUST.length,
   267→          shouldCount: architecture.SHOULD.length,
   268→          couldCount: architecture.COULD.length,
   269→          outCount: architecture.OUT.length,
   270→          mvpItems: architecture.mvp_definition.items.length,
   271→        },
   272→        {
   273→          source: "product_architecture_engine",
   274→          sourceType: "AGENT",
   275→          confidence: 0.8,
   276→          truthLevel: "ESTIMATED",
   277→          scope: "agent",
   278→        }
   279→      );
   280→
   281→      await this.tracer.succeed(executionId, {
   282→        mustCount: architecture.MUST.length,
   283→        outCount: architecture.OUT.length,
   284→      });
   285→
   286→      return architecture;
   287→    } catch (error) {
   288→      const err = error instanceof Error ? error : new Error(String(error));
   289→      await this.tracer.fail(executionId, { message: err.message, outputs: {} });
   290→
   291→      // Fallback: produce a minimal architecture from the opportunity
   292→      return this.fallbackArchitecture(opportunity, decision, evidence);
   293→    }
   294→  }

   298→  /** Parse the AI response into structured architecture parts */
   299→  private parseArchitectureResponse(text: string): Partial<ProductArchitecture> {
   300→    try {
   301→      // Try to extract JSON from the response
   302→      const jsonMatch = text.match(/\{[\s\S]*\}/);
   303→      if (jsonMatch) {
   304→        const parsed = JSON.parse(jsonMatch[0]);
   305→        return parsed;
   306→      }
   307→    } catch {
   308→      // Parse failed — will use fallback
   309→    }
   310→    return {};
   311→  }

   313→  /** Build the complete ProductArchitecture from parsed parts */
   314→  private buildArchitecture(
   315→    opportunity: Opportunity,
   316→    decision: ProductDecision,
   317→    evidence: ProductEvidence[],
   318→    parsed: Partial<ProductArchitecture>
   319→  ): ProductArchitecture {
   320→    const evidenceIds = evidence.map((e) => e.id);
   321→
   322→    // JTBD
   323→    const jtbd: JobToBeDone[] = (parsed.jtbd || []).map((j: Record<string, unknown>) => ({
   324→      statement: `When ${j.situation || "context"}, I want to ${j.motivation || "action"}, so I can ${j.outcome || "result"}`,
   325→      situation: String(j.situation || ""),
   326→      motivation: String(j.motivation || ""),
   327→      outcome: String(j.outcome || ""),
   328→      priority: (["primary", "secondary", "tertiary"].includes(String(j.priority)) ? j.priority : "primary") as JobToBeDone["priority"],
   329→    }));
   330→
   331→    if (jtbd.length === 0) {
   332→      jtbd.push({
   333→        statement: `When facing ${opportunity.problem}, I want to ${opportunity.solution}, so I can solve my problem effectively`,
   334→        situation: opportunity.problem,
   335→        motivation: opportunity.solution,
   336→        outcome: "Effective problem resolution",
   337→        priority: "primary",
   338→      });
   339→    }
   340→
   341→    // MoSCoW items (AC-015)
   342→    const MUST: MoscowItem[] = this.buildMoscowItems(parsed.MUST, [
   343→      {
   344→        description: `Core problem solution: ${opportunity.solution}`,
   345→        rationale: "Solves the primary problem identified in the opportunity",
   346→        evidenceIds,
   347→        effort: "weeks",
   348→        inMvp: true,
   349→      },
   350→    ], true, evidenceIds);
   351→
   352→    const SHOULD: MoscowItem[] = this.buildMoscowItems(parsed.SHOULD, [], false, evidenceIds);
   353→
   354→    const COULD: MoscowItem[] = this.buildMoscowItems(parsed.COULD, [], false, evidenceIds);
   355→
   356→    // OUT items — AC-015: explicitly excluded to prevent scope creep
   357→    const OUT: MoscowItem[] = this.buildMoscowItems(parsed.OUT, [
   358→      {
   359→        description: "Features not related to the core problem statement",
   360→        rationale: "Scope boundary — anything not directly addressing the core problem is OUT to prevent scope creep (AC-015)",
   361→        evidenceIds: [],
   362→        effort: "weeks",
   363→        inMvp: false,
   364→      },
   365→      {
   366→        description: "Multi-language support (unless core problem is language-specific)",
   367→        rationale: "Internationalization is a post-MVP concern. Including it now violates scope control (AC-015)",
   368→        evidenceIds: [],
   369→        effort: "months",
   370→        inMvp: false,
   371→      },
   372→    ], false, evidenceIds);
   373→
   374→    // MVP definition — subset of MUST items only (AC-015)
   375→    const mvpItems = MUST.filter((item) => item.inMvp);
   376→    const mvp_definition = {
   377→      scope: `MVP: Core problem solution for ${opportunity.audience}`,
   378→      items: mvpItems.length > 0 ? mvpItems : MUST.slice(0, 1),
   379→      timebox: this.estimateMvpTimebox(opportunity, decision),
   380→      successMetrics: (parsed.success_criteria as string[]) || [
   381→        "Users can solve the core problem end-to-end",
   382→        "Positive feedback from target audience",
   383→        "Engagement metrics meet minimum threshold",
   384→      ],
   385→    };
   386→
   387→    return {
   388→      client: opportunity.audience,
   389→      problem: opportunity.problem,
   390→      jtbd,
   391→      desired_outcome: parsed.desired_outcome || `Effectively solve ${opportunity.problem} for ${opportunity.audience}`,
   392→      value_proposition: parsed.value_proposition || `${opportunity.solution} — the most effective way for ${opportunity.audience} to solve ${opportunity.problem}`,
   393→      format: opportunity.format || "unspecified",
   394→      MUST,
   395→      SHOULD,
   396→      COULD,
   397→      OUT,
   398→      ux_description: parsed.ux_description || `Clean, focused interface that guides ${opportunity.audience} through solving ${opportunity.problem} with minimal friction`,
   399→      differentiation: parsed.differentiation || this.inferDifferentiation(evidence),
   400→      monetization: parsed.monetization || this.inferMonetization(opportunity, evidence),
   401→      mvp_definition,
   402→      dependencies: this.inferDependencies(opportunity, decision),
   403→      risks: (parsed.risks as string[]) || this.inferRisks(evidence),
   404→      assumptions: (parsed.assumptions as string[]) || [
   405→        `The problem (${opportunity.problem}) is real and painful for ${opportunity.audience}`,
   406→        `The proposed solution (${opportunity.solution}) effectively addresses the problem`,
   407→        `${opportunity.audience} is willing to adopt a new solution`,
   408→      ],
   409→      unknowns: (parsed.unknowns as string[]) || decision.uncertainties,
   410→      validation_requirements: this.buildValidationRequirements(decision, evidence),
   411→      build_requirements: this.buildBuildRequirements(opportunity, decision, MUST),
   412→      success_criteria: (parsed.success_criteria as string[]) || [
   413→        "Core problem is solved for target audience",
   414→        "User satisfaction score >= 7/10",
   415→        "Retention rate >= 40% after 30 days",
   416→        "Time-to-value < 5 minutes for new users",
   417→      ],
   418→      architectedAt: new Date().toISOString(),
   419→    };
   420→  }

   422→  /** Build MoSCoW items from parsed data with defaults */
   423→  private buildMoscowItems(
   424→    parsedItems: unknown,
   425→    defaults: MoscowItem[],
   426→    isMust: boolean,
   427→    evidenceIds: string[]
   428→  ): MoscowItem[] {
   429→    if (Array.isArray(parsedItems) && parsedItems.length > 0) {
   430→      return parsedItems.map((item: Record<string, unknown>) => ({
   431→        description: String(item.description || ""),
   432→        rationale: String(item.rationale || "Derived from product analysis"),
   433→        evidenceIds: (item.evidenceIds as string[]) || evidenceIds.slice(0, 3),
   434→        effort: (["hours", "days", "weeks", "months"].includes(String(item.effort)) ? item.effort : "days") as MoscowItem["effort"],
   435→        inMvp: isMust ? (item.inMvp !== false) : false,
   436→      }));
   437→    }
   438→    return defaults;
   439→  }

   441→  /** Estimate MVP timebox based on product format */
   442→  private estimateMvpTimebox(opportunity: Opportunity, decision: ProductDecision): string {
   443→    const format = (opportunity.format || "").toLowerCase();
   444→
   445→    if (["ebook", "guía", "guide", "template"].some((f) => format.includes(f))) {
   446→      return "1-2 weeks";
   447→    }
   448→    if (["workbook", "checklist", "course"].some((f) => format.includes(f))) {
   449→      return "2-3 weeks";
   450→    }
   451→    if (["app", "web app", "tool", "landing page"].some((f) => format.includes(f))) {
   452→      return "4-6 weeks";
   453→    }
   454→    if (["saas", "platform", "api"].some((f) => format.includes(f))) {
   455→      return "8-12 weeks";
   456→    }
   457→    return "4-6 weeks";
   458→  }

   460→  /** Infer differentiation from evidence */
   461→  private inferDifferentiation(evidence: ProductEvidence[]): string {
   462→    const diffEvidence = evidence.filter((e) => e.type === "DIFFERENTIATION");
   463→    if (diffEvidence.length > 0) {
   464→      return diffEvidence.map((e) => e.content).join("; ");
   465→    }
   466→    return "Unique approach to solving the identified problem — differentiation to be refined based on competitive analysis";
   467→  }

   469→  /** Infer monetization model from opportunity and evidence */
   470→  private inferMonetization(opportunity: Opportunity, evidence: ProductEvidence[]): string {
   471→    const pricingEvidence = evidence.filter((e) => e.type === "PRICING");
   472→    if (pricingEvidence.length > 0) {
   473→      return `Based on market pricing evidence: ${pricingEvidence[0].content}`;
   474→    }
   475→
   476→    const format = (opportunity.format || "").toLowerCase();
   477→    if (["ebook", "guía", "guide", "template"].some((f) => format.includes(f))) {
   478→      return "One-time purchase (digital product)";
   479→    }
   480→    if (["course", "curso"].some((f) => format.includes(f))) {
   481→      return "Course enrollment fee (one-time or installment)";
   482→    }
   483→    if (["saas", "app"].some((f) => format.includes(f))) {
   484→      return "Subscription (monthly/annual) with free tier";
   485→    }
   486→    return "To be determined based on market research and validation";
   487→  }

   489→  /** Infer dependencies */
   490→  private inferDependencies(opportunity: Opportunity, decision: ProductDecision): string[] {
   491→    const deps: string[] = [];
   492→    const format = (opportunity.format || "").toLowerCase();
   493→
   494→    if (["app", "web app", "saas", "platform", "api"].some((f) => format.includes(f))) {
   495→      deps.push("Hosting/infrastructure");
   496→      deps.push("Authentication system");
   497→      deps.push("Database");
   498→    }
   499→
   500→    if (decision.constructor?.secondary) {
   501→      deps.push(`${decision.constructor.secondary} availability for technical implementation`);
   502→    }
   503→
   504→    deps.push("Content creation tools");
   505→    return deps;
   506→  }

   508→  /** Infer risks from evidence */
   509→  private inferRisks(evidence: ProductEvidence[]): string[] {
   510→    const risks: string[] = [];
   511→    const compEvidence = evidence.filter((e) => e.type === "COMPETITION");
   512→
   513→    if (compEvidence.length >= 5) {
   514→      risks.push("Crowded market — may struggle to differentiate");
   515→    }
   516→    if (compEvidence.some((e) => e.confidence >= 0.8)) {
   517→      risks.push("Strong incumbent competitor with high market share");
   518→    }
   519→
   520→    risks.push("Market timing risk — conditions may change during build");
   521→    risks.push("Adoption risk — target audience may resist change");
   522→
   523→    return risks;
   524→  }

   530→  ): ValidationRequirement[] {

   532→
   533→    // Problem validation
   534→    requirements.push({
   535→      what: "Problem intensity and frequency",
   536→      how: "User interviews with 10+ target users",
   537→      successCriteria: "7+ out of 10 users confirm the problem is real and painful",
   538→      threshold: "10 interviews",
   539→      priority: "critical",
   540→    });
   541→
   542→    // Solution validation
   543→    requirements.push({
   544→      what: "Solution desirability",
   545→      how: "Present solution concept to target users, measure interest",
   546→      successCriteria: "60%+ express strong interest in the solution",
   547→      threshold: "20 responses",
   548→      priority: "critical",
   549→    });
   550→
   551→    // Payment validation
   552→    const pricingEvidence = evidence.filter((e) => e.type === "PRICING");
   553→    if (pricingEvidence.length === 0) {
   554→      requirements.push({
   555→        what: "Willingness to pay",
   556→        how: "Pre-order or deposit test at proposed price point",
   557→        successCriteria: "5+ pre-orders or 20%+ expressing purchase intent",
   558→        threshold: "50 survey responses",
   559→        priority: "high",
   560→      });
   561→    }
   562→
   563→    // Usability validation
   564→    requirements.push({
   565→      what: "Usability and time-to-value",
   566→      how: "Usability test with MVP prototype",
   567→      successCriteria: "80%+ can complete core task without help",
   568→      threshold: "5 usability tests",
   569→      priority: "high",
   570→    });
   571→
   572→    return requirements;
   573→  }

   575→  /** Build requirements for construction */
   576→  private buildBuildRequirements(
   577→    opportunity: Opportunity,
   578→    decision: ProductDecision,
   579→    mustItems: MoscowItem[]
   580→  ): BuildRequirement[] {
   581→    const requirements: BuildRequirement[] = [];
   582→    const format = (opportunity.format || "").toLowerCase();
   583→

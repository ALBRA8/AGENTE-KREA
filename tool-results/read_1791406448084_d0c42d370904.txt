   186→/**
   187→ * Generate a launch package from product data.
   188→ *
   189→ * Assembles all elements from existing data:
   190→ * - Product info, offer, positioning, customer, pricing
   191→ * - FAQ generated from common objections + product features
   192→ * - Objection responses with honest evidence tags
   193→ * - CTA based on readiness state
   194→ * - Proof requirements with honest availability status
   195→ * - Integrity check catching all dishonest patterns
   196→ * - Disclaimers when evidence is insufficient
   197→ *
   198→ * CRITICAL: Never fabricates missing data. If something is missing,
   199→ * it's marked as empty/missing, never invented.
   200→ */
   201→export function generateLaunchPackage(
   202→  productDef: ProductDefinition,
   203→  packaging: Packaging | undefined,
   204→  offerArch: OfferArchitecture | undefined,
   205→  assets: LaunchAssets | undefined,
   206→  qaReport: ProductQAReport | undefined,
   207→  readiness: CommercialReadiness | undefined
   208→): LaunchPackage {
   209→  const packageId = `lp_${randomUUID().replace(/-/g, "").substring(0, 20)}`;
   210→  const generatedAt = new Date();
   211→
   212→  // ── Product Info ──────────────────────────────────────────────────────────
   213→  const product: LaunchProductInfo = {
   214→    name: productDef.name,
   215→    type: productDef.type,
   216→    description: productDef.description,
   217→    version: productDef.version,
   218→  };
   219→
   220→  // ── Offer ─────────────────────────────────────────────────────────────────
   221→  const offer: OfferInfo = offerArch
   222→    ? {
   223→        problem: offerArch.problem,
   224→        promise: offerArch.promise,
   225→        mechanism: offerArch.mechanism,
   226→      }
   227→    : {
   228→        problem: productDef.painPoints.join("; ") || "Not defined",
   229→        promise: productDef.valueProposition || "Not defined",
   230→        mechanism: "Not defined",
   231→      };
   232→
   233→  // ── Positioning ───────────────────────────────────────────────────────────
   234→  const positioning = buildPositioning(productDef, offerArch);
   235→
   236→  // ── Target Customer ───────────────────────────────────────────────────────
   237→  const targetCustomer: CustomerInfo = {
   238→    description: productDef.targetCustomer || "Not defined",
   239→    painPoints: productDef.painPoints.length > 0
   240→      ? productDef.painPoints
   241→      : ["Not defined"],
   242→  };
   243→
   244→  // ── Core Promise ──────────────────────────────────────────────────────────
   245→  const corePromise = offer.promise;
   246→
   247→  // ── Benefits / Included / Bonuses ─────────────────────────────────────────
   248→  const benefits = offerArch?.benefits || [productDef.valueProposition || "Not defined"];
   249→  const included = offerArch?.included || [];
   250→  const bonuses = offerArch?.bonuses || [];
   251→
   252→  // ── Price ─────────────────────────────────────────────────────────────────
   253→  const price: PriceInfo = {
   254→    amount: productDef.price.amount,
   255→    currency: productDef.price.currency,
   256→    model: productDef.price.model,
   257→  };
   258→
   259→  // ── Delivery ──────────────────────────────────────────────────────────────
   260→  const delivery: DeliveryInfo = packaging
   261→    ? {
   262→        method: packaging.deliveryFormat || "Not defined",
   263→        format: packaging.deliveryFormat ? [packaging.deliveryFormat] : [],
   264→        access: packaging.accessMethod || "Not defined",
   265→      }
   266→    : {
   267→        method: "Not defined",
   268→        format: [],
   269→        access: "Not defined",
   270→      };
   271→
   272→  // ── FAQ ───────────────────────────────────────────────────────────────────
   273→  const faq = generateFAQ(productDef, offerArch, packaging);
   274→
   275→  // ── Objection Responses ───────────────────────────────────────────────────
   276→  const objections = generateObjectionResponses(productDef, readiness);
   277→
   278→  // ── CTA ───────────────────────────────────────────────────────────────────
   279→  const cta = generateCTA(readiness?.state);
   280→
   281→  // ── Assets ────────────────────────────────────────────────────────────────
   282→  const assetInfos: AssetInfo[] = assets?.files || [];
   283→
   284→  // ── Proof Requirements ────────────────────────────────────────────────────
   285→  const proofRequirements = generateProofRequirements(
   286→    productDef,
   287→    readiness
   288→  );
   289→
   290→  // ── Integrity Check ───────────────────────────────────────────────────────
   291→  const integrityCheck = runIntegrityCheck(productDef, offerArch);
   292→
   293→  // ── Disclaimers ───────────────────────────────────────────────────────────
   294→  const disclaimers = generateDisclaimers(readiness, integrityCheck, proofRequirements);
   295→
   296→  // ── Evidence ──────────────────────────────────────────────────────────────
   297→  const hasVerifiedData = readiness && readiness.score.overall > 0.5;
   298→  const evidence: EvidenceTag = hasVerifiedData ? "VERIFIED" : "INFERRED";
   299→
   300→  return {
   301→    packageId,
   302→    productId: productDef.name.toLowerCase().replace(/\s+/g, "-"),
   303→    version: productDef.version,
   304→    product,
   305→    offer,
   306→    positioning,
   307→    targetCustomer,
   308→    corePromise,
   309→    benefits,
   310→    included,
   311→    bonuses,
   312→    price,
   313→    delivery,
   314→    faq,
   315→    objections,
   316→    cta,
   317→    assets: assetInfos,
   318→    proofRequirements,
   319→    disclaimers,
   320→    integrityCheck,
   321→    evidence,
   322→    generatedAt,
   323→  };
   324→}

   326→// ─── Validation ───────────────────────────────────────────────────────────────
   327→
   328→/**
   329→ * Validate a launch package.
   330→ *
   331→ * Checks:
   332→ * - All required fields are populated
   333→ * - Integrity passes
   334→ * - No fake claims detected
   335→ */
   336→export function validateLaunchPackage(
   337→  pkg: LaunchPackage
   338→): { valid: boolean; issues: string[] } {
   339→  const issues: string[] = [];
   340→
   341→  // Required string fields must be non-empty
   342→  const requiredStrings: Array<{ field: string; value: string }> = [
   343→    { field: "product.name", value: pkg.product.name },
   344→    { field: "product.type", value: pkg.product.type },
   345→    { field: "product.description", value: pkg.product.description },
   346→    { field: "product.version", value: pkg.product.version },
   347→    { field: "offer.problem", value: pkg.offer.problem },
   348→    { field: "offer.promise", value: pkg.offer.promise },
   349→    { field: "offer.mechanism", value: pkg.offer.mechanism },
   350→    { field: "positioning", value: pkg.positioning },
   351→    { field: "targetCustomer.description", value: pkg.targetCustomer.description },
   352→    { field: "corePromise", value: pkg.corePromise },
   353→    { field: "cta", value: pkg.cta },
   354→  ];
   355→
   356→  for (const { field, value } of requiredStrings) {
   357→    if (!value || value === "Not defined") {
   358→      issues.push(`Required field "${field}" is empty or undefined`);
   359→    }
   360→  }
   361→
   362→  // Price must be positive
   363→  if (pkg.price.amount <= 0) {
   364→    issues.push("Price amount must be positive");
   365→  }
   366→
   367→  // Benefits must be non-empty
   368→  if (pkg.benefits.length === 0) {
   369→    issues.push("Product must have at least one benefit");
   370→  }
   371→
   372→  // Integrity must pass
   373→  if (!pkg.integrityCheck.passes) {
   374→    issues.push(`Integrity check failed: ${pkg.integrityCheck.issues.join("; ")}`);
   375→  }
   376→
   377→  // Check for fake claims in the package
   378→  const fakeClaimIssues = getIntegrityIssues(pkg);
   379→  if (fakeClaimIssues.length > 0) {
   380→    issues.push(`Fake claims detected: ${fakeClaimIssues.join("; ")}`);
   381→  }
   382→
   383→  // Delivery must be defined
   384→  if (pkg.delivery.method === "Not defined") {
   385→    issues.push("Delivery method is not defined");
   386→  }
   387→
   388→  return {
   389→    valid: issues.length === 0,
   390→    issues,
   391→  };
   392→}
   393→
   394→// ─── Integrity Check ──────────────────────────────────────────────────────────
   395→
   396→/**
   397→ * Detect integrity issues in a launch package.
   398→ *
   399→ * Checks for:
   400→ * - Fake testimonials
   401→ * - Guaranteed results
   402→ * - Invented statistics
   403→ * - False authority claims
   404→ * - False scarcity
   405→ */
   406→export function getIntegrityIssues(pkg: LaunchPackage): string[] {
   407→  const issues: string[] = [];
   408→
   409→  // Check all text fields for dishonest patterns
   410→  const textFields = [
   411→    pkg.product.description,
   412→    pkg.offer.problem,
   413→    pkg.offer.promise,
   414→    pkg.offer.mechanism,
   415→    pkg.positioning,
   416→    pkg.corePromise,
   417→    ...pkg.benefits,
   418→    ...pkg.included,
   419→    ...pkg.bonuses,
   420→    ...pkg.faq.map((f) => `${f.question} ${f.answer}`),
   421→    ...pkg.objections.map((o) => `${o.objection} ${o.response}`),
   422→    pkg.cta,
   423→    ...pkg.disclaimers,
   424→  ];
   425→
   426→  const fullText = textFields.join(" ");
   427→
   428→  // Check for fake testimonials
   429→  for (const pattern of FAKE_TESTIMONIAL_PATTERNS) {
   430→    if (pattern.test(fullText)) {
   431→      issues.push("Fake testimonial pattern detected in product text");
   432→      break;
   433→    }
   434→  }
   435→
   436→  // Check for guaranteed results
   437→  for (const pattern of GUARANTEED_RESULT_PATTERNS) {
   438→    if (pattern.test(fullText)) {
   439→      issues.push("Guaranteed-result claim detected in product text");
   440→      break;
   441→    }
   442→  }
   443→
   444→  // Check for invented statistics
   445→  for (const pattern of INVENTED_STATISTICS_PATTERNS) {
   446→    if (pattern.test(fullText)) {
   447→      issues.push("Invented-statistics pattern detected in product text");
   448→      break;
   449→    }
   450→  }
   451→
   452→  // Check for false authority
   453→  for (const pattern of FALSE_AUTHORITY_PATTERNS) {
   454→    if (pattern.test(fullText)) {
   455→      issues.push("False authority claim detected in product text");
   456→      break;
   457→    }
   458→  }
   459→
   460→  // Check for false scarcity
   461→  for (const pattern of FALSE_SCARCITY_PATTERNS) {
   462→    if (pattern.test(fullText)) {
   463→      issues.push("False scarcity pattern detected in product text");
   464→      break;
   465→    }
   466→  }
   467→
   468→  return issues;
   469→}

   471→// ─── Internal: Integrity Check for ProductDefinition ─────────────────────────
   472→
   473→function runIntegrityCheck(
   474→  productDef: ProductDefinition,
   475→  offerArch: OfferArchitecture | undefined
   476→): IntegrityCheck {
   477→  const issues: string[] = [];
   478→
   479→  // Check product definition fields
   480→  const textsToCheck = [
   481→    productDef.description,
   482→    productDef.valueProposition,
   483→    productDef.differentiation,
   484→    ...productDef.painPoints,
   485→  ];
   486→  if (offerArch) {
   487→    textsToCheck.push(
   488→      offerArch.problem,
   489→      offerArch.promise,
   490→      offerArch.mechanism,
   491→      ...offerArch.benefits,
   492→      ...offerArch.bonuses
   493→    );
   494→  }
   495→
   496→  const allText = textsToCheck.join(" ");
   497→
   498→  // Run all pattern checks
   499→  for (const pattern of FAKE_TESTIMONIAL_PATTERNS) {
   500→    if (pattern.test(allText)) {
   501→      issues.push("Fake testimonial language detected");
   502→      break;
   503→    }
   504→  }
   505→
   506→  for (const pattern of GUARANTEED_RESULT_PATTERNS) {
   507→    if (pattern.test(allText)) {
   508→      issues.push("Guaranteed-result claim detected");
   509→      break;
   510→    }
   511→  }
   512→
   513→  for (const pattern of INVENTED_STATISTICS_PATTERNS) {
   514→    if (pattern.test(allText)) {
   515→      issues.push("Invented statistics detected");
   516→      break;
   517→    }
   518→  }
   519→
   520→  for (const pattern of FALSE_AUTHORITY_PATTERNS) {
   521→    if (pattern.test(allText)) {
   522→      issues.push("False authority claim detected");
   523→      break;
   524→    }
   525→  }
   526→
   527→  for (const pattern of FALSE_SCARCITY_PATTERNS) {
   528→    if (pattern.test(allText)) {
   529→      issues.push("False scarcity pattern detected");
   530→      break;
   531→    }
   532→  }
   533→
   534→  return {
   535→    passes: issues.length === 0,
   536→    issues,
   537→  };
   538→}

   540→// ─── Internal: FAQ Generation ─────────────────────────────────────────────────
   541→
   542→function generateFAQ(
   543→  productDef: ProductDefinition,
   544→  offerArch: OfferArchitecture | undefined,
   545→  packaging: Packaging | undefined
   546→): FAQItem[] {
   547→  const faq: FAQItem[] = [];
   548→
   549→  // FAQ from common objections
   550→  faq.push({
   551→    question: `What is ${productDef.name}?`,
   552→    answer: productDef.description,
   553→  });
   554→
   555→  if (productDef.targetCustomer) {
   556→    faq.push({
   557→      question: `Who is ${productDef.name} for?`,
   558→      answer: `${productDef.name} is designed for ${productDef.targetCustomer}.`,
   559→    });
   560→  }
   561→
   562→  if (offerArch?.mechanism) {
   563→    faq.push({
   564→      question: `How does ${productDef.name} work?`,
   565→      answer: offerArch.mechanism,
   566→    });
   567→  }
   568→
   569→  faq.push({
   570→    question: `How much does ${productDef.name} cost?`,
   571→    answer: `${productDef.name} is ${productDef.price.currency} ${productDef.price.amount} (${productDef.price.model}).`,
   572→  });
   573→
   574→  if (packaging?.deliveryFormat) {
   575→    faq.push({
   576→      question: `How will I receive ${productDef.name}?`,
   577→      answer: `You'll receive ${productDef.name} as ${packaging.deliveryFormat}${packaging.accessMethod ? `, accessible via ${packaging.accessMethod}` : ""}.`,
   578→    });
   579→  }
   580→
   581→  if (packaging?.prerequisites && packaging.prerequisites.length > 0) {
   582→    faq.push({
   583→      question: `What do I need before using ${productDef.name}?`,
   584→      answer: `Prerequisites: ${packaging.prerequisites.join(", ")}.`,
   585→    });
   586→  }
   587→
   588→  if (productDef.differentiation) {
   589→    faq.push({
   590→      question: `How is ${productDef.name} different from alternatives?`,
   591→      answer: productDef.differentiation,
   592→    });
   593→  }
   594→
   595→  // Standard objection-based FAQs
   596→  faq.push({
   597→    question: "What if it doesn't work for me?",
   598→    answer: "We encourage you to review the product details carefully before purchasing to ensure it meets your needs.",
   599→  });
   600→
   601→  return faq;
   602→}

   604→// ─── Internal: Objection Responses ───────────────────────────────────────────
   605→
   606→function generateObjectionResponses(
   607→  productDef: ProductDefinition,
   608→  readiness: CommercialReadiness | undefined
   609→): ObjectionResponse[] {
   610→  const objections: ObjectionResponse[] = [];
   611→
   612→  // Price objection
   613→  objections.push({
   614→    objection: "It's too expensive",
   615→    response: `At ${productDef.price.currency} ${productDef.price.amount}, ${productDef.name} is priced to reflect the value it delivers. Consider the cost of the problem it solves: ${productDef.painPoints[0] || "the challenges you're facing"}.`,
   616→    evidence: readiness && readiness.score.commercialClarity.score > 0.5
   617→      ? "VERIFIED"
   618→      : "INFERRED",
   619→  });
   620→
   621→  // Time objection
   622→  objections.push({
   623→    objection: "I don't have time for this",
   624→    response: `${productDef.name} is designed to be ${productDef.type === "ebook" || productDef.type === "guide" ? "read at your own pace" : "implemented incrementally"}. You can start seeing results with focused effort.`,
   625→    evidence: "INFERRED",
   626→  });
   627→
   628→  // Skepticism objection
   629→  objections.push({
   630→    objection: "I've tried similar things before and they didn't work",
   631→    response: productDef.differentiation
   632→      ? `${productDef.name} is different because: ${productDef.differentiation}`
   633→      : "We understand your skepticism. We encourage you to review the product details to see if it's right for your specific situation.",
   634→    evidence: productDef.differentiation ? "VERIFIED" : "INFERRED",
   635→  });
   636→
   637→  // Relevance objection
   638→  objections.push({
   639→    objection: "This isn't relevant to me",
   640→    response: productDef.targetCustomer
   641→      ? `${productDef.name} is specifically designed for ${productDef.targetCustomer}. If that describes you, this product is likely relevant to your situation.`
   642→      : "Please review the product description to determine if it's relevant to your situation.",
   643→    evidence: productDef.targetCustomer ? "VERIFIED" : "INFERRED",
   644→  });
   645→
   646→  // Complexity objection
   647→  objections.push({
   648→    objection: "This looks too complicated",
   649→    response: `${productDef.name} is structured to be straightforward${productDef.type === "guide" || productDef.type === "ebook" ? " with clear sections and step-by-step guidance" : ""}. We recommend starting with the core material and expanding from there.`,
   650→    evidence: "INFERRED",
   651→  });
   652→
   653→  return objections;
   654→}

   675→// ─── Internal: Proof Requirements ────────────────────────────────────────────
   676→
   677→function generateProofRequirements(
   678→  productDef: ProductDefinition,
   679→  readiness: CommercialReadiness | undefined
   680→): ProofStatus[] {
   681→  const proofs: ProofStatus[] = [];
   682→
   683→  // Problem validation
   684→  const hasProblem = productDef.painPoints.length > 0;
   685→  proofs.push({
   686→    type: "problem_validation",
   687→    status: hasProblem ? "AVAILABLE" : "MISSING",
   688→    description: hasProblem
   689→      ? "Problem statement defined with specific pain points"
   690→      : "Problem validation missing — no pain points defined",
   691→  });
   692→
   693→  // Solution demonstration
   694→  const hasSolution = !!productDef.valueProposition;
   695→  proofs.push({
   696→    type: "solution_demonstration",
   697→    status: hasSolution ? "AVAILABLE" : "MISSING",
   698→    description: hasSolution
   699→      ? "Solution/value proposition is stated"
   700→      : "Solution demonstration missing — no value proposition defined",
   701→  });
   702→
   703→  // Market evidence
   704→  const hasMarket = readiness && readiness.score.commercialClarity.score > 0.5;
   705→  proofs.push({
   706→    type: "market_evidence",
   707→    status: hasMarket
   708→      ? "AVAILABLE"
   709→      : readiness
   710→        ? "PARTIAL"
   711→        : "MISSING",
   712→    description: hasMarket
   713→      ? "Market evidence available from commercial clarity"
   714→      : "Market evidence is insufficient or missing",
   715→  });
   716→
   717→  // Quality evidence
   718→  const hasQuality = readiness && readiness.score.quality.score > 0.5;
   719→  proofs.push({
   720→    type: "quality_evidence",
   721→    status: hasQuality
   722→      ? "AVAILABLE"
   723→      : readiness
   724→        ? "PARTIAL"
   725→        : "MISSING",
   726→    description: hasQuality
   727→      ? "Product has passed quality assurance"
   728→      : "Quality evidence is insufficient or missing",
   729→  });
   730→
   731→  // Differentiation evidence
   732→  const hasDiff = !!productDef.differentiation;
   733→  proofs.push({
   734→    type: "differentiation_evidence",
   735→    status: hasDiff ? "AVAILABLE" : "MISSING",
   736→    description: hasDiff
   737→      ? "Differentiation from alternatives is stated"
   738→      : "Differentiation evidence is missing",
   739→  });
   740→
   741→  // Testimonial evidence
   742→  proofs.push({
   743→    type: "testimonial_evidence",
   744→    status: "MISSING",
   745→    description: "Testimonial evidence must be provided externally — never fabricated",
   746→  });
   747→
   748→  return proofs;
   749→}

   751→// ─── Internal: Disclaimers ───────────────────────────────────────────────────
   752→
   753→function generateDisclaimers(
   754→  readiness: CommercialReadiness | undefined,
   755→  integrityCheck: IntegrityCheck,
   756→  proofRequirements: ProofStatus[]
   757→): string[] {
   758→  const disclaimers: string[] = [];
   759→
   760→  // Add disclaimer if readiness is low
   761→  if (!readiness || readiness.score.overall < 0.5) {
   762→    disclaimers.push(
   763→      "This product is still in development. Features and content may change before final release."
   764→    );
   765→  }
   766→
   767→  // Add disclaimer if integrity issues exist
   768→  if (!integrityCheck.passes) {
   769→    disclaimers.push(
   770→      "Some marketing claims in this product have been flagged for review. Please verify independently."
   771→    );
   772→  }
   773→
   774→  // Add disclaimer for missing proof
   775→  const missingProofs = proofRequirements.filter(
   776→    (p) => p.status === "MISSING"
   777→  );
   778→  if (missingProofs.length > 0) {
   779→    disclaimers.push(
   780→      `Evidence is not yet available for: ${missingProofs.map((p) => p.type.replace(/_/g, " ")).join(", ")}. Claims should be verified independently.`
   781→    );
   782→  }
   783→
   784→  // Add disclaimer for partial proof
   785→  const partialProofs = proofRequirements.filter(
   786→    (p) => p.status === "PARTIAL"
   787→  );
   788→  if (partialProofs.length > 0) {
   789→    disclaimers.push(
   790→      `Evidence is partial for: ${partialProofs.map((p) => p.type.replace(/_/g, " ")).join(", ")}. Additional verification is recommended.`
   791→    );
   792→  }
   793→
   794→  // Standard results disclaimer
   795→  disclaimers.push(
   796→    "Results may vary. This product provides tools and information; outcomes depend on individual effort and circumstances."
   797→  );
   798→
   799→  return disclaimers;
   800→}

   802→// ─── Internal: Positioning ───────────────────────────────────────────────────
   803→
   804→function buildPositioning(
   805→  productDef: ProductDefinition,
   806→  offerArch: OfferArchitecture | undefined
   807→): string {
   808→  const parts: string[] = [];
   809→
   810→  if (productDef.valueProposition) {
   811→    parts.push(productDef.valueProposition);
   812→  }
   813→
   814→  if (productDef.differentiation) {
   815→    parts.push(`Unlike alternatives, ${productDef.differentiation}`);
   816→  }
   817→
   818→  if (productDef.targetCustomer) {
   819→    parts.push(`Designed for ${productDef.targetCustomer}`);
   820→  }
   821→
   822→  if (offerArch?.mechanism) {
   823→    parts.push(`Through ${offerArch.mechanism}`);
   824→  }
   825→
   826→  return parts.length > 0 ? parts.join(". ") + "." : "Product positioning not fully defined.";
   827→}

     1→/**
     2→ * MCP Contract — ADN GENERAL DEL AGENTE V1.0
     3→ *
     4→ * Model Context Protocol (MCP) support for AGENTE-KREA.
     5→ * MCP allows agents to expose their capabilities as tools/resources
     6→ * and consume capabilities from other agents/servers.
     7→ *
     8→ * Key separation:
     9→ * - KREA OFFERS: What capabilities KREA exposes to other agents
    10→ * - KREA CONSUMES: What capabilities KREA uses from external sources
    11→ *
    12→ * No direct coupling to other agent implementations — all communication
    13→ * goes through the MCP protocol abstraction.
    14→ */

   251→/**
   252→ * MCPServer — Exposes KREA's capabilities to other agents via MCP.
   253→ * This is WHAT KREA OFFERS.
   254→ */
   255→export class MCPServer {
   256→  private config: MCPServerConfig;
   257→  private toolHandlers: Map<string, (params: Record<string, unknown>) => Promise<MCPToolCallResult>> = new Map();
   258→  private resourceHandlers: Map<string, () => Promise<unknown>> = new Map();
   259→  private initialized: boolean = false;
   260→
   261→  constructor(config: MCPServerConfig) {
   262→    this.config = config;
   263→  }
   264→
   265→  /** Register a handler for a tool */
   266→  registerToolHandler(toolName: string, handler: (params: Record<string, unknown>) => Promise<MCPToolCallResult>): void {
   267→    this.toolHandlers.set(toolName, handler);
   268→  }
   269→
   270→  /** Register a handler for a resource */
   271→  registerResourceHandler(uri: string, handler: () => Promise<unknown>): void {
   272→    this.resourceHandlers.set(uri, handler);
   273→  }
   274→
   275→  /** Handle an incoming MCP request */
   276→  async handleRequest(request: MCPRequest): Promise<MCPResponse> {
   277→    try {
   278→      switch (request.method) {
   279→        case "initialize":
   280→          return this.handleInitialize(request);
   281→        case "tools/list":
   282→          return this.handleToolsList(request);
   283→        case "tools/call":
   284→          return await this.handleToolsCall(request);
   285→        case "resources/list":
   286→          return this.handleResourcesList(request);
   287→        case "resources/read":
   288→          return await this.handleResourcesRead(request);
   289→        case "prompts/list":
   290→          return this.handlePromptsList(request);
   291→        default:
   292→          return {
   293→            jsonrpc: "2.0",
   294→            id: request.id,
   295→            error: { code: -32601, message: `Method not found: ${request.method}` },
   296→          };
   297→      }
   298→    } catch (error) {
   299→      return {
   300→        jsonrpc: "2.0",
   301→        id: request.id,
   302→        error: {
   303→          code: -32603,
   304→          message: error instanceof Error ? error.message : "Internal error",
   305→        },
   306→      };
   307→    }
   308→  }
   309→
   310→  /** Get the server config */
   311→  getConfig(): MCPServerConfig {
   312→    return this.config;
   313→  }
   314→
   315→  /** Get available tools */
   316→  getAvailableTools(): MCPToolOffering[] {
   317→    return this.config.tools.filter((t) => t.available);
   318→  }
   319→
   320→  /** Get available resources */
   321→  getAvailableResources(): MCPResourceOffering[] {
   322→    return this.config.resources;
   323→  }
   324→
   325→  // ── Private Handlers ──────────────────────────────────────────────────
   326→
   327→  private handleInitialize(request: MCPRequest): MCPResponse {
   328→    this.initialized = true;
   329→    return {
   330→      jsonrpc: "2.0",
   331→      id: request.id,
   332→      result: {
   333→        protocolVersion: this.config.protocolVersion,
   334→        capabilities: this.config.capabilities,
   335→        serverInfo: {
   336→          name: this.config.name,
   337→          version: this.config.version,
   338→        },
   339→      },
   340→    };
   341→  }
   342→
   343→  private handleToolsList(request: MCPRequest): MCPResponse {
   344→    return {
   345→      jsonrpc: "2.0",
   346→      id: request.id,
   347→      result: {
   348→        tools: this.config.tools.map((tool) => ({
   349→          name: tool.name,
   350→          description: tool.description,
   351→          inputSchema: tool.inputSchema,
   352→        })),
   353→      },
   354→    };
   355→  }
   356→
   357→  private async handleToolsCall(request: MCPRequest): Promise<MCPResponse> {
   358→    const params = request.params || {};
   359→    const toolName = params.name as string;
   360→    const toolArgs = (params.arguments as Record<string, unknown>) || {};
   361→
   362→    const tool = this.config.tools.find((t) => t.name === toolName);
   363→    if (!tool) {
   364→      return {
   365→        jsonrpc: "2.0",
   366→        id: request.id,
   367→        error: { code: -32602, message: `Unknown tool: ${toolName}` },
   368→      };
   369→    }
   370→
   371→    if (!tool.available) {
   372→      return {
   373→        jsonrpc: "2.0",
   374→        id: request.id,
   375→        error: { code: -32603, message: `Tool not available: ${toolName}` },
   376→      };
   377→    }
   378→
   379→    const handler = this.toolHandlers.get(toolName);
   380→    if (!handler) {
   381→      return {
   382→        jsonrpc: "2.0",
   383→        id: request.id,
   384→        error: { code: -32603, message: `No handler for tool: ${toolName}` },
   385→      };
   386→    }
   387→
   388→    const result = await handler(toolArgs);
   389→    return {
   390→      jsonrpc: "2.0",
   391→      id: request.id,
   392→      result: {
   393→        content: result.content,
   394→        isError: result.isError,
   395→      },
   396→    };
   397→  }
   398→
   399→  private handleResourcesList(request: MCPRequest): MCPResponse {
   400→    return {
   401→      jsonrpc: "2.0",
   402→      id: request.id,
   403→      result: {
   404→        resources: this.config.resources.map((res) => ({
   405→          uri: res.uri,
   406→          name: res.name,
   407→          description: res.description,
   408→          mimeType: res.mimeType,
   409→        })),
   410→      },
   411→    };
   412→  }
   413→
   414→  private async handleResourcesRead(request: MCPRequest): Promise<MCPResponse> {
   415→    const params = request.params || {};
   416→    const uri = params.uri as string;
   417→
   418→    const resource = this.config.resources.find((r) => r.uri === uri);
   419→    if (!resource) {
   420→      return {
   421→        jsonrpc: "2.0",
   422→        id: request.id,
   423→        error: { code: -32602, message: `Unknown resource: ${uri}` },
   424→      };
   425→    }
   426→
   427→    const handler = this.resourceHandlers.get(uri);
   428→    if (!handler) {
   429→      return {
   430→        jsonrpc: "2.0",
   431→        id: request.id,
   432→        error: { code: -32603, message: `No handler for resource: ${uri}` },
   433→      };
   434→    }
   435→
   436→    const contents = await handler();
   437→    return {
   438→      jsonrpc: "2.0",
   439→      id: request.id,
   440→      result: {
   441→        contents: [
   442→          {
   443→            uri,
   444→            mimeType: resource.mimeType,
   445→            text: typeof contents === "string" ? contents : JSON.stringify(contents),
   446→          },
   447→        ],
   448→      },
   449→    };
   450→  }
   451→
   452→  private handlePromptsList(request: MCPRequest): MCPResponse {
   453→    return {
   454→      jsonrpc: "2.0",
   455→      id: request.id,
   456→      result: {
   457→        prompts: this.config.prompts.map((prompt) => ({
   458→          name: prompt.name,
   459→          description: prompt.description,
   460→          arguments: prompt.arguments,
   461→        })),
   462→      },
   463→    };
   464→  }
   465→}

   469→/**
   470→ * MCPClient — Consumes capabilities from external MCP servers.
   471→ * This is WHAT KREA CONSUMES.
   472→ */
   473→export class MCPClient {
   474→  private remoteServers: Map<string, MCPRemoteServer> = new Map();
   475→  private callLog: Array<{
   476→    timestamp: string;
   477→    serverId: string;
   478→    toolName: string;
   479→    success: boolean;
   480→    durationMs: number;
   481→  }> = [];
   482→
   483→  /** Register a remote MCP server */
   484→  registerRemoteServer(server: MCPRemoteServer): void {
   485→    this.remoteServers.set(server.serverId, server);
   486→  }
   487→
   488→  /** Remove a remote MCP server */
   489→  removeRemoteServer(serverId: string): void {
   490→    this.remoteServers.delete(serverId);
   491→  }
   492→
   493→  /** Get all remote servers */
   494→  getRemoteServers(): MCPRemoteServer[] {
   495→    return Array.from(this.remoteServers.values());
   496→  }
   497→
   498→  /** Get a remote server by ID */
   499→  getRemoteServer(serverId: string): MCPRemoteServer | undefined {
   500→    return this.remoteServers.get(serverId);
   501→  }
   502→
   503→  /** Find a remote tool by name across all servers */
   504→  findRemoteTool(toolName: string): MCPRemoteTool | undefined {
   505→    for (const server of Array.from(this.remoteServers.values())) {
   506→      const tool = server.remoteTools.find((t) => t.name === toolName && t.available);
   507→      if (tool) return tool;
   508→    }
   509→    return undefined;
   510→  }
   511→
   512→  /** Find all tools matching a capability across all servers */
   513→  findToolsByCapability(capability: string): MCPRemoteTool[] {
   514→    const results: MCPRemoteTool[] = [];
   515→    for (const server of Array.from(this.remoteServers.values())) {
   516→      if (!server.connected) continue;
   517→      for (const tool of server.remoteTools) {
   518→        if (tool.available && tool.description.toLowerCase().includes(capability.toLowerCase())) {
   519→          results.push(tool);
   520→        }
   521→      }
   522→    }
   523→    return results;
   524→  }
   525→
   526→  /**
   527→   * Call a tool on a remote MCP server.
   528→   * This is a stub — the actual transport (HTTP, SSE, etc.) must be provided.
   529→   */
   530→  async callTool(
   531→    serverId: string,
   532→    toolName: string,
   533→    args: Record<string, unknown>,
   534→    transport?: (request: MCPRequest) => Promise<MCPResponse>,
   535→  ): Promise<MCPToolCallResult> {
   536→    const server = this.remoteServers.get(serverId);
   537→    const startTime = Date.now();
   538→
   539→    if (!server) {
   540→      return {
   541→        success: false,
   542→        toolName,
   543→        content: [{ type: "text", text: `Unknown server: ${serverId}` }],
   544→        isError: true,
   545→      };
   546→    }
   547→
   548→    if (!server.connected) {
   549→      return {
   550→        success: false,
   551→        toolName,
   552→        content: [{ type: "text", text: `Server ${serverId} is not connected` }],
   553→        isError: true,
   554→      };
   555→    }
   556→
   557→    if (!transport) {
   558→      return {
   559→        success: false,
   560→        toolName,
   561→        content: [{ type: "text", text: `No transport configured for server ${serverId}` }],
   562→        isError: true,
   563→      };
   564→    }
   565→
   566→    try {
   567→      const request: MCPRequest = {
   568→        jsonrpc: "2.0",
   569→        id: randomUUID(),
   570→        method: "tools/call",
   571→        params: { name: toolName, arguments: args },
   572→      };
   573→
   574→      const response = await transport(request);
   575→      const durationMs = Date.now() - startTime;
   576→
   577→      if (response.error) {
   578→        this.callLog.push({
   579→          timestamp: new Date().toISOString(),
   580→          serverId,
   581→          toolName,
   582→          success: false,
   583→          durationMs,
   584→        });
   585→
   586→        return {
   587→          success: false,
   588→          toolName,
   589→          content: [{ type: "text", text: response.error.message }],
   590→          isError: true,
   591→          meta: { durationMs, creditsConsumed: 0, evidenceIds: [] },
   592→        };
   593→      }
   594→
   595→      const result = response.result as { content?: MCPToolCallResult["content"]; isError?: boolean };
   596→
   597→      this.callLog.push({
   598→        timestamp: new Date().toISOString(),
   599→        serverId,
   600→        toolName,
   601→        success: true,
   602→        durationMs,
   603→      });
   604→
   605→      return {
   606→        success: true,
   607→        toolName,
   608→        content: result.content || [{ type: "text", text: "No content returned" }],
   609→        isError: result.isError || false,
   610→        meta: { durationMs, creditsConsumed: 0, evidenceIds: [] },
   611→      };
   612→    } catch (error) {
   613→      const durationMs = Date.now() - startTime;
   614→      this.callLog.push({
   615→        timestamp: new Date().toISOString(),
   616→        serverId,
   617→        toolName,
   618→        success: false,
   619→        durationMs,
   620→      });
   621→
   622→      return {
   623→        success: false,
   624→        toolName,
   625→        content: [{ type: "text", text: error instanceof Error ? error.message : String(error) }],
   626→        isError: true,
   627→        meta: { durationMs, creditsConsumed: 0, evidenceIds: [] },
   628→      };
   629→    }
   630→  }
   631→
   632→  /** Get call log */
   633→  getCallLog(limit?: number): Array<{ timestamp: string; serverId: string; toolName: string; success: boolean; durationMs: number }> {
   634→    const log = [...this.callLog];
   635→    return limit ? log.slice(-limit) : log;
   636→  }
   637→
   638→  /** Get client stats */
   639→  getStats(): {
   640→    totalCalls: number;
   641→    successful: number;
   642→    failed: number;
   643→    connectedServers: number;
   644→    totalRemoteTools: number;
   645→  } {
   646→    const totalCalls = this.callLog.length;
   647→    const successful = this.callLog.filter((c) => c.success).length;
   648→    const servers = Array.from(this.remoteServers.values());
   649→    const connectedServers = servers.filter((s) => s.connected).length;
   650→    const totalRemoteTools = servers.reduce((sum, s) => sum + s.remoteTools.length, 0);
   651→
   652→    return {
   653→      totalCalls,
   654→      successful,
   655→      failed: totalCalls - successful,
   656→      connectedServers,
   657→      totalRemoteTools,
   658→    };
   659→  }
   660→}

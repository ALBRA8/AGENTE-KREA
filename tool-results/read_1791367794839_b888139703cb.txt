   194→  // These are periodically flushed to the database via audit events
   195→  private providerCalls: ProviderCallRecord[] = [];

   197→  private skillCalls: SkillCallRecord[] = [];
   198→  private memoryOps: MemoryOperationRecord[] = [];

   203→  // ─── Tracking Methods ───────────────────────────────────────────────
   204→
   205→  /**
   206→   * Track an execution completion.
   207→   * Records key metrics to the audit log.
   208→   */
   209→  async trackExecution(execution: {
   210→    id: string;
   211→    task: string;
   212→    status: string;
   213→    cost: number;
   214→    startedAt: Date | null;
   215→    completedAt: Date | null;
   216→    agentId?: string;
   217→    userId?: string;
   218→  }): Promise<void> {
   219→    const latencyMs =
   220→      execution.startedAt && execution.completedAt
   221→        ? execution.completedAt.getTime() - execution.startedAt.getTime()
   222→        : null;
   223→
   224→    await db.auditEvent.create({
   225→      data: {
   226→        agentId: execution.agentId || this.defaultAgentId,
   227→        userId: execution.userId || null,
   228→        action: "execution_completed",
   229→        resource: execution.id,
   230→        details: JSON.stringify({
   231→          task: execution.task,
   232→          status: execution.status,
   233→          cost: execution.cost,
   234→          latencyMs,
   235→        }),
   236→        result: execution.status === "SUCCEEDED" ? "SUCCESS" : execution.status === "FAILED" ? "FAILURE" : "PARTIAL",
   237→      },
   238→    });
   239→  }
   240→
   241→  /**
   242→   * Track a provider (e.g., image generation API) call.
   243→   */
   244→  trackProviderUsage(providerId: string, type: string, latencyMs: number, success: boolean): void {
   245→    this.providerCalls.push({
   246→      providerId,
   247→      type,
   248→      latencyMs,
   249→      success,
   250→      timestamp: new Date(),
   251→    });
   252→  }
   253→
   254→  /**
   255→   * Track a tool usage during an execution.
   256→   */
   257→  trackToolUsage(toolId: string, executionId: string): void {
   258→    this.toolCalls.push({ toolId, executionId, timestamp: new Date() });
   259→  }

   261→  /**
   262→   * Track a skill usage during an execution.
   263→   */
   264→  trackSkillUsage(skillId: string, executionId: string): void {

   266→  }

   296→  // ─── Flush Buffers to Database ──────────────────────────────────────
   297→
   298→  /**
   299→   * Flush in-memory metric buffers to the database as audit events.
   300→   * Should be called periodically (e.g., every 30 seconds).
   301→   */
   302→  async flushBuffers(): Promise<{
   303→    flushed: {
   304→      providerCalls: number;
   305→      toolCalls: number;
   306→      skillCalls: number;
   307→      memoryOps: number;
   308→      mcpActivity: number;
   309→      errors: number;
   310→      learningEvents: number;
   311→    };
   312→  }> {
   313→    const counts = {
   314→      providerCalls: 0,
   315→      toolCalls: 0,
   316→      skillCalls: 0,
   317→      memoryOps: 0,
   318→      mcpActivity: 0,
   319→      errors: 0,
   320→      learningEvents: 0,
   321→    };
   322→
   323→    // Flush provider calls
   324→    if (this.providerCalls.length > 0) {
   325→      const calls = this.providerCalls.splice(0);
   326→      counts.providerCalls = calls.length;
   327→
   328→      // Aggregate by provider+type
   329→      const aggregated = new Map<string, { calls: ProviderCallRecord[]; successes: number; failures: number; totalLatency: number }>();
   330→      for (const call of calls) {
   331→        const key = `${call.providerId}|${call.type}`;
   332→        const entry = aggregated.get(key) || { calls: [], successes: 0, failures: 0, totalLatency: 0 };
   333→        entry.calls.push(call);
   334→        if (call.success) entry.successes++;
   335→        else entry.failures++;
   336→        entry.totalLatency += call.latencyMs;
   337→        aggregated.set(key, entry);
   338→      }
   339→
   340→      for (const [key, entry] of aggregated) {
   341→        const [providerId, type] = key.split("|");
   342→        await db.auditEvent.create({
   343→          data: {
   344→            agentId: this.defaultAgentId,
   345→            action: "provider_usage_batch",
   346→            details: JSON.stringify({
   347→              providerId,
   348→              type,
   349→              calls: entry.calls.length,
   350→              successes: entry.successes,
   351→              failures: entry.failures,
   352→              avgLatencyMs: Math.round(entry.totalLatency / entry.calls.length),
   353→            }),
   354→            result: entry.failures === 0 ? "SUCCESS" : entry.successes === 0 ? "FAILURE" : "PARTIAL",
   355→          },
   356→        });
   357→      }
   358→    }
   359→
   360→    // Flush tool calls
   361→    if (this.toolCalls.length > 0) {
   362→      const calls = this.toolCalls.splice(0);
   363→      counts.toolCalls = calls.length;
   364→
   365→      const aggregated = new Map<string, number>();
   366→      for (const call of calls) {
   367→        aggregated.set(call.toolId, (aggregated.get(call.toolId) || 0) + 1);
   368→      }
   369→
   370→      await db.auditEvent.create({
   371→        data: {
   372→          agentId: this.defaultAgentId,
   373→          action: "tool_usage_batch",
   374→          details: JSON.stringify(Object.fromEntries(aggregated)),
   375→          result: "SUCCESS",
   376→        },
   377→      });
   378→    }
   379→
   380→    // Flush skill calls
   381→    if (this.skillCalls.length > 0) {
   382→      const calls = this.skillCalls.splice(0);
   383→      counts.skillCalls = calls.length;
   384→
   385→      const aggregated = new Map<string, number>();
   386→      for (const call of calls) {
   387→        aggregated.set(call.skillId, (aggregated.get(call.skillId) || 0) + 1);
   388→      }
   389→
   390→      await db.auditEvent.create({
   391→        data: {
   392→          agentId: this.defaultAgentId,
   393→          action: "skill_usage_batch",
   394→          details: JSON.stringify(Object.fromEntries(aggregated)),
   395→          result: "SUCCESS",
   396→        },
   397→      });
   398→    }
   399→
   400→    // Flush memory operations
   401→    if (this.memoryOps.length > 0) {
   402→      const ops = this.memoryOps.splice(0);
   403→      counts.memoryOps = ops.length;
   404→
   405→      const aggregated = new Map<string, number>();
   406→      for (const op of ops) {
   407→        const key = `${op.operation}|${op.domain}`;
   408→        aggregated.set(key, (aggregated.get(key) || 0) + 1);
   409→      }
   410→
   411→      await db.auditEvent.create({
   412→        data: {
   413→          agentId: this.defaultAgentId,
   414→          action: "memory_activity_batch",
   415→          details: JSON.stringify(Object.fromEntries(aggregated)),
   416→          result: "SUCCESS",
   417→        },
   418→      });
   419→    }
   420→
   421→    // Flush MCP activity
   422→    if (this.mcpActivity.length > 0) {
   423→      const activity = this.mcpActivity.splice(0);
   424→      counts.mcpActivity = activity.length;
   425→
   426→      const aggregated = new Map<string, { calls: number; successes: number }>();
   427→      for (const a of activity) {
   428→        const key = `${a.direction}|${a.tool}`;
   429→        const entry = aggregated.get(key) || { calls: 0, successes: 0 };
   430→        entry.calls++;
   431→        if (a.result) entry.successes++;
   432→        aggregated.set(key, entry);
   433→      }
   434→
   435→      await db.auditEvent.create({
   436→        data: {
   437→          agentId: this.defaultAgentId,
   438→          action: "mcp_activity_batch",
   439→          details: JSON.stringify(Object.fromEntries(aggregated)),
   440→          result: "SUCCESS",
   441→        },
   442→      });
   443→    }
   444→
   445→    // Flush errors
   446→    if (this.errors.length > 0) {
   447→      const errs = this.errors.splice(0);
   448→      counts.errors = errs.length;
   449→
   450→      const aggregated = new Map<string, { count: number; lastError: string }>();
   451→      for (const e of errs) {
   452→        const entry = aggregated.get(e.source) || { count: 0, lastError: "" };
   453→        entry.count++;
   454→        entry.lastError = e.error;
   455→        aggregated.set(e.source, entry);
   456→      }
   457→
   458→      for (const [source, entry] of aggregated) {
   459→        await db.auditEvent.create({
   460→          data: {
   461→            agentId: this.defaultAgentId,
   462→            action: "error_batch",
   463→            resource: source,
   464→            details: JSON.stringify({ count: entry.count, lastError: entry.lastError }),
   465→            result: "FAILURE",
   466→          },
   467→        });
   468→      }
   469→    }
   470→
   471→    // Flush learning events
   472→    if (this.learningEvents.length > 0) {
   473→      const events = this.learningEvents.splice(0);
   474→      counts.learningEvents = events.length;
   475→
   476→      await db.auditEvent.create({
   477→        data: {
   478→          agentId: this.defaultAgentId,
   479→          action: "learning_events_batch",
   480→          details: JSON.stringify({ count: events.length, ids: events.map((e) => e.learningId) }),
   481→          result: "SUCCESS",
   482→        },
   483→      });
   484→    }
   485→
   486→    return { flushed: counts };
   487→  }

   489→  // ─── Metrics Retrieval ──────────────────────────────────────────────
   490→
   491→  /**
   492→   * Get aggregated metrics for a time range.
   493→   * All metrics are derived from real database data — never fabricated.
   494→   */
   495→  async getMetrics(timeRange: TimeRange, agentId?: string): Promise<AggregatedMetrics> {
   496→    const effectiveAgentId = agentId || this.defaultAgentId;
   497→
   498→    // ── Execution Metrics ──
   499→    const executions = await db.execution.findMany({
   500→      where: {
   501→        agentId: effectiveAgentId,
   502→        createdAt: { gte: timeRange.start, lte: timeRange.end },
   503→      },
   504→    });
   505→
   506→    const succeeded = executions.filter((e) => e.status === "SUCCEEDED").length;
   507→    const failed = executions.filter((e) => e.status === "FAILED").length;
   508→    const cancelled = executions.filter((e) => e.status === "CANCELLED").length;
   509→    const pending = executions.filter((e) => e.status === "PENDING").length;
   510→    const running = executions.filter((e) => e.status === "RUNNING").length;
   511→
   512→    const completedExecutions = executions.filter(
   513→      (e) => e.startedAt && e.completedAt
   514→    );
   515→    const totalLatency = completedExecutions.reduce(
   516→      (sum, e) => sum + (e.completedAt!.getTime() - e.startedAt!.getTime()),
   517→      0
   518→    );
   519→    const totalCost = executions.reduce((sum, e) => sum + e.cost, 0);
   520→
   521→    const executionMetrics: ExecutionMetric = {
   522→      total: executions.length,
   523→      succeeded,
   524→      failed,
   525→      cancelled,
   526→      pending,
   527→      running,
   528→      successRate: executions.length > 0 ? succeeded / executions.length : 0,
   529→      avgLatencyMs: completedExecutions.length > 0 ? totalLatency / completedExecutions.length : 0,
   530→      totalCost,
   531→      isEstimated: false, // Real data from DB
   532→    };
   533→
   534→    // ── Provider Usage Metrics ──
   535→    const providerAuditEvents = await db.auditEvent.findMany({
   536→      where: {
   537→        agentId: effectiveAgentId,
   538→        action: "provider_usage_batch",
   539→        createdAt: { gte: timeRange.start, lte: timeRange.end },
   540→      },
   541→    });
   542→
   543→    const providerMap = new Map<string, { calls: number; successes: number; failures: number; totalLatency: number }>();
   544→    for (const evt of providerAuditEvents) {
   545→      if (!evt.details) continue;
   546→      try {
   547→        const d = JSON.parse(evt.details);
   548→        const key = `${d.providerId}|${d.type}`;
   549→        const existing = providerMap.get(key) || { calls: 0, successes: 0, failures: 0, totalLatency: 0 };
   550→        existing.calls += d.calls || 0;
   551→        existing.successes += d.successes || 0;
   552→        existing.failures += d.failures || 0;
   553→        existing.totalLatency += (d.avgLatencyMs || 0) * (d.calls || 1);
   554→        providerMap.set(key, existing);
   555→      } catch { /* ignore */ }
   556→    }
   557→
   558→    const providerMetrics: ProviderUsageMetric[] = Array.from(providerMap.entries()).map(([key, v]) => {
   559→      const [providerId, type] = key.split("|");
   560→      return {
   561→        providerId,
   562→        type,
   563→        calls: v.calls,
   564→        successes: v.successes,
   565→        failures: v.failures,
   566→        avgLatencyMs: v.calls > 0 ? v.totalLatency / v.calls : 0,
   567→        successRate: v.calls > 0 ? v.successes / v.calls : 0,
   568→        isEstimated: false,
   569→      };
   570→    });
   571→
   572→    // ── Tool Usage Metrics ──
   573→    const toolUsage: Map<string, number> = new Map();
   574→    for (const e of executions) {
   575→      if (e.toolsUsed) {
   576→        try {
   577→          const tools: string[] = JSON.parse(e.toolsUsed);
   578→          for (const t of tools) {
   579→            toolUsage.set(t, (toolUsage.get(t) || 0) + 1);
   580→          }
   581→        } catch { /* ignore */ }
   582→      }
   583→    }
   584→
   585→    const toolMetrics: ToolUsageMetric[] = Array.from(toolUsage.entries()).map(([toolId, count]) => ({
   586→      toolId,
   587→      executions: count,
   588→      successRate: 0, // Would need per-tool success tracking — mark as estimated
   589→      isEstimated: true,
   590→    }));
   591→
   592→    // ── Skill Usage Metrics ──
   593→    const skills = await db.skill.findMany({
   594→      where: { agentId: effectiveAgentId, status: "ACTIVE" },
   595→    });
   596→
   597→    const skillUsageMap: Map<string, number> = new Map();
   598→    for (const e of executions) {
   599→      if (e.skillsUsed) {
   600→        try {
   601→          const usedSkills: string[] = JSON.parse(e.skillsUsed);
   602→          for (const s of usedSkills) {
   603→            skillUsageMap.set(s, (skillUsageMap.get(s) || 0) + 1);
   604→          }
   605→        } catch { /* ignore */ }
   606→      }
   607→    }
   608→
   609→    const skillMetrics: SkillUsageMetric[] = skills.map((s) => ({
   610→      skillId: s.id,
   611→      skillName: s.name,
   612→      executions: skillUsageMap.get(s.name) || 0,
   613→      successRate: s.successRate,
   614→      confidence: s.confidence,
   615→      isEstimated: s.successRate === 0 && skillUsageMap.get(s.name) === 0, // Estimated if no data
   616→    }));
   617→
   618→    // ── Memory Activity Metrics ──
   619→    const memoryAuditEvents = await db.auditEvent.findMany({
   620→      where: {
   621→        agentId: effectiveAgentId,
   622→        action: "memory_activity_batch",
   623→        createdAt: { gte: timeRange.start, lte: timeRange.end },
   624→      },
   625→    });
   626→
   627→    const memoryMap = new Map<string, number>();
   628→    for (const evt of memoryAuditEvents) {
   629→      if (!evt.details) continue;
   630→      try {
   631→        const d = JSON.parse(evt.details) as Record<string, number>;
   632→        for (const [key, count] of Object.entries(d)) {
   633→          memoryMap.set(key, (memoryMap.get(key) || 0) + count);
   634→        }
   635→      } catch { /* ignore */ }
   636→    }
   637→
   638→    const memoryMetrics: MemoryActivityMetric[] = Array.from(memoryMap.entries()).map(([key, count]) => {
   639→      const [operation, domain] = key.split("|");
   640→      return { operation, domain, count, isEstimated: false };
   641→    });
   642→
   643→    // ── MCP Activity Metrics ──
   644→    const mcpAuditEvents = await db.auditEvent.findMany({
   645→      where: {
   646→        agentId: effectiveAgentId,
   647→        action: "mcp_activity_batch",
   648→        createdAt: { gte: timeRange.start, lte: timeRange.end },
   649→      },
   650→    });
   651→
   652→    const mcpMap = new Map<string, { calls: number; successes: number }>();
   653→    for (const evt of mcpAuditEvents) {
   654→      if (!evt.details) continue;
   655→      try {
   656→        const d = JSON.parse(evt.details) as Record<string, { calls: number; successes: number }>;
   657→        for (const [key, val] of Object.entries(d)) {
   658→          const existing = mcpMap.get(key) || { calls: 0, successes: 0 };
   659→          existing.calls += val.calls || 0;
   660→          existing.successes += val.successes || 0;
   661→          mcpMap.set(key, existing);
   662→        }
   663→      } catch { /* ignore */ }
   664→    }
   665→
   666→    const mcpMetrics: MCPActivityMetric[] = Array.from(mcpMap.entries()).map(([key, v]) => {
   667→      const [direction, tool] = key.split("|");
   668→      return {
   669→        direction: direction as "inbound" | "outbound",
   670→        tool,
   671→        calls: v.calls,
   672→        successRate: v.calls > 0 ? v.successes / v.calls : 0,
   673→        isEstimated: false,
   674→      };
   675→    });
   676→
   677→    // ── Error Metrics ──
   678→    const errorAuditEvents = await db.auditEvent.findMany({
   679→      where: {
   680→        agentId: effectiveAgentId,
   681→        action: "error_batch",
   682→        createdAt: { gte: timeRange.start, lte: timeRange.end },
   683→      },
   684→    });
   685→
   686→    const errorMap = new Map<string, { count: number; lastOccurrence: Date }>();
   687→    for (const evt of errorAuditEvents) {
   688→      const source = evt.resource || "unknown";
   689→      const existing = errorMap.get(source) || { count: 0, lastOccurrence: evt.createdAt };
   690→      try {
   691→        const d = JSON.parse(evt.details || "{}");
   692→        existing.count += d.count || 1;
   693→      } catch {
   694→        existing.count += 1;
   695→      }
   696→      existing.lastOccurrence = evt.createdAt;
   697→      errorMap.set(source, existing);
   698→    }
   699→
   700→    const errorMetrics: ErrorMetric[] = Array.from(errorMap.entries()).map(([source, v]) => ({
   701→      source,
   702→      count: v.count,
   703→      lastOccurrence: v.lastOccurrence,
   704→      isEstimated: false,
   705→    }));
   706→
   707→    // ── Learning Metrics ──
   708→    const feedbacks = await db.feedback.findMany({
   709→      where: {
   710→        agentId: effectiveAgentId,
   711→        createdAt: { gte: timeRange.start, lte: timeRange.end },
   712→      },
   713→    });
   714→
   715→    const learningMetrics: LearningMetric = {
   716→      totalFeedback: feedbacks.length,
   717→      pendingValidation: feedbacks.filter((f) => f.validationStatus === "PENDING").length,
   718→      validated: feedbacks.filter((f) => f.validationStatus === "VALIDATED").length,
   719→      invalidated: feedbacks.filter((f) => f.validationStatus === "INVALIDATED").length,
   720→      learningApplied: feedbacks.filter((f) => f.validationStatus === "VALIDATED" && f.learning).length,
   721→      isEstimated: false,
   722→    };
   723→
   724→    return {
   725→      timeRange,
   726→      executions: executionMetrics,
   727→      providerUsage: providerMetrics,
   728→      toolUsage: toolMetrics,
   729→      skillUsage: skillMetrics,
   730→      memoryActivity: memoryMetrics,
   731→      mcpActivity: mcpMetrics,
   732→      errors: errorMetrics,
   733→      learning: learningMetrics,
   734→      generatedAt: new Date(),
   735→    };
   736→  }

   738→  /**
   739→   * Get dashboard-ready data with the most relevant metrics.
   740→   */
   741→  async getDashboard(agentId?: string): Promise<DashboardData> {
   742→    const effectiveAgentId = agentId || this.defaultAgentId;
   743→
   744→    // Overview stats
   745→    const executionStats = await db.execution.aggregate({
   746→      where: { agentId: effectiveAgentId },
   747→      _count: true,
   748→      _avg: { cost: true },
   749→    });
   750→
   751→    const succeededCount = await db.execution.count({
   752→      where: { agentId: effectiveAgentId, status: "SUCCEEDED" },
   753→    });
   754→
   755→    const totalExecutions = executionStats._count;
   756→    const successRate = totalExecutions > 0 ? succeededCount / totalExecutions : 0;
   757→
   758→    // Calculate average latency
   759→    const completedExecutions = await db.execution.findMany({
   760→      where: {
   761→        agentId: effectiveAgentId,
   762→        startedAt: { not: null },
   763→        completedAt: { not: null },
   764→      },
   765→      select: { startedAt: true, completedAt: true },
   766→      take: 100,
   767→    });
   768→
   769→    const avgLatencyMs =
   770→      completedExecutions.length > 0
   771→        ? completedExecutions.reduce((sum, e) => {
   772→            return sum + (e.completedAt!.getTime() - e.startedAt!.getTime());
   773→          }, 0) / completedExecutions.length
   774→        : 0;
   775→
   776→    // Active skills count
   777→    const activeSkillsCount = await db.skill.count({
   778→      where: { agentId: effectiveAgentId, status: "ACTIVE" },
   779→    });
   780→
   781→    // Total memories
   782→    const totalMemories = await db.memory.count({
   783→      where: { agentId: effectiveAgentId },
   784→    });
   785→
   786→    // Pending learning
   787→    const pendingLearning = await db.feedback.count({
   788→      where: { agentId: effectiveAgentId, validationStatus: "PENDING" },
   789→    });
   790→
   791→    // Recent executions
   792→    const recentExecutions = await db.execution.findMany({
   793→      where: { agentId: effectiveAgentId },
   794→      orderBy: { createdAt: "desc" },
   795→      take: 10,
   796→      select: {
   797→        id: true,
   798→        task: true,
   799→        status: true,
   800→        createdAt: true,
   801→        completedAt: true,
   802→        cost: true,
   803→      },
   804→    });
   805→
   806→    // Top tools (from execution data)
   807→    const executionsWithTools = await db.execution.findMany({
   808→      where: { agentId: effectiveAgentId, toolsUsed: { not: null } },
   809→      select: { toolsUsed: true, status: true },
   810→    });
   811→
   812→    const toolCountMap = new Map<string, { executions: number; successes: number }>();
   813→    for (const e of executionsWithTools) {
   814→      if (!e.toolsUsed) continue;
   815→      try {
   816→        const tools: string[] = JSON.parse(e.toolsUsed);
   817→        for (const t of tools) {
   818→          const entry = toolCountMap.get(t) || { executions: 0, successes: 0 };
   819→          entry.executions++;
   820→          if (e.status === "SUCCEEDED") entry.successes++;
   821→          toolCountMap.set(t, entry);
   822→        }
   823→      } catch { /* ignore */ }
   824→    }
   825→
   826→    const topTools: ToolUsageMetric[] = Array.from(toolCountMap.entries())
   827→      .map(([toolId, v]) => ({
   828→        toolId,
   829→        executions: v.executions,
   830→        successRate: v.executions > 0 ? v.successes / v.executions : 0,
   831→        isEstimated: false,
   832→      }))
   833→      .sort((a, b) => b.executions - a.executions)
   834→      .slice(0, 10);
   835→
   836→    // Top skills
   837→    const activeSkills = await db.skill.findMany({
   838→      where: { agentId: effectiveAgentId, status: "ACTIVE" },
   839→      orderBy: { successRate: "desc" },
   840→      take: 10,
   841→    });
   842→
   843→    const topSkills: SkillUsageMetric[] = activeSkills.map((s) => ({
   844→      skillId: s.id,
   845→      skillName: s.name,
   846→      executions: 0, // Would need to count from executions
   847→      successRate: s.successRate,
   848→      confidence: s.confidence,
   849→      isEstimated: s.successRate === 0,
   850→    }));
   851→
   852→    // Recent errors
   853→    const recentErrors = await db.auditEvent.findMany({
   854→      where: {
   855→        agentId: effectiveAgentId,
   856→        result: "FAILURE",
   857→      },
   858→      orderBy: { createdAt: "desc" },
   859→      take: 10,
   860→      select: {
   861→        id: true,
   862→        action: true,
   863→        details: true,
   864→        createdAt: true,
   865→      },
   866→    });
   867→
   868→    // Feedback summary
   869→    const feedbacks = await db.feedback.findMany({
   870→      where: { agentId: effectiveAgentId },
   871→    });
   872→
   873→    const feedbackByType: Record<string, number> = {};
   874→    for (const f of feedbacks) {
   875→      feedbackByType[f.feedbackType] = (feedbackByType[f.feedbackType] || 0) + 1;
   876→    }
   877→
   878→    // Memory summary
   879→    const memories = await db.memory.findMany({
   880→      where: { agentId: effectiveAgentId },
   881→    });
   882→
   883→    const memoryByType: Record<string, number> = {};
   884→    let totalConfidence = 0;
   885→    for (const m of memories) {
   886→      memoryByType[m.type] = (memoryByType[m.type] || 0) + 1;
   887→      totalConfidence += m.confidence;
   888→    }
   889→
   890→    return {
   891→      overview: {
   892→        totalExecutions,
   893→        successRate,
   894→        avgLatencyMs,
   895→        totalCost: (executionStats._avg.cost || 0) * totalExecutions,
   896→        activeSkills: activeSkillsCount,
   897→        totalMemories,
   898→        pendingLearning,
   899→      },
   900→      recentExecutions,
   901→      topTools,
   902→      topSkills,
   903→      recentErrors,
   904→      feedbackSummary: {
   905→        total: feedbacks.length,
   906→        byType: feedbackByType,
   907→        pendingValidation: feedbacks.filter((f) => f.validationStatus === "PENDING").length,
   908→      },
   909→      memorySummary: {
   910→        total: memories.length,
   911→        byType: memoryByType,
   912→        avgConfidence: memories.length > 0 ? totalConfidence / memories.length : 0,
   913→      },
   914→    };
   915→  }

   944→  /**
   945→   * Get audit log with filters.
   946→   */
   947→  async getAuditLog(filters: {
   948→    agentId?: string;
   949→    userId?: string;
   950→    action?: string;
   951→    startDate?: Date;
   952→    endDate?: Date;
   953→    limit?: number;
   954→  } = {}): Promise<unknown[]> {
   955→    const where: Record<string, unknown> = {};
   956→    if (filters.agentId) where.agentId = filters.agentId;
   957→    if (filters.userId) where.userId = filters.userId;
   958→    if (filters.action) where.action = { contains: filters.action };
   959→
   960→    if (filters.startDate || filters.endDate) {
   961→      const createdAt: Record<string, Date> = {};
   962→      if (filters.startDate) createdAt.gte = filters.startDate;
   963→      if (filters.endDate) createdAt.lte = filters.endDate;
   964→      where.createdAt = createdAt;
   965→    }
   966→
   967→    const events = await db.auditEvent.findMany({
   968→      where,
   969→      orderBy: { createdAt: "desc" },
   970→      take: filters.limit ?? 100,
   971→    });
   972→
   973→    return events.map((e) => ({
   974→      ...e,
   975→      details: e.details ? JSON.parse(e.details) : null,
   976→      evidence: e.evidence ? JSON.parse(e.evidence) : null,
   977→    }));
   978→  }

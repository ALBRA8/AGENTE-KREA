   195→/**
   196→ * Create a Commercial Product from a Product Dossier, Architecture, and Economics.
   197→ *
   198→ * Uses ZAI to help define the commercial product (customer profile, value prop,
   199→ * positioning), but ALL market data is tagged INFERRED or ESTIMATED.
   200→ *
   201→ * Price ranges come from domain heuristics — never invented benchmarks.
   202→ * Objections are generated from target customer analysis.
   203→ * Proof requirements start as MISSING (honest default).
   204→ */
   205→export async function createCommercialProduct(
   206→  dossier: ProductDossierData,
   207→  architecture: ProductArchitecture,
   208→  economics: ProductEconomics
   209→): Promise<CommercialProduct> {
   210→  const productId = generateId("cprod");
   211→  const productVersion = "1.0.0";
   212→  const productType = mapProductType(architecture.productType);
   213→  const productName = dossier.title;
   214→
   215→  // Get price heuristics for this product type
   216→  const priceHeuristic = PRICE_HEURISTICS[productType] || PRICE_HEURISTICS.ebook;
   217→
   218→  // Build the price range from heuristics (ALWAYS ESTIMATED for new products)
   219→  const priceRange: PriceRange = {
   220→    min: priceHeuristic.min,
   221→    max: priceHeuristic.max,
   222→    currency: economics.estimatedCost.currency,
   223→    recommended: Math.round((priceHeuristic.min + priceHeuristic.max) / 2),
   224→    evidence: "ESTIMATED", // NEVER VERIFIED for new product prices
   225→  };
   226→
   227→  // Build the price strategy
   228→  const priceStrategy: PriceStrategy = {
   229→    model: priceHeuristic.model,
   230→    reasoning: `Based on ${productType} product type heuristics. Actual pricing should be validated with market research.`,
   231→    evidence: "ESTIMATED",
   232→  };
   233→
   234→  // Generate proof requirements (all start as MISSING — honest default)
   235→  const proofRequirements: ProofRequirement[] = generateProofRequirements(productType);
   236→
   237→  // Try LLM-assisted commercial definition
   238→  try {
   239→    const zai = await getZAI();
   240→    const prompt = buildCommercialPrompt(dossier, architecture, economics, productType, priceRange);
   241→
   242→    const response = await zai.chat.completions.create({
   243→      messages: [
   244→        { role: "system", content: COMMERCIAL_SYSTEM_PROMPT },
   245→        { role: "user", content: prompt },

   247→    });
   248→
   249→    const content = response.choices[0]?.message?.content || "";
   250→    const parseResult = parseStructuredResponse(content, commercialResponseSchema);
   251→
   252→    if (parseResult.success) {
   253→      const data = parseResult.data;
   254→
   255→      const commercialProduct: CommercialProduct = {

   257→        productVersion,

   259→        productType,
   260→        targetCustomer: data.targetCustomer,
   261→        problem: dossier.opportunity?.problemStatement || dossier.description,
   262→        jobToBeDone: data.jobToBeDone,
   263→        desiredOutcome: data.desiredOutcome,

   267→        includedAssets: inferIncludedAssets(productType, architecture),
   268→        excludedScope: inferExcludedScope(productType, architecture),
   269→        priceStrategy,
   270→        priceRange,
   271→        monetizationModel: priceStrategy.model,
   272→        distributionStrategy: DISTRIBUTION_DEFAULTS[productType] || "Direct digital delivery",
   273→        positioning: data.positioning,
   274→        keyBenefits: data.keyBenefits,
   275→        objections: data.objections,
   276→        proofRequirements,
   277→        commercialReadiness: evaluateCommercialReadiness({
   278→          productId,
   279→          productVersion,
   280→          productName,
   281→          productType,
   282→          targetCustomer: data.targetCustomer,
   283→          problem: dossier.opportunity?.problemStatement || dossier.description,
   284→          jobToBeDone: data.jobToBeDone,
   285→          desiredOutcome: data.desiredOutcome,
   286→          valueProposition: data.valueProposition,
   287→          differentiation: data.differentiation,
   288→          format: inferFormat(productType, architecture),
   289→          includedAssets: inferIncludedAssets(productType, architecture),
   290→          excludedScope: inferExcludedScope(productType, architecture),
   291→          priceStrategy,
   292→          priceRange,
   293→          monetizationModel: priceStrategy.model,

   295→          positioning: data.positioning,

   300→          launchRequirements: [],
   301→          assumptions: [],
   302→          risks: economics.riskFactors,
   303→          evidence: "INFERRED",
   304→          provenance: `commercial-product:${productId}`,

   337→/**
   338→ * Evaluate commercial readiness of a product.
   339→ *
   340→ * Each dimension is scored 0-1 with evidence.
   341→ * Overall = weighted average.
   342→ * Warnings are added when data is insufficient.
   343→ * NEVER inflates scores — honest assessment only.
   344→ */
   345→export function evaluateCommercialReadiness(
   346→  product: CommercialProduct,
   347→  qaReport?: { passed: boolean; issues: string[] },
   348→  assets?: { id: string; type: string; verified: boolean }[]
   349→): CommercialReadinessScore {
   350→  // Product completeness: do we have a well-defined product?
   351→  const productCompleteness = scoreProductCompleteness(product);
   352→
   353→  // Content completeness: is content ready?
   354→  const contentCompleteness = scoreContentCompleteness(product, assets);
   355→
   356→  // Quality: has QA been done?
   357→  const quality = scoreQuality(qaReport);
   358→
   359→  // Packaging: is the product packaged for delivery?
   360→  const packaging = scorePackaging(product);
   361→
   362→  // Delivery: can we deliver the product?
   363→  const delivery = scoreDelivery(product);
   364→
   365→  // Differentiation: is the product differentiated?
   366→  const differentiation = scoreDifferentiation(product);
   367→
   368→  // Evidence: how strong is our evidence?
   369→  const evidence = scoreEvidence(product);
   370→
   371→  // Commercial clarity: is the offer clear?
   372→  const commercialClarity = scoreCommercialClarity(product);
   373→
   374→  // Weighted average — never inflate
   375→  const weights = {
   376→    productCompleteness: 0.15,
   377→    contentCompleteness: 0.15,
   378→    quality: 0.15,
   379→    packaging: 0.10,
   380→    delivery: 0.10,
   381→    differentiation: 0.10,
   382→    evidence: 0.15,
   383→    commercialClarity: 0.10,
   384→  };
   385→
   386→  const overall = Math.min(
   387→    1,
   388→    weights.productCompleteness * productCompleteness.score +
   389→      weights.contentCompleteness * contentCompleteness.score +
   390→      weights.quality * quality.score +
   391→      weights.packaging * packaging.score +
   392→      weights.delivery * delivery.score +
   393→      weights.differentiation * differentiation.score +
   394→      weights.evidence * evidence.score +
   395→      weights.commercialClarity * commercialClarity.score
   396→  );
   397→
   398→  return {
   399→    overall: roundScore(overall),
   400→    productCompleteness,
   401→    contentCompleteness,
   402→    quality,
   403→    packaging,
   404→    delivery,
   405→    differentiation,
   406→    evidence,
   407→    commercialClarity,
   408→  };
   409→}

   413→function scoreProductCompleteness(product: CommercialProduct): DimensionScore {
   414→  const warnings: string[] = [];
   415→  let score = 0;
   416→
   417→  // Check core fields
   418→  if (product.problem && product.problem.length > 10) score += 0.15;
   419→  else warnings.push("Problem statement is missing or too short");
   420→
   421→  if (product.jobToBeDone && product.jobToBeDone.length > 10) score += 0.15;
   422→  else warnings.push("Job to be done is missing or too short");
   423→
   424→  if (product.valueProposition && product.valueProposition.length > 10) score += 0.2;
   425→  else warnings.push("Value proposition is missing or too short");
   426→
   427→  if (product.targetCustomer.description.length > 10) score += 0.15;
   428→  else warnings.push("Target customer is not well defined");
   429→
   430→  if (product.keyBenefits.length > 0) score += 0.15;
   431→  else warnings.push("No key benefits defined");
   432→
   433→  if (product.format) score += 0.1;
   434→  else warnings.push("Product format not specified");
   435→
   436→  if (product.differentiation && product.differentiation.length > 5) score += 0.1;
   437→  else warnings.push("Differentiation is not defined");
   438→
   439→  return {
   440→    score: roundScore(score),
   441→    evidence: "INFERRED",
   442→    confidence: score > 0.7 ? 0.8 : score > 0.4 ? 0.5 : 0.2,
   443→    warnings,
   444→  };
   445→}
   446→
   447→function scoreContentCompleteness(
   448→  product: CommercialProduct,
   449→  assets?: { id: string; type: string; verified: boolean }[]
   450→): DimensionScore {
   451→  const warnings: string[] = [];
   452→
   453→  if (!assets || assets.length === 0) {
   454→    warnings.push("No assets available — content cannot be verified");
   455→    return {
   456→      score: 0.1,
   457→      evidence: "NOT_VERIFIED",
   458→      confidence: 0.1,
   459→      warnings,
   460→    };
   461→  }
   462→
   463→  const verifiedAssets = assets.filter((a) => a.verified).length;
   464→  const totalAssets = assets.length;
   465→  const verifiedRatio = totalAssets > 0 ? verifiedAssets / totalAssets : 0;
   466→
   467→  if (verifiedRatio < 0.5) {
   468→    warnings.push(`Only ${verifiedAssets}/${totalAssets} assets are verified`);
   469→  }
   470→
   471→  return {
   472→    score: roundScore(Math.min(1, verifiedRatio * 1.2)),
   473→    evidence: verifiedRatio > 0.8 ? "VERIFIED" : verifiedRatio > 0.5 ? "INFERRED" : "NOT_VERIFIED",
   474→    confidence: verifiedRatio > 0.8 ? 0.9 : verifiedRatio > 0.5 ? 0.6 : 0.3,
   475→    warnings,
   476→  };
   477→}

   479→function scoreQuality(
   480→  qaReport?: { passed: boolean; issues: string[] }
   481→): DimensionScore {
   482→  const warnings: string[] = [];
   483→
   484→  if (!qaReport) {
   485→    warnings.push("No QA report available — quality cannot be verified");
   486→    return {
   487→      score: 0.2,
   488→      evidence: "NOT_VERIFIED",
   489→      confidence: 0.1,
   490→      warnings,
   491→    };
   492→  }
   493→
   494→  if (!qaReport.passed) {
   495→    warnings.push(`QA did not pass: ${qaReport.issues.length} issues found`);
   496→    return {
   497→      score: 0.3,
   498→      evidence: "VERIFIED",
   499→      confidence: 0.8,
   500→      warnings,
   501→    };
   502→  }
   503→
   504→  if (qaReport.issues.length > 0) {
   505→    warnings.push(`QA passed with ${qaReport.issues.length} minor issues`);
   506→  }
   507→
   508→  return {
   509→    score: qaReport.issues.length === 0 ? 0.95 : 0.75,
   510→    evidence: "VERIFIED",
   511→    confidence: 0.9,
   512→    warnings,
   513→  };
   514→}

   516→function scorePackaging(product: CommercialProduct): DimensionScore {
   517→  const warnings: string[] = [];
   518→  let score = 0;
   519→
   520→  if (product.format) score += 0.25;
   521→  else warnings.push("Format not specified");
   522→
   523→  if (product.distributionStrategy) score += 0.25;
   524→  else warnings.push("Distribution strategy not defined");
   525→
   526→  if (product.includedAssets.length > 0) score += 0.25;
   527→  else warnings.push("No included assets defined");
   528→
   529→  if (product.priceRange.recommended > 0) score += 0.25;
   530→  else warnings.push("Price not set");
   531→
   532→  return {
   533→    score: roundScore(score),
   534→    evidence: "INFERRED",
   535→    confidence: score > 0.7 ? 0.7 : 0.4,
   536→    warnings,
   537→  };
   538→}

   540→function scoreDelivery(product: CommercialProduct): DimensionScore {
   541→  const warnings: string[] = [];
   542→  let score = 0;
   543→
   544→  if (product.distributionStrategy) score += 0.4;
   545→  else warnings.push("No distribution strategy");
   546→
   547→  if (product.format) score += 0.3;
   548→  else warnings.push("No delivery format specified");
   549→
   550→  if (product.priceStrategy.model) score += 0.3;
   551→  else warnings.push("No monetization model");
   552→
   553→  return {
   554→    score: roundScore(score),
   555→    evidence: "INFERRED",
   556→    confidence: 0.5,
   557→    warnings,
   558→  };
   559→}
   560→
   561→function scoreDifferentiation(product: CommercialProduct): DimensionScore {
   562→  const warnings: string[] = [];
   563→
   564→  if (!product.differentiation || product.differentiation.length < 10) {
   565→    warnings.push("Differentiation is weak or undefined");
   566→    return {
   567→      score: 0.2,
   568→      evidence: "NOT_VERIFIED",
   569→      confidence: 0.2,
   570→      warnings,
   571→    };
   572→  }
   573→
   574→  // Basic heuristic: longer, more specific differentiation statements score higher
   575→  const lengthScore = Math.min(1, product.differentiation.length / 200);
   576→  const hasSpecifics = product.keyBenefits.length > 2;
   577→
   578→  const score = Math.min(1, lengthScore * 0.6 + (hasSpecifics ? 0.4 : 0.1));
   579→
   580→  if (!hasSpecifics) {
   581→    warnings.push("Few key benefits — differentiation may not be convincing");
   582→  }
   583→
   584→  return {
   585→    score: roundScore(score),
   586→    evidence: "INFERRED",
   587→    confidence: 0.5,
   588→    warnings,
   589→  };
   590→}

   592→function scoreEvidence(product: CommercialProduct): DimensionScore {
   593→  const warnings: string[] = [];
   594→
   595→  // Check how much evidence is actually verified
   596→  const overallEvidence = product.evidence;
   597→  const priceEvidence = product.priceRange.evidence;
   598→  const strategyEvidence = product.priceStrategy.evidence;
   599→
   600→  // Count proof requirements that are available
   601→  const availableProofs = product.proofRequirements.filter(
   602→    (p) => p.status === "AVAILABLE"
   603→  ).length;
   604→  const totalProofs = product.proofRequirements.length;
   605→  const proofRatio = totalProofs > 0 ? availableProofs / totalProofs : 0;
   606→
   607→  let score = 0;
   608→
   609→  // Overall evidence quality
   610→  if (overallEvidence === "VERIFIED") score += 0.3;
   611→  else if (overallEvidence === "INFERRED") score += 0.15;
   612→  else warnings.push("Overall product evidence is not verified");
   613→
   614→  // Price evidence
   615→  if (priceEvidence === "VERIFIED") score += 0.2;
   616→  else if (priceEvidence === "ESTIMATED" || priceEvidence === "INFERRED") score += 0.1;
   617→  else warnings.push("Price evidence is not established");
   618→
   619→  // Strategy evidence
   620→  if (strategyEvidence === "VERIFIED") score += 0.2;
   621→  else if (strategyEvidence === "ESTIMATED" || strategyEvidence === "INFERRED") score += 0.1;
   622→  else warnings.push("Strategy evidence is not established");
   623→
   624→  // Proof availability
   625→  score += proofRatio * 0.3;
   626→
   627→  if (proofRatio < 0.5) {
   628→    warnings.push(
   629→      `Only ${availableProofs}/${totalProofs} proof requirements are available — most are MISSING`
   630→    );
   631→  }
   632→
   633→  return {
   634→    score: roundScore(score),
   635→    evidence: overallEvidence === "VERIFIED" ? "VERIFIED" : "INFERRED",
   636→    confidence: proofRatio > 0.7 ? 0.8 : proofRatio > 0.3 ? 0.5 : 0.2,
   637→    warnings,
   638→  };
   639→}

   641→function scoreCommercialClarity(product: CommercialProduct): DimensionScore {
   642→  const warnings: string[] = [];
   643→  let score = 0;
   644→
   645→  if (product.valueProposition.length > 20) score += 0.2;
   646→  else warnings.push("Value proposition is too short to be compelling");
   647→
   648→  if (product.positioning.length > 10) score += 0.2;
   649→  else warnings.push("Positioning is missing");
   650→
   651→  if (product.objections.length > 0) score += 0.15;
   652→  else warnings.push("No customer objections identified — may indicate incomplete customer analysis");
   653→
   654→  if (product.keyBenefits.length >= 3) score += 0.2;
   655→  else if (product.keyBenefits.length > 0) score += 0.1;
   656→  else warnings.push("No key benefits defined");
   657→
   658→  if (product.priceRange.recommended > 0) score += 0.15;
   659→  else warnings.push("No recommended price set");
   660→
   661→  if (product.targetCustomer.painPoints.length > 0) score += 0.1;
   662→  else warnings.push("No customer pain points identified");
   663→
   664→  return {
   665→    score: roundScore(score),
   666→    evidence: "INFERRED",
   667→    confidence: score > 0.7 ? 0.7 : 0.4,
   668→    warnings,
   669→  };
   670→}

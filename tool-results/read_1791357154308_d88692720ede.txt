   356→   356→// ─── EditorialDesign Class ─────────────────────────────────────────────
   357→   357→
   358→   358→/**
   359→   359→ * EditorialDesign — produces editorial specifications and deterministic layouts.
   360→   360→ *
   361→   361→ * Two main operations:
   362→   362→ * 1. design() — produces the EditorialSpecification (AC-038)
   363→   363→ * 2. layout() — produces a deterministic LayoutResult (AC-039)
   364→   364→ *
   365→   365→ * Both operations derive from the same ArtDirectionSpec, ensuring
   366→   366→ * visual system coherence throughout (AC-040).
   367→   367→ */
   368→   368→export class EditorialDesign {
   369→   369→  /**
   370→   370→   * Produce the editorial specification from art direction and book structure.
   371→   371→   *
   372→   372→   * @param artDirection - The visual system specification
   373→   373→   * @param bookArchitecture - The book's editorial structure
   374→   374→   * @returns Complete EditorialSpecification (AC-038)
   375→   375→   */
   376→   376→  design(
   377→   377→    artDirection: ArtDirectionSpec,
   378→   378→    bookArchitecture: BookArchitectureInput
   379→   379→  ): EditorialSpecification {
   380→   380→    const margins = this.computeMargins(artDirection);
   381→   381→    const grid = this.buildGrid(artDirection);
   382→   382→    const headingStyles = this.buildHeadingStyles(artDirection);
   383→   383→    const paragraphStyle = this.buildParagraphStyle(artDirection);
   384→   384→    const quoteStyle = this.buildQuoteStyle(artDirection);
   385→   385→    const listStyle = this.buildListStyle(artDirection);
   386→   386→    const tableStyle = this.buildTableStyle(artDirection);
   387→   387→    const imageStyle = this.buildImageStyle(artDirection);
   388→   388→    const captionStyle = this.buildCaptionStyle(artDirection);
   389→   389→    const chapterStyle = this.buildChapterStyle(artDirection, headingStyles[0]);
   390→   390→    const pageBreakRules = this.buildPageBreakRules();
   391→   391→    const numbering = this.buildNumberingConfig();
   392→   392→    const indexConfig = this.buildIndexConfig(bookArchitecture);
   393→   393→    const hierarchyRules = this.buildHierarchyRules();
   394→   394→
   395→   395→    return {
   396→   396→      id: randomUUID(),
   397→   397→      createdAt: new Date().toISOString(),
   398→   398→      visual_system_id: artDirection.visual_system.id,
   399→   399→      margins,
   400→   400→      grid,
   401→   401→      hierarchy_rules: hierarchyRules,
   402→   402→      heading_styles: headingStyles,
   403→   403→      paragraph_style: paragraphStyle,
   404→   404→      quote_style: quoteStyle,
   405→   405→      list_style: listStyle,
   406→   406→      table_style: tableStyle,
   407→   407→      image_style: imageStyle,
   408→   408→      caption_style: captionStyle,
   409→   409→      chapter_style: chapterStyle,
   410→   410→      page_break_rules: pageBreakRules,
   411→   411→      numbering,
   412→   412→      index_config: indexConfig,
   413→   413→    };
   414→   414→  }
   415→   415→
   416→   416→  /**
   417→   417→   * Produce a deterministic layout from content and specifications.
   418→   418→   *
   419→   419→   * AC-039: Layout is deterministic — same inputs always produce same output.
   420→   420→   * This is achieved by processing content sequentially with fixed rules,
   421→   421→   * no randomization, and a deterministic page-breaking algorithm.
   422→   422→   *
   423→   423→   * @param content - The book content to lay out
   424→   424→   * @param editorial - The editorial specification
   425→   425→   * @param artDirection - The visual system (AC-040)
   426→   426→   * @returns Deterministic LayoutResult (AC-039)
   427→   427→   */
   428→   428→  layout(
   429→   429→    content: BookContent,
   430→   430→    editorial: EditorialSpecification,
   431→   431→    artDirection: ArtDirectionSpec
   432→   432→  ): LayoutResult {
   433→   433→    const inputHash = this.computeInputHash(content, editorial);
   434→   434→    const { width } = artDirection.visual_system.page_dimensions;
   435→   435→    const contentWidth = width - editorial.margins.inner - editorial.margins.outer;
   436→   436→    const contentHeight = artDirection.visual_system.page_dimensions.height
   437→   437→      - editorial.margins.top
   438→   438→      - editorial.margins.bottom;
   439→   439→
   440→   440→    const pages: LayoutPage[] = [];
   441→   441→    const toc: LayoutResult["table_of_contents"] = [];
   442→   442→
   443→   443→    let currentPage = 0;
   444→   444→    let currentY = 0;
   445→   445→    let currentChapter: number | null = null;
   446→   446→    let currentPageElements: LayoutElement[] = [];
   447→   447→    let remainingSpace = contentHeight;
   448→   448→
   449→   449→    // Process each chapter
   450→   450→    for (const chapter of content.chapters) {
   451→   451→      // Chapter opening — always starts new page
   452→   452→      if (currentPage > 0 || currentPageElements.length > 0) {
   453→   453→        // Finalize previous page
   454→   454→        pages.push(this.finalizePage(
   455→   455→          currentPage,
   456→   456→          currentChapter,
   457→   457→          currentPageElements,
   458→   458→          remainingSpace,
   459→   459→          editorial
   460→   460→        ));
   461→   461→        currentPage++;
   462→   462→        currentPageElements = [];
   463→   463→        currentY = 0;
   464→   464→        remainingSpace = contentHeight;
   465→   465→      }
   466→   466→
   467→   467→      currentChapter = chapter.chapter_number;
   468→   468→
   469→   469→      // Chapter heading
   470→   470→      const chHeadingHeight = this.estimateElementHeight(
   471→   471→        "heading",
   472→   472→        chapter.title,
   473→   473→        editorial.chapter_style.title_style,
   474→   474→        contentWidth
   475→   475→      );
   476→   476→
   477→   477→      toc.push({
   478→   478→        title: chapter.title,
   479→   479→        page: currentPage + 1, // 1-indexed for display
   480→   480→        level: 1,
   481→   481→      });
   482→   482→
   483→   483→      currentPageElements.push({
   484→   484→        type: "heading",
   485→   485→        page: currentPage,
   486→   486→        y: currentY,
   487→   487→        x: editorial.margins.inner,
   488→   488→        width: contentWidth,
   489→   489→        height: chHeadingHeight,
   490→   490→        content_ref: `chapter-${chapter.chapter_number}-title`,
   491→   491→        style: { ...editorial.chapter_style.title_style },
   492→   492→      });
   493→   493→
   494→   494→      currentY += chHeadingHeight;
   495→   495→      remainingSpace -= chHeadingHeight;
   496→   496→
   497→   497→      // Process sections
   498→   498→      for (const section of chapter.sections) {
   499→   499→        // Section heading
   500→   500→        if (section.heading) {
   501→   501→          const headingStyle = editorial.heading_styles.find(
   502→   502→            (h) => h.level === section.level
   503→   503→          ) ?? editorial.heading_styles[editorial.heading_styles.length - 1];
   504→   504→
   505→   505→          const headingHeight = this.estimateElementHeight(
   506→   506→            "heading",
   507→   507→            section.heading,
   508→   508→            headingStyle,
   509→   509→            contentWidth
   510→   510→          );
   511→   511→
   512→   512→          if (headingHeight > remainingSpace) {
   513→   513→            pages.push(this.finalizePage(
   514→   514→              currentPage, currentChapter, currentPageElements, remainingSpace, editorial
   515→   515→            ));
   516→   516→            currentPage++;
   517→   517→            currentPageElements = [];
   518→   518→            currentY = 0;
   519→   519→            remainingSpace = contentHeight;
   520→   520→          }
   521→   521→
   522→   522→          toc.push({
   523→   523→            title: section.heading,
   524→   524→            page: currentPage + 1,
   525→   525→            level: section.level,
   526→   526→          });
   527→   527→
   528→   528→          currentPageElements.push({
   529→   529→            type: "heading",
   530→   530→            page: currentPage,
   531→   531→            y: currentY,
   532→   532→            x: editorial.margins.inner,
   533→   533→            width: contentWidth,
   534→   534→            height: headingHeight,
   535→   535→            content_ref: `chapter-${chapter.chapter_number}-section-${section.level}`,
   536→   536→            style: { ...headingStyle },
   537→   537→          });
   538→   538→
   539→   539→          currentY += headingHeight;
   540→   540→          remainingSpace -= headingHeight;
   541→   541→        }
   542→   542→
   543→   543→        // Paragraphs
   544→   544→        for (let pi = 0; pi < section.paragraphs.length; pi++) {
   545→   545→          const para = section.paragraphs[pi];
   546→   546→          const paraHeight = this.estimateElementHeight(
   547→   547→            "paragraph",
   548→   548→            para,
   549→   549→            editorial.paragraph_style,
   550→   550→            contentWidth
   551→   551→          );
   552→   552→
   553→   553→          if (paraHeight > remainingSpace) {
   554→   554→            pages.push(this.finalizePage(
   555→   555→              currentPage, currentChapter, currentPageElements, remainingSpace, editorial
   556→   556→            ));
   557→   557→            currentPage++;
   558→   558→            currentPageElements = [];
   559→   559→            currentY = 0;
   560→   560→            remainingSpace = contentHeight;
   561→   561→          }
   562→   562→
   563→   563→          currentPageElements.push({
   564→   564→            type: "paragraph",
   565→   565→            page: currentPage,
   566→   566→            y: currentY,
   567→   567→            x: editorial.margins.inner,
   568→   568→            width: contentWidth,
   569→   569→            height: Math.min(paraHeight, remainingSpace),
   570→   570→            content_ref: `chapter-${chapter.chapter_number}-para-${pi}`,
   571→   571→            style: { ...editorial.paragraph_style },
   572→   572→          });
   573→   573→
   574→   574→          currentY += paraHeight;
   575→   575→          remainingSpace -= paraHeight;
   576→   576→        }
   577→   577→      }
   578→   578→    }
   579→   579→
   580→   580→    // Finalize last page
   581→   581→    if (currentPageElements.length > 0) {
   582→   582→      pages.push(this.finalizePage(
   583→   583→        currentPage, currentChapter, currentPageElements, remainingSpace, editorial
   584→   584→      ));
   585→   585→    }
   586→   586→
   587→   587→    return {
   588→   588→      id: randomUUID(),
   589→   589→      createdAt: new Date().toISOString(),
   590→   590→      input_hash: inputHash,
   591→   591→      page_count: pages.length,
   592→   592→      pages,
   593→   593→      table_of_contents: toc,
   594→   594→      visual_system_id: artDirection.visual_system.id,
   595→   595→    };
   596→   596→  }

   857→   857→   * Estimate the height of a layout element in points.
   858→   858→   * Used by the deterministic layout algorithm.
   859→   859→   */
   860→   860→  private estimateElementHeight(
   861→   861→    type: "heading" | "paragraph",
   862→   862→    text: string,
   863→   863→    style: { size: number; line_height: number; space_before?: number; space_after?: number },
   864→   864→    contentWidth: number
   865→   865→  ): number {
   866→   866→    // Estimate characters per line based on font size and width
   867→   867→    const charsPerLine = Math.floor(contentWidth / (style.size * 0.5));
   868→   868→    const lines = Math.max(1, Math.ceil(text.length / charsPerLine));
   869→   869→    const lineHeight = style.size * style.line_height;
   870→   870→    const height = lines * lineHeight;
   871→   871→
   872→   872→    const spaceBefore = ("space_before" in style && typeof style.space_before === "number")
   873→   873→      ? style.space_before : 0;
   874→   874→    const spaceAfter = ("space_after" in style && typeof style.space_after === "number")
   875→   875→      ? style.space_after : 0;
   876→   876→
   877→   877→    return height + spaceBefore + spaceAfter;
   878→   878→  }

   908→   908→  /**
   909→   909→   * Compute a deterministic hash of layout inputs for reproducibility check.
   910→   910→   * AC-039: Same inputs → same hash → same layout.
   911→   911→   */
   912→   912→  private computeInputHash(
   913→   913→    content: BookContent,
   914→   914→    editorial: EditorialSpecification
   915→   915→  ): string {
   916→   916→    // Simple deterministic hash — concatenate key inputs and hash
   917→   917→    const data = JSON.stringify({
   918→   918→      title: content.title,
   919→   919→      chapterCount: content.chapters.length,
   920→   920→      totalWords: content.total_word_count,
   921→   921→      editorialId: editorial.id,
   922→   922→      margins: editorial.margins,
   923→   923→      grid: editorial.grid,
   924→   924→    });
   925→   925→
   926→   926→    // Simple hash function (djb2)
   927→   927→    let hash = 5381;
   928→   928→    for (let i = 0; i < data.length; i++) {
   929→   929→      hash = ((hash << 5) + hash + data.charCodeAt(i)) & 0xffffffff;
   930→   930→    }
   931→   931→    return hash.toString(16).padStart(8, "0");
   932→   932→  }

   152→   152→// ─── VisualQA Class ────────────────────────────────────────────────────
   153→   153→
   154→   154→/**
   155→   155→ * VisualQA — inspects PDF/layout results for visual quality issues.
   156→   156→ *
   157→   157→ * Three main operations:
   158→   158→ * 1. inspect() — detect all visual quality issues (AC-043)
   159→   159→ * 2. diagnose() — determine root cause for each issue
   160→   160→ * 3. repair() — attempt to fix an issue
   161→   161→ *
   162→   162→ * AC-043: Never trust the renderer output blindly.
   163→   163→ * AC-044: Every issue has severity, location, description, suggested_fix.
   164→   164→ * AC-045: Repair loop with max 5 iterations.
   165→   165→ */
   166→   166→export class VisualQA {
   167→   167→  /**
   168→   168→   * Inspect a PDF/layout result for visual quality issues.
   169→   169→   *
   170→   170→   * AC-043: This inspection is MANDATORY. The renderer finishing does NOT
   171→   171→   * mean the PDF is done — we must actively verify visual quality.
   172→   172→   *
   173→   173→   * @param pdfSpec - The PDF specification and layout to inspect
   174→   174→   * @param content - The source content that was rendered
   175→   175→   * @param editorial - The editorial specification used
   176→   176→   * @returns QAResult with all detected issues
   177→   177→   */
   178→   178→  inspect(
   179→   179→    pdfSpec: PDFSpecForQA,
   180→   180→    content: BookContent,
   181→   181→    editorial: EditorialSpecification
   182→   182→  ): QAResult {
   183→   183→    const startTime = Date.now();
   184→   184→    const issues: QAIssue[] = [];
   185→   185→
   186→   186→    // Run all inspection checks
   187→   187→    issues.push(...this.checkCutText(pdfSpec, content, editorial));
   188→   188→    issues.push(...this.checkOverflow(pdfSpec, editorial));
   189→   189→    issues.push(...this.checkOffPage(pdfSpec));
   190→   190→    issues.push(...this.checkEmptyPages(pdfSpec));
   191→   191→    issues.push(...this.checkBadHierarchy(pdfSpec, content));
   192→   192→    issues.push(...this.checkContrastProblems(pdfSpec, editorial));
   193→   193→    issues.push(...this.checkTypographicInconsistency(pdfSpec, editorial));
   194→   194→    issues.push(...this.checkNumberingErrors(pdfSpec, content));
   195→   195→    issues.push(...this.checkLegibilityProblems(pdfSpec, editorial));
   196→   196→
   197→   197→    // Compute summaries
   198→   198→    const byCategory: Record<QAIssueCategory, number> = {
   199→   199→      cut_text: 0, overflow: 0, off_page: 0, distorted_images: 0,
   200→   200→      empty_pages: 0, bad_hierarchy: 0, contrast_problems: 0,
   201→   201→      typographic_inconsistency: 0, cover_errors: 0, numbering_errors: 0,
   202→   202→      index_errors: 0, legibility_problems: 0,
   203→   203→    };
   204→   204→
   205→   205→    const bySeverity: Record<QAIssueSeverity, number> = {
   206→   206→      critical: 0, warning: 0, info: 0,
   207→   207→    };
   208→   208→
   209→   209→    for (const issue of issues) {
   210→   210→      byCategory[issue.category]++;
   211→   211→      bySeverity[issue.severity]++;
   212→   212→    }
   213→   213→
   214→   214→    const duration = Date.now() - startTime;
   215→   215→
   216→   216→    return {
   217→   217→      passed: bySeverity.critical === 0,
   218→   218→      issue_count: issues.length,
   219→   219→      issues,
   220→   220→      by_category: byCategory,
   221→   221→      by_severity: bySeverity,
   222→   222→      inspectedAt: new Date().toISOString(),
   223→   223→      inspection_duration_ms: duration,
   224→   224→    };
   225→   225→  }
   226→   226→
   227→   227→  /**
   228→   228→   * Diagnose the root cause of QA issues.
   229→   229→   *
   230→   230→   * @param issues - The issues to diagnose
   231→   231→   * @returns Array of diagnoses with repair strategies
   232→   232→   */
   233→   233→  diagnose(issues: QAIssue[]): Diagnosis[] {
   234→   234→    return issues.map((issue) => this.diagnoseSingle(issue));
   235→   235→  }
   236→   236→
   237→   237→  /**
   238→   238→   * Attempt to repair a single QA issue.
   239→   239→   *
   240→   240→   * @param issue - The issue to repair
   241→   241→   * @param content - Current book content
   242→   242→   * @param editorial - Current editorial specification
   243→   243→   * @returns RepairedContent with the repair result
   244→   244→   */
   245→   245→  repair(
   246→   246→    issue: QAIssue,
   247→   247→    content: BookContent,
   248→   248→    editorial: EditorialSpecification
   249→   249→  ): RepairedContent {
   250→   250→    if (!issue.auto_repairable) {
   251→   251→      return {
   252→   252→        repaired: false,
   253→   253→        issue_id: issue.id,
   254→   254→        repair_description: "Issue is not auto-repairable — requires manual intervention",
   255→   255→        remaining_issues: [issue.id],
   256→   256→      };
   257→   257→    }
   258→   258→
   259→   259→    switch (issue.category) {
   260→   260→      case "cut_text":
   261→   261→        return this.repairCutText(issue, content, editorial);
   262→   262→      case "overflow":
   263→   263→        return this.repairOverflow(issue, content, editorial);
   264→   264→      case "empty_pages":
   265→   265→        return this.repairEmptyPages(issue, content, editorial);
   266→   266→      case "bad_hierarchy":
   267→   267→        return this.repairBadHierarchy(issue, content, editorial);
   268→   268→      case "numbering_errors":
   269→   269→        return this.repairNumberingErrors(issue, content, editorial);
   270→   270→      default:
   271→   271→        return {
   272→   272→          repaired: false,
   273→   273→          issue_id: issue.id,
   274→   274→          repair_description: `Auto-repair not implemented for category: ${issue.category}`,
   275→   275→          remaining_issues: [issue.id],
   276→   276→        };
   277→   277→    }
   278→   278→  }
   279→   279→
   280→   280→  // ─── Inspection Checks ─────────────────────────────────────────────
   281→   281→
   282→   282→  /**
   283→   283→   * Check for cut/truncated text.
   284→   284→   */
   285→   285→  private checkCutText(
   286→   286→    pdfSpec: PDFSpecForQA,
   287→   287→    content: BookContent,
   288→   288→    editorial: EditorialSpecification
   289→   289→  ): QAIssue[] {
   290→   290→    const issues: QAIssue[] = [];
   291→   291→
   292→   292→    for (const page of pdfSpec.layout.pages) {
   293→   293→      for (const element of page.elements) {
   294→   294→        if (element.type === "paragraph" || element.type === "heading") {
   295→   295→          // Check if element height was constrained by remaining space
   296→   296→          const contentRef = element.content_ref;
   297→   297→          const sourceContent = this.findSourceContent(content, contentRef);
   298→   298→
   299→   299→          if (sourceContent && element.height < this.estimateMinHeight(sourceContent, editorial)) {
   300→   300→            issues.push({
   301→   301→              id: `qa-cut-${page.page_number}-${element.content_ref}`,
   302→   302→              category: "cut_text",
   303→   303→              severity: "critical",
   304→   304→              location: {
   305→   305→                page: page.page_number,
   306→   306→                element_ref: element.content_ref,
   307→   307→                bounds: { x: element.x, y: element.y, width: element.width, height: element.height },
   308→   308→              },
   309→   309→              description: `Text appears truncated on page ${page.page_number + 1}: element height (${element.height.toFixed(1)}pt) is less than estimated minimum`,
   310→   310→              suggested_fix: "Reduce font size, increase line height, or split content across pages",
   311→   311→              auto_repairable: true,
   312→   312→            });
   313→   313→          }
   314→   314→        }
   315→   315→      }
   316→   316→    }
   317→   317→
   318→   318→    return issues;
   319→   319→  }
   320→   320→
   321→   321→  /**
   322→   322→   * Check for content overflow.
   323→   323→   */
   324→   324→  private checkOverflow(
   325→   325→    pdfSpec: PDFSpecForQA,
   326→   326→    editorial: EditorialSpecification
   327→   327→  ): QAIssue[] {
   328→   328→    const issues: QAIssue[] = [];
   329→   329→    const pageHeight = pdfSpec.layout.pages[0]
   330→   330→      ? editorial.margins.top + editorial.margins.bottom +
   331→   331→        pdfSpec.layout.pages[0].remaining_space +
   332→   332→        pdfSpec.layout.pages[0].elements.reduce((sum, e) => sum + e.height, 0)
   333→   333→      : 0;
   334→   334→
   335→   335→    for (const page of pdfSpec.layout.pages) {
   336→   336→      for (const element of page.elements) {
   337→   337→        const elementBottom = element.y + element.height;
   338→   338→        const contentBottom = pageHeight - editorial.margins.bottom;
   339→   339→
   340→   340→        if (elementBottom > contentBottom) {
   341→   341→          issues.push({
   342→   342→            id: `qa-overflow-${page.page_number}-${element.content_ref}`,
   343→   343→            category: "overflow",
   344→   344→            severity: "critical",
   345→   345→            location: {
   346→   346→              page: page.page_number,
   347→   347→              element_ref: element.content_ref,
   348→   348→            },
   349→   349→            description: `Content overflows page bottom by ${(elementBottom - contentBottom).toFixed(1)}pt on page ${page.page_number + 1}`,
   350→   350→            suggested_fix: "Move overflowing content to next page or reduce element size",
   351→   351→            auto_repairable: true,
   352→   352→          });
   353→   353→        }
   354→   354→      }
   355→   355→    }
   356→   356→
   357→   357→    return issues;
   358→   358→  }

   379→   379→            suggested_fix: "Adjust margins or reposition element within page bounds",

   389→   389→  /**
   390→   390→   * Check for empty pages (pages with no meaningful content).
   391→   391→   */
   392→   392→  private checkEmptyPages(pdfSpec: PDFSpecForQA): QAIssue[] {
   393→   393→    const issues: QAIssue[] = [];
   394→   394→
   395→   395→    for (const page of pdfSpec.layout.pages) {
   396→   396→      const contentElements = page.elements.filter(
   397→   397→        (e) => e.type !== "page_break" && e.type !== "decoration"
   398→   398→      );
   399→   399→
   400→   400→      if (contentElements.length === 0 && pdfSpec.layout.pages.length > 1) {
   401→   401→        issues.push({
   402→   402→          id: `qa-empty-${page.page_number}`,
   403→   403→          category: "empty_pages",
   404→   404→          severity: "warning",
   405→   405→          location: { page: page.page_number },
   406→   406→          description: `Page ${page.page_number + 1} has no meaningful content`,
   407→   407→          suggested_fix: "Remove empty page or redistribute content from adjacent pages",
   408→   408→          auto_repairable: true,
   409→   409→        });
   410→   410→      }
   411→   411→    }
   412→   412→
   413→   413→    return issues;
   414→   414→  }

   416→   416→  /**
   417→   417→   * Check for broken heading hierarchy.
   418→   418→   */
   419→   419→  private checkBadHierarchy(
   420→   420→    pdfSpec: PDFSpecForQA,
   421→   421→    content: BookContent
   422→   422→  ): QAIssue[] {
   423→   423→    const issues: QAIssue[] = [];
   424→   424→    let lastHeadingLevel = 0;
   425→   425→
   426→   426→    for (const chapter of content.chapters) {
   427→   427→      for (const section of chapter.sections) {
   428→   428→        if (section.heading) {
   429→   429→          // Check for skipped levels (e.g., h1 → h3)
   430→   430→          if (lastHeadingLevel > 0 && section.level > lastHeadingLevel + 1) {
   431→   431→            issues.push({
   432→   432→              id: `qa-hierarchy-${chapter.chapter_number}-${section.level}`,
   433→   433→              category: "bad_hierarchy",
   434→   434→              severity: "warning",
   435→   435→              location: {
   436→   436→                page: 0,
   437→   437→                chapter: chapter.chapter_number,
   438→   438→              },
   439→   439→              description: `Heading hierarchy skip: level ${lastHeadingLevel} to ${section.level} in chapter ${chapter.chapter_number}`,
   440→   440→              suggested_fix: "Insert intermediate heading level or adjust hierarchy",
   441→   441→              auto_repairable: true,
   442→   442→            });
   443→   443→          }
   444→   444→          lastHeadingLevel = section.level;
   445→   445→        }
   446→   446→      }
   447→   447→    }
   448→   448→
   449→   449→    return issues;
   450→   450→  }

   475→   475→        auto_repairable: true,

   482→   482→  /**
   483→   483→   * Check for typographic inconsistency.
   484→   484→   */
   485→   485→  private checkTypographicInconsistency(
   486→   486→    pdfSpec: PDFSpecForQA,
   487→   487→    editorial: EditorialSpecification
   488→   488→  ): QAIssue[] {
   489→   489→    const issues: QAIssue[] = [];
   490→   490→    const usedFonts = new Set<string>();
   491→   491→    const usedSizes = new Set<number>();
   492→   492→

   773→   773→  /**
   774→   774→   * Find source content by reference key.
   775→   775→   */
   776→   776→  private findSourceContent(
   777→   777→    content: BookContent,
   778→   778→    contentRef: string
   779→   779→  ): string | null {
   780→   780→    const match = contentRef.match(/chapter-(\d+)-para-(\d+)/);
   781→   781→    if (match) {
   782→   782→      const chapterNum = parseInt(match[1], 10);
   783→   783→      const paraIdx = parseInt(match[2], 10);
   784→   784→      const chapter = content.chapters.find((c) => c.chapter_number === chapterNum);
   785→   785→      if (chapter) {
   786→   786→        let paraCount = 0;
   787→   787→        for (const section of chapter.sections) {
   788→   788→          for (let i = 0; i < section.paragraphs.length; i++) {
   789→   789→            if (paraCount === paraIdx) return section.paragraphs[i];
   790→   790→            paraCount++;
   791→   791→          }
   792→   792→        }
   793→   793→      }
   794→   794→    }
   795→   795→    return null;
   796→   796→  }

   801→   801→  private estimateMinHeight(text: string, editorial: EditorialSpecification): number {
   802→   802→    const charsPerLine = 60; // conservative estimate

   807→   807→  /**
   808→   808→   * Estimate WCAG contrast ratio (simplified).
   809→   809→   */
   810→   810→  private estimateContrast(foreground: string, background: string): number {
   811→   811→    const fgLum = this.relativeLuminance(foreground);
   812→   812→    const bgLum = this.relativeLuminance(background);
   813→   813→    const lighter = Math.max(fgLum, bgLum);
   814→   814→    const darker = Math.min(fgLum, bgLum);
   815→   815→    return (lighter + 0.05) / (darker + 0.05);
   816→   816→  }
   817→   817→
   818→   818→  private relativeLuminance(hex: string): number {
   819→   819→    const r = this.sRGBtoLinear(parseInt(hex.slice(1, 3), 16) / 255);
   820→   820→    const g = this.sRGBtoLinear(parseInt(hex.slice(3, 5), 16) / 255);
   821→   821→    const b = this.sRGBtoLinear(parseInt(hex.slice(5, 7), 16) / 255);
   822→   822→    return 0.2126 * r + 0.7152 * g + 0.0722 * b;
   823→   823→  }
   824→   824→
   825→   825→  private sRGBtoLinear(c: number): number {
   826→   826→    return c <= 0.04045 ? c / 12.92 : Math.pow((c + 0.055) / 1.055, 2.4);
   827→   827→  }

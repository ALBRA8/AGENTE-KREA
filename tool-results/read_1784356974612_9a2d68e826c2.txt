   172→  const [prompt, setPrompt] = useState("");

   175→  const [result, setResult] = useState<string | null>(null);
   176→  const [error, setError] = useState("");
   177→
   178→  const generate = async () => {
   179→    setLoading(true); setError(""); setResult(null);
   180→    try {
   181→      const res = await fetch("/api/generate/image", {
   182→        method: "POST", headers: { "Content-Type": "application/json", Authorization: `Bearer ${user.id}` },
   183→        body: JSON.stringify({ prompt, style }),
   184→      });
   185→      const data = await res.json();
   186→      if (!res.ok) { setError(data.error); return; }

   190→  };
   191→
   192→  const styles = ["fotográfico, alta calidad", "ilustración digital", "estilo minimalista", "3D render", "arte conceptual", "diseño publicitario"];

   247→    copy: { title: "Copywriting", desc: "Textos persuasivos que venden", placeholder: "Ej: Escribe un headline para una campaña de lanzamiento de un curso de marketing digital para emprendedores", label: "Copy / Publicidad" },

   249→    script: { title: "Scripts de Video", desc: "Guiones para YouTube, TikTok, Reels", placeholder: "Ej: Escribe un script de 60 segundos para un reel que promocione una app de productividad", label: "Detalles del video" },
   250→    email: { title: "Email Marketing", desc: "Emails que abren y convierten", placeholder: "Ej: Escribe una secuencia de 3 emails de bienvenida para nuevos suscriptores de un newsletter de negocios", label: "Objetivo del email" },
   251→  };
   252→  const cfg = configs[type] || configs.copy;
   253→  const [prompt, setPrompt] = useState("");
   254→  const [loading, setLoading] = useState(false);
   255→  const [result, setResult] = useState("");
   256→  const [error, setError] = useState("");
   257→
   258→  const generate = async () => {
   259→    setLoading(true); setError(""); setResult("");
   260→    try {
   261→      const res = await fetch("/api/generate/text", {
   262→        method: "POST", headers: { "Content-Type": "application/json", Authorization: `Bearer ${user.id}` },
   263→        body: JSON.stringify({ prompt, type }),
   264→      });
   265→      const data = await res.json();
   266→      if (!res.ok) { setError(data.error); return; }
   267→      setResult(data.text); onUpdateCredits(data.credits);

   285→          </div>
   286→          <button onClick={generate} disabled={loading || !prompt} className="w-full py-3.5 rounded-xl bg-gradient-to-b from-[#3b82f6] to-[#1e40af] text-white font-bold text-sm shadow-[0_10px_24px_rgba(30,64,175,0.45)] hover:brightness-110 active:scale-[0.98] transition-all disabled:opacity-50 flex items-center justify-center gap-2">

   289→          {error && <p className="text-red-400 text-xs">{error}</p>}
   290→        </div>
   291→        <div>
   292→          {result ? (
   293→            <div className="rounded-2xl border border-white/[0.06] bg-[#0f1629] h-full flex flex-col">
   294→              <div className="flex items-center justify-between p-3 border-b border-white/[0.06]">

   304→          )}
   305→        </div>
   306→      </div>

   522→    { id: "text", label: "Textos" },

   525→    { id: "subtitle", label: "Subtítulos" },
   526→  ];

   539→      {loading ? (
   540→        <div className="flex items-center justify-center py-20 text-white/30"><Loader2 className="w-6 h-6 animate-spin mr-2" /> Cargando...</div>
   541→      ) : gens.length === 0 ? (

   569→/* ════════════ Main App ════════════ */
   570→interface AppShellProps { onLogout: () => void }
   571→
   572→export default function AppPage({ onLogout }: AppShellProps) {
   573→  const [user, setUser] = useState<User | null>(null);
   574→  const [page, setPage] = useState("dashboard");
   575→  const [sidebarOpen, setSidebarOpen] = useState(false);
   576→  const [loading, setLoading] = useState(true);
   577→  const [generations, setGenerations] = useState<Generation[]>([]);
   578→
   579→  const loadUser = useCallback(async () => {
   580→    const token = localStorage.getItem("p360_token");
   581→    if (!token) { onLogout(); return; }
   582→    try {
   583→      const res = await fetch("/api/auth/me", { headers: { Authorization: `Bearer ${token}` } });
   584→      if (!res.ok) { onLogout(); return; }
   585→      const data = await res.json();
   586→      setUser(data);
   587→      // Load recent generations
   588→      const gRes = await fetch("/api/generations?limit=10", { headers: { Authorization: `Bearer ${token}` } });
   589→      if (gRes.ok) setGenerations(await gRes.json());
   590→    } catch { onLogout(); }
   591→    setLoading(false);
   592→  }, []);
   593→
   594→  useEffect(() => { loadUser(); }, [loadUser]);
   595→
   596→  const logout = () => { localStorage.removeItem("p360_token"); localStorage.removeItem("p360_user"); onLogout(); };
   597→  const refreshUser = () => loadUser();
   598→  const onUpdateCredits = (c: number) => { if (user) setUser({ ...user, credits: c }); };
   599→
   600→  if (loading) return (
   601→    <div className="min-h-screen bg-[#080c16] flex items-center justify-center">
   602→      <Loader2 className="w-8 h-8 animate-spin text-[#3b82f6]" />
   603→    </div>
   604→  );
   605→
   606→  if (!user) return null;
   607→
   608→  const renderPage = () => {
   609→    switch (page) {
   610→      case "dashboard": return <Dashboard user={user} generations={generations} />;
   611→      case "images": return <ImageGenerator user={user} onUpdateCredits={onUpdateCredits} />;
   612→      case "copy": case "social": case "script": case "email": return <TextGenerator type={page} user={user} onUpdateCredits={onUpdateCredits} />;
   613→      case "voice": return <VoiceGenerator user={user} onUpdateCredits={onUpdateCredits} />;
   614→      case "ebook": return <EbookGenerator user={user} onUpdateCredits={onUpdateCredits} />;
   615→      case "subtitle": return <SubtitleGenerator user={user} onUpdateCredits={onUpdateCredits} />;
   616→      case "library": return <Library user={user} />;
   617→      default: return <Dashboard user={user} generations={generations} />;
   618→    }
   619→  };
   620→
   621→  return (
   622→    <AuthCtx.Provider value={{ user, logout, refreshUser }}>
   623→      <div className="min-h-screen bg-[#080c16] flex">
   624→        <Sidebar active={page} onNav={setPage} open={sidebarOpen} onClose={() => setSidebarOpen(false)} user={user} />
   625→        <div className="flex-1 flex flex-col min-h-screen">
   626→          {/* Top bar */}
   627→          <header className="sticky top-0 z-30 flex items-center gap-4 px-4 sm:px-6 py-3 border-b border-white/[0.06] backdrop-blur-xl bg-[#080c16]/80">
   628→            <button onClick={() => setSidebarOpen(true)} className="lg:hidden text-white/50 hover:text-white">
   629→              <Menu className="w-5 h-5" />
   630→            </button>
   631→            <div className="flex-1" />
   632→            <div className="flex items-center gap-3">
   633→              <div className="flex items-center gap-1.5 px-3 py-1.5 rounded-lg bg-yellow-400/10 text-yellow-400 text-xs font-bold">
   634→                <Coins className="w-3.5 h-3.5" /> {user.credits}
   635→              </div>
   636→              <button onClick={logout} className="p-2 rounded-lg hover:bg-white/5 text-white/40 hover:text-white/70 transition-colors" title="Cerrar sesión">
   637→                <LogOut className="w-4 h-4" />
   638→              </button>
   639→            </div>
   640→          </header>
   641→          {/* Content */}
   642→          <main className="flex-1 p-4 sm:p-6 lg:p-8 overflow-y-auto">
   643→            {renderPage()}
   644→          </main>
   645→        </div>
   646→      </div>
   647→    </AuthCtx.Provider>
   648→  );

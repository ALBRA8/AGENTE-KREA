   156→
   157→function StatusBadge({ status }: { status: string }) {
   158→  const upper = status.toUpperCase();
   159→  const colors = STATUS_COLORS[upper] ?? STATUS_COLORS.UNKNOWN;
   160→  const Icon = STATUS_ICONS[upper] ?? HelpCircle;
   161→  return (
   162→    <span className={`inline-flex items-center gap-1.5 px-2.5 py-0.5 rounded-lg text-[11px] font-semibold border ${colors}`}>
   163→      <Icon className="w-3 h-3" />
   164→      {status}
   165→    </span>
   166→  );
   167→}

   265→/* ════════════ Main Component ════════════ */
   266→
   267→export default function AdnDashboard({ user }: { user: User }) {
   268→  const [activeTab, setActiveTab] = useState<TabId>("identity");
   269→  const [loading, setLoading] = useState(true);
   270→  const [doctorRunning, setDoctorRunning] = useState(false);
   271→
   272→  /* Data state */
   273→  const [identity, setIdentity] = useState<Identity | null>(null);
   274→  const [mission, setMission] = useState<Mission | null>(null);
   275→  const [capabilities, setCapabilities] = useState<Capability[]>([]);
   276→  const [doctor, setDoctor] = useState<DoctorResult | null>(null);
   277→  const [memory, setMemory] = useState<MemoryStats[]>([]);
   278→  const [skills, setSkills] = useState<Skill[]>([]);
   279→  const [observability, setObservability] = useState<Observability | null>(null);
   280→  const [mcp, setMcp] = useState<McpData | null>(null);
   281→  const [security, setSecurity] = useState<SecurityData | null>(null);
   282→  const [agentComm, setAgentComm] = useState<AgentComm | null>(null);
   283→
   284→  /* ── Fetch helpers ── */
   285→  const fetchJson = useCallback(
   286→    async (path: string): Promise<unknown> => {
   287→      try {
   288→        const res = await fetch(path);
   289→        if (!res.ok) return null;
   290→        return await res.json();
   291→      } catch {
   292→        return null;
   293→      }
   294→    },
   295→    []
   296→  );
   297→
   298→  /* ── Load all data ── */
   299→  const loadAll = useCallback(async () => {
   300→    setLoading(true);
   301→    const [
   302→      identRes,
   303→      missionRes,
   304→      capRes,
   305→      docRes,
   306→      memRes,
   307→      skillsRes,
   308→      obsRes,
   309→      mcpRes,
   310→      secRes,
   311→      commRes,
   312→    ] = await Promise.all([
   313→      fetchJson("/api/adn/identity"),
   314→      fetchJson("/api/adn/mission"),
   315→      fetchJson("/api/adn/capabilities"),
   316→      fetchJson("/api/adn/doctor"),
   317→      fetchJson("/api/adn/memory?limit=50"),
   318→      fetchJson("/api/adn/skills"),
   319→      fetchJson("/api/adn/observability?range=24h"),
   320→      fetchJson("/api/adn/mcp"),
   321→      fetchJson("/api/adn/security"),
   322→      fetchJson("/api/adn/agent-comm?limit=10"),
   323→    ]);
   324→
   325→    setIdentity((identRes as Identity) ?? null);
   326→    setMission((missionRes as Mission) ?? null);
   327→    setCapabilities(
   328→      Array.isArray(capRes) ? (capRes as Capability[]) : (capRes as { capabilities?: Capability[] })?.capabilities ?? []
   329→    );
   330→    setDoctor((docRes as DoctorResult) ?? null);
   331→    setMemory(
   332→      Array.isArray(memRes) ? (memRes as MemoryStats[]) : (memRes as { types?: MemoryStats[] })?.types ?? []
   333→    );
   334→    setSkills(
   335→      Array.isArray(skillsRes) ? (skillsRes as Skill[]) : (skillsRes as { skills?: Skill[] })?.skills ?? []
   336→    );
   337→    setObservability((obsRes as Observability) ?? null);
   338→    setMcp((mcpRes as McpData) ?? null);
   339→    setSecurity((secRes as SecurityData) ?? null);
   340→    setAgentComm((commRes as AgentComm) ?? null);
   341→    setLoading(false);
   342→  }, [fetchJson]);
   343→
   344→  useEffect(() => {
   345→    loadAll();
   346→  }, [loadAll]);
   347→
   348→  /* ── Doctor run ── */
   349→  const runDoctor = useCallback(async () => {
   350→    setDoctorRunning(true);
   351→    const res = await fetchJson("/api/adn/doctor");
   352→    if (res) setDoctor(res as DoctorResult);
   353→    setDoctorRunning(false);
   354→  }, [fetchJson]);
   355→
   356→  /* ════════════ Sections ════════════ */
   357→
   358→  /* ── Identity ── */
   359→  function IdentitySection() {
   360→    if (!identity) return <EmptyState icon={Shield} text="Sin datos de identidad" />;
   361→    return (
   362→      <SectionCard>
   363→        <SectionHeading icon={Shield} title="Identidad del Agente" />
   364→        <div>
   365→          <FieldRow icon={Bot} label="Nombre" value={<span className="font-semibold gradient-text">{identity.name}</span>} />
   366→          <FieldRow icon={Hash} label="Versión" value={identity.version} />
   367→          <FieldRow icon={Globe} label="Dominio" value={identity.domain} />
   368→          <FieldRow icon={Target} label="Misión" value={identity.mission} />
   369→          <FieldRow icon={Activity} label="Status" value={<StatusBadge status={identity.status} />} />
   370→          <FieldRow icon={Network} label="Ecosistema" value={identity.ecosystem} />
   371→        </div>
   372→      </SectionCard>
   373→    );
   374→  }
   375→
   376→  /* ── Mission ── */
   377→  function MissionSection() {
   378→    if (!mission) return <EmptyState icon={Target} text="Sin datos de misión" />;
   379→    return (
   380→      <div className="space-y-4">
   381→        {/* In Scope */}
   382→        <SectionCard>
   383→          <SectionHeading icon={CheckCircle2} title="En Scope (In-Scope)" />
   384→          {mission.in_scope && mission.in_scope.length > 0 ? (
   385→            <div className="space-y-2">
   386→              {mission.in_scope.map((item, i) => (
   387→                <div
   388→                  key={i}
   389→                  className="flex items-center justify-between p-2.5 rounded-xl bg-white/[0.02] border border-white/[0.04]"
   390→                >
   391→                  <span className="text-[13px] text-white/80">{item.name}</span>
   392→                  <StatusBadge status={item.status} />
   393→                </div>
   394→              ))}
   395→            </div>
   396→          ) : (
   397→            <p className="text-xs text-white/30">Ningún item en scope</p>
   398→          )}
   399→        </SectionCard>
   400→
   401→        {/* Out of Scope */}
   402→        <SectionCard>
   403→          <SectionHeading icon={XCircle} title="Fuera de Scope" />
   404→          {mission.out_of_scope && mission.out_of_scope.length > 0 ? (
   405→            <div className="flex flex-wrap gap-2">
   406→              {mission.out_of_scope.map((item, i) => (
   407→                <span
   408→                  key={i}
   409→                  className="inline-flex items-center gap-1.5 px-3 py-1.5 rounded-lg text-[11px] font-medium bg-red-500/10 text-red-400 border border-red-500/20"
   410→                >
   411→                  <XCircle className="w-3 h-3" />
   412→                  {item}
   413→                </span>
   414→              ))}
   415→            </div>
   416→          ) : (
   417→            <p className="text-xs text-white/30">Nada fuera de scope</p>
   418→          )}
   419→        </SectionCard>
   420→
   421→        {/* Delegatable */}
   422→        <SectionCard>
   423→          <SectionHeading icon={ArrowRightLeft} title="Delegable" />
   424→          {mission.delegatable && mission.delegatable.length > 0 ? (
   425→            <div className="flex flex-wrap gap-2">
   426→              {mission.delegatable.map((item, i) => (
   427→                <span
   428→                  key={i}
   429→                  className="inline-flex items-center gap-1.5 px-3 py-1.5 rounded-lg text-[11px] font-medium bg-blue-500/10 text-blue-400 border border-blue-500/20"
   430→                >
   431→                  <ArrowRight className="w-3 h-3" />
   432→                  {item}
   433→                </span>
   434→              ))}
   435→            </div>
   436→          ) : (
   437→            <p className="text-xs text-white/30">Nada delegable</p>
   438→          )}
   439→        </SectionCard>
   440→      </div>
   441→    );
   442→  }
   443→

   451→          icon={Zap}
   452→          title="Capacidades"

   454→            <span className="text-[11px] text-white/30">
   455→              {capabilities.length} total
   456→            </span>

   461→            <div
   462→              key={i}
   463→              className="p-3 rounded-xl bg-white/[0.02] border border-white/[0.04] hover:border-white/[0.08] transition-colors"
   464→            >
   465→              <div className="flex items-center justify-between mb-1.5">
   466→                <span className="text-[13px] font-semibold text-white/90">
   467→                  {cap.name}
   468→                </span>
   469→                <StatusBadge status={cap.status} />
   470→              </div>
   471→              <div className="flex flex-wrap items-center gap-3 text-[11px] text-white/40">

   474→                    <ShieldAlert className="w-3 h-3" />

   486→                    <Wrench className="w-3 h-3" />
   487→                    Tools:
   488→                    <span className="text-white/60">

   491→                  </span>

   493→              </div>
   494→            </div>

   496→        </div>
   497→      </SectionCard>
   498→    );
   499→  }
   500→
   501→  /* ── Doctor ── */
   502→  function DoctorSection() {
   503→    return (
   504→      <SectionCard>
   505→        <SectionHeading
   506→          icon={Stethoscope}
   507→          title="Health Check"
   508→          action={
   509→            <button
   510→              onClick={runDoctor}
   511→              disabled={doctorRunning}
   512→              className="inline-flex items-center gap-1.5 px-3 py-1.5 rounded-lg text-[11px] font-semibold bg-gradient-to-r from-[#3b82f6] to-[#7c3aed] text-white hover:opacity-90 transition-opacity disabled:opacity-50"
   513→            >
   514→              {doctorRunning ? (
   515→                <Loader2 className="w-3 h-3 animate-spin" />
   516→              ) : (
   517→                <RefreshCw className="w-3 h-3" />
   518→              )}
   519→              {doctorRunning ? "Ejecutando..." : "Run Doctor"}
   520→            </button>
   521→          }
   522→        />
   523→
   524→        {doctor ? (
   525→          <div className="space-y-3">
   526→            {/* Overall status */}
   527→            <div className="flex items-center justify-between p-3 rounded-xl bg-white/[0.02] border border-white/[0.04]">
   528→              <span className="text-[13px] text-white/70">Estado General</span>
   529→              <StatusBadge status={doctor.status} />
   530→            </div>
   531→
   532→            {doctor.timestamp && (
   533→              <div className="flex items-center gap-2 text-[11px] text-white/30">
   534→                <Clock className="w-3 h-3" />
   535→                Último check: {new Date(doctor.timestamp).toLocaleString("es-ES")}
   536→              </div>
   537→            )}
   538→
   539→            {/* Individual checks */}
   540→            {doctor.checks && doctor.checks.length > 0 && (
   541→              <div className="space-y-2 mt-2">
   542→                {doctor.checks.map((check, i) => (
   543→                  <div
   544→                    key={i}
   545→                    className="flex items-center justify-between p-2.5 rounded-xl bg-white/[0.02] border border-white/[0.04]"
   546→                  >
   547→                    <div className="flex items-center gap-2">
   548→                      <StatusBadge status={check.status} />
   549→                      <span className="text-[12px] text-white/70">
   550→                        {check.name}
   551→                      </span>
